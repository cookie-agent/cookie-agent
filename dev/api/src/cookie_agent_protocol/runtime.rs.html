62impl RuntimeSnapshotV1 {
63    pub fn validate(&self) -> Result<(), &'static str> {
64        if self.catalog_source == CatalogSource::Bootstrap && !self.catalog_state.stale {
65            return Err("bootstrap catalog state must be stale");
66        }
67        if self.catalog_source == CatalogSource::Network && self.catalog_state.stale {
68            return Err("network catalog state cannot be stale");
69        }
70        if self.providers.len() > 4096
71            || self
72                .providers
73                .windows(2)
74                .any(|pair| pair[0].id >= pair[1].id)
75        {
76            return Err("providers must be strictly sorted and unique with at most 4096 entries");
77        }
78        if self.models.len() > 4096
79            || self
80                .models
81                .windows(2)
82                .any(|pair| pair[0].key >= pair[1].key)
83        {
84            return Err("models must be strictly sorted and unique with at most 4096 entries");
85        }
86        if self.agents.len() > 4096
87            || self
88                .agents
89                .windows(2)
90                .any(|pair| (&pair[0].preset, &pair[0].id) >= (&pair[1].preset, &pair[1].id))
91            || self.agents.iter().any(|agent| agent.validate().is_err())
92        {
93            return Err(
94                "agents must be strictly sorted and unique by preset and ID with at most 4096 entries",
95            );
96        }
97        Ok(())
98    }
99}

166#[derive(Clone, Debug, Eq, JsonSchema, PartialEq, Serialize, TS)]
167#[serde(deny_unknown_fields)]
168pub struct RuntimeChangedNotification {
169    #[serde(deserialize_with = "crate::deserialize_required_option")]
170    #[schemars(with = "crate::NullableSchema<RuntimeRevision>", required)]
171    #[ts(type = "RuntimeRevision | null")]
172    pub previous_revision: Option<RuntimeRevision>,
173    pub snapshot: RuntimeSnapshotV1,
174    #[schemars(length(min = 1, max = 9))]
175    pub reasons: Vec<RuntimeChangeReason>,
176}
177impl<'de> Deserialize<'de> for RuntimeChangedNotification {
178    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
179    where
180        D: serde::Deserializer<'de>,
181    {
182        #[derive(Deserialize)]
183        #[serde(deny_unknown_fields)]
184        struct Wire {
185            #[serde(deserialize_with = "crate::deserialize_required_option")]
186            previous_revision: Option<RuntimeRevision>,
187            snapshot: RuntimeSnapshotV1,
188            reasons: Vec<RuntimeChangeReason>,
189        }
190        let wire = Wire::deserialize(deserializer)?;
191        if wire.reasons.is_empty()
192            || wire.reasons.len() > 9
193            || wire.reasons.windows(2).any(|pair| pair[0] >= pair[1])
194        {
195            return Err(serde::de::Error::custom(
196                "runtime change reasons must be a nonempty strictly sorted unique set",
197            ));
198        }
199        Ok(Self {
200            previous_revision: wire.previous_revision,
201            snapshot: wire.snapshot,
202            reasons: wire.reasons,
203        })
204    }
205}
