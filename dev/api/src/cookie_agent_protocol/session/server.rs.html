36/// Per-connection facilities available to a server implementation.
37#[derive(Clone)]
38pub struct ServerContext {
39    notifications: mpsc::Sender<Value>,
40    shutdown: CancellationToken,
41}
42
43impl ServerContext {
44    /// Emit one JSON-RPC notification in connection order.
45    pub async fn notify<T: Serialize>(
46        &self,
47        method: &str,
48        params: &T,
49    ) -> Result<(), TransportError> {
50        let params = serde_json::to_value(params)?;
51        let notification = serde_json::to_value(Notification::new(method, Some(params)))?;
52        tokio::select! {
53            () = self.shutdown.cancelled() => Err(TransportError::Closed),
54            result = self.notifications.send(notification) => result.map_err(|_| TransportError::Closed),
55        }
56    }
57
58    /// A cancellation token scoped to this protocol connection.
59    #[must_use]
60    pub fn shutdown(&self) -> CancellationToken {
61        self.shutdown.clone()
62    }
63}

249/// Run one complete server-side protocol session over a transport.
250pub async fn serve<T, S>(
251    server: Arc<S>,
252    mut transport: T,
253    shutdown: CancellationToken,
254) -> Result<(), TransportError>
255where
256    T: Transport,
257    S: ServerProtocol,
258{
259    let (notifications, mut notification_rx) = mpsc::channel(OUTBOUND_QUEUE_CAPACITY);
260    let context = ServerContext {
261        notifications,
262        shutdown,
263    };
264    let _guard = ConnectionShutdown(context.shutdown());
265    let mut handshaken = false;
266    loop {
267        tokio::select! {
268            () = context.shutdown.cancelled() => return Ok(()),
269            incoming = transport.recv() => {
270                let Some(frame) = incoming? else { return Ok(()); };
271                let incoming = match parse_incoming(frame).and_then(classify_incoming) {
272                    Ok(incoming) => incoming,
273                    Err(fault) => {
274                        transport.send(MessageFrame::Value(error_response(None, fault)?)).await?;
275                        continue;
276                    }
277                };
278                match incoming {
279                    Incoming::Request { id, method, params } => {
280                        let (response, connected) = if !handshaken {
281                            if method != "handshake" {
282                                (error_response(Some(id), handshake_required())?, false)
283                            } else {
284                                match decode_handshake(params) {
285                                    Ok(()) => {
286                                        handshaken = true;
287                                        (success_response(id, &ServerHello { protocol_version: crate::ProtocolVersion::current() })?, true)
288                                    }
289                                    Err(fault) => (error_response(Some(id), fault)?, false),
290                                }
291                            }
292                        } else {
293                            let result = if method == "handshake" {
294                                decode_handshake(params).map(|()| {
295                                    serde_json::to_value(ServerHello {
296                                        protocol_version: crate::ProtocolVersion::current(),
297                                    })
298                                    .expect("server hello serializes")
299                                })
300                            } else {
301                                dispatch(server.as_ref(), &method, params, &context, true).await
302                            };
303                            (match result {
304                                Ok(value) => success_response(id, &value)?,
305                                Err(fault) => error_response(Some(id), fault)?,
306                            }, false)
307                        };
308                        transport.send(MessageFrame::Value(response)).await?;
309                        if connected {
310                            server.connected(context.clone()).await;
311                        }
312                    }
313                    Incoming::Notification { method, params } => {
314                        if handshaken {
315                            if method == "handshake" {
316                                let _ = decode_handshake(params);
317                            } else {
318                                let _ = dispatch(server.as_ref(), &method, params, &context, false).await;
319                            }
320                        }
321                    }
322                }
323            }
324            Some(notification) = notification_rx.recv() => {
325                transport.send(MessageFrame::Value(notification)).await?;
326            }
327        }
328    }
329}

331async fn dispatch<S: ServerProtocol>(
332    server: &S,
333    method: &str,
334    params: Option<Value>,
335    context: &ServerContext,
336    has_request_id: bool,
337) -> Result<Value, ServerFault> {
338    match method {
339        "session.create" => value(server.create_session(decode(params)?).await?),
340        "session.list" => value(server.list_sessions(decode_default(params)?).await?),
341        "session.get" => value(server.get_session(decode(params)?).await?),
342        "session.usage" => value(server.session_usage(decode(params)?).await?),
343        "agent.usage" => value(server.agent_usage(decode(params)?).await?),
344        "usage.global" => value(server.global_usage(decode_default(params)?).await?),
345        "session.children" => value(server.session_children(decode(params)?).await?),
346        "session.tree" => value(server.session_tree(decode(params)?).await?),
347        "session.resume" => value(server.resume_session(decode(params)?).await?),
348        "session.rename" => value(server.rename_session(decode_rename(params)?).await?),
349        "session.set_permission_mode" => value(server.set_permission_mode(decode(params)?).await?),
350        "session.permission.get" => value(server.get_session_permissions(decode(params)?).await?),
351        "session.permission.set" => value(server.set_session_permission(decode(params)?).await?),
352        "session.permission.clear" => {
353            value(server.clear_session_permission(decode(params)?).await?)
354        }
355        "session.compact" => value(server.compact_session(decode(params)?).await?),
356        "session.revert" => value(server.revert_session(decode(params)?).await?),
357        "session.fork" => value(server.fork_session(decode(params)?).await?),
358        "run.start" => value(server.start_run(decode(params)?).await?),
359        "run.steer" => value(server.steer_run(decode(params)?).await?),
360        "run.recall_steer" => value(server.recall_steer(decode(params)?).await?),
361        "run.cancel" => value(server.cancel_run(decode(params)?).await?),
362        "run.tool_stdin" => value(server.tool_stdin(decode(params)?).await?),
363        "events.subscribe" => value(server.subscribe_events(decode(params)?, context).await?),
364        "approval.respond" => value(server.respond_approval(decode(params)?).await?),
365        "approval.list" => value(server.list_approvals(decode(params)?).await?),
366        "mcp.auth.begin" => value(server.begin_mcp_auth(decode(params)?).await?),
367        "mcp.auth.cancel" => value(server.cancel_mcp_auth(decode(params)?).await?),
368        "mcp.server.list" => value(server.list_mcp_servers(decode_default(params)?).await?),
369        "mcp.server.add" => value(server.add_mcp_server(decode(params)?).await?),
370        "mcp.server.edit" => value(server.edit_mcp_server(decode(params)?).await?),
371        "mcp.server.remove" => value(server.remove_mcp_server(decode(params)?).await?),
372        "mcp.server.set_enabled" => value(server.set_mcp_server_enabled(decode(params)?).await?),
373        "mcp.server.reconnect" => value(server.reconnect_mcp_server(decode(params)?).await?),
374        "mcp.server.persist" => value(server.persist_mcp_server(decode(params)?).await?),
375        crate::RUNTIME_SNAPSHOT_GET_METHOD if has_request_id => {
376            value(server.runtime_snapshot(decode_default(params)?).await?)
377        }
378        crate::PROVIDER_CONNECT_METHOD if has_request_id => {
379            value(server.connect_provider(decode(params)?).await?)
380        }
381        crate::PROVIDER_DISCONNECT_METHOD if has_request_id => {
382            value(server.disconnect_provider(decode(params)?).await?)
383        }
384        crate::RUNTIME_SNAPSHOT_GET_METHOD
385        | crate::PROVIDER_CONNECT_METHOD
386        | crate::PROVIDER_DISCONNECT_METHOD => {
387            Err(ServerFault::new(-32600, "request id required", None))
388        }
389        _ => Err(ServerFault::method_not_found()),
390    }
391}

393fn decode<T: serde::de::DeserializeOwned>(params: Option<Value>) -> Result<T, ServerFault> {
394    serde_json::from_value(params.unwrap_or(Value::Null)).map_err(|_| ServerFault::invalid_params())
395}

397fn decode_default<T>(params: Option<Value>) -> Result<T, ServerFault>
398where
399    T: serde::de::DeserializeOwned + Default,
400{
401    match params {
402        Some(params) => serde_json::from_value(params).map_err(|_| ServerFault::invalid_params()),
403        None => Ok(T::default()),
404    }

407#[derive(Deserialize)]
408#[serde(deny_unknown_fields)]
409struct RawSessionRenameParams {
410    session_id: SessionId,
411    client_rename_id: ClientRenameId,
412    change: RawSessionRenameChange,
413}
414
415#[derive(Deserialize)]
416#[serde(tag = "type", rename_all = "snake_case", deny_unknown_fields)]
417enum RawSessionRenameChange {
418    Set { title: String },
419    Clear,
420    Reset,
421}
422
423fn decode_rename(params: Option<Value>) -> Result<SessionRenameParams, ServerFault> {
424    let params = params.unwrap_or(Value::Null);
425    if let Ok(request) = serde_json::from_value::<SessionRenameParams>(params.clone()) {
426        return Ok(request);
427    }
428    if serde_json::to_vec(&params)
429        .map_err(|_| ServerFault::invalid_params())?
430        .len()
431        > MAX_RAW_RENAME_PARAMS_BYTES
432    {
433        return Err(ServerFault::invalid_params());
434    }
435    let raw: RawSessionRenameParams =
436        serde_json::from_value(params).map_err(|_| ServerFault::invalid_params())?;
437    let invalid_title = rename_invalid_title(&raw);
438    let change = match raw.change {
439        RawSessionRenameChange::Set { title } => SessionRenameChange::Set {
440            title: SessionTitle::new(title).map_err(|_| invalid_title)?,
441        },
442        RawSessionRenameChange::Clear => SessionRenameChange::Clear,
443        RawSessionRenameChange::Reset => SessionRenameChange::Reset,
444    };
445    Ok(SessionRenameParams {
446        session_id: raw.session_id,
447        client_rename_id: raw.client_rename_id,
448        change,
449    })
450}
451
452fn rename_invalid_title(raw: &RawSessionRenameParams) -> ServerFault {
453    ServerFault::new(
454        -32602,
455        "session rename rejected",
456        serde_json::to_value(SessionRenameError {
457            code: SessionRenameErrorCode::InvalidTitle,
458            session_id: raw.session_id,
459            client_rename_id: raw.client_rename_id.clone(),
460        })
461        .ok(),
462    )
463}

481fn parse_incoming(frame: MessageFrame) -> Result<Value, ServerFault> {
482    match frame {
483        MessageFrame::Value(value) => Ok(value),
484        MessageFrame::Text(text) => {
485            serde_json::from_str(&text).map_err(|_| ServerFault::new(-32700, "parse error", None))
486        }
487    }
488}

490fn classify_incoming(value: Value) -> Result<Incoming, ServerFault> {
491    let object = value.as_object().ok_or_else(|| {
492        ServerFault::new(
493            -32600,
494            "invalid request",
495            Some(json!({ "detail": "a batch or scalar is not a request" })),
496        )
497    })?;
498    if object.contains_key("id") {
499        let request: Request = serde_json::from_value(value).map_err(|_| {
500            ServerFault::new(
501                -32600,
502                "invalid request",
503                Some(json!({ "detail": "invalid JSON-RPC request envelope" })),
504            )
505        })?;
506        if request.method.is_empty() {
507            return Err(ServerFault::new(
508                -32600,
509                "invalid request",
510                Some(json!({ "detail": "method must be non-empty" })),
511            ));
512        }
513        Ok(Incoming::Request {
514            id: request.id,
515            method: request.method,
516            params: request.params,
517        })
518    } else {
519        let notification: Notification = serde_json::from_value(value).map_err(|_| {
520            ServerFault::new(
521                -32600,
522                "invalid request",
523                Some(json!({ "detail": "invalid JSON-RPC notification envelope" })),
524            )
525        })?;
526        if notification.method.is_empty() {
527            return Err(ServerFault::new(
528                -32600,
529                "invalid request",
530                Some(json!({ "detail": "method must be non-empty" })),
531            ));
532        }
533        Ok(Incoming::Notification {
534            method: notification.method,
535            params: notification.params,
536        })
537    }
538}

550fn success_response<T: Serialize>(id: JsonRpcId, result: &T) -> Result<Value, TransportError> {
551    Ok(serde_json::to_value(Response::Success(SuccessResponse {
552        jsonrpc: JsonRpcVersion::current(),

558fn error_response(id: Option<JsonRpcId>, fault: ServerFault) -> Result<Value, TransportError> {
559    match id {
560        Some(id) => Ok(serde_json::to_value(Response::Error(ErrorResponse {
561            jsonrpc: JsonRpcVersion::current(),
562            id,
563            error: JsonRpcError {
564                code: fault.code,
565                message: fault.message.into(),
566                data: fault.data,
567            },
568        }))?),
569        None => Ok(json!({
570            "jsonrpc": "2.0",
571            "id": null,
572            "error": { "code": fault.code, "message": fault.message, "data": fault.data },
573        })),
574    }
