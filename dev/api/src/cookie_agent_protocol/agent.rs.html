117pub struct FrozenDelegationPolicy {
118    #[schemars(length(min = 1, max = 256))]
119    #[ts(type = "Array<AgentId>")]
120    pub targets: Vec<AgentId>,
121    pub effective_depth_ceiling: u32,
122}
123impl FrozenDelegationPolicy {
124    pub fn validate(&self) -> Result<(), AgentSchemaError> {
125        if self.targets.is_empty() || self.targets.len() > 256 {
126            return Err(AgentSchemaError::InvalidDelegationTargets);
127        }
128        if self.targets.windows(2).any(|pair| pair[0] >= pair[1]) {
129            return Err(AgentSchemaError::InvalidDelegationTargets);
130        }
131        Ok(())
132    }
133}

172    pub permissions: Vec<PermissionRule>,
173    #[serde(deserialize_with = "crate::deserialize_required_option")]
174    #[schemars(with = "crate::NullableSchema<FrozenDelegationPolicy>", required)]
175    pub delegation: Option<FrozenDelegationPolicy>,

177    pub fallback_chain: Vec<FrozenModelBinding>,
178    pub selected_suffix_start: u32,
179}

181impl AgentSnapshot {
182    pub const MAX_PROMPT_BYTES: usize = 128 * 1024;
183    pub fn validate(&self) -> Result<(), AgentSchemaError> {
184        if self.description.trim().is_empty()
185            || self.description.len() > 512
186            || self.description.chars().any(char::is_control)
187        {
188            return Err(AgentSchemaError::InvalidDescription);
189        }
190        if self.composed_prompt.trim().is_empty()
191            || self.composed_prompt.len() > Self::MAX_PROMPT_BYTES
192        {
193            return Err(AgentSchemaError::InvalidPrompt);
194        }
195        if self.tools.len() > 256
196            || self.permissions.len() > 256
197            || self.fallback_chain.is_empty()
198            || self.fallback_chain.len() > 256
199        {
200            return Err(AgentSchemaError::InvalidListBounds);
201        }
202        let mut tools = BTreeSet::new();
203        if !self.tools.iter().all(|tool| tools.insert(*tool)) {
204            return Err(AgentSchemaError::DuplicateTool);
205        }
206        let mut models = BTreeSet::<ModelKey>::new();
207        if !self
208            .fallback_chain
209            .iter()
210            .all(|binding| models.insert(binding.selection.model.clone()))
211        {
212            return Err(AgentSchemaError::DuplicateFallbackModel);
213        }
214        for binding in &self.fallback_chain {
215            binding
216                .validate()
217                .map_err(|_| AgentSchemaError::InvalidModelBinding)?;
218        }
219        if self.selected_suffix_start as usize >= self.fallback_chain.len() {
220            return Err(AgentSchemaError::InvalidSuffixStart);
221        }
222        if let Some(delegation) = &self.delegation {
223            delegation.validate()?;
224        }
225        Ok(())
226    }
227
228    pub fn validate_selection(&self, selection: &RunSelection) -> Result<(), AgentSchemaError> {
229        self.validate()?;
230        let start = self.selected_suffix_start as usize;
231        if selection.agent != self.agent
232            || self.fallback_chain[start].selection.model != selection.model.model
233        {
234            return Err(AgentSchemaError::SelectionMismatch);
235        }
236        Ok(())
237    }
238
239    pub fn validate_selected_suffix(
240        &self,
241        selection: &RunSelection,
242        selected_suffix: &[FrozenModelBinding],
243    ) -> Result<(), AgentSchemaError> {
244        self.validate_selection(selection)?;
245        let start = self.selected_suffix_start as usize;
246        let expected = &self.fallback_chain[start..];
247        if selected_suffix.len() != expected.len() || selected_suffix.is_empty() {
248            return Err(AgentSchemaError::SelectedSuffixMismatch);
249        }
250        let selected_head = &selected_suffix[0];
251        let fallback_head = &expected[0];
252        if selected_head.selection != selection.model
253            || selected_head.descriptor != fallback_head.descriptor
254        {
255            return Err(AgentSchemaError::SelectedSuffixMismatch);
256        }
257        if selection.model.variant == fallback_head.selection.variant
258            && selected_head != fallback_head
259        {
260            return Err(AgentSchemaError::SelectedSuffixMismatch);
261        }
262        if selected_suffix[1..] != expected[1..] {
263            return Err(AgentSchemaError::SelectedSuffixMismatch);
264        }
265        Ok(())
266    }
267}

368impl AgentDescriptor {
369    pub fn validate(&self) -> Result<(), AgentSchemaError> {
370        if self.description.trim().is_empty()
371            || self.description.len() > 512
372            || self.description.chars().any(char::is_control)
373        {
374            return Err(AgentSchemaError::InvalidDescription);
375        }
376        if self.resolved_fallback.len() > 256
377            || self.tools.len() > 256
378            || self.delegation_targets.len() > 256
379        {
380            return Err(AgentSchemaError::InvalidListBounds);
381        }
382        let mut models = BTreeSet::new();
383        if !self
384            .resolved_fallback
385            .iter()
386            .all(|selection| models.insert(selection.model.clone()))
387        {
388            return Err(AgentSchemaError::DuplicateFallbackModel);
389        }
390        if self
391            .delegation_targets
392            .windows(2)
393            .any(|pair| pair[0] >= pair[1])
394        {
395            return Err(AgentSchemaError::InvalidDelegationTargets);
396        }
397        if self.runnable_as_root
398            && (!self.enabled
399                || matches!(self.mode, AgentMode::Subagent | AgentMode::Internal)
400                || self.resolved_fallback.is_empty())
401        {
402            return Err(AgentSchemaError::InvalidRootRunnable);
403        }
404        Ok(())
405    }
406}

441#[derive(Clone, Copy, Debug, Eq, PartialEq)]
442pub enum AgentSchemaError {
443    InvalidDelegationTargets,
444    InvalidDescription,
445    InvalidPrompt,
446    InvalidListBounds,
447    DuplicateTool,
448    DuplicateFallbackModel,
449    InvalidModelBinding,
450    InvalidSuffixStart,
451    SelectionMismatch,
452    SelectedSuffixMismatch,
453    InvalidRootRunnable,
454}
455impl fmt::Display for AgentSchemaError {
456    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
457        f.write_str(match self{Self::InvalidDelegationTargets=>"delegation targets must be a nonempty, strictly sorted unique list of at most 256 IDs",Self::InvalidDescription=>"agent description must be nonblank, control-free, and 1..=512 bytes",Self::InvalidPrompt=>"composed prompt must be nonblank and at most 128 KiB",Self::InvalidListBounds=>"agent list exceeds bounds or frozen fallback is empty",Self::DuplicateTool=>"agent tools must be unique",Self::DuplicateFallbackModel=>"fallback chains may contain each model key at most once",Self::InvalidModelBinding=>"agent snapshot contains an invalid frozen model binding",Self::InvalidSuffixStart=>"selected_suffix_start must index the frozen fallback chain",Self::SelectionMismatch=>"run selection does not match the snapshot agent and selected fallback start",Self::SelectedSuffixMismatch=>"selected suffix does not exactly match the frozen fallback order and selected head",Self::InvalidRootRunnable=>"runnable_as_root contradicts enabled mode or fallback state"})
458    }
459}
