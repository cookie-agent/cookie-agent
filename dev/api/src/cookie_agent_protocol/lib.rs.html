13impl<T: JsonSchema> JsonSchema for NullableSchema<T> {
14    fn inline_schema() -> bool {
15        true
16    }
17    fn schema_name() -> Cow<'static, str> {
18        Cow::Owned(format!("Nullable{}", T::schema_name()))

21        generator.subschema_for::<Option<T>>()

33#[derive(Deserialize)]
34#[serde(deny_unknown_fields)]
35struct RequiredModelSelectionWire {
36    model: ModelKey,
37    #[serde(deserialize_with = "deserialize_required_option")]
38    variant: Option<VariantId>,

41pub(crate) fn deserialize_required_model_selection<'de, D>(
42    deserializer: D,
43) -> Result<ModelSelection, D::Error>
44where
45    D: serde::Deserializer<'de>,
46{
47    let wire = RequiredModelSelectionWire::deserialize(deserializer)?;
48    Ok(ModelSelection {
49        model: wire.model,
50        variant: wire.variant,
51    })
52}
53
54pub(crate) fn deserialize_required_model_selections<'de, D>(
55    deserializer: D,
56) -> Result<Vec<ModelSelection>, D::Error>
57where
58    D: serde::Deserializer<'de>,
59{
60    Vec::<RequiredModelSelectionWire>::deserialize(deserializer).map(|values| {
61        values
62            .into_iter()
63            .map(|wire| ModelSelection {
64                model: wire.model,
65                variant: wire.variant,
66            })
67            .collect()
68    })
69}

71pub(crate) struct RequiredModelSelectionSchema;
72
73impl JsonSchema for RequiredModelSelectionSchema {
74    fn inline_schema() -> bool {
75        true
76    }
77
78    fn schema_name() -> Cow<'static, str> {
79        ModelSelection::schema_name()
80    }
81
82    fn json_schema(generator: &mut SchemaGenerator) -> Schema {
83        let mut schema = ModelSelection::json_schema(generator);
84        schema
85            .ensure_object()
86            .insert("required".into(), serde_json::json!(["model", "variant"]));
87        schema
88    }
89}

144macro_rules! exact_numeric_wire_type {
145    ($name:ident, $value:expr, $ts:literal, $description:literal) => {
146        #[doc = $description]
147        #[derive(Clone, Copy, Debug, Default, Eq, Hash, PartialEq, TS)]
148        #[ts(type = $ts)]
149        pub struct $name(());
150
151        impl $name {
152            #[must_use]
153            pub const fn current() -> Self {
154                Self(())
155            }
156
157            #[must_use]
158            pub const fn value(self) -> u32 {
159                $value
160            }
161        }
162
163        impl Serialize for $name {
164            fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
165            where
166                S: serde::Serializer,
167            {
168                serializer.serialize_u32($value)
169            }
170        }
171
172        impl<'de> Deserialize<'de> for $name {
173            fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
174            where
175                D: serde::Deserializer<'de>,
176            {
177                let value = u32::deserialize(deserializer)?;
178                if value == $value {
179                    Ok(Self::current())
180                } else {
181                    Err(serde::de::Error::custom(format!(
182                        "unsupported exact wire version {value}; expected {}",
183                        $value
184                    )))
185                }
186            }
187        }
188
189        impl JsonSchema for $name {
190            fn inline_schema() -> bool {
191                true
192            }
193
194            fn schema_name() -> Cow<'static, str> {
195                Cow::Borrowed(stringify!($name))
196            }
197
198            fn json_schema(_generator: &mut SchemaGenerator) -> Schema {
199                json_schema!({"type": "integer", "const": $value})
200            }
201        }
202    };
203}

206/// An event/session-JSONL schema accepted by this build.
207#[derive(Clone, Copy, Debug, Eq, Hash, PartialEq, TS)]
208#[ts(type = "15 | 16 | 17 | 18")]
209pub struct EventSchemaVersion(u32);
210
211impl EventSchemaVersion {
212    #[must_use]
213    pub const fn current() -> Self {
214        Self(EVENT_SCHEMA_VERSION)
215    }
216
217    #[must_use]
218    pub const fn value(self) -> u32 {
219        self.0
220    }
221}
222
223impl Default for EventSchemaVersion {
224    fn default() -> Self {
225        Self::current()
226    }
227}
228
229impl Serialize for EventSchemaVersion {
230    fn serialize<S>(&self, serializer: S) -> Result<S::Ok, S::Error>
231    where
232        S: serde::Serializer,
233    {
234        serializer.serialize_u32(self.0)
235    }
236}
237
238impl<'de> Deserialize<'de> for EventSchemaVersion {
239    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
240    where
241        D: serde::Deserializer<'de>,
242    {
243        let value = u32::deserialize(deserializer)?;
244        if matches!(value, 15..=18) {
245            Ok(Self(value))
246        } else {
247            Err(serde::de::Error::custom(format!(
248                "unsupported event schema version {value}; expected 15, 16, 17, or 18"
249            )))
250        }
251    }
252}

261    }

263    fn json_schema(_generator: &mut SchemaGenerator) -> Schema {
264        json_schema!({"type": "integer", "enum": [15, 16, 17, 18]})
265    }
