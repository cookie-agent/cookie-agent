383pub struct PersistedToolResult {
384    pub title: SafeDisplayText,
385    pub output: String,
386    pub metadata: Value,
387    #[serde(deserialize_with = "crate::deserialize_required_option")]
388    #[schemars(with = "crate::NullableSchema<ToolOutputTruncation>", required)]
389    pub truncation: Option<ToolOutputTruncation>,
390    #[schemars(length(max = 256))]
391    pub attachments: Vec<ToolAttachment>,
392}
393impl PersistedToolResult {
394    pub const MAX_OUTPUT_BYTES: usize = 2 * 1024 * 1024;
395    pub const MAX_METADATA_BYTES: usize = 2 * 1024 * 1024;
396    pub fn validate(&self) -> Result<(), EventSchemaError> {
397        if self.output.len() > Self::MAX_OUTPUT_BYTES {
398            return Err(EventSchemaError::ToolOutputTooLarge);
399        }
400        if serde_json::to_vec(&self.metadata)
401            .map_err(|_| EventSchemaError::InvalidJson)?
402            .len()
403            > Self::MAX_METADATA_BYTES
404        {
405            return Err(EventSchemaError::ToolMetadataTooLarge);
406        }
407        if self.attachments.len() > 256 {
408            return Err(EventSchemaError::TooManyAttachments);
409        }
410        for attachment in &self.attachments {
411            attachment.reference.validate()?;
412            if attachment.filename.as_ref().is_some_and(|name| {
413                name.is_empty() || name.len() > 512 || name.chars().any(char::is_control)
414            }) {
415                return Err(EventSchemaError::InvalidAttachmentFilename);
416            }
417        }
418        if let Some(truncation) = &self.truncation {
419            truncation.retained.validate()?;
420        }
421        Ok(())
422    }
423}

2304/// Decodes an event payload, replacing only leaves which Serde identifies as
2305/// nullable fields. Required-field and unknown-variant failures remain errors.
2306pub fn deserialize_event_payload_best_effort(mut value: Value) -> Result<EventPayloadRead, String> {
2307    let mut degraded_fields = Vec::new();
2308    let mut previous_path = None;
2309    for _ in 0..=MAX_DEGRADED_FIELDS {
2310        match serde_path_to_error::deserialize::<_, EventPayload>(value.clone().into_deserializer())
2311        {
2312            Ok(payload) => {
2313                return Ok(EventPayloadRead {
2314                    payload,
2315                    degraded_fields,
2316                });
2317            }
2318            Err(error) => {
2319                let path = error.path().to_string();
2320                if path != "." && previous_path.as_deref() == Some(path.as_str()) {
2321                    return Err(error.to_string());
2322                }
2323                let degraded = if path == "." {
2324                    replace_degradable_leaf(&mut value, &error.to_string())
2325                } else if replace_path_with_null(&mut value, error.path()) {
2326                    Some(path.clone())
2327                } else {
2328                    None
2329                };
2330                let Some(degraded) = degraded else {
2331                    return Err(error.to_string());
2332                };
2333                previous_path = Some(path.clone());
2334                degraded_fields.push(degraded);
2335            }
2336        }
2337    }
2338    Err(format!(
2339        "event payload exceeds the {MAX_DEGRADED_FIELDS}-field degradation limit"
2340    ))

2349fn replace_degradable_leaf(value: &mut Value, original_error: &str) -> Option<String> {
2350    let mut paths = Vec::new();
2351    collect_leaf_paths(value, &mut Vec::new(), &mut paths);
2352    for path in paths {
2353        let mut candidate = value.clone();
2354        if !set_value_path(&mut candidate, &path, Value::Null) {
2355            continue;
2356        }
2357        match serde_path_to_error::deserialize::<_, EventPayload>(
2358            candidate.clone().into_deserializer(),
2359        ) {
2360            Ok(_) => {
2361                *value = candidate;
2362                return Some(display_value_path(&path));
2363            }
2364            Err(error)
2365                if error.to_string() != original_error
2366                    && !error.inner().to_string().contains("invalid type: null") =>
2367            {
2368                *value = candidate;
2369                return Some(display_value_path(&path));
2370            }
2371            Err(_) => {}
2372        }
2373    }
2374    let missing = original_error
2375        .split("missing field `")
2376        .nth(1)
2377        .and_then(|suffix| suffix.split('`').next())?;
2378    let mut object_paths = Vec::new();
2379    collect_object_paths(value, &mut Vec::new(), &mut object_paths);
2380    for mut path in object_paths {
2381        path.push(ValuePathSegment::Key(missing.to_owned()));
2382        let mut candidate = value.clone();
2383        if !set_value_path(&mut candidate, &path, Value::Null) {
2384            continue;
2385        }
2386        if serde_json::from_value::<EventPayload>(candidate.clone()).is_ok() {
2387            *value = candidate;
2388            return Some(display_value_path(&path));
2389        }
2390    }
2391    None
2392}

2446fn set_value_path(value: &mut Value, path: &[ValuePathSegment], replacement: Value) -> bool {
2447    let Some((leaf, parents)) = path.split_last() else {
2448        return false;
2449    };
2450    let mut current = value;
2451    for segment in parents {
2452        let Some(next) = (match segment {
2453            ValuePathSegment::Key(key) => current.get_mut(key),
2454            ValuePathSegment::Index(index) => current.get_mut(*index),
2455        }) else {
2456            return false;
2457        };
2458        current = next;
2459    }
2460    match leaf {
2461        ValuePathSegment::Key(key) => current
2462            .as_object_mut()
2463            .map(|object| object.insert(key.clone(), replacement))
2464            .is_some(),
2465        ValuePathSegment::Index(index) => current
2466            .as_array_mut()
2467            .and_then(|array| array.get_mut(*index))
2468            .map(|value| *value = replacement)
2469            .is_some(),
2470    }
2471}

2483            ValuePathSegment::Index(index) => result.push_str(&format!("[{index}]")),

2490    let segments = path
2491        .iter()
2492        .filter(|segment| !matches!(segment, Segment::Enum { .. }))
2493        .collect::<Vec<_>>();

2495        return false;

2498    for segment in parents {
2499        let next = match segment {
2500            Segment::Map { key } => current.get_mut(key),
2501            Segment::Seq { index } => current.get_mut(*index),
2502            Segment::Enum { .. } | Segment::Unknown => None,
2503        };
2504        let Some(next) = next else {
2505            return false;
2506        };
2507        current = next;
2508    }

2510        Segment::Map { key } => {
2511            let Some(object) = current.as_object_mut() else {
2512                return false;
2513            };

2516            true
2517        }
2518        Segment::Seq { index } => current
2519            .as_array_mut()
2520            .and_then(|array| array.get_mut(*index))
2521            .map(|leaf| *leaf = Value::Null)
2522            .is_some(),
2523        Segment::Enum { .. } | Segment::Unknown => false,
2524    }
