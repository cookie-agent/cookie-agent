106    fn test_digest(value: &str) -> cookie_agent_models::Sha256Digest {
107        // A deterministic 64-hex stand-in; config fingerprints are opaque to
108        // the TUI and only carried through to the engine.
109        let base = value.to_owned();
110        let mut hex = String::with_capacity(64);
111        while hex.len() < 64 {
112            hex.push_str(
113                &base
114                    .bytes()
115                    .map(|byte| format!("{byte:02x}"))
116                    .collect::<String>(),
117            );
118        }
119        hex.truncate(64);
120        let hex = hex
121            .chars()
122            .map(|character| {
123                if character.is_ascii_hexdigit() {
124                    character
125                } else {
126                    'a'
127                }
128            })
129            .collect::<String>()
130            .to_lowercase();
131        cookie_agent_models::Sha256Digest::new(hex).expect("test digest")
132    }

274    fn approval_request() -> ApprovalRequest {
275        let resource = PreparedApprovalResource {
276            capability: PermissionAction::Bash,
277            canonical: PreparedResourceIdentity::new("command:git-status")
278                .expect("prepared resource identity"),
279            binding_digest: PreparedResourceDigest::from_canonical_binding_bytes(b"git status"),
280            binding_lifetime: PreparedBindingLifetime::ProcessLocal,
281            boundary: ApprovalBoundary::CommandPrefix {
282                prefix: "git status".into(),
283            },
284            source: ApprovalResourceSource::ModelRequest,
285        };
286        let resource_digest = resource.binding_digest.clone();
287        let operation = PreparedOperationIdentity::new(
288            Sha256Digest::of_bytes(b"normalized arguments"),
289            vec![ApprovalCapability {
290                action: PermissionAction::Bash,
291                operation: PreparedCapabilityOperation::new("execute")
292                    .expect("prepared capability operation"),
293            }],
294            vec![resource],
295            Sha256Digest::of_bytes(b"execution context"),
296        )
297        .expect("prepared operation");
298        ApprovalRequest::new(
299            ApprovalId::new_v7(),
300            7,
301            ApprovalTrigger::PermissionPolicy,
302            operation,
303            vec![ApprovalEvaluation {
304                resource_digest,
305                effect: PermissionEffect::Ask,
306                trace: DecisionTrace {
307                    action: PermissionAction::Bash,
308                    normalized_resource: "git status".into(),
309                    candidates: vec![MatchedPermissionRule {
310                        source_layer: SafeCode::new("test").expect("safe code"),
311                        action: PermissionAction::Bash,
312                        resource: cookie_agent_protocol::WildcardPattern::new("git status")
313                            .expect("wildcard"),
314                        effect: PermissionEffect::Ask,
315                    }],
316                    effect: PermissionEffect::Ask,
317                    precedence_reason: "test".into(),
318                },
319            }],
320            ApprovalConstraints {
321                allow_once: true,
322                allow_tree_grant: false,
323                cancellable: true,
324                expires_at: None,
325            },
326        )
327        .expect("approval request")
328    }

611                mode: cookie_agent_protocol::PermissionMode::Ask,
612            })
613            .await
614            .expect("ask mode");
615
616        let observer = Client::connect_in_process(server.clone());
617        observer.handshake().await.expect("observer handshake");
618        let mut deliveries = observer
619            .subscribe_deliveries()
620            .expect("observer deliveries");
621        observer
622            .subscribe_events(session.session_id, Some(1))
623            .await
624            .expect("subscribe events");
625        let runner = Client::connect_in_process(server);
626        runner.handshake().await.expect("runner handshake");
627        let started = match runner
628            .start_run(cookie_agent_protocol::RunStartParams {
629                session_id: session.session_id,
630                client_run_id: cookie_agent_protocol::ClientRunId::new(format!(
631                    "approval-e2e-{}",
632                    uuid::Uuid::now_v7()
633                ))
634                .expect("client run ID"),
635                selection: test_run_selection(),
636                input: cookie_agent_protocol::encode_skill_submission("fresh-skill", "value"),
637            })
638            .await
639        {
640            Ok(started) => started,
641            Err(crate::ClientError::Rpc(error)) => {
642                let debug_code = error
643                    .data
644                    .as_ref()
645                    .and_then(|data| data.get("debug_code"))
646                    .and_then(serde_json::Value::as_str)
647                    .unwrap_or("missing_debug_code");
648                panic!(
649                    "start direct skill run failed: rpc_code={}, debug_code={debug_code}",
650                    error.code
651                );
652            }
653            Err(_) => panic!("start direct skill run failed with a non-RPC client error"),
654        };
655
656        let request = tokio::time::timeout(STAGE_TIMEOUT, async {
657            let mut request = None;
658            loop {
659                let event = match deliveries.recv().await.expect("escalation delivery") {
660                    crate::ClientDelivery::Live { message, .. } => match message.as_ref() {
661                        cookie_agent_protocol::EventSubscriptionMessage::Event { event } => {
662                            Some(event.clone())
663                        }
664                        _ => None,
665                    },
666                    crate::ClientDelivery::ReplayEvent { event, .. } => Some(event),
667                    _ => None,
668                };
669                let Some(event) = event else { continue };
670                if let EventPayload::ApprovalRequested { request: approval } = &event.payload {
671                    request = Some(approval.clone());
672                }
673                if let EventPayload::ApprovalEscalated { approval_id, .. } = event.payload {
674                    let request = request.expect("escalation followed its approval request");
675                    assert_eq!(request.approval_id(), approval_id);
676                    break request;
677                }
678            }
679        })
680        .await
681        .expect("approval escalation event timed out");
682
683        let request_revision = serde_json::to_value(&request).expect("approval JSON")["revision"]
684            .as_u64()
685            .expect("approval revision");
686        // Production clones its subscribed client for the response RPC while
687        // the UI task keeps draining the original delivery receiver.
688        let approver = observer.clone();
689        let response_request = request.clone();
690        let response = tokio::spawn(async move {
691            approver
692                .respond_approval(cookie_agent_protocol::ApprovalRespondParams {
693                    session_id: session.session_id,
694                    approval_id: response_request.approval_id(),
695                    request_revision,
696                    operation_fingerprint: response_request.operation_fingerprint().clone(),
697                    client_response_id: cookie_agent_protocol::ClientResponseId::new(
698                        uuid::Uuid::now_v7().to_string(),
699                    )
700                    .expect("response ID"),
701                    decision: cookie_agent_protocol::ApprovalUserDecision::ApproveOnce,
702                    feedback: None,
703                })
704                .await
705        });
706        let response_stage = tokio::time::timeout(STAGE_TIMEOUT, response);
707        let finalization_stage = tokio::time::timeout(STAGE_TIMEOUT, async {
708            loop {
709                let event = match deliveries.recv().await.expect("finalization delivery") {
710                    crate::ClientDelivery::Live { message, .. } => match message.as_ref() {
711                        cookie_agent_protocol::EventSubscriptionMessage::Event { event } => {
712                            Some(event.clone())
713                        }
714                        _ => None,
715                    },
716                    crate::ClientDelivery::ReplayEvent { event, .. } => Some(event),
717                    _ => None,
718                };
719                let Some(event) = event else { continue };
720                if let EventPayload::ApprovalFinalized {
721                    approval_id,
722                    decision,
723                } = event.payload
724                    && approval_id == request.approval_id()
725                {
726                    assert_eq!(
727                        decision.outcome,
728                        cookie_agent_protocol::ApprovalFinalOutcome::Approved
729                    );
730                    break;
731                }
732            }
733        });
734        let (response, finalized) = tokio::join!(response_stage, finalization_stage);
735        response
736            .expect("same-connection approval response RPC timed out")
737            .expect("approval response task panicked")
738            .expect("approve direct skill");
739        finalized.expect("approval finalization event timed out");
740
741        tokio::time::timeout(STAGE_TIMEOUT, async {
742            loop {
743                let event = match deliveries.recv().await.expect("skill delivery") {
744                    crate::ClientDelivery::Live { message, .. } => match message.as_ref() {
745                        cookie_agent_protocol::EventSubscriptionMessage::Event { event } => {
746                            Some(event.clone())
747                        }
748                        _ => None,
749                    },
750                    crate::ClientDelivery::ReplayEvent { event, .. } => Some(event),
751                    _ => None,
752                };
753                let Some(event) = event else { continue };
754                if matches!(event.payload, EventPayload::SkillLoaded { ref name, .. } if name == "fresh-skill") {
755                    break;
756                }
757            }
758        })
759        .await
760        .expect("skill-loaded event timed out after approval");
761        runner
762            .cancel_run(cookie_agent_protocol::RunCancelParams {
763                run_id: started.run_id,
764            })
765            .await
766            .expect("cancel test run");
767    }
