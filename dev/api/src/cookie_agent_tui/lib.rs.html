102    fn test_digest(value: &str) -> cookie_agent_models::Sha256Digest {
103        // A deterministic 64-hex stand-in; config fingerprints are opaque to
104        // the TUI and only carried through to the engine.
105        let base = value.to_owned();
106        let mut hex = String::with_capacity(64);
107        while hex.len() < 64 {
108            hex.push_str(
109                &base
110                    .bytes()
111                    .map(|byte| format!("{byte:02x}"))
112                    .collect::<String>(),
113            );
114        }
115        hex.truncate(64);
116        let hex = hex
117            .chars()
118            .map(|character| {
119                if character.is_ascii_hexdigit() {
120                    character
121                } else {
122                    'a'
123                }
124            })
125            .collect::<String>()
126            .to_lowercase();
127        cookie_agent_models::Sha256Digest::new(hex).expect("test digest")
128    }

159    fn in_process_server() -> (tempfile::TempDir, Arc<Server>) {
160        let directory = tempfile::tempdir().expect("temporary data directory");
161        #[cfg(unix)]
162        {
163            use std::os::unix::fs::PermissionsExt as _;
164            fs::set_permissions(directory.path(), fs::Permissions::from_mode(0o700))
165                .expect("private test root");
166        }
167        let providers = test_providers();
168        let provider_store = directory.path().join("provider-store");
169        fs::create_dir(&provider_store).expect("provider store directory");
170        #[cfg(unix)]
171        fs::set_permissions(&provider_store, fs::Permissions::from_mode(0o700))
172            .expect("private provider store");
173        let revision =
174            cookie_agent_protocol::CatalogRevision::new(format!("sha256:{}", "0".repeat(64)))
175                .expect("catalog revision");
176        let now = Timestamp::now();
177        let catalog = Arc::new(CatalogSnapshot {
178            revision,
179            source: CatalogSource::Bootstrap,
180            state: CatalogRuntimeState {
181                availability: CatalogAvailability::Bootstrap,
182                age: CatalogAgeState::Current,
183                last_error: None,
184            },
185            validated_at: now,
186            last_checked_at: now,
187            etag: None,
188            providers: BTreeMap::new(),
189            canonical_models: BTreeMap::new(),
190            quarantine: Vec::new(),
191        });
192        let model_manager = Arc::new(
193            ModelManager::new(
194                providers.clone(),
195                catalog,
196                ProviderStore::open(&provider_store).expect("provider store"),
197            )
198            .expect("model manager"),
199        );
200        let agent_id = AgentId::new("primary").expect("agent id");
201        let config = LoadedConfiguration {
202            runtime: RuntimeConfig {
203                schema_version: ConfigSchemaVersion,
204                server: ServerConfig::default(),
205                tool_output: ToolOutputConfig::default(),
206                approval: ApprovalConfig::default(),
207                context_compaction: ContextCompactionConfig::default(),
208                session_title: SessionTitleConfig::default(),
209                delegation: cookie_agent_config::DelegationConfig::default(),
210                providers,
211            },
212            agents: BTreeMap::from([(agent_id.clone(), test_agent_document(&agent_id))]),
213        };
214        let engine = Engine::open(EngineOptions {
215            data_dir: directory.path().join("data"),
216            cwd: directory.path().to_owned(),
217            config,
218            model_manager: Arc::clone(&model_manager),
219            tools: Vec::new(),
220        })
221        .expect("open engine");
222        (directory, Arc::new(Server::new(engine)))
223    }

229    fn approval_request() -> ApprovalRequest {
230        let resource = PreparedApprovalResource {
231            capability: PermissionAction::Bash,
232            canonical: PreparedResourceIdentity::new("command:git-status")
233                .expect("prepared resource identity"),
234            binding_digest: PreparedResourceDigest::from_canonical_binding_bytes(b"git status"),
235            binding_lifetime: PreparedBindingLifetime::ProcessLocal,
236            boundary: ApprovalBoundary::CommandPrefix {
237                prefix: "git status".into(),
238            },
239            source: ApprovalResourceSource::ModelRequest,
240        };
241        let resource_digest = resource.binding_digest.clone();
242        let operation = PreparedOperationIdentity::new(
243            Sha256Digest::of_bytes(b"normalized arguments"),
244            vec![ApprovalCapability {
245                action: PermissionAction::Bash,
246                operation: PreparedCapabilityOperation::new("execute")
247                    .expect("prepared capability operation"),
248            }],
249            vec![resource],
250            Sha256Digest::of_bytes(b"execution context"),
251        )
252        .expect("prepared operation");
253        ApprovalRequest::new(
254            ApprovalId::new_v7(),
255            7,
256            ApprovalTrigger::PermissionPolicy,
257            operation,
258            vec![ApprovalEvaluation {
259                resource_digest,
260                effect: PermissionEffect::Ask,
261                trace: DecisionTrace {
262                    action: PermissionAction::Bash,
263                    normalized_resource: "git status".into(),
264                    candidates: vec![MatchedPermissionRule {
265                        source_layer: SafeCode::new("test").expect("safe code"),
266                        action: PermissionAction::Bash,
267                        resource: cookie_agent_protocol::WildcardPattern::new("git status")
268                            .expect("wildcard"),
269                        effect: PermissionEffect::Ask,
270                    }],
271                    effect: PermissionEffect::Ask,
272                    precedence_reason: "test".into(),
273                },
274            }],
275            ApprovalConstraints {
276                allow_once: true,
277                allow_tree_grant: false,
278                cancellable: true,
279                expires_at: None,
280            },
281        )
282        .expect("approval request")
283    }

379        let models = client.runtime_snapshot().await.expect("runtime snapshot");

382            "custom.gateway/arbitrary-model"
383        );
384        let session = client
385            .create_session(SessionCreateParams {
386                selection: test_run_selection(),
387            })
388            .await
389            .expect("session")
390            .session;
391        let mut app = App::new(client).await.expect("TUI config");
392        let request = approval_request();
393        let approval_id = request.approval_id();
394        app.store.apply_event(StoredEvent {
395            event_schema_version: EventSchemaVersion::current(),
396            session_id: session.session_id,
397            run_id: Some(cookie_agent_protocol::RunId::new_v7()),
398            seq: 2,
399            timestamp: Timestamp::now(),
400            payload: EventPayload::ApprovalRequested { request },
401        });
402        app.store.apply_event(StoredEvent {
403            event_schema_version: EventSchemaVersion::current(),
404            session_id: session.session_id,
405            run_id: Some(cookie_agent_protocol::RunId::new_v7()),
406            seq: 3,
407            timestamp: Timestamp::now(),
408            payload: EventPayload::ApprovalEscalated {
409                approval_id,
410                reason_code: cookie_agent_protocol::ApprovalReasonCode::Escalated,
411            },
412        });
413        for (width, height) in [(40, 12), (80, 24), (160, 50)] {
414            let backend = TestBackend::new(width, height);
415            let mut terminal = Terminal::new(backend).expect("test terminal");
416            terminal
417                .draw(|frame| app.draw_for_test(frame))
418                .expect("app render");
419            let buffer = terminal.backend().buffer();
420            let rendered = (0..buffer.area.height)
421                .flat_map(|y| {
422                    (0..buffer.area.width).map(move |x| buffer[(x, y)].symbol().to_owned())
423                })
424                .collect::<String>();
425            assert!(rendered.contains("Approval"));
426            assert!(rendered.contains("Agents"));
427            if width >= 80 {
428                assert!(rendered.contains("Conversation"));
429            }
430            if (width, height) == (80, 24) {
431                assert!(rendered.contains("git status"));
432                assert!(rendered.contains("primary • custom.gateway/arbitrary-model[base]"));
433            }
434        }
435    }

438    async fn client_round_trips_handshake_create_and_event_replay_in_process() {
439        let (_directory, server) = in_process_server();
440        let client = Client::connect_in_process(server);
441        client.handshake().await.expect("handshake");
442        let session = client
443            .create_session(SessionCreateParams {
444                selection: test_run_selection(),
445            })
446            .await
447            .expect("create session")
448            .session;
449        let mut deliveries = client.subscribe_deliveries().expect("delivery receiver");
450        client
451            .subscribe_events(session.session_id, None)
452            .await
453            .expect("subscribe events");
454        let mut replayed = Vec::new();
455        loop {
456            match deliveries.recv().await.expect("delivery") {
457                crate::ClientDelivery::ReplayEvent { event, .. } => replayed.push(*event),
458                crate::ClientDelivery::ReplayEnd { .. } => break,
459                _ => {}
460            }
461        }
462        assert!(!replayed.is_empty());
463        assert!(
464            replayed
465                .iter()
466                .all(|event| event.session_id == session.session_id)
467        );
468    }

472        let (_directory, server) = in_process_server();
473        let setup = Client::connect_in_process(server.clone());
474        setup.handshake().await.expect("setup handshake");
475        let existing = setup
476            .create_session(SessionCreateParams {
477                selection: test_run_selection(),
478            })
479            .await
480            .expect("existing session")
481            .session;
482
483        let attached = Client::connect_in_process(server.clone());
484        attached.handshake().await.expect("attach handshake");
485        let attached_app = App::new(attached).await.expect("attached TUI config");
486        assert_eq!(attached_app.store.sessions.len(), 1);
487        assert!(
488            attached_app
489                .store
490                .sessions
491                .contains_key(&existing.session_id)
492        );
493
494        let local = Client::connect_in_process(server);
495        local.handshake().await.expect("local handshake");
496        let local_app = App::new_with_new_session(local)
497            .await
498            .expect("local TUI config");
499        assert_eq!(local_app.store.sessions.len(), 1);
500        let local_session = local_app
501            .store
502            .sessions
503            .keys()
504            .next()
505            .copied()
506            .expect("new local root session");
507        assert_ne!(local_session, existing.session_id);
