105    fn test_digest(value: &str) -> cookie_agent_models::Sha256Digest {
106        // A deterministic 64-hex stand-in; config fingerprints are opaque to
107        // the TUI and only carried through to the engine.
108        let base = value.to_owned();
109        let mut hex = String::with_capacity(64);
110        while hex.len() < 64 {
111            hex.push_str(
112                &base
113                    .bytes()
114                    .map(|byte| format!("{byte:02x}"))
115                    .collect::<String>(),
116            );
117        }
118        hex.truncate(64);
119        let hex = hex
120            .chars()
121            .map(|character| {
122                if character.is_ascii_hexdigit() {
123                    character
124                } else {
125                    'a'
126                }
127            })
128            .collect::<String>()
129            .to_lowercase();
130        cookie_agent_models::Sha256Digest::new(hex).expect("test digest")
131    }

266    fn approval_request() -> ApprovalRequest {
267        let resource = PreparedApprovalResource {
268            capability: PermissionAction::Bash,
269            canonical: PreparedResourceIdentity::new("command:git-status")
270                .expect("prepared resource identity"),
271            binding_digest: PreparedResourceDigest::from_canonical_binding_bytes(b"git status"),
272            binding_lifetime: PreparedBindingLifetime::ProcessLocal,
273            boundary: ApprovalBoundary::CommandPrefix {
274                prefix: "git status".into(),
275            },
276            source: ApprovalResourceSource::ModelRequest,
277        };
278        let resource_digest = resource.binding_digest.clone();
279        let operation = PreparedOperationIdentity::new(
280            Sha256Digest::of_bytes(b"normalized arguments"),
281            vec![ApprovalCapability {
282                action: PermissionAction::Bash,
283                operation: PreparedCapabilityOperation::new("execute")
284                    .expect("prepared capability operation"),
285            }],
286            vec![resource],
287            Sha256Digest::of_bytes(b"execution context"),
288        )
289        .expect("prepared operation");
290        ApprovalRequest::new(
291            ApprovalId::new_v7(),
292            7,
293            ApprovalTrigger::PermissionPolicy,
294            operation,
295            vec![ApprovalEvaluation {
296                resource_digest,
297                effect: PermissionEffect::Ask,
298                trace: DecisionTrace {
299                    action: PermissionAction::Bash,
300                    normalized_resource: "git status".into(),
301                    candidates: vec![MatchedPermissionRule {
302                        source_layer: SafeCode::new("test").expect("safe code"),
303                        action: PermissionAction::Bash,
304                        resource: cookie_agent_protocol::WildcardPattern::new("git status")
305                            .expect("wildcard"),
306                        effect: PermissionEffect::Ask,
307                    }],
308                    effect: PermissionEffect::Ask,
309                    precedence_reason: "test".into(),
310                },
311            }],
312            ApprovalConstraints {
313                allow_once: true,
314                allow_tree_grant: false,
315                cancellable: true,
316                expires_at: None,
317            },
318        )
319        .expect("approval request")
320    }

606            .subscribe_deliveries()
607            .expect("observer deliveries");
608        observer
609            .subscribe_events(session.session_id, Some(1))
610            .await
611            .expect("subscribe events");
612        let runner = Client::connect_in_process(server);
613        runner.handshake().await.expect("runner handshake");
614        let started = match runner
615            .start_run(cookie_agent_protocol::RunStartParams {

617                client_run_id: cookie_agent_protocol::ClientRunId::new(format!(
618                    "approval-e2e-{}",
619                    uuid::Uuid::now_v7()

621                .expect("client run ID"),
622                selection: test_run_selection(),
623                input: cookie_agent_protocol::encode_skill_submission("fresh-skill", "value"),
624            })
625            .await
626        {
627            Ok(started) => started,
628            Err(crate::ClientError::Rpc(error)) => {
629                let debug_code = error
630                    .data
631                    .as_ref()
632                    .and_then(|data| data.get("debug_code"))
633                    .and_then(serde_json::Value::as_str)
634                    .unwrap_or("missing_debug_code");
635                panic!(
636                    "start direct skill run failed: rpc_code={}, debug_code={debug_code}",
637                    error.code
638                );
639            }
640            Err(_) => panic!("start direct skill run failed with a non-RPC client error"),
641        };
642
643        let request = tokio::time::timeout(STAGE_TIMEOUT, async {
644            let mut request = None;
645            loop {
646                let event = match deliveries.recv().await.expect("escalation delivery") {
647                    crate::ClientDelivery::Live { message, .. } => match message.as_ref() {
648                        cookie_agent_protocol::EventSubscriptionMessage::Event { event } => {
649                            Some(event.clone())
650                        }
651                        _ => None,
652                    },
653                    crate::ClientDelivery::ReplayEvent { event, .. } => Some(event),
654                    _ => None,
655                };
656                let Some(event) = event else { continue };
657                if let EventPayload::ApprovalRequested { request: approval } = &event.payload {
658                    request = Some(approval.clone());
659                }
660                if let EventPayload::ApprovalEscalated { approval_id, .. } = event.payload {
661                    let request = request.expect("escalation followed its approval request");
662                    assert_eq!(request.approval_id(), approval_id);
663                    break request;
664                }
665            }
666        })
667        .await
668        .expect("approval escalation event timed out");
669
670        let request_revision = serde_json::to_value(&request).expect("approval JSON")["revision"]
671            .as_u64()
672            .expect("approval revision");
673        // Production clones its subscribed client for the response RPC while
674        // the UI task keeps draining the original delivery receiver.
675        let approver = observer.clone();
676        let response_request = request.clone();
677        let response = tokio::spawn(async move {
678            approver
679                .respond_approval(cookie_agent_protocol::ApprovalRespondParams {
680                    session_id: session.session_id,
681                    approval_id: response_request.approval_id(),
682                    request_revision,
683                    operation_fingerprint: response_request.operation_fingerprint().clone(),
684                    client_response_id: cookie_agent_protocol::ClientResponseId::new(
685                        uuid::Uuid::now_v7().to_string(),
686                    )
687                    .expect("response ID"),
688                    decision: cookie_agent_protocol::ApprovalUserDecision::ApproveOnce,
689                    feedback: None,
690                })
691                .await
692        });
693        let response_stage = tokio::time::timeout(STAGE_TIMEOUT, response);
694        let finalization_stage = tokio::time::timeout(STAGE_TIMEOUT, async {
695            loop {
696                let event = match deliveries.recv().await.expect("finalization delivery") {
697                    crate::ClientDelivery::Live { message, .. } => match message.as_ref() {
698                        cookie_agent_protocol::EventSubscriptionMessage::Event { event } => {
699                            Some(event.clone())
700                        }
701                        _ => None,
702                    },
703                    crate::ClientDelivery::ReplayEvent { event, .. } => Some(event),
704                    _ => None,
705                };
706                let Some(event) = event else { continue };
707                if let EventPayload::ApprovalFinalized {
708                    approval_id,
709                    decision,
710                } = event.payload
711                    && approval_id == request.approval_id()
712                {
713                    assert_eq!(
714                        decision.outcome,
715                        cookie_agent_protocol::ApprovalFinalOutcome::Approved
716                    );
717                    break;
718                }
719            }
720        });
721        let (response, finalized) = tokio::join!(response_stage, finalization_stage);
722        response
723            .expect("same-connection approval response RPC timed out")
724            .expect("approval response task panicked")
725            .expect("approve direct skill");
726        finalized.expect("approval finalization event timed out");
727
728        tokio::time::timeout(STAGE_TIMEOUT, async {
729            loop {
730                let event = match deliveries.recv().await.expect("skill delivery") {
731                    crate::ClientDelivery::Live { message, .. } => match message.as_ref() {
732                        cookie_agent_protocol::EventSubscriptionMessage::Event { event } => {
733                            Some(event.clone())
734                        }
735                        _ => None,
736                    },
737                    crate::ClientDelivery::ReplayEvent { event, .. } => Some(event),
738                    _ => None,
739                };
740                let Some(event) = event else { continue };
741                if matches!(event.payload, EventPayload::SkillLoaded { ref name, .. } if name == "fresh-skill") {
742                    break;
743                }
744            }
745        })
746        .await
747        .expect("skill-loaded event timed out after approval");
748        runner
749            .cancel_run(cookie_agent_protocol::RunCancelParams {
750                run_id: started.run_id,
751            })
752            .await
753            .expect("cancel test run");
754    }
