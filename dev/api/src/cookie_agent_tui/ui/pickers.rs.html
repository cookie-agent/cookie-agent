80/// Case-insensitive substring matching over provider ID and display name.
81pub(crate) fn provider_matches(provider: &ProviderDescriptor, query: &str) -> bool {
82    if query.trim().is_empty() {
83        return true;
84    }
85    let query = query.trim().to_lowercase();
86    provider
87        .display_name
88        .as_str()
89        .to_lowercase()
90        .contains(&query)
91        || provider.id.as_str().to_lowercase().contains(&query)
92}

94/// Session picker matching over title and the untitled placeholder.
95pub(crate) fn session_matches(session: &SessionMeta, query: &str) -> bool {
96    if query.trim().is_empty() {
97        return true;
98    }
99    let query = query.trim().to_lowercase();
100    session
101        .title
102        .as_ref()
103        .map_or("untitled", |title| title.as_str())
104        .to_lowercase()
105        .contains(&query)
106}

126/// Search sessions by title and group matching rows by their local activity day.
127pub(crate) fn session_search_rows(
128    sessions: &[SessionMeta],
129    query: &str,
130    now: Timestamp,
131    time_zone: &TimeZone,
132) -> Vec<SessionSearchRow> {
133    let today = now.to_zoned(time_zone.clone()).date();
134    let yesterday = today.yesterday().ok();
135    let mut sessions = sessions
136        .iter()
137        .filter(|session| session_matches(session, query))
138        .collect::<Vec<_>>();
139    sessions.sort_by_key(|session| std::cmp::Reverse(session.last_activity));
140
141    let mut rows = Vec::new();
142    let mut current_date = None;
143    for session in sessions {
144        let date = session.last_activity.to_zoned(time_zone.clone()).date();
145        if current_date != Some(date) {
146            rows.push(SessionSearchRow::Header(session_day_label(
147                date, today, yesterday,
148            )));
149            current_date = Some(date);
150        }
151        let title = session
152            .title
153            .as_ref()
154            .map_or_else(|| "untitled".to_owned(), ToString::to_string);
155        rows.push(SessionSearchRow::Session {
156            session_id: session.session_id,
157            label: format!(
158                "{title}  ({} · {})",
159                session.creation_selection.agent,
160                short_id(session.session_id)
161            ),
162        });
163    }
164    rows
165}

167fn session_day_label(date: Date, today: Date, yesterday: Option<Date>) -> String {
168    if date == today {
169        "Today".to_owned()
170    } else if Some(date) == yesterday {
171        "Yesterday".to_owned()
172    } else {
173        date.strftime("%b %-d").to_string()
174    }
175}

182/// A right-aligned, dimmed key hint on a picker panel's bottom border, so
183/// every chooser explains itself without docs.
184fn footer_hint(theme: &Theme, hint: Option<&str>) -> Option<Line<'static>> {
185    hint.map(|hint| Line::from(Span::styled(hint.to_owned(), theme.internal())).right_aligned())
186}

188/// Rows never hard-clip mid-word at the panel edge: each label is ellipsized
189/// to the space left after the selection marker's two columns.
190fn ellipsized(entries: Vec<String>, inner_width: u16) -> Vec<String> {
191    let available = usize::from(inner_width.saturating_sub(2));
192    entries
193        .into_iter()
194        .map(|entry| super::app::truncate_with_ellipsis(&entry, available))
195        .collect()
196}

198/// The textual chrome of a picker panel: its title, the message shown when
199/// there is nothing to list, and the bottom-border key hint.
200pub(crate) struct PickerChrome<'a> {
201    pub(crate) title: &'a str,
202    pub(crate) empty_message: Option<&'a str>,
203    pub(crate) hint: Option<&'a str>,
204}

206pub(crate) fn render(
207    frame: &mut Frame,
208    chrome: PickerChrome<'_>,
209    entries: Vec<String>,
210    area: Rect,
211    state: &mut ListState,
212    theme: &Theme,
213) -> Vec<(Rect, usize)> {
214    super::app::paint_panel(frame, area, theme);
215    let entry_count = entries.len();
216    if entry_count == 0 {
217        let content = chrome.empty_message.map_or_else(
218            || {
219                Line::from(vec![
220                    Span::styled("No matches. ", theme.muted()),
221                    Span::styled("Backspace or Ctrl-U clears the filter.", theme.internal()),
222                ])
223            },
224            |message| Line::from(Span::styled(message.to_owned(), theme.muted())),
225        );
226        let mut block = Block::default()
227            .borders(Borders::ALL)
228            .border_style(theme.panel_border())
229            .title(chrome.title);
230        if let Some(hint) = footer_hint(theme, chrome.hint) {
231            block = block.title_bottom(hint);
232        }
233        frame.render_widget(ratatui::widgets::Paragraph::new(content).block(block), area);
234        return Vec::new();
235    }
236    let inner = inner_rect(area);
237    let entries = ellipsized(entries, inner.width);
238    let mut block = Block::default()
239        .borders(Borders::ALL)
240        .border_style(theme.panel_border())
241        .title(chrome.title);
242    if let Some(hint) = footer_hint(theme, chrome.hint) {
243        block = block.title_bottom(hint);
244    }
245    frame.render_stateful_widget(
246        List::new(entries.into_iter().map(ListItem::new).collect::<Vec<_>>())
247            .highlight_symbol("> ")
248            .highlight_style(theme.selected())
249            .block(block),
250        area,
251        state,
252    );
253    (state.offset()..entry_count)
254        .take(usize::from(inner.height))
255        .enumerate()
256        .map(|(row, index)| {
257            (
258                Rect::new(
259                    inner.x,
260                    inner.y + u16::try_from(row).unwrap_or(u16::MAX),
261                    inner.width,
262                    1,
263                ),
264                index,
265            )
266        })
267        .collect()
268}

270pub(crate) fn move_selection(state: &mut ListState, len: usize, up: bool) {
271    if len == 0 {
272        state.select(None);
273        return;
274    }
275    let selected = state.selected().unwrap_or(0);
276    state.select(Some(if up {
277        selected.saturating_sub(1)
278    } else {
279        (selected + 1).min(len - 1)
280    }));
281}
282
283pub(crate) fn cycle_selection(state: &mut ListState, len: usize, backward: bool) {
284    if len == 0 {
285        state.select(None);
286        return;
287    }
288    let selected = state.selected().unwrap_or(0) % len;
289    state.select(Some(if backward {
290        (selected + len - 1) % len
291    } else {
292        (selected + 1) % len
293    }));
294}

296pub(crate) fn clamp_tree_view(
297    selection: &mut usize,
298    offset: &mut usize,
299    entry_count: usize,
300    viewport_height: usize,
301) {
302    if entry_count == 0 {
303        *selection = 0;
304        *offset = 0;
305        return;
306    }
307    *selection = (*selection).min(entry_count - 1);
308    if viewport_height == 0 {
309        *offset = 0;
310        return;
311    }
312    let max_offset = entry_count.saturating_sub(viewport_height);
313    *offset = (*offset).min(max_offset);
314    if *selection < *offset {
315        *offset = *selection;
316    } else if *selection >= *offset + viewport_height {
317        *offset = (*selection + 1).saturating_sub(viewport_height);
318    }
319}

321/// Depth-first flattening of the delegation tree with each node's depth.
322pub(crate) fn flatten_tree(
323    tree: &SessionTree,
324    depth: usize,
325    collapsed: &HashSet<SessionId>,
326    entries: &mut Vec<(SessionId, SessionMeta, usize)>,
327) {
328    entries.push((tree.session.session_id, tree.session.clone(), depth));
329    if !collapsed.contains(&tree.session.session_id) {
330        for child in &tree.children {
331            flatten_tree(child, depth + 1, collapsed, entries);
332        }
333    }
334}

336fn inner_rect(area: Rect) -> Rect {
337    Rect::new(
338        area.x.saturating_add(1),
339        area.y.saturating_add(1),
340        area.width.saturating_sub(2),
341        area.height.saturating_sub(2),
342    )
343}
