80/// Case-insensitive substring matching over provider ID and display name.
81pub(crate) fn provider_matches(provider: &ProviderDescriptor, query: &str) -> bool {
82    if query.trim().is_empty() {
83        return true;
84    }
85    let query = query.trim().to_lowercase();
86    provider
87        .display_name
88        .as_str()
89        .to_lowercase()
90        .contains(&query)
91        || provider.id.as_str().to_lowercase().contains(&query)
92}

94/// Session picker matching over title and the untitled placeholder.
95pub(crate) fn session_matches(session: &SessionMeta, query: &str) -> bool {
96    if query.trim().is_empty() {
97        return true;
98    }
99    let query = query.trim().to_lowercase();
100    session
101        .title
102        .as_ref()
103        .map_or("untitled", |title| title.as_str())
104        .to_lowercase()
105        .contains(&query)
106}

126/// Search sessions by title and group matching rows by their local activity day.
127pub(crate) fn session_search_rows(
128    sessions: &[SessionMeta],
129    query: &str,
130    now: Timestamp,
131    time_zone: &TimeZone,
132) -> Vec<SessionSearchRow> {
133    let today = now.to_zoned(time_zone.clone()).date();
134    let yesterday = today.yesterday().ok();
135    let mut sessions = sessions
136        .iter()
137        .filter(|session| session_matches(session, query))
138        .collect::<Vec<_>>();
139    sessions.sort_by_key(|session| std::cmp::Reverse(session.last_activity));
140
141    let mut rows = Vec::new();
142    let mut current_date = None;
143    for session in sessions {
144        let date = session.last_activity.to_zoned(time_zone.clone()).date();
145        if current_date != Some(date) {
146            rows.push(SessionSearchRow::Header(session_day_label(
147                date, today, yesterday,
148            )));
149            current_date = Some(date);
150        }
151        let title = session
152            .title
153            .as_ref()
154            .map_or_else(|| "untitled".to_owned(), ToString::to_string);
155        let degraded = if session.skipped_events.is_empty() {
156            ""
157        } else {
158            " !"
159        };
160        rows.push(SessionSearchRow::Session {
161            session_id: session.session_id,
162            label: format!(
163                "{title}{degraded}  ({} · {})",
164                session.creation_selection.agent,
165                short_id(session.session_id)
166            ),
167        });
168    }
169    rows
170}

172fn session_day_label(date: Date, today: Date, yesterday: Option<Date>) -> String {
173    if date == today {
174        "Today".to_owned()
175    } else if Some(date) == yesterday {
176        "Yesterday".to_owned()
177    } else {
178        date.strftime("%b %-d").to_string()
179    }

193/// Rows never hard-clip mid-word at the panel edge: each label is ellipsized
194/// to the space left after the selection marker's two columns.
195fn ellipsized(entries: Vec<String>, inner_width: u16) -> Vec<String> {
196    let available = usize::from(inner_width.saturating_sub(2));
197    entries
198        .into_iter()
199        .map(|entry| super::app::truncate_with_ellipsis(&entry, available))
200        .collect()

211pub(crate) fn render(
212    frame: &mut Frame,
213    chrome: PickerChrome<'_>,
214    entries: Vec<String>,
215    area: Rect,
216    state: &mut ListState,
217    theme: &Theme,
218) -> Vec<(Rect, usize)> {
219    super::app::paint_panel(frame, area, theme);
220    let entry_count = entries.len();
221    if entry_count == 0 {
222        let content = chrome.empty_message.map_or_else(
223            || {
224                Line::from(vec![
225                    Span::styled("No matches. ", theme.muted()),
226                    Span::styled("Backspace or Ctrl-U clears the filter.", theme.internal()),
227                ])
228            },
229            |message| Line::from(Span::styled(message.to_owned(), theme.muted())),
230        );
231        let mut block = Block::default()
232            .borders(Borders::ALL)
233            .border_style(theme.panel_border())
234            .title(chrome.title);
235        if let Some(hint) = footer_hint(theme, chrome.hint) {
236            block = block.title_bottom(hint);
237        }
238        frame.render_widget(ratatui::widgets::Paragraph::new(content).block(block), area);
239        return Vec::new();
240    }
241    let inner = inner_rect(area);
242    let entries = ellipsized(entries, inner.width);
243    let mut block = Block::default()
244        .borders(Borders::ALL)
245        .border_style(theme.panel_border())
246        .title(chrome.title);
247    if let Some(hint) = footer_hint(theme, chrome.hint) {
248        block = block.title_bottom(hint);
249    }
250    frame.render_stateful_widget(
251        List::new(entries.into_iter().map(ListItem::new).collect::<Vec<_>>())
252            .highlight_symbol("> ")
253            .highlight_style(theme.selected())
254            .block(block),
255        area,
256        state,
257    );
258    (state.offset()..entry_count)
259        .take(usize::from(inner.height))
260        .enumerate()
261        .map(|(row, index)| {
262            (
263                Rect::new(
264                    inner.x,
265                    inner.y + u16::try_from(row).unwrap_or(u16::MAX),
266                    inner.width,
267                    1,
268                ),
269                index,
270            )
271        })
272        .collect()
273}

275pub(crate) fn move_selection(state: &mut ListState, len: usize, up: bool) {
276    if len == 0 {
277        state.select(None);
278        return;
279    }
280    let selected = state.selected().unwrap_or(0);
281    state.select(Some(if up {
282        selected.saturating_sub(1)
283    } else {
284        (selected + 1).min(len - 1)
285    }));
286}

288pub(crate) fn cycle_selection(state: &mut ListState, len: usize, backward: bool) {
289    if len == 0 {
290        state.select(None);
291        return;
292    }
293    let selected = state.selected().unwrap_or(0) % len;
294    state.select(Some(if backward {
295        (selected + len - 1) % len
296    } else {
297        (selected + 1) % len
298    }));
299}

301pub(crate) fn clamp_tree_view(
302    selection: &mut usize,
303    offset: &mut usize,
304    entry_count: usize,
305    viewport_height: usize,
306) {
307    if entry_count == 0 {
308        *selection = 0;
309        *offset = 0;
310        return;
311    }
312    *selection = (*selection).min(entry_count - 1);
313    if viewport_height == 0 {
314        *offset = 0;
315        return;
316    }
317    let max_offset = entry_count.saturating_sub(viewport_height);
318    *offset = (*offset).min(max_offset);
319    if *selection < *offset {
320        *offset = *selection;
321    } else if *selection >= *offset + viewport_height {
322        *offset = (*selection + 1).saturating_sub(viewport_height);
323    }
324}

326/// Depth-first flattening of the delegation tree with each node's depth.
327pub(crate) fn flatten_tree(
328    tree: &SessionTree,
329    depth: usize,
330    collapsed: &HashSet<SessionId>,
331    states: &HashMap<SessionId, SessionState>,
332    entries: &mut Vec<(SessionId, SessionMeta, usize)>,
333) {
334    entries.push((tree.session.session_id, tree.session.clone(), depth));
335    if !collapsed.contains(&tree.session.session_id) {
336        let mut children = tree.children.iter().collect::<Vec<_>>();
337        children.sort_by(|left, right| {
338            let left_state = states.get(&left.session.session_id);
339            let right_state = states.get(&right.session.session_id);
340            right_state
341                .and_then(|state| state.last_agent_activity)
342                .cmp(&left_state.and_then(|state| state.last_agent_activity))
343                .then_with(|| {
344                    match (
345                        left_state.and_then(|state| state.created_at),
346                        right_state.and_then(|state| state.created_at),
347                    ) {
348                        (Some(left), Some(right)) => left.cmp(&right),
349                        _ => left.session.session_id.cmp(&right.session.session_id),
350                    }
351                })
352                .then_with(|| left.session.session_id.cmp(&right.session.session_id))
353        });
354        for child in children {
355            flatten_tree(child, depth + 1, collapsed, states, entries);
356        }
357    }
358}

363        area.y.saturating_add(1),

365        area.height.saturating_sub(2),
366    )
