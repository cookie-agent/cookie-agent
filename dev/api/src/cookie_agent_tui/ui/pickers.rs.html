84/// Case-insensitive substring matching over provider ID and display name.
85pub(crate) fn provider_matches(provider: &ProviderDescriptor, query: &str) -> bool {
86    if query.trim().is_empty() {
87        return true;
88    }
89    let query = query.trim().to_lowercase();
90    provider
91        .display_name
92        .as_str()
93        .to_lowercase()
94        .contains(&query)
95        || provider.id.as_str().to_lowercase().contains(&query)
96}

98/// Case-insensitive substring matching over model display names and canonical keys.
99pub(crate) fn model_matches(
100    selection: &ModelSelection,
101    descriptor: Option<&AvailableModelDescriptor>,
102    query: &str,
103) -> bool {
104    if query.trim().is_empty() {
105        return true;
106    }
107    let query = query.trim().to_lowercase();
108    selection.model.to_string().to_lowercase().contains(&query)
109        || descriptor.is_some_and(|descriptor| {
110            descriptor.display_name.to_lowercase().contains(&query)
111                || selection.variant.as_ref().is_some_and(|selected| {
112                    descriptor.variants.iter().any(|variant| {
113                        variant.id == *selected
114                            && variant.display_name.to_lowercase().contains(&query)
115                    })
116                })
117        })
118}

120/// Session picker matching over title and the untitled placeholder.
121pub(crate) fn session_matches(session: &SessionMeta, query: &str) -> bool {
122    if query.trim().is_empty() {
123        return true;
124    }
125    let query = query.trim().to_lowercase();
126    session
127        .title
128        .as_ref()
129        .map_or("untitled", |title| title.as_str())
130        .to_lowercase()
131        .contains(&query)
132}

152/// Search sessions by title and group matching rows by their local activity day.
153pub(crate) fn session_search_rows(
154    sessions: &[SessionMeta],
155    query: &str,
156    now: Timestamp,
157    time_zone: &TimeZone,
158) -> Vec<SessionSearchRow> {
159    let today = now.to_zoned(time_zone.clone()).date();
160    let yesterday = today.yesterday().ok();
161    let mut sessions = sessions
162        .iter()
163        .filter(|session| session_matches(session, query))
164        .collect::<Vec<_>>();
165    sessions.sort_by_key(|session| std::cmp::Reverse(session.last_activity));
166
167    let mut rows = Vec::new();
168    let mut current_date = None;
169    for session in sessions {
170        let date = session.last_activity.to_zoned(time_zone.clone()).date();
171        if current_date != Some(date) {
172            rows.push(SessionSearchRow::Header(session_day_label(
173                date, today, yesterday,
174            )));
175            current_date = Some(date);
176        }
177        let title = session
178            .title
179            .as_ref()
180            .map_or_else(|| "untitled".to_owned(), ToString::to_string);
181        let degraded = if session.skipped_events.is_empty() {
182            ""
183        } else {
184            " !"
185        };
186        rows.push(SessionSearchRow::Session {
187            session_id: session.session_id,
188            label: format!(
189                "{title}{degraded}  ({} · {})",
190                session.creation_selection.agent,
191                short_id(session.session_id)
192            ),
193        });
194    }
195    rows
196}

198fn session_day_label(date: Date, today: Date, yesterday: Option<Date>) -> String {
199    if date == today {
200        "Today".to_owned()
201    } else if Some(date) == yesterday {
202        "Yesterday".to_owned()
203    } else {
204        date.strftime("%b %-d").to_string()
205    }

219/// Rows never hard-clip mid-word at the panel edge: each label is ellipsized
220/// to the space left after the selection marker's two columns.
221fn ellipsized(entries: Vec<String>, inner_width: u16) -> Vec<String> {
222    let available = usize::from(inner_width.saturating_sub(2));
223    entries

257pub(crate) fn render_lines(
258    frame: &mut Frame,
259    chrome: PickerChrome<'_>,
260    entries: Vec<Line<'static>>,
261    area: Rect,
262    state: &mut ListState,
263    theme: &Theme,
264    highlight_style: Style,
265) -> Vec<(Rect, usize)> {
266    super::app::paint_panel(frame, area, theme);
267    let entry_count = entries.len();
268    if entry_count == 0 {
269        let content = chrome.empty_message.map_or_else(
270            || {
271                Line::from(vec![
272                    Span::styled("No matches. ", theme.muted()),
273                    Span::styled("Backspace or Ctrl-U clears the filter.", theme.internal()),
274                ])
275            },
276            |message| Line::from(Span::styled(message.to_owned(), theme.muted())),
277        );
278        let mut block = Block::default()
279            .borders(Borders::ALL)
280            .border_style(theme.panel_border())
281            .title(chrome.title);
282        if let Some(hint) = footer_hint(theme, chrome.hint) {
283            block = block.title_bottom(hint);
284        }
285        frame.render_widget(ratatui::widgets::Paragraph::new(content).block(block), area);
286        return Vec::new();
287    }
288    let inner = inner_rect(area);
289    let mut block = Block::default()
290        .borders(Borders::ALL)
291        .border_style(theme.panel_border())
292        .title(chrome.title);
293    if let Some(hint) = footer_hint(theme, chrome.hint) {
294        block = block.title_bottom(hint);
295    }
296    frame.render_stateful_widget(
297        List::new(entries.into_iter().map(ListItem::new).collect::<Vec<_>>())
298            .highlight_symbol("> ")
299            .highlight_style(highlight_style)
300            .block(block),
301        area,
302        state,
303    );
304    (state.offset()..entry_count)
305        .take(usize::from(inner.height))
306        .enumerate()
307        .map(|(row, index)| {
308            (
309                Rect::new(
310                    inner.x,
311                    inner.y + u16::try_from(row).unwrap_or(u16::MAX),
312                    inner.width,
313                    1,
314                ),
315                index,
316            )
317        })
318        .collect()
319}

321pub(crate) fn move_selection(state: &mut ListState, len: usize, up: bool) {
322    if len == 0 {
323        state.select(None);
324        return;
325    }
326    let selected = state.selected().unwrap_or(0);
327    state.select(Some(if up {
328        selected.saturating_sub(1)
329    } else {
330        (selected + 1).min(len - 1)
331    }));
332}
333
334pub(crate) fn cycle_selection(state: &mut ListState, len: usize, backward: bool) {
335    if len == 0 {
336        state.select(None);
337        return;
338    }
339    let selected = state.selected().unwrap_or(0) % len;
340    state.select(Some(if backward {
341        (selected + len - 1) % len
342    } else {
343        (selected + 1) % len
344    }));
345}

347pub(crate) fn clamp_tree_view(
348    selection: &mut usize,
349    offset: &mut usize,
350    entry_count: usize,
351    viewport_height: usize,
352) {
353    if entry_count == 0 {
354        *selection = 0;
355        *offset = 0;
356        return;
357    }
358    *selection = (*selection).min(entry_count - 1);
359    if viewport_height == 0 {
360        *offset = 0;
361        return;
362    }
363    let max_offset = entry_count.saturating_sub(viewport_height);
364    *offset = (*offset).min(max_offset);
365    if *selection < *offset {
366        *offset = *selection;
367    } else if *selection >= *offset + viewport_height {
368        *offset = (*selection + 1).saturating_sub(viewport_height);
369    }
370}

372/// Depth-first flattening of the delegation tree with each node's depth.
373pub(crate) fn flatten_tree(
374    tree: &SessionTree,
375    depth: usize,
376    collapsed: &HashSet<SessionId>,
377    states: &HashMap<SessionId, SessionState>,
378    entries: &mut Vec<(SessionId, SessionMeta, usize)>,
379) {
380    entries.push((tree.session.session_id, tree.session.clone(), depth));
381    if !collapsed.contains(&tree.session.session_id) {
382        let mut children = tree.children.iter().collect::<Vec<_>>();
383        children.sort_by(|left, right| {
384            let left_state = states.get(&left.session.session_id);
385            let right_state = states.get(&right.session.session_id);
386            right_state
387                .and_then(|state| state.last_agent_activity)
388                .cmp(&left_state.and_then(|state| state.last_agent_activity))
389                .then_with(|| {
390                    match (
391                        left_state.and_then(|state| state.created_at),
392                        right_state.and_then(|state| state.created_at),
393                    ) {
394                        (Some(left), Some(right)) => left.cmp(&right),
395                        _ => left.session.session_id.cmp(&right.session.session_id),
396                    }
397                })
398                .then_with(|| left.session.session_id.cmp(&right.session.session_id))
399        });
400        for child in children {
401            flatten_tree(child, depth + 1, collapsed, states, entries);
402        }
403    }
404}

411        area.height.saturating_sub(2),
412    )

687        // The long row ellipsizes instead of clipping mid-word…
688        assert!(text.contains('…'), "{text}");
689        assert!(!text.contains("Display Name"), "{text}");
690        // …and never spills over the right border.
691        assert_eq!(buffer[(29, 1)].symbol(), "│");
692        // The bottom border carries the key hints.
693        assert!(text.contains("enter: select"), "{text}");
694        assert!(text.contains("esc: close"), "{text}");
695    }
