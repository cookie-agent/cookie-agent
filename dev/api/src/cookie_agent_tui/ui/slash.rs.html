176pub(crate) fn entries(input: &str) -> Vec<&'static CommandSpec> {
177    let query = input
178        .strip_prefix('/')
179        .unwrap_or_default()
180        .split_whitespace()
181        .next()
182        .unwrap_or_default()
183        .to_ascii_lowercase();
184    COMMANDS
185        .iter()
186        .filter(|spec| query.is_empty() || spec.name.contains(&query))
187        .collect()
188}

192    COMMANDS
193        .iter()
194        .find(|spec| spec.name == name || spec.aliases.contains(&name))
195}

202pub(crate) fn parse_submission_with_skills(
203    input: &str,
204    skills: &[String],
205) -> Result<Submission, String> {
206    // Commands are deliberately single-line. Multiline text beginning with
207    // `/` is always sent verbatim as a prompt, so a pasted block cannot
208    // accidentally execute a client command.
209    if input.contains('\n') {
210        return Ok(Submission::Prompt(input.to_owned()));
211    }
212    if let Some(prompt) = input.strip_prefix("//") {
213        return Ok(Submission::Prompt(format!("/{prompt}")));
214    }
215    let Some(command_line) = input.strip_prefix('/') else {
216        return Ok(Submission::Prompt(input.to_owned()));
217    };
218    let parts = command_line.split_whitespace().collect::<Vec<_>>();
219    if parts.first().is_none_or(|name| {
220        !COMMANDS
221            .iter()
222            .any(|spec| spec.name == *name || spec.aliases.contains(name))
223            && !skills.iter().any(|skill| skill == name)
224    }) {
225        return Err(format!("unknown command: /{command_line}"));
226    }
227    let exact_builtin = match parts.as_slice() {
228        ["show", "agent", "panel"] => Some(SlashCommand::ShowAgentPanel),
229        ["hide", "agent", "panel"] => Some(SlashCommand::HideAgentPanel),
230        ["approve", "once"] => Some(SlashCommand::Approve(ApprovalUserDecision::ApproveOnce)),
231        ["approve", "all"] => Some(SlashCommand::Approve(ApprovalUserDecision::ApproveTree)),
232        ["approve", "reject"] => Some(SlashCommand::Approve(ApprovalUserDecision::Reject)),
233        ["approve", "cancel"] => Some(SlashCommand::Approve(ApprovalUserDecision::Cancel)),
234        ["events", "debug"] => Some(SlashCommand::Events(crate::state::EventLevel::Debug)),
235        ["events", "info"] => Some(SlashCommand::Events(crate::state::EventLevel::Info)),
236        ["events", "warning"] => Some(SlashCommand::Events(crate::state::EventLevel::Warning)),
237        ["events", "error"] => Some(SlashCommand::Events(crate::state::EventLevel::Error)),
238        _ => None,
239    };
240    let command = if let Some(command) = exact_builtin {
241        command
242    } else if parts.first() == Some(&"goal") {
243        let remainder = command_line
244            .trim_start()
245            .strip_prefix("goal")
246            .map(str::trim)
247            .unwrap_or_default();
248        SlashCommand::Goal(match remainder {
249            "" => return Err("usage: /goal <objective|pause|resume|cancel>".to_owned()),
250            "status" => return Err("goal status is shown in the goal bar".to_owned()),
251            "pause" => GoalCommand::Pause,
252            "resume" => GoalCommand::Resume,
253            "cancel" => GoalCommand::Cancel,
254            objective => GoalCommand::Objective(objective.to_owned()),
255        })
256    } else if parts.first() == Some(&"compact") {
257        let focus = command_line
258            .strip_prefix("compact")
259            .map(str::trim)
260            .filter(|focus| !focus.is_empty())
261            .map(str::to_owned);
262        SlashCommand::Compact(focus)
263    } else if let Some(name) = parts
264        .first()
265        .filter(|name| skills.iter().any(|skill| skill == **name))
266    {
267        let args = command_line
268            .strip_prefix(name)
269            .map(str::trim)
270            .unwrap_or_default()
271            .to_owned();
272        SlashCommand::Skill {
273            name: (*name).to_owned(),
274            args,
275        }
276    } else {
277        match parts.as_slice() {
278            ["quit"] | ["q"] => SlashCommand::Quit,
279            ["new"] => SlashCommand::New,
280            ["preset"] => SlashCommand::Preset,
281            ["connect"] => SlashCommand::Connect,
282            ["mcp"] => SlashCommand::Mcp,
283            ["permissions"] | ["perms"] => SlashCommand::Permissions,
284            ["skills"] => SlashCommand::Skills,
285            ["sessions"] => SlashCommand::Sessions,
286            ["usage"] => SlashCommand::Usage,
287            ["cancel"] => SlashCommand::Cancel,
288            ["help"] => SlashCommand::Help,
289            _ => return Err(format!("invalid command: /{command_line}")),
290        }
291    };
292    Ok(Submission::Command(command))
293}

511pub(crate) fn move_selection(state: &mut ListState, len: usize, up: bool) {
512    if len == 0 {
513        state.select(Some(0));
514        return;
515    }
516    let selected = state.selected().unwrap_or(0);
517    state.select(Some(if up {
518        selected.saturating_sub(1)
519    } else {
520        (selected + 1).min(len - 1)
521    }));
522}

524pub(crate) fn render(
525    frame: &mut Frame,
526    query: &str,
527    entries: Vec<String>,
528    area: Rect,
529    state: &mut ListState,
530    theme: &crate::theme::Theme,
531) -> Vec<(Rect, usize)> {
532    let inner = inner_rect(area);
533    let list_area = Rect::new(
534        inner.x,
535        inner.y.saturating_add(2),
536        inner.width,
537        inner.height.saturating_sub(2),
538    );
539    super::app::paint_panel(frame, area, theme);
540    frame.render_widget(
541        Block::default()
542            .borders(Borders::ALL)
543            .border_style(theme.panel_border())
544            .title("Commands")
545            .title_bottom(
546                ratatui::text::Line::from(ratatui::text::Span::styled(
547                    "↑↓ move · enter: choose · esc: dismiss",
548                    theme.internal(),
549                ))
550                .right_aligned(),
551            ),
552        area,
553    );
554    frame.render_widget(
555        Paragraph::new(format!("/{query}")).block(
556            Block::default()
557                .borders(Borders::BOTTOM)
558                .border_style(theme.panel_border())
559                .title("Search"),
560        ),
561        Rect::new(inner.x, inner.y, inner.width, inner.height.min(2)),
562    );
563    let entry_count = entries.len();
564    if entry_count == 0 {
565        frame.render_widget(
566            Paragraph::new("No matching commands").style(theme.muted()),
567            list_area,
568        );
569        state.select(None);
570        return Vec::new();
571    }
572    // Rows ellipsize instead of hard-clipping at the panel edge; the two
573    // selection-marker columns stay reserved on every row.
574    let available = usize::from(list_area.width.saturating_sub(2));
575    let entries = entries
576        .into_iter()
577        .map(|entry| super::app::truncate_with_ellipsis(&entry, available))
578        .collect::<Vec<_>>();
579    frame.render_stateful_widget(
580        List::new(entries.into_iter().map(ListItem::new).collect::<Vec<_>>())
581            .highlight_symbol("> ")
582            .highlight_style(theme.selected()),
583        list_area,
584        state,
585    );
586    (state.offset()..entry_count)
587        .take(usize::from(list_area.height))
588        .enumerate()
589        .map(|(row, index)| {
590            (
591                Rect::new(
592                    list_area.x,
593                    list_area.y + u16::try_from(row).unwrap_or(u16::MAX),
594                    list_area.width,
595                    1,
596                ),
597                index,
598            )
599        })
600        .collect()
601}
602
603fn inner_rect(area: Rect) -> Rect {
604    Rect::new(
605        area.x.saturating_add(1),
606        area.y.saturating_add(1),
607        area.width.saturating_sub(2),
608        area.height.saturating_sub(2),
609    )
610}
