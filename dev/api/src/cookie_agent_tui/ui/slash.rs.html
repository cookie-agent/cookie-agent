138#[derive(Debug, Eq, PartialEq)]
139pub(crate) enum Submission {
140    Prompt(String),
141    Command(SlashCommand),
142}

144pub(crate) fn entries(input: &str) -> Vec<&'static CommandSpec> {
145    let query = input
146        .strip_prefix('/')
147        .unwrap_or_default()
148        .split_whitespace()
149        .next()
150        .unwrap_or_default()
151        .to_ascii_lowercase();
152    COMMANDS
153        .iter()
154        .filter(|spec| query.is_empty() || spec.name.contains(&query))
155        .collect()
156}

158#[cfg(test)]
159pub(crate) fn command_spec(name: &str) -> Option<&'static CommandSpec> {
160    COMMANDS
161        .iter()
162        .find(|spec| spec.name == name || spec.aliases.contains(&name))
163}

170pub(crate) fn parse_submission_with_skills(
171    input: &str,
172    skills: &[String],
173) -> Result<Submission, String> {
174    // Commands are deliberately single-line. Multiline text beginning with
175    // `/` is always sent verbatim as a prompt, so a pasted block cannot
176    // accidentally execute a client command.
177    if input.contains('\n') {
178        return Ok(Submission::Prompt(input.to_owned()));
179    }
180    if let Some(prompt) = input.strip_prefix("//") {
181        return Ok(Submission::Prompt(format!("/{prompt}")));
182    }
183    let Some(command_line) = input.strip_prefix('/') else {
184        return Ok(Submission::Prompt(input.to_owned()));
185    };
186    let parts = command_line.split_whitespace().collect::<Vec<_>>();
187    if parts.first().is_none_or(|name| {
188        !COMMANDS
189            .iter()
190            .any(|spec| spec.name == *name || spec.aliases.contains(name))
191            && !skills.iter().any(|skill| skill == name)
192    }) {
193        return Err(format!("unknown command: /{command_line}"));
194    }
195    let command = if let Some(name) = parts
196        .first()
197        .filter(|name| skills.iter().any(|skill| skill == **name))
198    {
199        let args = command_line
200            .strip_prefix(name)
201            .map(str::trim)
202            .unwrap_or_default()
203            .to_owned();
204        SlashCommand::Skill {
205            name: (*name).to_owned(),
206            args,
207        }
208    } else if parts.first() == Some(&"compact") {
209        let focus = command_line
210            .strip_prefix("compact")
211            .map(str::trim)
212            .filter(|focus| !focus.is_empty())
213            .map(str::to_owned);
214        SlashCommand::Compact(focus)
215    } else {
216        match parts.as_slice() {
217            ["quit"] | ["q"] => SlashCommand::Quit,
218            ["new"] => SlashCommand::New,
219            ["preset"] => SlashCommand::Preset,
220            ["connect"] => SlashCommand::Connect,
221            ["mcp"] => SlashCommand::Mcp,
222            ["permissions"] | ["perms"] => SlashCommand::Permissions,
223            ["skills"] => SlashCommand::Skills,
224            ["sessions"] => SlashCommand::Sessions,
225            ["usage"] => SlashCommand::Usage,
226            ["cancel"] => SlashCommand::Cancel,
227            ["approve", "once"] => SlashCommand::Approve(ApprovalUserDecision::ApproveOnce),
228            ["approve", "all"] => SlashCommand::Approve(ApprovalUserDecision::ApproveTree),
229            ["approve", "reject"] => SlashCommand::Approve(ApprovalUserDecision::Reject),
230            ["approve", "cancel"] => SlashCommand::Approve(ApprovalUserDecision::Cancel),
231            ["events", "debug"] => SlashCommand::Events(crate::state::EventLevel::Debug),
232            ["events", "info"] => SlashCommand::Events(crate::state::EventLevel::Info),
233            ["events", "warning"] => SlashCommand::Events(crate::state::EventLevel::Warning),
234            ["events", "error"] => SlashCommand::Events(crate::state::EventLevel::Error),
235            ["help"] => SlashCommand::Help,
236            _ => return Err(format!("invalid command: /{command_line}")),
237        }
238    };
239    Ok(Submission::Command(command))
240}

304pub(crate) fn move_selection(state: &mut ListState, len: usize, up: bool) {
305    if len == 0 {
306        state.select(Some(0));
307        return;
308    }
309    let selected = state.selected().unwrap_or(0);
310    state.select(Some(if up {
311        selected.saturating_sub(1)
312    } else {
313        (selected + 1).min(len - 1)
314    }));
315}

317pub(crate) fn render(
318    frame: &mut Frame,
319    query: &str,
320    entries: Vec<String>,
321    area: Rect,
322    state: &mut ListState,
323    theme: &crate::theme::Theme,
324) -> Vec<(Rect, usize)> {
325    let inner = inner_rect(area);
326    let list_area = Rect::new(
327        inner.x,
328        inner.y.saturating_add(2),
329        inner.width,
330        inner.height.saturating_sub(2),
331    );
332    super::app::paint_panel(frame, area, theme);
333    frame.render_widget(
334        Block::default()
335            .borders(Borders::ALL)
336            .border_style(theme.panel_border())
337            .title("Commands")
338            .title_bottom(
339                ratatui::text::Line::from(ratatui::text::Span::styled(
340                    "↑↓ move · enter: choose · esc: dismiss",
341                    theme.internal(),
342                ))
343                .right_aligned(),
344            ),
345        area,
346    );
347    frame.render_widget(
348        Paragraph::new(format!("/{query}")).block(
349            Block::default()
350                .borders(Borders::BOTTOM)
351                .border_style(theme.panel_border())
352                .title("Search"),
353        ),
354        Rect::new(inner.x, inner.y, inner.width, inner.height.min(2)),
355    );
356    let entry_count = entries.len();
357    if entry_count == 0 {
358        frame.render_widget(
359            Paragraph::new("No matching commands").style(theme.muted()),
360            list_area,
361        );
362        state.select(None);
363        return Vec::new();
364    }
365    // Rows ellipsize instead of hard-clipping at the panel edge; the two
366    // selection-marker columns stay reserved on every row.
367    let available = usize::from(list_area.width.saturating_sub(2));
368    let entries = entries
369        .into_iter()
370        .map(|entry| super::app::truncate_with_ellipsis(&entry, available))
371        .collect::<Vec<_>>();
372    frame.render_stateful_widget(
373        List::new(entries.into_iter().map(ListItem::new).collect::<Vec<_>>())
374            .highlight_symbol("> ")
375            .highlight_style(theme.selected()),
376        list_area,
377        state,
378    );
379    (state.offset()..entry_count)
380        .take(usize::from(list_area.height))
381        .enumerate()
382        .map(|(row, index)| {
383            (
384                Rect::new(
385                    list_area.x,
386                    list_area.y + u16::try_from(row).unwrap_or(u16::MAX),
387                    list_area.width,
388                    1,
389                ),
390                index,
391            )
392        })
393        .collect()
394}
395
396fn inner_rect(area: Rect) -> Rect {
397    Rect::new(
398        area.x.saturating_add(1),
399        area.y.saturating_add(1),
400        area.width.saturating_sub(2),
401        area.height.saturating_sub(2),
402    )
403}

409    #[test]
410    fn palette_rows_ellipsize_and_the_footer_explains_the_keys() {
411        let theme = crate::theme::Theme::default();
412        let mut terminal = Terminal::new(TestBackend::new(34, 10)).expect("terminal");
413        let mut state = ListState::default().with_selected(Some(0));
414        terminal
415            .draw(|frame| {
416                super::render(
417                    frame,
418                    "",
419                    vec![
420                        "/approve once|tree|reject|cancel — answer an approval".to_owned(),
421                        "/quit — exit the TUI".to_owned(),
422                    ],
423                    frame.area(),
424                    &mut state,
425                    &theme,
426                );
427            })
428            .expect("render palette");
429        let buffer = terminal.backend().buffer();
430        let text = (0..10)
431            .flat_map(|y| (0..34).map(move |x| buffer[(x, y)].symbol().to_owned()))
432            .collect::<String>();
433        assert!(text.contains('…'), "{text}");
434        assert!(!text.contains("answer an approval"), "{text}");
435        // No row spills over the right border.
436        assert_eq!(buffer[(33, 3)].symbol(), "│");
437        assert!(text.contains("enter: choose"), "{text}");
438        assert!(text.contains("esc: dismiss"), "{text}");
439    }
