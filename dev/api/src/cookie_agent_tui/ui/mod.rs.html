16const MAX_AGENT_HEIGHT: u16 = 10;
17/// Composer ceiling in total rows: five text rows plus top/bottom borders.
18const MAX_INPUT_HEIGHT: u16 = 7;
19
20#[derive(Clone, Copy, Debug, Eq, PartialEq)]
21pub(crate) struct UiLayout {
22    pub(crate) agent: Rect,
23    pub(crate) conversation: Rect,
24    /// The steer-queue strip between the conversation pane and the status
25    /// line; zero height while the selected session's queue is empty.
26    pub(crate) queue: Rect,
27    pub(crate) status: Rect,
28    /// The persistent goal bar directly above the composer; zero height when
29    /// it is not visible or the terminal has no space for it.
30    pub(crate) goal: Rect,
31    pub(crate) input: Rect,
32    pub(crate) bar: Rect,
33}
34
35/// Resolve the single top-to-bottom pane geometry for a known visible tree
36/// row count, queue-strip row demand, goal visibility, and composer text-row
37/// demand. The Agents panel is hidden when `visible_tree_row_count` is zero or
38/// one; otherwise it is exactly `clamp(visible_tree_row_count, 1, 8)` text rows
39/// with its borders outside that count, so the conversation starts immediately
40/// below it. The composer takes one text row by default and grows with its
41/// content up to five; every added composer row is reclaimed from the
42/// conversation pane. The goal bar takes exactly one row when it is visible
43/// and space remains above the composer, with the ephemeral status line
44/// yielding first on short terminals. The queue strip reclaims conversation
45/// rows the same way and leaves the conversation at least one row whenever
46/// content space remains after fixed bars.
47pub(crate) fn terminal_layout_with_tree_rows(
48    area: Rect,
49    visible_tree_rows: usize,
50    queue_rows: u16,
51    goal_visible: bool,
52    input_text_rows: u16,
53) -> UiLayout {
54    let agent_height = if visible_tree_rows <= 1 {
55        0
56    } else {
57        (visible_tree_rows.clamp(1, 8) as u16)
58            .saturating_add(2)
59            .min(MAX_AGENT_HEIGHT)
60    };
61    let bar_height = u16::from(area.height > 0);
62    let input_height = input_text_rows
63        .clamp(1, input::MAX_TEXT_ROWS)
64        .saturating_add(2)
65        .min(MAX_INPUT_HEIGHT)
66        .min(area.height.saturating_sub(bar_height));
67    let available_height = area
68        .height
69        .saturating_sub(input_height)
70        .saturating_sub(bar_height);
71    let goal_height = u16::from(goal_visible && available_height > 0);
72    let status_height = u16::from(available_height.saturating_sub(goal_height) >= 3);
73    let content_height = available_height
74        .saturating_sub(goal_height)
75        .saturating_sub(status_height);
76    let agent_height = if content_height > 1 {
77        agent_height.min(content_height - 1)
78    } else {
79        content_height
80    };
81    let conversation_available = content_height.saturating_sub(agent_height);
82    // The strip never starves the conversation completely: on a cramped
83    // terminal it shrinks away rather than taking the last row.
84    let queue_height = queue_rows.min(conversation_available.saturating_sub(1));
85    let conversation_height = conversation_available.saturating_sub(queue_height);
86    let agent = Rect::new(area.x, area.y, area.width, agent_height);
87    let conversation = Rect::new(
88        area.x,
89        area.y.saturating_add(agent_height),
90        area.width,
91        conversation_height,
92    );
93    let queue = Rect::new(
94        area.x,
95        conversation.y.saturating_add(conversation_height),
96        area.width,
97        queue_height,
98    );
99    let status = Rect::new(
100        area.x,
101        queue.y.saturating_add(queue_height),
102        area.width,
103        status_height,
104    );
105    let goal = Rect::new(
106        area.x,
107        status.y.saturating_add(status_height),
108        area.width,
109        goal_height,
110    );
111    let input = Rect::new(
112        area.x,
113        area.y
114            .saturating_add(area.height.saturating_sub(bar_height + input_height)),
115        area.width,
116        input_height,
117    );
118    let bar = Rect::new(
119        area.x,
120        area.y
121            .saturating_add(area.height.saturating_sub(bar_height)),
122        area.width,
123        bar_height,
124    );
125    UiLayout {
126        agent,
127        conversation,
128        queue,
129        status,
130        goal,
131        input,
132        bar,
133    }
134}

136#[cfg(test)]
137mod tests {
138    use super::terminal_layout_with_tree_rows;
139    use ratatui::layout::Rect;

141    #[test]
142    fn hidden_goal_preserves_existing_geometry_without_a_blank_row() {
143        for width in [1, 3, 7, 8, 20, 40, 80] {
144            for height in 0..=24 {
145                let area = Rect::new(2, 3, width, height);
146                let layout = terminal_layout_with_tree_rows(area, 3, 5, false, 2);
147
148                assert_eq!(layout.goal.height, 0, "{width}x{height}");
149                assert_eq!(layout.goal.y, layout.input.y, "{width}x{height}");
150                assert_eq!(
151                    layout.status.y + layout.status.height,
152                    layout.input.y,
153                    "{width}x{height}"
154                );
155            }
156        }
157    }
158
159    #[test]
160    fn visible_goal_is_one_row_when_space_exists_and_never_overlaps() {
161        for width in [1, 3, 7, 8, 20, 40, 80] {
162            for height in 0..=24 {
163                let area = Rect::new(2, 3, width, height);
164                let hidden = terminal_layout_with_tree_rows(area, 3, 5, false, 1);
165                let shown = terminal_layout_with_tree_rows(area, 3, 5, true, 1);
166                let expected_goal_height = u16::from(
167                    area.height
168                        .saturating_sub(hidden.input.height)
169                        .saturating_sub(hidden.bar.height)
170                        > 0,
171                );
172
173                assert_eq!(shown.goal.height, expected_goal_height, "{width}x{height}");
174                assert_eq!(shown.goal.y + shown.goal.height, shown.input.y);
175                assert_eq!(shown.input, hidden.input, "{width}x{height}");
176                assert_eq!(shown.bar, hidden.bar, "{width}x{height}");
177
178                let panes = [
179                    shown.agent,
180                    shown.conversation,
181                    shown.queue,
182                    shown.status,
183                    shown.goal,
184                    shown.input,
185                    shown.bar,
186                ];
187                for pane in panes {
188                    assert_eq!(pane.x, area.x, "{width}x{height}: {pane:?}");
189                    assert_eq!(pane.width, area.width, "{width}x{height}: {pane:?}");
190                    assert!(
191                        pane.y >= area.y && pane.y + pane.height <= area.y + area.height,
192                        "{width}x{height}: {pane:?} outside {area:?}"
193                    );
194                }
195                for pair in panes.windows(2) {
196                    assert!(
197                        pair[0].y + pair[0].height <= pair[1].y,
198                        "{width}x{height}: {:?} overlaps {:?}",
199                        pair[0],
200                        pair[1]
201                    );
202                }
203            }
204        }
205    }

207    #[test]
208    fn visible_goal_is_reserved_before_ephemeral_status() {
209        let area = Rect::new(0, 0, 20, 7);
210        let hidden = terminal_layout_with_tree_rows(area, 0, 0, false, 1);
211        let shown = terminal_layout_with_tree_rows(area, 0, 0, true, 1);
212
213        assert_eq!(hidden.status.height, 1);
214        assert_eq!(hidden.goal.height, 0);
215        assert_eq!(shown.status.height, 0);
216        assert_eq!(shown.goal.height, 1);
217        assert_eq!(shown.goal.y + shown.goal.height, shown.input.y);
218        assert_eq!(shown.input, hidden.input);
219        assert_eq!(shown.bar, hidden.bar);
220    }
