16const MAX_AGENT_HEIGHT: u16 = 10;
17/// Composer ceiling in total rows: five text rows plus top/bottom borders.
18const MAX_INPUT_HEIGHT: u16 = 7;
19
20#[derive(Clone, Copy, Debug, Eq, PartialEq)]
21pub(crate) struct UiLayout {
22    pub(crate) agent: Rect,
23    pub(crate) conversation: Rect,
24    /// The steer-queue strip between the conversation pane and the status
25    /// line; zero height while the selected session's queue is empty.
26    pub(crate) queue: Rect,
27    pub(crate) input: Rect,
28    pub(crate) status: Rect,
29    pub(crate) bar: Rect,
30}
31
32/// Resolve the single top-to-bottom pane geometry for a known visible tree
33/// row count, queue-strip row demand, and composer text-row demand. The
34/// Agents panel is exactly `clamp(visible_tree_row_count, 1, 8)` text rows
35/// with its borders outside that count, so the conversation starts
36/// immediately below it. The composer takes one text row by default and
37/// grows with its content up to five; every added composer row is reclaimed
38/// from the conversation pane. The queue strip reclaims conversation rows
39/// the same way, keeping the status line and composer pinned and always
40/// leaving the conversation at least one row.
41pub(crate) fn terminal_layout_with_tree_rows(
42    area: Rect,
43    visible_tree_rows: usize,
44    queue_rows: u16,
45    input_text_rows: u16,
46) -> UiLayout {
47    let agent_text_rows = visible_tree_rows.clamp(1, 8) as u16;
48    let agent_height = agent_text_rows.saturating_add(2).min(MAX_AGENT_HEIGHT);
49    let bar_height = u16::from(area.height > 0);
50    let input_height = input_text_rows
51        .clamp(1, input::MAX_TEXT_ROWS)
52        .saturating_add(2)
53        .min(MAX_INPUT_HEIGHT)
54        .min(area.height.saturating_sub(bar_height));
55    let available_height = area
56        .height
57        .saturating_sub(input_height)
58        .saturating_sub(bar_height);
59    let status_height = u16::from(available_height >= 3);
60    let content_height = available_height.saturating_sub(status_height);
61    let agent_height = if content_height > 1 {
62        agent_height.min(content_height - 1)
63    } else {
64        content_height
65    };
66    let conversation_available = content_height.saturating_sub(agent_height);
67    // The strip never starves the conversation completely: on a cramped
68    // terminal it shrinks away rather than taking the last row.
69    let queue_height = queue_rows.min(conversation_available.saturating_sub(1));
70    let conversation_height = conversation_available.saturating_sub(queue_height);
71    let agent = Rect::new(area.x, area.y, area.width, agent_height);
72    let conversation = Rect::new(
73        area.x,
74        area.y.saturating_add(agent_height),
75        area.width,
76        conversation_height,
77    );
78    let queue = Rect::new(
79        area.x,
80        conversation.y.saturating_add(conversation_height),
81        area.width,
82        queue_height,
83    );
84    let status = Rect::new(
85        area.x,
86        queue.y.saturating_add(queue_height),
87        area.width,
88        status_height,
89    );
90    let input = Rect::new(
91        area.x,
92        area.y
93            .saturating_add(area.height.saturating_sub(bar_height + input_height)),
94        area.width,
95        input_height,
96    );
97    let bar = Rect::new(
98        area.x,
99        area.y
100            .saturating_add(area.height.saturating_sub(bar_height)),
101        area.width,
102        bar_height,
103    );
104    UiLayout {
105        agent,
106        conversation,
107        queue,
108        status,
109        input,
110        bar,
111    }
112}
