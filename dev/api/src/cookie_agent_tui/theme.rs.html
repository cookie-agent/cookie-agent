1//! Semantic terminal themes and capability-aware color quantization.
2//!
3//! The default theme is a warm, light "cookie" palette: one cream surface
4//! across the whole frame, espresso-brown text, and caramel/cinnamon/sage
5//! accents. Every foreground is chosen for WCAG-AA-or-better contrast
6//! against the cream surface; state is never conveyed by color alone
7//! (bold/italic/underline and text markers accompany every semantic color).
8//! `Mono` drops all color, `HighContrast` keeps bright ANSI colors on the
9//! terminal's own background.
10
11use ratatui::style::{Color, Modifier, Style};

16    Mono,
17    HighContrast,
18}

21pub enum ColorLevel {

25    TrueColor,

42// --- The bakery palette (true color) ---------------------------------------
43//
44// Surfaces: a single cream base, parchment code blocks, toasted selection,
45// glaze hover, mustard borders. Text: espresso body with cocoa and latte
46// derivatives. Accents: honey focus, caramel, sage, cinnamon, cranberry,
47// maple, slate, plum, terracotta.

49const CREAM: (u8, u8, u8) = (0xFB, 0xF4, 0xE6);
50const PARCHMENT: (u8, u8, u8) = (0xF2, 0xE5, 0xCC);
51const TOASTED: (u8, u8, u8) = (0xEA, 0xD7, 0xB4);
52const GLAZE: (u8, u8, u8) = (0xF0, 0xDF, 0xC0);
53const CRUST: (u8, u8, u8) = (0xC9, 0xAE, 0x85);
54// Pane chrome (conversation, tree, pickers, unfocused input) sits one step
55// darker and yellower than the crust selection wash: a muted mustard that
56// reads clearly against the cream surface without turning reddish.
57const BORDER: (u8, u8, u8) = (0x8B, 0x70, 0x48);
58const ESPRESSO: (u8, u8, u8) = (0x46, 0x30, 0x1F);
59const COCOA: (u8, u8, u8) = (0x6E, 0x4E, 0x38);
60const LATTE: (u8, u8, u8) = (0x86, 0x69, 0x4F);
61const FAWN: (u8, u8, u8) = (0x93, 0x76, 0x5B);
62const CARAMEL: (u8, u8, u8) = (0x9C, 0x5A, 0x10);
63const SAGE: (u8, u8, u8) = (0x4E, 0x7A, 0x34);
64const OLIVE: (u8, u8, u8) = (0x3F, 0x6B, 0x2A);
65const CINNAMON: (u8, u8, u8) = (0xA8, 0x5B, 0x17);
66const CRANBERRY: (u8, u8, u8) = (0xAE, 0x33, 0x27);
67const HONEY: (u8, u8, u8) = (0x8F, 0x64, 0x08);
68const MAPLE: (u8, u8, u8) = (0x8A, 0x4A, 0x0B);
69const SLATE: (u8, u8, u8) = (0x3D, 0x6A, 0x8C);

78// --- xterm-256 background bands --------------------------------------------
79//
80// The naive RGB→cube quantizer drags every warm beige into the pink
81// (255, 215, 215) cell 224, so each light band names its cube index by hand.
82// The ladder steps from light to deep: cream surface, parchment code band,
83// glaze hover, toasted selection.
84const CREAM_256: u8 = 231; // (255, 255, 255)
85const PARCHMENT_256: u8 = 230; // (255, 255, 215)
86const GLAZE_256: u8 = 223; // (255, 215, 175)
87const TOASTED_256: u8 = 222; // (255, 215, 135)
88const CRUST_256: u8 = 180; // (215, 175, 135)
89// Hand-picked border cell: the nearest-cube match for BORDER is the
90// pinkish (135, 95, 95), so the border keeps the closest cell that stays
91// warm yellow even though it runs lighter than the true-color original.
92const BORDER_256: u8 = 137; // (175, 135, 95)

107        let requested = std::env::var("COOKIE_THEME").unwrap_or_default();
108        let no_color = std::env::var_os("NO_COLOR").is_some();
109        let term = std::env::var("TERM").unwrap_or_default();

114    /// Apply a theme kind chosen by the TUI config file while still honoring
115    /// `NO_COLOR`/`TERM=dumb` and detected terminal color capability.
116    pub fn with_kind_from_env(kind: ThemeKind) -> Self {
117        let requested = match kind {
118            ThemeKind::Default => "default",
119            ThemeKind::Mono => "mono",
120            ThemeKind::HighContrast => "high-contrast",
121        };
122        let no_color = std::env::var_os("NO_COLOR").is_some();
123        let term = std::env::var("TERM").unwrap_or_default();
124        let colorterm = std::env::var("COLORTERM").unwrap_or_default();
125        Self::from_environment(requested, no_color, &term, &colorterm)
126    }

128    pub fn from_environment(requested: &str, no_color: bool, term: &str, colorterm: &str) -> Self {
129        let requested_kind = match requested.to_ascii_lowercase().as_str() {
130            "mono" | "monochrome" => ThemeKind::Mono,
131            "high-contrast" | "high_contrast" | "contrast" => ThemeKind::HighContrast,
132            _ => ThemeKind::Default,
133        };
134        let colors =
135            if requested_kind == ThemeKind::Mono || no_color || term.eq_ignore_ascii_case("dumb") {
136                ColorLevel::None
137            } else if requested_kind == ThemeKind::HighContrast {
138                ColorLevel::Ansi16
139            } else if colorterm.eq_ignore_ascii_case("truecolor")
140                || colorterm.eq_ignore_ascii_case("24bit")
141            {
142                ColorLevel::TrueColor
143            } else if term.contains("256color") {
144                ColorLevel::Ansi256
145            } else {
146                ColorLevel::Ansi16
147            };
148        let kind = if colors == ColorLevel::None {
149            ThemeKind::Mono
150        } else {
151            requested_kind
152        };
153        Self::new(kind, colors)
154    }

156    pub const fn new(kind: ThemeKind, colors: ColorLevel) -> Self {
157        let colors = match (kind, colors) {
158            (ThemeKind::Mono, _) | (ThemeKind::HighContrast, ColorLevel::None) => ColorLevel::None,
159            (ThemeKind::HighContrast, _) => ColorLevel::Ansi16,
160            (ThemeKind::Default, colors) => colors,
161        };
162        Self {
163            key: ThemeKey { kind, colors },
164        }
165    }

171    /// Base surface painted beneath the whole default-theme UI: a warm cream
172    /// background with an espresso foreground, so even unstyled text stays
173    /// readable. Mono and high-contrast themes keep the terminal background.
174    pub fn surface(&self) -> Style {
175        if self.key.kind != ThemeKind::Default {
176            return Style::default();
177        }
178        match self.key.colors {
179            ColorLevel::None => Style::default(),
180            ColorLevel::Ansi16 => Style::default().fg(Color::Black).bg(Color::White),
181            ColorLevel::Ansi256 => {
182                let mut style = Style::default().bg(Color::Indexed(CREAM_256));
183                if let Some(foreground) = self.quantize_rgb(ESPRESSO.0, ESPRESSO.1, ESPRESSO.2) {
184                    style = style.fg(foreground);
185                }
186                style
187            }
188            ColorLevel::TrueColor => Style::default()
189                .fg(Color::Rgb(ESPRESSO.0, ESPRESSO.1, ESPRESSO.2))
190                .bg(Color::Rgb(CREAM.0, CREAM.1, CREAM.2)),
191        }
192    }

194    /// Panels (overlays, pickers, the focused input box, the bottom bar)
195    /// share the one cream surface — borders, not a second fill, delineate
196    /// them. Foreground is left alone so content keeps whatever the surface
197    /// painted.
198    pub fn panel(&self) -> Style {
199        self.background_color(CREAM, Color::White, CREAM_256, None)
200            .map_or_else(Style::default, |background| Style::default().bg(background))
201    }

203    /// Mustard border for pane chrome (conversation, tree, pickers):
204    /// darker and yellower than the pale crust wash so the frame stays
205    /// visible on the cream surface.
206    pub fn panel_border(&self) -> Style {
207        let style = self.semantic(BORDER, Color::DarkGray, Color::White, Modifier::DIM);
208        if self.key.kind == ThemeKind::Default && self.key.colors == ColorLevel::Ansi256 {
209            // Foregrounds normally trust the quantizer, but it grays out
210            // this warm yellow — use the hand-picked cell.
211            style.fg(Color::Indexed(BORDER_256))
212        } else {
213            style
214        }
215    }

217    /// Keyboard-selected row in pickers, palettes, and the session tree: a
218    /// toasted background with bold espresso text. Reverse video carries the
219    /// selection where no color is available.
220    pub fn selected(&self) -> Style {
221        let style = Style::default().add_modifier(Modifier::BOLD);
222        match self.key.colors {
223            ColorLevel::None => Style::default().add_modifier(Modifier::REVERSED),
224            _ if self.key.kind == ThemeKind::HighContrast => {
225                style.fg(Color::Black).bg(Color::LightYellow)
226            }
227            ColorLevel::Ansi16 => style.fg(Color::Black).bg(Color::Gray),
228            ColorLevel::Ansi256 => {
229                let mut style = style.bg(Color::Indexed(TOASTED_256));
230                if let Some(foreground) = self.quantize_rgb(ESPRESSO.0, ESPRESSO.1, ESPRESSO.2) {
231                    style = style.fg(foreground);
232                }
233                style
234            }
235            ColorLevel::TrueColor => style
236                .fg(Color::Rgb(ESPRESSO.0, ESPRESSO.1, ESPRESSO.2))
237                .bg(Color::Rgb(TOASTED.0, TOASTED.1, TOASTED.2)),
238        }
239    }

241    /// Mouse-drag text selection: a deeper crust wash than the keyboard
242    /// `selected` row and never bold, so the two selection kinds never read
243    /// alike. Where subtle color exists (ANSI-256, true color) only the
244    /// background is set — patched cells keep their foregrounds, so
245    /// selected code keeps its syntax colors. ANSI-16 and high-contrast
246    /// targets are the one documented exception: their text is always a
247    /// bright color, which a light-cyan wash would swallow, so the pair is
248    /// pinned to black-on-light-cyan. Cruder targets add bold on top of
249    /// reverse video to stay distinct.
250    pub fn text_selection(&self) -> Style {
251        match self.key.colors {
252            ColorLevel::None => Style::default().add_modifier(Modifier::REVERSED | Modifier::BOLD),
253            _ if self.key.kind == ThemeKind::HighContrast => {
254                Style::default().fg(Color::Black).bg(Color::LightCyan)
255            }
256            ColorLevel::Ansi16 => Style::default().fg(Color::Black).bg(Color::LightCyan),
257            ColorLevel::Ansi256 => Style::default().bg(Color::Indexed(CRUST_256)),
258            ColorLevel::TrueColor => Style::default().bg(Color::Rgb(CRUST.0, CRUST.1, CRUST.2)),
259        }
260    }

262    /// Hover affordance patched over interactive text cells: a quiet glaze
263    /// background — one calm step deeper than the cream surface, never a
264    /// reddish tint. Cruder targets (mono, ANSI-16, high contrast) underline
265    /// as well, so the affordance never relies on a subtle background shift
266    /// alone. Existing foreground colors are preserved by the patch.
267    pub fn hover(&self) -> Style {
268        let quiet_background_only = matches!(
269            (self.key.kind, self.key.colors),
270            (
271                ThemeKind::Default,
272                ColorLevel::Ansi256 | ColorLevel::TrueColor
273            )
274        );
275        let style = if quiet_background_only {
276            Style::default()
277        } else {
278            Style::default().add_modifier(Modifier::UNDERLINED)
279        };
280        self.background_color(GLAZE, Color::Gray, GLAZE_256, Some(Color::DarkGray))
281            .map_or(style, |background| style.bg(background))
282    }

300    /// One approval decision button. `active` (hover) fills the button with
301    /// a decision-tinted background; the border/label color and the button
302    /// glyphs carry the meaning, never color alone.
303    pub fn decision(&self, tone: DecisionTone, active: bool) -> Style {
304        if self.key.colors == ColorLevel::None {
305            let mut style = Style::default().add_modifier(Modifier::BOLD);
306            if active {
307                style = style.add_modifier(Modifier::REVERSED);
308            }
309            return style;
310        }
311        if self.key.kind == ThemeKind::HighContrast {
312            let foreground = match tone {
313                DecisionTone::Allow => Color::LightGreen,
314                DecisionTone::Deny => Color::LightRed,
315                DecisionTone::Neutral => Color::White,
316            };
317            let mut style = Style::default().fg(foreground).add_modifier(Modifier::BOLD);
318            if active {
319                style = style.add_modifier(Modifier::REVERSED);
320            }
321            return style;
322        }
323        if self.key.colors == ColorLevel::Ansi16 {
324            if active {
325                let background = match tone {
326                    DecisionTone::Allow => Color::LightGreen,
327                    DecisionTone::Deny => Color::LightRed,
328                    DecisionTone::Neutral => Color::Gray,
329                };
330                return Style::default()
331                    .fg(Color::Black)
332                    .bg(background)
333                    .add_modifier(Modifier::BOLD);
334            }
335            let foreground = match tone {
336                DecisionTone::Allow => Color::Green,
337                DecisionTone::Deny => Color::Red,
338                DecisionTone::Neutral => Color::Cyan,
339            };
340            return Style::default().fg(foreground).add_modifier(Modifier::BOLD);
341        }
342        let (foreground, tint) = match tone {
343            DecisionTone::Allow => (OLIVE, ALLOW_TINT),
344            DecisionTone::Deny => (CRANBERRY, DENY_TINT),
345            DecisionTone::Neutral => (COCOA, NEUTRAL_TINT),
346        };
347        let mut style = Style::default().add_modifier(Modifier::BOLD);
348        if let Some(foreground) = self.quantize_rgb(foreground.0, foreground.1, foreground.2) {
349            style = style.fg(foreground);
350        }
351        if active && let Some(background) = self.quantize_rgb(tint.0, tint.1, tint.2) {
352            style = style.bg(background);
353        }
354        style
355    }

456    pub fn quantize_rgb(&self, red: u8, green: u8, blue: u8) -> Option<Color> {
457        match self.key.colors {
458            ColorLevel::None => None,
459            _ if self.key.kind == ThemeKind::HighContrast => {
460                Some(nearest_high_contrast(red, green, blue))
461            }
462            ColorLevel::TrueColor => Some(Color::Rgb(red, green, blue)),
463            ColorLevel::Ansi256 => Some(Color::Indexed(rgb_to_ansi256(red, green, blue))),
464            ColorLevel::Ansi16 => Some(nearest_ansi16(red, green, blue)),
465        }
466    }

468    /// A background color honoring kind/level fallbacks, with hand-picked
469    /// ANSI-16 choices that stay readable on a light surface and a
470    /// hand-picked xterm-256 cube index — the computed quantization drags
471    /// warm beiges into pink cells, so light bands never trust it.
472    fn background_color(
473        &self,
474        rgb: (u8, u8, u8),
475        ansi16: Color,
476        ansi256: u8,
477        high_contrast: Option<Color>,
478    ) -> Option<Color> {
479        match self.key.colors {
480            ColorLevel::None => None,
481            _ if self.key.kind == ThemeKind::HighContrast => high_contrast,
482            ColorLevel::Ansi16 => Some(ansi16),
483            ColorLevel::Ansi256 => Some(Color::Indexed(ansi256)),
484            ColorLevel::TrueColor => Some(Color::Rgb(rgb.0, rgb.1, rgb.2)),
485        }
486    }
487
488    fn semantic(
489        &self,
490        rgb: (u8, u8, u8),
491        ansi: Color,
492        high_contrast: Color,
493        modifier: Modifier,
494    ) -> Style {
495        let modifier = match self.key.kind {
496            ThemeKind::HighContrast => modifier | Modifier::BOLD,
497            ThemeKind::Default | ThemeKind::Mono => modifier,
498        };
499        let color = match self.key.colors {
500            ColorLevel::None => None,
501            _ if self.key.kind == ThemeKind::HighContrast => Some(high_contrast),
502            ColorLevel::Ansi16 => Some(ansi),
503            ColorLevel::Ansi256 | ColorLevel::TrueColor => self.quantize_rgb(rgb.0, rgb.1, rgb.2),
504        };
505        color.map_or_else(
506            || Style::default().add_modifier(modifier),
507            |color| Style::default().fg(color).add_modifier(modifier),
508        )
509    }
510}

512fn nearest_high_contrast(red: u8, green: u8, blue: u8) -> Color {
513    const COLORS: [(Color, (u8, u8, u8)); 7] = [
514        (Color::LightRed, (255, 0, 0)),
515        (Color::LightGreen, (0, 255, 0)),
516        (Color::LightYellow, (255, 255, 0)),
517        (Color::LightBlue, (0, 0, 255)),
518        (Color::LightMagenta, (255, 0, 255)),
519        (Color::LightCyan, (0, 255, 255)),
520        (Color::White, (255, 255, 255)),
521    ];
522    nearest_color(red, green, blue, &COLORS)
523}
524
525fn rgb_to_ansi256(red: u8, green: u8, blue: u8) -> u8 {
526    let component = |value: u8| ((u16::from(value) * 5 + 127) / 255) as u8;
527    16 + 36 * component(red) + 6 * component(green) + component(blue)
528}
529
530fn nearest_ansi16(red: u8, green: u8, blue: u8) -> Color {
531    const COLORS: [(Color, (u8, u8, u8)); 16] = [
532        (Color::Black, (0, 0, 0)),
533        (Color::Red, (128, 0, 0)),
534        (Color::Green, (0, 128, 0)),
535        (Color::Yellow, (128, 128, 0)),
536        (Color::Blue, (0, 0, 128)),
537        (Color::Magenta, (128, 0, 128)),
538        (Color::Cyan, (0, 128, 128)),
539        (Color::Gray, (192, 192, 192)),
540        (Color::DarkGray, (128, 128, 128)),
541        (Color::LightRed, (255, 0, 0)),
542        (Color::LightGreen, (0, 255, 0)),
543        (Color::LightYellow, (255, 255, 0)),
544        (Color::LightBlue, (0, 0, 255)),
545        (Color::LightMagenta, (255, 0, 255)),
546        (Color::LightCyan, (0, 255, 255)),
547        (Color::White, (255, 255, 255)),
548    ];
549    nearest_color(red, green, blue, &COLORS)
550}

552fn nearest_color(red: u8, green: u8, blue: u8, colors: &[(Color, (u8, u8, u8))]) -> Color {
553    colors
554        .iter()
555        .min_by_key(|(_, (candidate_red, candidate_green, candidate_blue))| {
556            let red = i32::from(red) - i32::from(*candidate_red);
557            let green = i32::from(green) - i32::from(*candidate_green);
558            let blue = i32::from(blue) - i32::from(*candidate_blue);
559            red * red + green * green + blue * blue
560        })
561        .map_or(Color::White, |(color, _)| *color)
562}

774        }
