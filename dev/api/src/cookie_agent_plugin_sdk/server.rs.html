98impl Handlers {
99    fn capabilities(&self) -> ExtensionPluginCapabilities {
100        let mut intercept = Vec::new();
101        if self.tool_before.is_some() {
102            intercept.push(ExtensionInterceptionHook::ToolBeforeCall);
103        }
104        if self.tool_after.is_some() {
105            intercept.push(ExtensionInterceptionHook::ToolAfterResult);
106        }
107        if self.agent_before.is_some() {
108            intercept.push(ExtensionInterceptionHook::AgentBeforeStart);
109        }
110        if self.compact.is_some() {
111            intercept.push(ExtensionInterceptionHook::SessionBeforeCompact);
112        }
113        ExtensionPluginCapabilities {
114            tools: !self.tools.is_empty(),
115            resources: false,
116            subscribe_events: self.event.is_some(),
117            subscribe_bus: self.bus.is_some(),
118            publish_bus: self.publish_bus,
119            publish_session_events: self.publish_session_events,
120            intercept,
121        }
122    }

152    async fn run_io<R, W>(self, reader: R, writer: W) -> Result<(), PluginError>
153    where
154        R: AsyncRead + Unpin + Send + 'static,
155        W: AsyncWrite + Unpin + Send + 'static,
156    {
157        let pending = Arc::new(Mutex::new(HashMap::new()));
158        let (outbound, outbound_rx) = mpsc::channel(128);
159        let state = Arc::new(ContextState {
160            grants: Mutex::new(HashMap::new()),
161            pending: Arc::clone(&pending),
162            outbound,
163            publishing: PublishingCapabilities {
164                bus: self.handlers.publish_bus,
165                session: self.handlers.publish_session_events,
166            },
167        });
168        let context = PluginContext { state, grant: None };
169        let mut writer_task = tokio::spawn(writer_loop(writer, outbound_rx, Arc::clone(&pending)));
170        let (inbound_tx, mut inbound_rx) = mpsc::channel(32);
171        let reader_task = tokio::spawn(reader_loop(reader, inbound_tx));
172
173        let mut dispatch_tasks: Vec<JoinHandle<()>> = Vec::new();
174        let result = loop {
175            tokio::select! {
176                inbound = inbound_rx.recv() => match inbound {
177                    Some(Ok(frame)) => {
178                        dispatch_tasks.retain(|task| !task.is_finished());
179                        match self.dispatch_message(
180                            &frame.message,
181                            frame.received_at,
182                            &context,
183                            &mut dispatch_tasks,
184                        ).await {
185                            Ok(Dispatch::Continue) => {}
186                            Ok(Dispatch::Shutdown) => break Ok(()),
187                            Err(error) => break Err(error),
188                        }
189                    }
190                    Some(Err(error)) => break Err(error),
191                    None => break Ok(()),
192                },
193                writer_result = writer_task_finished(&mut writer_task) => {
194                    break match writer_result {
195                        Ok(()) => Err(PluginError::TransportClosed),
196                        Err(error) => Err(error),
197                    };
198                }
199            }
200        };
201
202        reader_task.abort();
203        for task in dispatch_tasks {
204            task.abort();
205        }
206        fail_pending(&pending);
207        writer_task.abort();
208        let _ = reader_task.await;
209        let _ = writer_task.await;
210        result
211    }

213    async fn dispatch_message(
214        &self,
215        message: &str,
216        received_at: Instant,
217        context: &PluginContext,
218        tasks: &mut Vec<JoinHandle<()>>,
219    ) -> Result<Dispatch, PluginError> {
220        let value: Value = serde_json::from_str(message)?;
221        if value.get("method").is_none() {
222            return Ok(Dispatch::Continue);
223        }
224        if value.get("id").is_some() {
225            let request: Request = serde_json::from_value(value).map_err(|error| {
226                PluginError::Protocol(format!("malformed engine request: {error}"))
227            })?;
228            self.dispatch_request(request, context, tasks).await?;
229            return Ok(Dispatch::Continue);
230        }
231
232        let notification: Notification = serde_json::from_value(value).map_err(|error| {
233            PluginError::Protocol(format!("malformed engine notification: {error}"))
234        })?;
235        self.dispatch_notification(notification, received_at, context, tasks)
236            .await
237    }

239    async fn dispatch_request(
240        &self,
241        request: Request,
242        context: &PluginContext,
243        tasks: &mut Vec<JoinHandle<()>>,
244    ) -> Result<(), PluginError> {
245        match request.method.as_str() {
246            PLUGIN_INITIALIZE_METHOD => {
247                let _: ExtensionInitializeParams = parse_params(&request)?;
248                let result = ExtensionInitializeResult {
249                    protocol_version: ExtensionProtocolVersion::current(),
250                    name: self.name.clone(),
251                    version: self.version.clone(),
252                    capabilities: self.handlers.capabilities(),
253                    tools: self
254                        .handlers
255                        .tools
256                        .iter()
257                        .map(|tool| tool.declaration.clone())
258                        .collect(),
259                };
260                send_success(context, request.id, result).await?;
261            }
262            PLUGIN_PING_METHOD => {
263                let _: ExtensionPingParams = parse_params(&request)?;
264                send_success(context, request.id, ExtensionPingResult {}).await?;
265            }
266            PLUGIN_TOOLS_CALL_METHOD => {
267                let params: ExtensionToolCallParams = match parse_params(&request) {
268                    Ok(params) => params,
269                    Err(error) => {
270                        send_error(context, request.id, INVALID_PARAMS, error.to_string()).await?;
271                        return Ok(());
272                    }
273                };
274                let Some(tool) = self
275                    .handlers
276                    .tools
277                    .iter()
278                    .find(|tool| tool.declaration.name == params.tool)
279                    .cloned()
280                else {
281                    send_error(
282                        context,
283                        request.id,
284                        METHOD_NOT_FOUND,
285                        format!("unknown plugin tool `{}`", params.tool),
286                    )
287                    .await?;
288                    return Ok(());
289                };
290                let plugin_context =
291                    context.register_request(params.session_id, params.context_id.clone());
292                tasks.push(tokio::spawn(async move {
293                    let handler_context = plugin_context.clone();
294                    let handler_params = params.clone();
295                    let result =
296                        isolate(
297                            async move { (tool.handler)(handler_context, handler_params).await },
298                        )
299                        .await;
300                    plugin_context.revoke(&params.context_id);
301                    match result {
302                        Ok(Ok(output)) => {
303                            let _ = send_success(
304                                &plugin_context,
305                                request.id,
306                                ExtensionToolCallResult::from(output),
307                            )
308                            .await;
309                        }
310                        Ok(Err(error)) => {
311                            let _ =
312                                send_rpc_error(&plugin_context, request.id, error.into_rpc()).await;
313                        }
314                        Err(()) => {
315                            let _ = send_error(
316                                &plugin_context,
317                                request.id,
318                                INTERNAL_ERROR,
319                                "plugin tool handler panicked",
320                            )
321                            .await;
322                        }
323                    }
324                }));
325            }
326            PLUGIN_INTERCEPT_TOOL_BEFORE_CALL_METHOD => {
327                let handler = self.handlers.tool_before.clone();
328                dispatch_intercept::<ExtensionToolBeforeCallParams, _, _, _>(
329                    request,
330                    context,
331                    handler,
332                    tasks,
333                    |handler, ctx, params| handler(ctx, params),
334                    |value| value,
335                )
336                .await?;
337            }
338            PLUGIN_INTERCEPT_TOOL_AFTER_RESULT_METHOD => {
339                let handler = self.handlers.tool_after.clone();
340                dispatch_intercept::<ExtensionToolAfterResultParams, _, _, _>(
341                    request,
342                    context,
343                    handler,
344                    tasks,
345                    |handler, ctx, params| handler(ctx, params),
346                    |value| value,
347                )
348                .await?;
349            }
350            PLUGIN_INTERCEPT_AGENT_BEFORE_START_METHOD => {
351                let handler = self.handlers.agent_before.clone();
352                dispatch_intercept::<ExtensionAgentBeforeStartParams, _, _, _>(
353                    request,
354                    context,
355                    handler,
356                    tasks,
357                    |handler, ctx, params| handler(ctx, params),
358                    |value| ExtensionAgentBeforeStartResult {
359                        addendum: value.into_option(),
360                    },
361                )
362                .await?;
363            }
364            PLUGIN_INTERCEPT_SESSION_BEFORE_COMPACT_METHOD => {
365                let handler = self.handlers.compact.clone();
366                dispatch_intercept::<ExtensionSessionBeforeCompactParams, _, _, _>(
367                    request,
368                    context,
369                    handler,
370                    tasks,
371                    |handler, ctx, params| handler(ctx, params),
372                    |value| ExtensionSessionBeforeCompactResult {
373                        addendum: value.into_option(),
374                    },
375                )
376                .await?;
377            }
378            _ => {
379                send_error(
380                    context,
381                    request.id,
382                    METHOD_NOT_FOUND,
383                    "plugin method is not supported",
384                )
385                .await?;
386            }
387        }
388        Ok(())
389    }

391    async fn dispatch_notification(
392        &self,
393        notification: Notification,
394        received_at: Instant,
395        context: &PluginContext,
396        tasks: &mut Vec<JoinHandle<()>>,
397    ) -> Result<Dispatch, PluginError> {
398        match notification.method.as_str() {
399            PLUGIN_SHUTDOWN_METHOD => {
400                let _: ExtensionShutdownParams = parse_notification_params(&notification)?;
401                Ok(Dispatch::Shutdown)
402            }
403            PLUGIN_EVENT_METHOD => {
404                let params: ExtensionEventParams = parse_notification_params(&notification)?;
405                if let Some(handler) = self.handlers.event.clone() {
406                    let plugin_context = context.register_notification(
407                        params.session_id,
408                        params.context_id.clone(),
409                        received_at,
410                    );
411                    tasks.push(tokio::spawn(async move {
412                        if isolate(async move { handler(plugin_context, params).await })
413                            .await
414                            .is_err()
415                        {
416                            eprintln!("cookie-agent plugin event handler panicked");
417                        }
418                    }));
419                }
420                Ok(Dispatch::Continue)
421            }
422            PLUGIN_BUS_EVENT_METHOD => {
423                let params: ExtensionBusEventParams = parse_notification_params(&notification)?;
424                if let Some(handler) = self.handlers.bus.clone() {
425                    let plugin_context = params.context_id.as_ref().map_or_else(
426                        || context.clone(),
427                        |context_id| {
428                            context.register_notification(
429                                params.session_id,
430                                context_id.clone(),
431                                received_at,
432                            )
433                        },
434                    );
435                    tasks.push(tokio::spawn(async move {
436                        if isolate(async move { handler(plugin_context, params).await })
437                            .await
438                            .is_err()
439                        {
440                            eprintln!("cookie-agent plugin bus handler panicked");
441                        }
442                    }));
443                }
444                Ok(Dispatch::Continue)
445            }
446            PLUGIN_EMIT_RESULT_METHOD => {
447                let result: ExtensionEmitResultParams = parse_notification_params(&notification)?;
448                resolve_emit(&context.state.pending, result);
449                Ok(Dispatch::Continue)
450            }
451            _ => Ok(Dispatch::Continue),
452        }
453    }
454}

601/// Context passed to plugin handlers for publishing correlated events.

608impl PluginContext {
609    /// Publishes a non-durable bus event using this handler's one-shot context grant.
610    ///
611    /// The server must be configured with
612    /// [`PluginServerBuilder::enable_bus_publishing`]. Notification grants expire locally after
613    /// four seconds, one second before the engine's authoritative deadline.
614    pub async fn emit_bus(
615        &self,
616        session_id: SessionId,
617        name: impl Into<String>,
618        payload: Value,
619    ) -> Result<ExtensionEmitStatus, PluginError> {
620        self.emit(session_id, name.into(), payload, EmitTarget::Bus)
621            .await
622    }

625    ///
626    /// The server must be configured with
627    /// [`PluginServerBuilder::enable_session_publishing`]. Notification grants expire locally
628    /// after four seconds, one second before the engine's authoritative deadline.
629    pub async fn emit_session(
630        &self,
631        session_id: SessionId,
632        name: impl Into<String>,
633        payload: Value,
634    ) -> Result<ExtensionEmitStatus, PluginError> {
635        self.emit(session_id, name.into(), payload, EmitTarget::Session)
636            .await

638
639    async fn emit(
640        &self,
641        session_id: SessionId,
642        name: String,
643        payload: Value,
644        target: EmitTarget,
645    ) -> Result<ExtensionEmitStatus, PluginError> {
646        if !self.state.publishing.enabled(target) {
647            return Err(PluginError::PublishingNotEnabled(target.name()));
648        }
649        let context_id = self.consume(session_id)?;
650        let params = ExtensionEmitParams {
651            session_id,
652            context_id,
653            name: name.clone(),
654            payload,
655        };
656        let value = serde_json::to_value(Notification::new(
657            PLUGIN_EMIT_METHOD,
658            Some(serde_json::to_value(params)?),
659        ))?;
660        let (reply, receive) = oneshot::channel();
661        self.state
662            .outbound
663            .send(Outbound::Emit { value, name, reply })
664            .await
665            .map_err(|_| PluginError::TransportClosed)?;
666        let result = receive.await.map_err(|_| PluginError::TransportClosed)??;
667        Ok(match target {
668            EmitTarget::Bus => result.bus,
669            EmitTarget::Session => result.durable,
670        })
671    }

673    fn register_notification(
674        &self,
675        session_id: SessionId,
676        context_id: String,
677        received_at: Instant,
678    ) -> Self {
679        self.register(
680            session_id,
681            context_id,
682            GrantExpiry::At(received_at + NOTIFICATION_CONTEXT_LIFETIME),
683        )
684    }

686    fn register_request(&self, session_id: SessionId, context_id: String) -> Self {
687        self.register(session_id, context_id, GrantExpiry::Request)
688    }
689
690    fn register(&self, session_id: SessionId, context_id: String, expiry: GrantExpiry) -> Self {
691        let now = Instant::now();
692        let mut grants = self
693            .state
694            .grants
695            .lock()
696            .unwrap_or_else(|poisoned| poisoned.into_inner());
697        grants.retain(
698            |_, grant| !matches!(grant.expiry, GrantExpiry::At(deadline) if deadline <= now),
699        );
700        grants.insert(context_id.clone(), Grant { session_id, expiry });
701        Self {
702            state: Arc::clone(&self.state),
703            grant: Some(ScopedGrant {
704                session_id,
705                context_id,
706            }),
707        }
708    }
709
710    fn consume(&self, session_id: SessionId) -> Result<String, PluginError> {
711        let Some(scope) = &self.grant else {
712            return Err(PluginError::ContextUnavailable(session_id));
713        };
714        if scope.session_id != session_id {
715            return Err(PluginError::ContextUnavailable(session_id));
716        }
717        let mut grants = self
718            .state
719            .grants
720            .lock()
721            .unwrap_or_else(|poisoned| poisoned.into_inner());
722        let Some(grant) = grants.get(&scope.context_id) else {
723            return Err(PluginError::ContextUnavailable(session_id));
724        };
725        if grant.session_id != session_id
726            || matches!(grant.expiry, GrantExpiry::At(deadline) if deadline <= Instant::now())
727        {
728            grants.remove(&scope.context_id);
729            return Err(PluginError::ContextUnavailable(session_id));
730        }
731        grants.remove(&scope.context_id);
732        Ok(scope.context_id.clone())
733    }
734
735    fn revoke(&self, context_id: &str) {
736        self.state
737            .grants
738            .lock()
739            .unwrap_or_else(|poisoned| poisoned.into_inner())
740            .remove(context_id);
741    }
742}

840async fn dispatch_intercept<P, H, O, R>(
841    request: Request,
842    context: &PluginContext,
843    handler: Option<H>,
844    tasks: &mut Vec<JoinHandle<()>>,
845    call: impl FnOnce(H, PluginContext, P) -> HandlerFuture<O> + Send + 'static,
846    convert: impl FnOnce(O) -> R + Send + 'static,
847) -> Result<(), PluginError>
848where
849    P: InterceptParams,
850    H: Send + 'static,
851    O: Send + 'static,
852    R: Serialize + Send + 'static,
853{
854    let Some(handler) = handler else {
855        send_error(
856            context,
857            request.id,
858            METHOD_NOT_FOUND,
859            "interception hook is not registered",
860        )
861        .await?;
862        return Ok(());
863    };
864    let params: P = match parse_params(&request) {
865        Ok(params) => params,
866        Err(error) => {
867            send_error(context, request.id, INVALID_PARAMS, error.to_string()).await?;
868            return Ok(());
869        }
870    };
871    let session_id = params.session_id();
872    let context_id = params.context_id().to_owned();
873    let plugin_context = context.register_request(session_id, context_id.clone());
874    tasks.push(tokio::spawn(async move {
875        let handler_context = plugin_context.clone();
876        let result = isolate(async move { call(handler, handler_context, params).await }).await;
877        plugin_context.revoke(&context_id);
878        match result {
879            Ok(result) => {
880                let _ = send_success(&plugin_context, request.id, convert(result)).await;
881            }
882            Err(()) => {
883                let _ = send_error(
884                    &plugin_context,
885                    request.id,
886                    INTERNAL_ERROR,
887                    "plugin interception handler panicked",
888                )
889                .await;
890            }
891        }
892    }));
893    Ok(())
894}

966async fn reader_loop<R>(reader: R, sender: mpsc::Sender<Result<InboundFrame, PluginError>>)
967where
968    R: AsyncRead + Unpin,
969{
970    let mut reader = BufReader::new(reader);
971    loop {
972        match read_frame(&mut reader).await {
973            Ok(Some(frame)) => {
974                let frame = InboundFrame {
975                    message: frame,
976                    received_at: Instant::now(),
977                };
978                if sender.send(Ok(frame)).await.is_err() {
979                    return;
980                }
981            }
982            Ok(None) => return,
983            Err(error) => {
984                let _ = sender.send(Err(error)).await;
985                return;
986            }
987        }
988    }
989}

991async fn writer_loop<W>(
992    mut writer: W,
993    mut receiver: mpsc::Receiver<Outbound>,
994    pending: PendingEmits,
995) -> Result<(), PluginError>
996where
997    W: AsyncWrite + Unpin,
998{
999    while let Some(outbound) = receiver.recv().await {
1000        let (value, emit) = match outbound {
1001            Outbound::Message(value) => (value, None),
1002            Outbound::Emit { value, name, reply } => (value, Some((name, reply))),
1003        };
1004        let bytes = serde_json::to_vec(&value)?;
1005        if bytes.len() > MAX_FRAME_BYTES {
1006            if let Some((_, reply)) = emit {
1007                let _ = reply.send(Err(PluginError::Protocol(format!(
1008                    "plugin frame exceeds {MAX_FRAME_BYTES} bytes"
1009                ))));
1010                continue;
1011            }
1012            return Err(PluginError::Protocol(format!(
1013                "plugin frame exceeds {MAX_FRAME_BYTES} bytes"
1014            )));
1015        }
1016        if let Some((name, reply)) = emit {
1017            pending
1018                .lock()
1019                .unwrap_or_else(|poisoned| poisoned.into_inner())
1020                .entry(name)
1021                .or_default()
1022                .push_back(reply);
1023        }
1024        writer.write_all(&bytes).await?;
1025        writer.write_all(b"\n").await?;
1026        writer.flush().await?;
1027    }
1028    Ok(())
1029}

1032    task: &mut JoinHandle<Result<(), PluginError>>,
1033) -> Result<(), PluginError> {
1034    match task.await {
1035        Ok(result) => result,
1036        Err(_) => Err(PluginError::TransportClosed),
1037    }

1040fn resolve_emit(pending: &PendingEmits, result: ExtensionEmitResultParams) {
1041    let reply = {
1042        let mut pending = pending
1043            .lock()
1044            .unwrap_or_else(|poisoned| poisoned.into_inner());
1045        let reply = pending.get_mut(&result.name).and_then(VecDeque::pop_front);
1046        if pending.get(&result.name).is_some_and(VecDeque::is_empty) {
1047            pending.remove(&result.name);
1048        }
1049        reply
1050    };
1051    if let Some(reply) = reply {
1052        let _ = reply.send(Ok(result));
1053    }
1054}
1055
1056fn fail_pending(pending: &PendingEmits) {
1057    let replies = pending
1058        .lock()
1059        .unwrap_or_else(|poisoned| poisoned.into_inner())
1060        .drain()
1061        .flat_map(|(_, replies)| replies)
1062        .collect::<Vec<_>>();
1063    for reply in replies {
1064        let _ = reply.send(Err(PluginError::TransportClosed));
1065    }
1066}

1068fn validate_tool(tool: &ToolDecl, duplicate: bool) -> Result<(), PluginError> {
1069    if duplicate {
1070        return Err(PluginError::InvalidTool(format!(
1071            "duplicate tool name `{}`",
1072            tool.name
1073        )));
1074    }
1075    if !is_snake_case(&tool.name) {
1076        return Err(PluginError::InvalidTool(format!(
1077            "tool name `{}` must be snake_case",
1078            tool.name
1079        )));
1080    }
1081    if tool.description.is_empty() {
1082        return Err(PluginError::InvalidTool(format!(
1083            "tool `{}` description must not be empty",
1084            tool.name
1085        )));
1086    }
1087    if !is_snake_case(&tool.permission_name) {
1088        return Err(PluginError::InvalidTool(format!(
1089            "tool `{}` permission_name must be snake_case",
1090            tool.name
1091        )));
1092    }
1093    if !tool.parameters.is_object() {
1094        return Err(PluginError::InvalidTool(format!(
1095            "tool `{}` parameters must be a JSON Schema object",
1096            tool.name
1097        )));
1098    }
1099    jsonschema::draft202012::meta::validate(&tool.parameters).map_err(|error| {
1100        PluginError::InvalidTool(format!(
1101            "tool `{}` has invalid JSON Schema: {error}",
1102            tool.name
1103        ))
1104    })?;
1105    if let Some(primary) = &tool.primary_resource_param
1106        && !tool
1107            .parameters
1108            .get("properties")
1109            .and_then(Value::as_object)
1110            .is_some_and(|properties| properties.contains_key(primary))
1111    {
1112        return Err(PluginError::InvalidTool(format!(
1113            "tool `{}` primary_resource_param `{primary}` is not declared in properties",
1114            tool.name
1115        )));
1116    }
1117    Ok(())
1118}

1120fn is_snake_case(value: &str) -> bool {
1121    !value.is_empty()
1122        && value.split('_').all(|word| {
1123            !word.is_empty()
1124                && word
1125                    .bytes()
1126                    .all(|byte| byte.is_ascii_lowercase() || byte.is_ascii_digit())
1127        })
1128        && value.as_bytes()[0].is_ascii_lowercase()
1129}
