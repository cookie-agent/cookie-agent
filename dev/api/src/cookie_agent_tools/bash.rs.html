52struct ProcessGroupChild {
53    child: Option<Child>,
54    process_group: i32,
55    complete: bool,
56}
57
58impl ProcessGroupChild {
59    fn kill_group(&mut self) {
60        unsafe {
61            libc::kill(-self.process_group, libc::SIGKILL);
62        }
63        if let Some(child) = &mut self.child {
64            let _ = child.start_kill();
65        }
66    }
67
68    async fn kill_and_reap(&mut self) {
69        self.kill_group();
70        if let Some(child) = &mut self.child {
71            let _ = child.wait().await;
72        }
73        self.complete = true;
74    }
75}
76
77impl Drop for ProcessGroupChild {
78    fn drop(&mut self) {
79        if !self.complete {
80            self.kill_group();
81            if let Some(mut child) = self.child.take()
82                && let Ok(runtime) = tokio::runtime::Handle::try_current()
83            {
84                runtime.spawn(async move {
85                    let _ = child.wait().await;
86                });
87            }
88        }
89    }
90}

107impl ToolProvider for BashTool {
108    fn tools_for_session(&self, _: &SessionToolContext) -> Result<Vec<ToolSpec>, ToolError> {
109        Ok(vec![ToolSpec {
110            name: "bash".into(),
111            description: "Execute one prepared shell command.".into(),
112            parameters: schema::<BashArgs>(),
113        }])
114    }
115
116    fn get_primary_argument(
117        &self,
118        name: &str,
119        arguments: &serde_json::Value,
120    ) -> Result<String, ToolError> {
121        if name != "bash" {
122            return Err(ToolError::execution("bash provider received another tool"));
123        }
124        let args: BashArgs = parse_args("bash", arguments.clone())?;
125        if args.command.trim().is_empty() {
126            return Err(ToolError::execution("command must not be empty"));
127        }
128        Ok(args.command)
129    }
130
131    fn get_display_argument(
132        &self,
133        name: &str,
134        arguments: &serde_json::Value,
135    ) -> Result<String, ToolError> {
136        let command = self.get_primary_argument(name, arguments)?;
137        Ok(compact_command_line(&command))
138    }
139
140    async fn prepare(
141        &self,
142        ctx: ToolPreparationContext,
143        call: ToolCall,
144    ) -> Result<PreparedTool, ToolError> {
145        let mut args: BashArgs = parse_args("bash", call.arguments)?;
146        if args.command.trim().is_empty() {
147            return Err(ToolError::execution("command must not be empty"));
148        }
149        if args.interactive {
150            return Err(ToolError::unsupported_security(
151                "interactive prepared bash is not supported",
152            ));
153        }
154        if args.timeout == 0 {
155            args.timeout = default_timeout();
156        }
157        let executable_path = resolve_executable("bash")?;
158        let executable = fs_cap::prepare_existing(Path::new("/"), &executable_path)?;
159        if executable.directory || executable.identity.mode & 0o111 == 0 {
160            return Err(ToolError::unsupported_security(
161                "prepared bash executable is not an executable regular file",
162            ));
163        }
164        let cwd = fs_cap::prepare_existing(std::path::Path::new("/"), &ctx.cwd)?;
165        if !cwd.directory {
166            return Err(ToolError::unsupported_security(
167                "bash cwd is not a directory",
168            ));
169        }
170        let mut executable_binding = executable.manifest_bytes()?;
171        executable_binding.extend_from_slice(executable_path.as_os_str().as_encoded_bytes());
172        let command = args.command.clone();
173        let mut binding = command.as_bytes().to_vec();
174        binding.extend_from_slice(&executable_binding);
175        let resources = vec![prepared_resource(
176            PermissionAction::Bash,
177            "command",
178            command.as_bytes(),
179            &binding,
180            PreparedBindingLifetime::ProcessLocal,
181            ApprovalResourceSource::PrimaryOperation,
182        )?];
183        let policy_labels = vec![command.clone()];
184        let mut context = cwd.manifest_bytes()?;
185        context.extend_from_slice(&executable_binding);
186        let operation = prepared_operation(
187            "bash",
188            &args,
189            vec![(PermissionAction::Bash, "execute")],
190            resources,
191            &context,
192        )?;
193        let normalized_arguments = serde_json::json!({
194            "command": command,
195        });
196        PreparedTool::new(
197            operation,
198            normalized_arguments,
199            None,
200            Box::new(BashExecutor {
201                args,
202                cwd,
203                executable,
204            }),
205        )?
206        .with_policy_labels(policy_labels)
207    }
208}

214#[async_trait]
215impl PreparedExecutor for BashExecutor {
216    async fn revalidate(&self) -> Result<(), ToolError> {
217        self.cwd.revalidate()?;
218        self.executable.revalidate()
219    }
220
221    async fn execute(
222        self: Box<Self>,
223        context: ToolExecutionContext,
224    ) -> Result<ToolResult, ToolError> {
225        self.cwd.revalidate()?;
226        self.executable.revalidate()?;
227        if context.cancellation.is_cancelled() {
228            return Err(ToolError::execution("prepared bash cancelled"));
229        }
230        let mut command = Command::new(self.executable.proc_fd_path());
231        command
232            .arg("-lc")
233            .arg(&self.args.command)
234            .current_dir(self.cwd.proc_fd_path())
235            .stdin(Stdio::null())
236            .stdout(Stdio::piped())
237            .stderr(Stdio::piped())
238            .kill_on_drop(true);
239        unsafe {
240            command.pre_exec(|| {
241                if libc::setsid() == -1 {
242                    Err(std::io::Error::last_os_error())
243                } else {
244                    Ok(())
245                }
246            });
247        }
248        let mut child = command
249            .spawn()
250            .map_err(|error| ToolError::execution(error.to_string()))?;
251        let process_group = child
252            .id()
253            .and_then(|id| i32::try_from(id).ok())
254            .ok_or_else(|| ToolError::execution("prepared bash child has no process id"))?;
255        let mut stdout = child
256            .stdout
257            .take()
258            .ok_or_else(|| ToolError::execution("bash stdout pipe missing"))?;
259        let mut stderr = child
260            .stderr
261            .take()
262            .ok_or_else(|| ToolError::execution("bash stderr pipe missing"))?;
263        let stdout_task = tokio::spawn(async move {
264            let mut bytes = Vec::new();
265            stdout.read_to_end(&mut bytes).await.map(|_| bytes)
266        });
267        let stderr_task = tokio::spawn(async move {
268            let mut bytes = Vec::new();
269            stderr.read_to_end(&mut bytes).await.map(|_| bytes)
270        });
271        let mut grouped = ProcessGroupChild {
272            child: Some(child),
273            process_group,
274            complete: false,
275        };
276        enum WaitOutcome {
277            Finished(std::io::Result<std::process::ExitStatus>),
278            TimedOut,
279            Cancelled,
280        }
281        let outcome = {
282            let wait = grouped
283                .child
284                .as_mut()
285                .expect("prepared child exists")
286                .wait();
287            tokio::pin!(wait);
288            tokio::select! {
289                result = tokio::time::timeout(Duration::from_millis(self.args.timeout), &mut wait) => {
290                    match result {
291                        Ok(result) => WaitOutcome::Finished(result),
292                        Err(_) => WaitOutcome::TimedOut,
293                    }
294                }
295                _ = context.cancellation.cancelled() => WaitOutcome::Cancelled,
296            }
297        };
298        let status = match outcome {
299            WaitOutcome::Finished(result) => {
300                grouped.complete = true;
301                result.map_err(|error| ToolError::execution(error.to_string()))?
302            }
303            WaitOutcome::TimedOut => {
304                grouped.kill_and_reap().await;
305                return Err(ToolError::execution("bash timed out"));
306            }
307            WaitOutcome::Cancelled => {
308                grouped.kill_and_reap().await;
309                return Err(ToolError::execution("prepared bash cancelled"));
310            }
311        };
312        grouped.kill_group();
313        let stdout = stdout_task
314            .await
315            .map_err(|error| ToolError::execution(error.to_string()))?
316            .map_err(|error| ToolError::execution(error.to_string()))?;
317        let stderr = stderr_task
318            .await
319            .map_err(|error| ToolError::execution(error.to_string()))?
320            .map_err(|error| ToolError::execution(error.to_string()))?;
321        context.progress.output(OutputStream::Stdout, &stdout);
322        context.progress.output(OutputStream::Stderr, &stderr);
323        let stdout = String::from_utf8_lossy(&stdout);
324        let stderr = String::from_utf8_lossy(&stderr);
325        Ok(ToolResult {
326            title: crate::safe_title("Bash"),
327            output: format!("{stdout}{stderr}"),
328            metadata: serde_json::json!({"status":status.code(),"success":status.success()}),
329            truncation: None,
330            attachments: Vec::new(),
331        })
332    }
333}

335fn resolve_executable(name: &str) -> Result<PathBuf, ToolError> {
336    let path =
337        env::var_os("PATH").unwrap_or_else(|| OsString::from("/usr/local/bin:/usr/bin:/bin"));
338    resolve_executable_in_path(name, &path)
339}
340
341fn resolve_executable_in_path(name: &str, path: &std::ffi::OsStr) -> Result<PathBuf, ToolError> {
342    for directory in env::split_paths(path) {
343        let candidate = directory.join(name);
344        let Ok(metadata) = std::fs::symlink_metadata(&candidate) else {
345            continue;
346        };
347        if !metadata.file_type().is_symlink() && metadata.is_file() && metadata.mode() & 0o111 != 0
348        {
349            return candidate
350                .canonicalize()
351                .map_err(|error| ToolError::execution(error.to_string()));
352        }
353    }
354    Err(ToolError::execution(format!(
355        "unable to resolve executable `{name}` from PATH during preparation"
356    )))
357}

639        let shell = resolve_executable_in_path(
640            "bash",

642        )
643        .expect("shell");
644        let mut command = tokio::process::Command::new(shell);
645        command
646            .arg("-c")
647            .arg(format!(
648                "sleep 30 & echo $! > '{}'; wait",
649                pid_file.display()
650            ))
651            .stdout(std::process::Stdio::null())
652            .stderr(std::process::Stdio::null());
653        unsafe {
654            command.pre_exec(|| {
655                if libc::setsid() == -1 {
656                    Err(std::io::Error::last_os_error())
657                } else {
658                    Ok(())
659                }
660            });
661        }
662        let child = command.spawn().expect("spawn group");
663        let process_group = i32::try_from(child.id().expect("pid")).expect("pid fits");
664        let mut grouped = super::ProcessGroupChild {
665            child: Some(child),
666            process_group,
667            complete: false,
668        };
669        for _ in 0..100 {
670            if pid_file.exists() {
671                break;
672            }
673            tokio::time::sleep(std::time::Duration::from_millis(5)).await;
674        }
675        let descendant: i32 = fs::read_to_string(&pid_file)
676            .expect("descendant pid")
677            .trim()
678            .parse()
679            .expect("numeric pid");
680        grouped.kill_and_reap().await;
681        for _ in 0..100 {
682            let alive = unsafe { libc::kill(descendant, 0) == 0 };
683            if !alive {
684                return;
685            }
686            tokio::time::sleep(std::time::Duration::from_millis(5)).await;
687        }
688        panic!("descendant process survived process-group cancellation");
689    }
