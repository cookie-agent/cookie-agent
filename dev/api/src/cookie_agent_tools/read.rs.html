83    fn get_permission_resource(
84        &self,
85        name: &str,
86        arguments: &serde_json::Value,
87    ) -> Result<(&'static str, Option<String>), ToolError> {
88        let permission_name = Self::get_permission_name(name)?;
89        let args: ReadArgs = parse_args("read", arguments.clone())?;
90        if args.file_path.is_empty() {
91            return Err(ToolError::execution("filePath must not be empty"));
92        }
93        Ok((
94            permission_name,
95            Some(crate::permission_path_label(
96                &args.file_path,
97                &self.workspace,
98            )),
99        ))
100    }

102    fn get_display_argument(
103        &self,
104        name: &str,
105        arguments: &serde_json::Value,
106    ) -> Result<String, ToolError> {
107        if name != "read" {
108            return Err(ToolError::execution("read provider received another tool"));
109        }
110        let args: ReadArgs = parse_args("read", arguments.clone())?;
111        if args.file_path.is_empty() {
112            return Err(ToolError::execution("filePath must not be empty"));
113        }
114        let path = crate::permission_path_label(&args.file_path, &self.workspace);
115        let mut display = crate::abbreviated_display_path(&path, &self.workspace);
116        let window = match (args.offset, args.limit) {
117            (Some(offset), Some(limit)) => Some(format!("offset={offset}, limit={limit}")),
118            (Some(offset), None) => Some(format!("offset={offset}")),
119            (None, Some(limit)) => Some(format!("limit={limit}")),
120            (None, None) => None,
121        };
122        if let Some(window) = window {
123            display.push_str(" [");
124            display.push_str(&window);
125            display.push(']');
126        }
127        Ok(display)
128    }

130    async fn prepare(
131        &self,
132        ctx: ToolPreparationContext,
133        call: ToolCall,
134    ) -> Result<PreparedTool, ToolError> {
135        if call.name != "read" {
136            return Err(ToolError::execution("read provider received another tool"));
137        }
138        let mut args: ReadArgs = parse_args("read", call.arguments)?;
139        let offset = args.offset.unwrap_or(0);
140        let limit = args.limit.unwrap_or(DEFAULT_LIMIT);
141        if limit == 0 {
142            return Err(ToolError::execution("limit must be positive"));
143        }
144        args.offset = Some(offset);
145        args.limit = Some(limit);
146        let target = fs_cap::prepare_existing(&ctx.cwd, std::path::Path::new(&args.file_path))?;
147        let binding = target.manifest_bytes()?;
148        let (resources, policy_labels) = prepared_path_resources(
149            PermissionAction::Read,
150            if target.directory {
151                "directory"
152            } else {
153                "file"
154            },
155            &target.display_path,
156            &self.workspace,
157            &binding,
158        )?;
159        let context = fs_cap::cwd_context_bytes(&ctx.cwd)?;
160        let operation = prepared_operation(
161            "read",
162            &args,
163            vec![(PermissionAction::Read, "read")],
164            resources,
165            &context,
166        )?;
167        let normalized_arguments = serde_json::json!({
168            "filePath": target.display_path,
169            "offset": offset,
170            "limit": limit,
171        });
172        PreparedTool::new(
173            operation,
174            normalized_arguments,
175            None,
176            Box::new(ReadExecutor {
177                target,
178                offset,
179                limit,
180            }),
181        )?
182        .with_policy_labels(policy_labels)
183    }

186#[async_trait]
187impl PreparedExecutor for ReadExecutor {
188    async fn revalidate(&self) -> Result<(), ToolError> {
189        self.target.revalidate()
190    }
191
192    async fn execute(
193        self: Box<Self>,
194        context: ToolExecutionContext,
195    ) -> Result<ToolResult, ToolError> {
196        if context.cancellation.is_cancelled() {
197            return Err(ToolError::execution(
198                "prepared read cancelled before execution",
199            ));
200        }
201        self.target.revalidate()?;
202        if self.target.directory {
203            let entries = self.target.directory_entries()?;
204            let snapshot = serde_json::to_vec(&entries)
205                .map_err(|error| ToolError::execution(error.to_string()))?;
206            if cookie_agent_protocol::Sha256Digest::of_bytes(&snapshot)
207                != self.target.content_digest
208            {
209                return Err(ToolError::operation_changed(
210                    "prepared directory snapshot changed",
211                ));
212            }
213            let page = directory_page(&entries, self.offset, self.limit).collect::<Vec<_>>();
214            let mut output = format!(
215                "<path>{}</path>\n<type>directory</type>\n<entries>\n",
216                self.target.display_path.display()
217            );
218            for (name, directory) in &page {
219                output.push_str(name);
220                if *directory {
221                    output.push('/');
222                }
223                output.push('\n');
224            }
225            output.push_str("</entries>");
226            return Ok(ToolResult {
227                title: crate::safe_title(format!(
228                    "Read directory {}",
229                    self.target.display_path.display()
230                )),
231                output,
232                metadata: serde_json::json!({"kind":"directory","shown":page.len(),"total_entries":entries.len()}),
233                truncation: None,
234                attachments: Vec::new(),
235                additional_messages: Vec::new(),
236            });
237        }
238        let bytes = self.target.verified_bytes()?;
239        if let Some(mime) = approved_media_type(&self.target.display_path, &bytes)? {
240            let gate = gate_attachment(
241                context.turn_context.adapter_family,
242                &context.turn_context.capabilities,
243                mime,
244                &bytes,
245            );
246            if let Some(error) = attachment_gate_error(
247                gate,
248                mime,
249                &context.turn_context.model,
250                context.turn_context.adapter,
251            ) {
252                return Err(ToolError::execution(error));
253            }
254            let attachment = context.retain_validated_attachment(
255                mime,
256                self.target
257                    .display_path
258                    .file_name()
259                    .map(|name| name.to_string_lossy().into_owned()),
260                &bytes,
261            )?;
262            let sha256 = attachment.sha256.clone();
263            let (output, attachments, additional_messages) = match gate {
264                AttachmentGate::AttachToolResult => (
265                    format!("Attached {mime} ({} bytes).", bytes.len()),
266                    vec![attachment],
267                    Vec::new(),
268                ),
269                AttachmentGate::DeliverViaUserTurn => (
270                    format!(
271                        "Attached {mime} ({} bytes), delivered in the following message.",
272                        bytes.len()
273                    ),
274                    Vec::new(),
275                    vec![
276                        ToolEmittedMessage::new(
277                            ToolEmittedMessageRole::User,
278                            vec![ToolEmittedContent::File(attachment)],
279                        )
280                        .map_err(|error| ToolError::execution(error.to_string()))?,
281                    ],
282                ),
283                AttachmentGate::RejectUnsupportedModel
284                | AttachmentGate::RejectUnsupportedFamily
285                | AttachmentGate::RejectTooLarge { .. } => {
286                    unreachable!("rejected attachment gates returned an error")
287                }
288            };
289            return Ok(ToolResult {
290                title: crate::safe_title(format!(
291                    "Read attachment {}",
292                    self.target.display_path.display()
293                )),
294                output,
295                metadata: serde_json::json!({"kind":"attachment","mime_type":mime,"sha256":sha256}),
296                truncation: None,
297                attachments,
298                additional_messages,
299            });
300        }
301        let text = std::str::from_utf8(&bytes)
302            .map_err(|_| ToolError::execution("read supports UTF-8 text or approved media"))?;
303        Ok(text_result(
304            &self.target.display_path,
305            text,
306            self.offset,
307            self.limit,
308        ))
309    }
310}

312fn text_result(path: &std::path::Path, text: &str, offset: usize, limit: usize) -> ToolResult {
313    let lines = text.lines().collect::<Vec<_>>();
314    let mut output = format!(
315        "<path>{}</path>\n<type>file</type>\n<content>\n",
316        path.display()
317    );
318    for (index, line) in text_page(&lines, offset, limit) {
319        output.push_str(&format!("{}: {line}\n", index + 1));
320    }
321    output.push_str("</content>");
322    ToolResult {
323        title: crate::safe_title(format!("Read file {}", path.display())),
324        output,
325        metadata: serde_json::json!({"kind":"text","offset":offset,"limit":limit,"total_lines":lines.len()}),
326        truncation: None,
327        attachments: Vec::new(),
328        additional_messages: Vec::new(),
329    }
330}

332fn text_page<'a>(
333    lines: &'a [&'a str],
334    offset: usize,
335    limit: usize,
336) -> impl Iterator<Item = (usize, &'a str)> + 'a {
337    lines.iter().copied().enumerate().skip(offset).take(limit)
338}
339
340fn directory_page<T>(entries: &[T], offset: usize, limit: usize) -> impl Iterator<Item = &T> {
341    entries.iter().skip(offset).take(limit)
342}
