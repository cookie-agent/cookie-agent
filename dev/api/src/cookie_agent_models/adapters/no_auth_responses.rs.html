16const ADAPTER_RECIPE_ID: &str = "oven.openai.responses";
17
18pub(crate) fn build(
19    provider_id: &str,
20    model_id: &str,
21    endpoint: String,
22    headers: BTreeMap<String, String>,
23    capabilities: oven_sdk::ModelCapabilities,
24) -> Result<ConstructedAdapter, ModelBuildError> {
25    let descriptor = LanguageModelDescriptor::new(
26        ModelIdentity::new(
27            ProviderId::new(provider_id.to_owned()),
28            ModelId::new(model_id.to_owned()),
29        )?,
30        oven_sdk::AdapterId::new(ADAPTER_RECIPE_ID),
31        capabilities,
32    )?;
33    let endpoint = url::Url::parse(&endpoint)
34        .map_err(|_| ModelError::invalid_request("invalid no-auth Responses endpoint"))?;
35    let headers = TemplateHeaderProvider::new(&headers)?;
36    let client = reqwest_oven::Client::builder()
37        .build()
38        .map_err(|_| ModelError::transport("could not construct no-auth Responses client"))?;
39    Ok(ConstructedAdapter {
40        model: Arc::new(NoAuthResponsesModel {
41            descriptor,
42            endpoint,
43            headers: Arc::new(headers),
44            client,
45        }),
46        provider_options: BTreeMap::new(),
47    })
48}

50#[derive(Clone)]
51struct NoAuthResponsesModel {
52    descriptor: LanguageModelDescriptor,
53    endpoint: url::Url,
54    headers: Arc<TemplateHeaderProvider>,
55    client: reqwest_oven::Client,
56}
57
58impl LanguageModel for NoAuthResponsesModel {
59    fn descriptor(&self) -> &LanguageModelDescriptor {
60        &self.descriptor
61    }
62
63    fn validate_request(&self, request: &Request) -> Result<(), ModelError> {
64        request.validate_for(&self.descriptor.capabilities)?;
65        if !request.tools.is_empty() || request.native_context.is_some() {
66            return Err(ModelError::unsupported(
67                "no-auth OpenAI Responses supports the reviewed text-only profile",
68            ));
69        }
70        Ok(())
71    }
72
73    fn stream<'a>(
74        &'a self,
75        request: Request,
76        abort: AbortSignal,
77    ) -> BoxFuture<'a, Result<StreamResponse, ModelError>> {
78        Box::pin(async move {
79            self.validate_request(&request)?;
80            if abort.is_aborted() {
81                return Err(ModelError::abort("request was aborted before dispatch"));
82            }
83            let body = encode_request(&request, self.descriptor.identity.model_id.as_str())
84                .map_err(|error| *error)?;
85            let headers = self
86                .headers
87                .resolved_headers(&request.header_context)
88                .map_err(ModelError::invalid_request)?;
89            let send = self
90                .client
91                .post(format!(
92                    "{}/responses",
93                    self.endpoint.as_str().trim_end_matches('/')
94                ))
95                .headers(headers)
96                .json(&body)
97                .send();
98            let response = tokio::select! {
99                response = send => response.map_err(|_| ModelError::transport("no-auth OpenAI Responses request failed"))?,
100                _ = abort.aborted() => return Err(ModelError::abort("request was aborted before response headers")),
101            };
102            if !response.status().is_success() {
103                return Err(ModelError::invalid_response(format!(
104                    "no-auth OpenAI Responses returned HTTP {}",
105                    response.status().as_u16()
106                )));
107            }
108            let body = response
109                .text()
110                .await
111                .map_err(|_| ModelError::transport("could not read no-auth Responses stream"))?;
112            let parts = decode_sse(&body).map_err(|error| *error)?;
113            Ok(StreamResponse::new(Box::pin(stream::iter(
114                parts.into_iter().map(Ok),
115            ))))
116        })
117    }
118}

120fn encode_request(request: &Request, model_id: &str) -> Result<Value, Box<ModelError>> {
121    let mut input = Vec::new();
122    for turn in &request.history {
123        match turn {
124            HistoryTurn::System(message) => {
125                let text = message
126                    .content
127                    .iter()
128                    .filter_map(|part| match part {
129                        SystemPart::Text(part) => Some(part.text.as_str()),
130                        SystemPart::Custom(_) => None,
131                    })
132                    .collect::<Vec<_>>()
133                    .join("\n");
134                if !text.is_empty() {
135                    input.push(json!({"type":"message","role":"developer","content":[{"type":"input_text","text":text}]}));
136                }
137            }
138            HistoryTurn::User(message) => {
139                let content = message
140                    .content
141                    .iter()
142                    .map(|part| match part {
143                        InputPart::Text(part) => Ok(json!({"type":"input_text","text":part.text})),
144                        InputPart::File(_) | InputPart::Custom(_) => {
145                            Err(Box::new(ModelError::unsupported(
146                                "no-auth OpenAI Responses supports text input only",
147                            )))
148                        }
149                    })
150                    .collect::<Result<Vec<_>, _>>()?;
151                if !content.is_empty() {
152                    input.push(json!({"type":"message","role":"user","content":content}));
153                }
154            }
155            HistoryTurn::Assistant(turn) => {
156                let text = turn
157                    .message
158                    .content
159                    .iter()
160                    .filter_map(|part| match part {
161                        AssistantPart::Text(part) => Some(part.text.as_str()),
162                        _ => None,
163                    })
164                    .collect::<String>();
165                if !text.is_empty() {
166                    input.push(json!({"type":"message","role":"assistant","content":[{"type":"output_text","text":text}]}));
167                }
168            }
169            HistoryTurn::Tool(_) => {
170                return Err(Box::new(ModelError::unsupported(
171                    "no-auth OpenAI Responses does not support tool history",
172                )));
173            }
174        }
175    }
176    let mut body = json!({
177        "model": model_id,
178        "input": input,
179        "stream": true,
180        "store": false
181    });
182    if let Some(value) = request.inference.max_output_tokens {
183        body["max_output_tokens"] = value.into();
184    }
185    if let Some(value) = request.inference.temperature {
186        body["temperature"] = value.into();
187    }
188    if let Some(value) = request.inference.top_p {
189        body["top_p"] = value.into();
190    }
191    Ok(body)
192}

194fn decode_sse(body: &str) -> Result<Vec<StreamPart>, Box<ModelError>> {
195    let mut parts = vec![StreamPart::StreamStart {
196        warnings: Vec::new(),
197    }];
198    let mut text_id = None;
199    let mut completed = false;
200    for event in body.replace("\r\n", "\n").split("\n\n") {
201        let data = event
202            .lines()
203            .filter_map(|line| line.strip_prefix("data:"))
204            .map(str::trim_start)
205            .collect::<Vec<_>>()
206            .join("\n");
207        if data.is_empty() || data == "[DONE]" {
208            continue;
209        }
210        let value: Value = serde_json::from_str(&data).map_err(|_| {
211            Box::new(ModelError::invalid_response(
212                "invalid no-auth Responses SSE event",
213            ))
214        })?;
215        match value.get("type").and_then(Value::as_str).unwrap_or("") {
216            "response.output_text.delta" => {
217                let id = value
218                    .get("item_id")
219                    .and_then(Value::as_str)
220                    .unwrap_or("response-text")
221                    .to_owned();
222                if text_id.is_none() {
223                    parts.push(StreamPart::TextStart {
224                        id: id.clone(),
225                        metadata: PartMetadata::default(),
226                    });
227                    text_id = Some(id.clone());
228                }
229                parts.push(StreamPart::TextDelta {
230                    id,
231                    delta: value
232                        .get("delta")
233                        .and_then(Value::as_str)
234                        .unwrap_or_default()
235                        .to_owned(),
236                    metadata: PartMetadata::default(),
237                });
238            }
239            "response.output_text.done" => {
240                if let Some(id) = text_id.take() {
241                    parts.push(StreamPart::TextEnd {
242                        id,
243                        metadata: PartMetadata::default(),
244                    });
245                }
246            }
247            "response.completed" => {
248                if let Some(id) = text_id.take() {
249                    parts.push(StreamPart::TextEnd {
250                        id,
251                        metadata: PartMetadata::default(),
252                    });
253                }
254                let response = value.get("response").unwrap_or(&value);
255                let usage = response.get("usage").unwrap_or(&Value::Null);
256                parts.push(StreamPart::Finish {
257                    finish: Finish::new(
258                        Usage {
259                            input_tokens: usage.get("input_tokens").and_then(Value::as_u64),
260                            output_tokens: usage.get("output_tokens").and_then(Value::as_u64),
261                            ..Usage::default()
262                        },
263                        FinishReason::stop(),
264                    ),
265                });
266                completed = true;
267            }
268            "response.failed" | "error" => {
269                return Err(Box::new(ModelError::invalid_response(
270                    "no-auth OpenAI Responses reported an error",
271                )));
272            }
273            _ => {}
274        }
275    }
276    if completed {
277        Ok(parts)
278    } else {
279        Err(Box::new(ModelError::unexpected_eof(
280            "no-auth OpenAI Responses ended without response.completed",
281        )))
282    }
283}
