107    pub fn build(&self) -> Result<ConstructedAdapter, ModelBuildError> {
108        let provider = self.provider()?;
109        let mut capabilities = self.capabilities.clone();
110        capabilities.compaction = match &self.adapter {
111            AdapterConfig::OpenaiResponses { settings, .. }
112                if matches!(
113                    settings.compaction,
114                    OpenAiResponsesCompactionConfig::OpenAiResponsesCompact
115                ) =>
116            {
117                oven_sdk::CompactionCapability::Native
118            }
119            AdapterConfig::AzureResponses { settings, .. }
120                if matches!(
121                    settings.compaction,
122                    AzureResponsesCompactionConfig::AzureResponsesCompact { .. }
123                ) =>
124            {
125                oven_sdk::CompactionCapability::Native
126            }
127            _ => oven_sdk::CompactionCapability::Unsupported,
128        };
129        let declaration = ModelDeclaration::new(ModelId::new(self.model_id.clone()), capabilities)?;
130        let (model, provider_options): (Arc<dyn LanguageModel>, BTreeMap<String, Value>) =
131            match &self.adapter {
132                AdapterConfig::Anthropic { settings, options } => (
133                    Arc::new(AnthropicModel::new(ModelConfig::new(
134                        provider.with_auth(self.anthropic_auth()?),
135                        declaration,
136                        settings.to_oven()?,
137                    ))?),
138                    namespace("anthropic", options.to_oven()?)?,
139                ),
140                AdapterConfig::AnthropicCompatible { settings, options } => (
141                    Arc::new(AnthropicCompatibleModel::new(ModelConfig::new(
142                        provider.with_auth(self.anthropic_compatible_auth()?),
143                        declaration,
144                        settings.to_oven()?,
145                    ))?),
146                    namespace("anthropic", options.to_oven()?)?,
147                ),
148                AdapterConfig::OpenaiChat { settings, options } => (
149                    Arc::new(OpenAiChatModel::new(ModelConfig::new(
150                        provider.with_auth(self.openai_auth()?),
151                        declaration,
152                        settings.to_openai_chat(),
153                    ))?),
154                    namespace("openai", json!({ "chat": options }))?,
155                ),
156                AdapterConfig::OpenaiResponses { settings, options } => (
157                    Arc::new(OpenAiResponsesModel::new(ModelConfig::new(
158                        provider.with_auth(self.openai_auth()?),
159                        declaration,
160                        settings.to_openai_responses(),
161                    ))?),
162                    namespace("openai", json!({ "responses": options }))?,
163                ),
164                AdapterConfig::OpenaiCompatible { settings, options } => (
165                    Arc::new(OpenAiCompatibleChatModel::new(ModelConfig::new(
166                        provider.with_auth(self.compatible_auth()?),
167                        declaration,
168                        settings.to_oven(),
169                    ))?),
170                    namespace(
171                        "openai_compatible",

173                            extra_body: options.extra_body.clone(),

175                    )?,
176                ),
177                AdapterConfig::Google { settings, options } => (
178                    Arc::new(GoogleModel::new(ModelConfig::new(
179                        provider.with_auth(self.google_auth()?),
180                        declaration,

183                    namespace("google", options.to_oven())?,
184                ),

189                        Arc::new(GoogleVertexModel::new(ModelConfig::new(
190                            provider,

196                }
197                AdapterConfig::Bedrock { settings, options } => (
198                    Arc::new(BedrockModel::new(ModelConfig::new(
199                        provider.with_auth(self.bedrock_auth()?),
200                        declaration,

207                        provider.with_auth(self.azure_auth()?),
208                        declaration,

213                AdapterConfig::AzureResponses { settings, options } => (

366/// Resolved authentication forms supported by static TOML configuration.
367#[derive(Clone, Deserialize, Serialize)]
368#[serde(tag = "type", rename_all = "snake_case", deny_unknown_fields)]
369pub enum AuthConfig {
370    /// No adapter-injected authentication.
371    None,
372    /// API-key authentication.
373    ApiKey { value: String },
374    /// Bearer authentication.
375    Bearer { token: String },
376    /// Caller-selected reviewed API-key header authentication.
377    HeaderApiKey { name: String, value: String },
378    /// Official OpenAI bearer authentication and account headers.
379    Openai {
380        api_key: String,
381        organization: Option<String>,
382        project: Option<String>,
383    },
384    /// Static OAuth access token.
385    AccessToken { token: String },
386    /// Static AWS credentials.
387    AwsStatic {
388        access_key_id: String,
389        secret_access_key: String,
390        session_token: Option<String>,
391    },
392}
393
394impl Drop for AuthConfig {
395    fn drop(&mut self) {
396        match self {
397            Self::None => {}
398            Self::ApiKey { value } => value.zeroize(),
399            Self::Bearer { token } | Self::AccessToken { token } => token.zeroize(),
400            Self::HeaderApiKey { value, .. } => value.zeroize(),
401            Self::Openai { api_key, .. } => api_key.zeroize(),
402            Self::AwsStatic {
403                access_key_id,
404                secret_access_key,
405                session_token,
406            } => {
407                access_key_id.zeroize();
408                secret_access_key.zeroize();
409                if let Some(session_token) = session_token {
410                    session_token.zeroize();
411                }
412            }
413        }
414    }
415}

510/// Concrete model construction error with redacted formatting.

512pub enum ModelBuildError {

551fn header_config(values: &BTreeMap<String, String>) -> Result<HeaderConfig, ModelBuildError> {
552    let mut headers = HeaderMap::new();
553    for (name, value) in values {
554        let parsed_name = HeaderName::from_bytes(name.as_bytes())
555            .map_err(|_| ModelBuildError::HeaderName(name.clone()))?;
556        let parsed_value =
557            HeaderValue::from_str(value).map_err(|_| ModelBuildError::HeaderValue(name.clone()))?;
558        headers.insert(parsed_name, parsed_value);
559    }
560    headers.insert(
561        http::header::USER_AGENT,
562        HeaderValue::from_static(CLIENT_USER_AGENT),
563    );
564    Ok(HeaderConfig {
565        static_headers: HeaderOverrides::new(headers),
566        dynamic_headers: None,
567    })
568}

570fn namespace(
571    name: &str,

574    let value = serde_json::to_value(value).map_err(ModelBuildError::ProviderOptions)?;
575    Ok([(name.to_owned(), value)].into_iter().collect())
