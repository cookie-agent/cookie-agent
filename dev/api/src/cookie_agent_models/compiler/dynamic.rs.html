156    pub fn compile_managed_with_headers(
157        &self,
158        catalog_revision: &str,
159        record: &CatalogProviderRecord,
160        authored: Option<&ModelsDevProvider>,
161        global_headers: &BTreeMap<HeaderName, SafeStaticHeaderValue>,
162    ) -> Result<CompiledDynamicProvider, DynamicCompileError> {
163        let family = self
164            .registry
165            .classify(record)
166            .ok_or(DynamicCompileError::UnsupportedProvider)?;
167        validate_managed_cache(record, authored, family)?;
168        if let Some(authored) = authored
169            && let Some(base_url) = authored.base_url.as_ref()
170        {
171            validate_managed_base_url(
172                crate::recipes::EndpointPolicy::DefaultWithAuthoredHttpsOverride {
173                    default: "https://invalid.example",
174                },
175                Some(base_url),
176            )
177            .map_err(|_| DynamicCompileError::Endpoint)?;
178        }
179        if authored.is_some_and(|value| {
180            value.base_url.is_some() && value.api_key.is_none() && value.auth_override.is_none()
181        }) {
182            return Err(DynamicCompileError::BaseUrlWithoutAuth);
183        }
184        if let Some(authored) = authored {
185            for id in authored.model_overrides.keys() {
186                if !record.models.contains_key(id) {
187                    return Err(DynamicCompileError::UnknownModelOverride);
188                }
189            }
190        }
191        let mut models = BTreeMap::new();
192        let mut unsupported_models = Vec::new();
193        for (table_id, entry) in &record.models {
194            let Some(model) = entry.record.as_ref() else {
195                continue;
196            };
197            if model.status == crate::catalog::CatalogModelStatus::Deprecated
198                || !model.modalities.output.iter().any(|value| value == "text")
199            {
200                continue;
201            }
202            let override_ = authored.and_then(|value| value.model_overrides.get(table_id));
203            if override_.and_then(|value| value.enabled) == Some(false) {
204                continue;
205            }
206            let resolved = match resolve_model(
207                record,
208                model,
209                authored.and_then(|value| value.shape),
210                override_.and_then(|value| value.shape),
211            ) {
212                Ok(resolved) => resolved,
213                Err(error) => {
214                    unsupported_models.push(UnsupportedModel {
215                        id: table_id.clone(),
216                        reason: error.to_string(),
217                    });
218                    continue;
219                }
220            };
221            match self.compile_managed_model(
222                catalog_revision,
223                &record.id,
224                model,
225                family,
226                &resolved,
227                authored,
228                override_,
229                global_headers,
230            ) {
231                Ok(compiled) => {
232                    models.insert(table_id.clone(), compiled);
233                }
234                Err(ModelLocalError::Unsupported(reason)) => {
235                    unsupported_models.push(UnsupportedModel {
236                        id: table_id.clone(),
237                        reason,
238                    });
239                }
240                Err(ModelLocalError::Provider(error)) => return Err(error),
241            }
242        }
243        unsupported_models.sort_by(|left, right| left.id.cmp(&right.id));
244        let provider_fingerprint = fingerprint(
245            "cookie-agent/dynamic-provider/v1",
246            &(
247                self.registry.revision(),
248                COMPILER_VERSION,
249                catalog_revision,
250                &record.id,
251                models
252                    .iter()
253                    .map(|(id, model)| (id, &model.behavior_fingerprint))
254                    .collect::<Vec<_>>(),
255                &unsupported_models,
256            ),
257        );
258        Ok(CompiledDynamicProvider {
259            id: record.id.clone(),
260            models,
261            unsupported_models,
262            fingerprint: provider_fingerprint,
263        })
264    }

266    #[allow(clippy::too_many_arguments)]
267    fn compile_managed_model(
268        &self,
269        catalog_revision: &str,
270        provider_id: &ProviderId,
271        model: &CatalogModelRecord,
272        provider_family: &'static FamilyRecipe,
273        resolved: &crate::recipes::ResolvedFamilyModel,
274        authored: Option<&ModelsDevProvider>,
275        override_: Option<&ManagedModelOverride>,
276        global_headers: &BTreeMap<HeaderName, SafeStaticHeaderValue>,
277    ) -> Result<CompiledDynamicModel, ModelLocalError> {
278        let adapter = resolved.adapter;
279        let adapter_id = match adapter {
280            OvenAdapterFamily::OpenaiCompatible => {
281                format!("oven.openai-compatible.chat.{}", provider_id.as_str())
282            }
283            OvenAdapterFamily::AnthropicCompatible => {
284                format!(
285                    "oven.anthropic-compatible.messages.{}",
286                    provider_id.as_str()
287                )
288            }
289            _ => adapter.protocol_recipe().to_owned(),
290        };
291        let mut capabilities = capabilities_from_catalog(model, adapter).map_err(|_| {
292            ModelLocalError::Unsupported("unsupported_model_capabilities".to_owned())
293        })?;
294        apply_compaction_config(
295            &mut capabilities,
296            adapter,
297            provider_id,
298            override_.map_or(crate::NativeCompactionConfig::Unsupported, |value| {
299                value.compaction
300            }),
301        )
302        .map_err(ModelLocalError::Provider)?;
303        if !validate_capability_shape(&capabilities)
304            || validate_capability_ceiling(adapter, &capabilities).is_err()
305        {
306            return Err(ModelLocalError::Unsupported(
307                "unsupported_model_capabilities".to_owned(),
308            ));
309        }
310        let template = authored
311            .and_then(|value| value.base_url.as_ref())
312            .map(crate::authoring::EndpointUrl::as_str)
313            .map(str::to_owned)
314            .or_else(|| resolved.endpoint_template.clone());
315        let (setup, endpoint) = resolved_managed_setup_and_endpoint(
316            provider_family,
317            resolved.recipe.family,
318            template.as_deref(),
319            authored,
320        )?;
321        let required_auth_method = match adapter {
322            OvenAdapterFamily::AwsBedrockConverse => Some("aws-sigv4-credentials-v1"),
323            OvenAdapterFamily::OpenaiResponses if resolved.recipe.family == FamilyKind::Bedrock => {
324                Some("bearer-api-key-v1")
325            }
326            _ => None,
327        };
328        let auth = managed_auth(
329            provider_family,
330            resolved.recipe,
331            required_auth_method,
332            authored,
333        )?;
334        let mut defaults = managed_defaults(model);
335        if let Some(override_) = override_ {
336            apply_partial_defaults(&mut defaults, &override_.defaults);
337        }
338        if !validate_defaults(&defaults, &capabilities) {
339            return Err(ModelLocalError::Unsupported(
340                "unsupported_model_capabilities".to_owned(),
341            ));
342        }
343        let (mut variants, variant_order, default_variant) =
344            managed_variants(&model.reasoning_options, override_, adapter).map_err(|_| {
345                ModelLocalError::Unsupported("unsupported_protocol_feature".to_owned())
346            })?;
347        if variants
348            .values()
349            .any(|variant| !validate_defaults(&variant.defaults, &capabilities))
350        {
351            return Err(ModelLocalError::Provider(DynamicCompileError::Variant));
352        }
353        let headers = merge_headers([
354            (global_headers, "global".to_owned()),
355            (
356                authored.map_or(&EMPTY_HEADERS, |value| &value.headers),
357                format!("provider `{provider_id}`"),
358            ),
359            (
360                override_.map_or(&EMPTY_HEADERS, |value| &value.headers),
361                format!("provider `{provider_id}` model `{}`", model.id),
362            ),
363        ])
364        .map_err(ModelLocalError::Provider)?;
365        for variant in variants.values_mut() {
366            variant.headers = merge_headers([
367                (
368                    &headers,
369                    format!("provider `{provider_id}` model `{}`", model.id),
370                ),
371                (
372                    &variant.headers,
373                    format!(
374                        "provider `{provider_id}` model `{}` variant `{}`",
375                        model.id, variant.id
376                    ),
377                ),
378            ])
379            .map_err(ModelLocalError::Provider)?;
380        }
381        let display_name = override_
382            .and_then(|value| value.display_name.clone())
383            .unwrap_or_else(|| model.name.clone());
384        let status = if setup.is_none() {
385            CompiledModelStatus::SetupUnavailable
386        } else if auth.source == AuthSourceCategory::Unavailable {
387            CompiledModelStatus::CredentialsUnavailable
388        } else {
389            CompiledModelStatus::Available
390        };
391        let options = ProviderOptions::default();
392        let behavior_fingerprint = fingerprint(
393            "cookie-agent/dynamic-model-behavior/v1",
394            &(
395                (
396                    self.registry.revision(),
397                    COMPILER_VERSION,
398                    catalog_revision,
399                    provider_id,
400                    &model.id,
401                    resolved.recipe.family.id(),
402                    &adapter_id,
403                    adapter,
404                ),
405                &endpoint,
406                &setup,
407                &auth,
408                &capabilities,
409                &defaults,
410                &options,
411                &headers,
412                &variants,
413                &variant_order,
414                &default_variant,
415                "managed_catalog",
416            ),
417        );
418        Ok(CompiledDynamicModel {
419            custom: false,
420            id: model.id.clone(),
421            display_name,
422            family_id: resolved.recipe.family.id().to_owned(),
423            effective_npm: resolved.npm.clone(),
424            adapter_id,
425            resolved_shape: match resolved.shape {
426                crate::recipes::ResolvedShape::Chat => "chat",
427                crate::recipes::ResolvedShape::Responses => "responses",
428            }
429            .to_owned(),
430            reasoning_field: match model.interleaved {
431                Some(crate::catalog::CatalogInterleaved::Reasoning) => "reasoning",
432                Some(crate::catalog::CatalogInterleaved::ReasoningContent)
433                | Some(crate::catalog::CatalogInterleaved::Default)
434                | None => "reasoning_content",
435            }
436            .to_owned(),
437            adapter,
438            endpoint,
439            setup,
440            auth,
441            capabilities,
442            defaults,
443            options,
444            headers,
445            cost: model.cost.clone(),
446            variants,
447            variant_order,
448            default_variant,
449            status,
450            behavior_fingerprint,
451        })
452    }

462    pub fn compile_custom_with_headers(
463        &self,
464        provider_id: &ProviderId,
465        provider: &CustomProvider,
466        global_headers: &BTreeMap<HeaderName, SafeStaticHeaderValue>,
467    ) -> Result<CompiledDynamicProvider, DynamicCompileError> {
468        let adapter = OvenAdapterFamily::parse(provider.adaptor.as_str())
469            .ok_or(DynamicCompileError::UnsupportedAdapter)?;
470        crate::authoring::validate_provider_cache(provider.cache.as_ref(), adapter.id())
471            .map_err(DynamicCompileError::Cache)?;
472        let wire = wire_adapter_for_custom(adapter);
473        validate_custom_endpoint(adapter, &provider.endpoint)
474            .map_err(|_| DynamicCompileError::Endpoint)?;
475        let setup_recipe = custom_setup_recipe(adapter);
476        let setup = validate_setup(setup_recipe, &provider.setup)
477            .map_err(|_| DynamicCompileError::Setup)?;
478        let auth_method = validate_auth_definition(&provider.auth, adapter.allowed_auth_methods())
479            .map_err(|_| DynamicCompileError::Auth)?;
480        let auth = custom_auth_shape(&provider.auth, auth_method);
481        let mut models = BTreeMap::new();
482        for (id, model) in &provider.models {
483            let capabilities = model.capabilities.clone();
484            if !validate_capability_shape(&capabilities)
485                || validate_capability_ceiling(adapter, &capabilities).is_err()
486                || !validate_defaults(&model.defaults, &capabilities)
487                || !validate_custom_options(&model.options, adapter)
488                || !validate_no_auth_profile(&auth, adapter, &model.capabilities, &model.options)
489            {
490                return Err(DynamicCompileError::CustomModel);
491            }
492            let (mut variants, variant_order, default_variant) =
493                custom_variants(&model.variants, model.default_variant.as_ref())
494                    .map_err(|_| DynamicCompileError::Variant)?;
495            if variants.values().any(|variant| {
496                !validate_defaults(&variant.defaults, &capabilities)
497                    || variant.reasoning.is_some() && !capabilities.reasoning
498                    || !validate_custom_options(&variant.options, adapter)
499                    || !reasoning_supported(variant.reasoning.as_ref(), adapter)
500            }) {
501                return Err(DynamicCompileError::Variant);
502            }
503            let headers = merge_headers([
504                (global_headers, "global".to_owned()),
505                (&provider.headers, format!("provider `{provider_id}`")),
506                (
507                    &model.headers,
508                    format!("provider `{provider_id}` model `{id}`"),
509                ),
510            ])?;
511            for variant in variants.values_mut() {
512                variant.headers = merge_headers([
513                    (&headers, format!("provider `{provider_id}` model `{id}`")),
514                    (
515                        &variant.headers,
516                        format!(
517                            "provider `{provider_id}` model `{id}` variant `{}`",
518                            variant.id
519                        ),
520                    ),
521                ])?;
522            }
523            if !model.enabled {
524                continue;
525            }
526            let endpoint = provider.endpoint.as_str().trim_end_matches('/').to_owned();
527            let options = model.options.clone();
528            let safe_headers = headers
529                .iter()
530                .map(|(name, value)| (name.as_str(), value.as_str()))
531                .collect::<Vec<_>>();
532            let behavior_fingerprint = fingerprint(
533                "cookie-agent/custom-model-behavior/v1",
534                &(
535                    self.registry.revision(),
536                    COMPILER_VERSION,
537                    provider_id,
538                    id,
539                    wire.adapter_id,
540                    adapter,
541                    &endpoint,
542                    &setup,
543                    &auth,
544                    &safe_headers,
545                    &capabilities,
546                    &model.defaults,
547                    &options,
548                    (&variants, &variant_order, &default_variant),
549                    "custom_authored",
550                ),
551            );
552            models.insert(
553                id.clone(),
554                CompiledDynamicModel {
555                    custom: true,
556                    id: id.clone(),
557                    display_name: model.display_name.clone(),
558                    family_id: "custom".into(),
559                    effective_npm: "custom".into(),
560                    adapter_id: wire.adapter_id.into(),
561                    resolved_shape: if adapter == OvenAdapterFamily::OpenaiResponses
562                        || adapter == OvenAdapterFamily::AzureOpenaiResponses
563                    {
564                        "responses"
565                    } else {
566                        "chat"
567                    }
568                    .into(),
569                    reasoning_field: "reasoning_content".into(),
570                    adapter,
571                    endpoint: Some(endpoint),
572                    setup: Some(setup.clone()),
573                    auth: auth.clone(),
574                    capabilities,
575                    defaults: model.defaults.clone(),
576                    options,
577                    headers,
578                    cost: None,
579                    variants,
580                    variant_order,
581                    default_variant,
582                    status: CompiledModelStatus::Available,
583                    behavior_fingerprint,
584                },
585            );
586        }
587        let safe_headers = provider
588            .headers
589            .iter()
590            .map(|(name, value)| (name.as_str(), value.as_str()))
591            .collect::<Vec<_>>();
592        let provider_fingerprint = fingerprint(
593            "cookie-agent/custom-provider/v1",
594            &(
595                self.registry.revision(),
596                COMPILER_VERSION,
597                provider_id,
598                provider.endpoint.as_str(),
599                adapter,
600                &setup,
601                &auth,
602                &safe_headers,
603                models
604                    .iter()
605                    .map(|(id, model)| (id, &model.behavior_fingerprint))
606                    .collect::<Vec<_>>(),
607            ),
608        );
609        Ok(CompiledDynamicProvider {
610            id: provider_id.clone(),
611            models,
612            unsupported_models: Vec::new(),
613            fingerprint: provider_fingerprint,
614        })
615    }

620fn merge_headers<'a>(
621    layers: impl IntoIterator<Item = (&'a BTreeMap<HeaderName, SafeStaticHeaderValue>, String)>,
622) -> Result<BTreeMap<HeaderName, SafeStaticHeaderValue>, DynamicCompileError> {
623    let mut merged = BTreeMap::new();
624    for (layer, scope) in layers {
625        validate_header_ownership(layer, scope).map_err(DynamicCompileError::StaticHeaders)?;
626        for (name, value) in layer {
627            if value.as_str().is_empty() {
628                merged.remove(name);
629            } else {
630                merged.insert(name.clone(), value.clone());
631            }
632        }
633    }
634    validate_header_limits(&merged).map_err(DynamicCompileError::StaticHeaders)?;
635    Ok(merged)
636}

638pub(crate) fn validate_managed_cache(
639    record: &CatalogProviderRecord,
640    authored: Option<&ModelsDevProvider>,
641    family: &'static FamilyRecipe,
642) -> Result<(), DynamicCompileError> {
643    let Some(authored) = authored else {
644        return Ok(());
645    };
646    let Some(cache) = authored.cache.as_ref() else {
647        return Ok(());
648    };
649    cache.validate_supported_shape().map_err(|error| {
650        DynamicCompileError::Cache(format!("provider `{}`: {error}", record.id))
651    })?;
652
653    let mut adapters = BTreeSet::from([managed_provider_adapter(
654        family.family,
655        authored.shape,
656        record.shape.as_deref(),
657    )]);
658    for (model_id, entry) in &record.models {
659        let Some(model) = entry.record.as_ref() else {
660            continue;
661        };
662        let override_ = authored.model_overrides.get(model_id);
663        if let Ok(resolved) = resolve_model(
664            record,
665            model,
666            authored.shape,
667            override_.and_then(|value| value.shape),
668        ) {
669            adapters.insert(resolved.adapter);
670        }
671    }
672    for adapter in adapters {
673        crate::authoring::validate_provider_cache(Some(cache), adapter.id()).map_err(|error| {
674            DynamicCompileError::Cache(format!("provider `{}`: {error}", record.id))
675        })?;
676    }
677    Ok(())
678}

680pub(crate) fn managed_provider_adapter(
681    family: FamilyKind,
682    authored_shape: Option<crate::ManagedModelShape>,
683    catalog_shape: Option<&str>,
684) -> OvenAdapterFamily {
685    let responses = matches!(authored_shape, Some(crate::ManagedModelShape::Responses))
686        || authored_shape.is_none() && catalog_shape == Some("responses")
687        || authored_shape.is_none() && catalog_shape.is_none() && family == FamilyKind::OpenAi;
688    match family {
689        FamilyKind::OpenAiCompatibleChat if responses => OvenAdapterFamily::OpenaiResponses,
690        FamilyKind::OpenAiCompatibleChat => OvenAdapterFamily::OpenaiCompatible,
691        FamilyKind::Anthropic => OvenAdapterFamily::AnthropicCompatible,
692        FamilyKind::OpenAi if responses => OvenAdapterFamily::OpenaiResponses,
693        FamilyKind::OpenAi => OvenAdapterFamily::OpenaiChat,
694        FamilyKind::Google => OvenAdapterFamily::GoogleGemini,
695        FamilyKind::Vertex | FamilyKind::VertexAnthropic => OvenAdapterFamily::GoogleVertexGemini,
696        FamilyKind::Bedrock if responses => OvenAdapterFamily::OpenaiResponses,
697        FamilyKind::Bedrock => OvenAdapterFamily::AwsBedrockConverse,
698        FamilyKind::Azure if responses => OvenAdapterFamily::AzureOpenaiResponses,
699        FamilyKind::Azure => OvenAdapterFamily::AzureOpenaiChat,
700        FamilyKind::Cohere => OvenAdapterFamily::CohereV2Chat,
701    }

705    capabilities: &mut ModelCapabilities,
706    adapter: OvenAdapterFamily,
707    provider_id: &ProviderId,

730fn resolved_managed_setup_and_endpoint(
731    _provider_family: &'static FamilyRecipe,
732    family: FamilyKind,
733    template: Option<&str>,
734    authored: Option<&ModelsDevProvider>,
735) -> Result<(Option<ValidatedSetup>, Option<String>), ModelLocalError> {
736    if template.is_none() && !matches!(family, FamilyKind::Vertex | FamilyKind::VertexAnthropic) {
737        return Ok((None, None));
738    }
739    let input = authored
740        .map(|value| &value.setup)
741        .cloned()
742        .unwrap_or_default();
743    let mut values = BTreeMap::new();
744    for (id, value) in &input {
745        let crate::authoring::SafeSetupValue::String(value) = value else {
746            return Err(ModelLocalError::Provider(DynamicCompileError::Setup));
747        };
748        values.insert(id.as_str().to_owned(), value.as_str().to_owned());
749    }
750    let mut required = template
751        .map(placeholders)
752        .unwrap_or_default()
753        .into_iter()
754        .map(|name| crate::recipes::setup_field_name(&name))
755        .collect::<Vec<_>>();
756    match family {
757        FamilyKind::Vertex | FamilyKind::VertexAnthropic => {
758            required.extend(["project".into(), "location".into()])
759        }
760        FamilyKind::Bedrock => required.push("region".into()),
761        FamilyKind::Azure => required.push("resource_name".into()),
762        _ => {}
763    }
764    required.sort();
765    required.dedup();
766    if required.iter().any(|field| !values.contains_key(field)) {
767        return Ok((None, None));
768    }
769    let endpoint = template
770        .and_then(|template| substitute_placeholders(template, &values))
771        .map(|value| value.trim_end_matches('/').to_owned())
772        .or_else(|| match family {
773            FamilyKind::Vertex | FamilyKind::VertexAnthropic => Some(format!(
774                "https://{}-aiplatform.googleapis.com/v1/projects/{}/locations/{}",
775                values.get("location")?,
776                values.get("project")?,
777                values.get("location")?
778            )),
779            _ => None,
780        });
781    let setup = ValidatedSetup {
782        recipe_id: "family-derived-setup-v1",
783        values,
784    };
785    Ok((Some(setup), endpoint))
786}

788fn managed_auth(
789    provider_family: &FamilyRecipe,
790    effective_recipe: &FamilyRecipe,
791    required_method: Option<&'static str>,
792    authored: Option<&ModelsDevProvider>,
793) -> Result<CompiledAuthShape, ModelLocalError> {
794    if let Some(authored) = authored {
795        if authored.api_key.is_some() {
796            let source_method = provider_family.default_auth_method;
797            let target_method =
798                compatible_model_auth(source_method, effective_recipe, required_method);
799            let Some(method) = target_method.and_then(auth_method) else {
800                return Ok(auth_shape(
801                    auth_method(effective_recipe.default_auth_method)
802                        .ok_or(ModelLocalError::Provider(DynamicCompileError::Auth))?,
803                    BTreeMap::new(),
804                    AuthSourceCategory::Unavailable,
805                ));
806            };
807            let required_api_key = method.credentials.len() == 1
808                && method.credentials[0].required
809                && method.credentials[0].name == "api_key";
810            if !required_api_key {
811                return Err(ModelLocalError::Provider(DynamicCompileError::Auth));
812            }
813            return Ok(auth_shape(
814                method,
815                BTreeMap::new(),
816                AuthSourceCategory::AuthoredApiKey,
817            ));
818        }
819        if let Some(auth) = &authored.auth_override {
820            let target_method =
821                compatible_model_auth(auth.method.as_str(), effective_recipe, required_method);
822            let Some(method) = target_method.and_then(auth_method) else {
823                return Ok(auth_shape(
824                    auth_method(effective_recipe.default_auth_method)
825                        .ok_or(ModelLocalError::Provider(DynamicCompileError::Auth))?,
826                    BTreeMap::new(),
827                    AuthSourceCategory::Unavailable,
828                ));
829            };
830            return Ok(auth_shape(
831                method,
832                BTreeMap::new(),
833                AuthSourceCategory::AuthoredOverride,
834            ));
835        }
836    }
837    let method = auth_method(required_method.unwrap_or(effective_recipe.default_auth_method))
838        .ok_or(ModelLocalError::Provider(DynamicCompileError::Auth))?;
839    Ok(auth_shape(
840        method,
841        BTreeMap::new(),
842        AuthSourceCategory::Unavailable,
843    ))
844}
845
846fn compatible_model_auth(
847    source_method: &str,
848    effective_recipe: &FamilyRecipe,
849    required_method: Option<&'static str>,
850) -> Option<&'static str> {
851    let mapped = crate::recipes::compatible_auth_method(source_method, effective_recipe)?;
852    required_method.map_or(Some(mapped), |required| {
853        (mapped == required).then_some(required)
854    })
855}

857fn custom_auth_shape(
858    auth: &AuthDefinition,
859    method: &'static crate::recipes::AuthMethodRecipe,
860) -> CompiledAuthShape {
861    let parameters = auth
862        .parameters
863        .iter()
864        .map(|(name, value)| (name.as_str().to_owned(), value.as_str().to_owned()))
865        .collect();
866    auth_shape(method, parameters, AuthSourceCategory::AuthoredCustom)
867}
868
869fn auth_shape(
870    method: &'static crate::recipes::AuthMethodRecipe,
871    safe_parameters: BTreeMap<String, String>,
872    source: AuthSourceCategory,
873) -> CompiledAuthShape {
874    let credential_fields = method
875        .credentials
876        .iter()
877        .map(|field| field.name.to_owned())
878        .collect();
879    let mut owned_headers = method
880        .owned_headers
881        .iter()
882        .map(|value| (*value).to_owned())
883        .collect::<BTreeSet<_>>();
884    if method.id == "api-key-header-v1"
885        && let Some(header) = safe_parameters.get("header_name")
886    {
887        owned_headers.insert(header.clone());
888    }
889    CompiledAuthShape {
890        method: method.id.to_owned(),
891        safe_parameters,
892        credential_fields,
893        owned_headers: owned_headers.into_iter().collect(),
894        source,
895    }
896}

912fn validate_custom_options(options: &ProviderOptions, adapter: OvenAdapterFamily) -> bool {
913    let has_openai =
914        options.organization.is_some() || options.project.is_some() || options.store.is_some();
915    let has_anthropic = !options.beta.is_empty();
916    let has_compatible = options.api_path.is_some();
917    let has_setup_leak = options.api_version.is_some()
918        || options.location.is_some()
919        || options.region.is_some()
920        || options.deployment.is_some();
921    !has_setup_leak
922        && match adapter {
923            OvenAdapterFamily::Anthropic | OvenAdapterFamily::AnthropicCompatible => {
924                !has_openai && !has_compatible
925            }
926            OvenAdapterFamily::OpenaiChat | OvenAdapterFamily::OpenaiResponses => {
927                !has_anthropic && !has_compatible
928            }
929            OvenAdapterFamily::OpenaiCompatible => !has_anthropic && !has_openai,
930            OvenAdapterFamily::GoogleGemini
931            | OvenAdapterFamily::GoogleVertexGemini
932            | OvenAdapterFamily::AwsBedrockConverse
933            | OvenAdapterFamily::AzureOpenaiChat
934            | OvenAdapterFamily::AzureOpenaiResponses
935            | OvenAdapterFamily::CohereV2Chat => !has_anthropic && !has_openai && !has_compatible,
936        }
937}

941    adapter: OvenAdapterFamily,
942) -> bool {

953                | OvenAdapterFamily::GoogleGemini
954                | OvenAdapterFamily::GoogleVertexGemini
955                | OvenAdapterFamily::CohereV2Chat
956        ),

972        && !capabilities.parallel_tool_calls

976        && capabilities.native_replay == crate::ReplayCapability::Unsupported
977        && options.store.is_none_or(|store| !store)
