15/// A held, descriptor-relative private directory.
16#[derive(Debug)]
17pub struct SecureDirectory {
18    pub(crate) directory: fs::File,
19    path: PathBuf,
20}
21
22impl SecureDirectory {
23    /// Opens `~/.cookie_agent/<relative>`, creating private components.
24    pub fn user_data(relative: impl AsRef<Path>) -> Result<Self, SecureStoreError> {
25        let root = paths::user_data_root().map_err(|_| SecureStoreError::HomeUnavailable)?;
26        #[cfg(unix)]
27        {
28            let home = root.parent().ok_or(SecureStoreError::UnsafePath)?;
29            let root_name = root.file_name().ok_or(SecureStoreError::UnsafePath)?;
30            let home_directory = open_absolute_directory(home, DirectoryPolicy::SafeAnchor)?;
31            let relative = Path::new(root_name).join(relative);
32            let mut directory = Self::open_private_in(&home_directory, &relative)?;
33            directory.path = home.join(relative);
34            Ok(directory)
35        }
36        #[cfg(not(unix))]
37        {
38            let _ = (root, relative);
39            Err(SecureStoreError::UnsupportedPlatform)
40        }
41    }
42
43    /// Opens a private path below a trusted existing anchor.
44    pub fn open_in(
45        anchor: impl AsRef<Path>,
46        relative: impl AsRef<Path>,
47    ) -> Result<Self, SecureStoreError> {
48        #[cfg(unix)]
49        {
50            let anchor_path = anchor.as_ref().to_path_buf();
51            let anchor = open_absolute_directory(&anchor_path, DirectoryPolicy::SafeAnchor)?;
52            let mut directory = Self::open_private_in(&anchor, relative.as_ref())?;
53            directory.path = anchor_path.join(relative.as_ref());
54            Ok(directory)
55        }
56        #[cfg(not(unix))]
57        {
58            let _ = (anchor, relative);
59            Err(SecureStoreError::UnsupportedPlatform)
60        }
61    }

63    /// Opens a private path below an existing, potentially shared project anchor.
64    ///
65    /// The anchor itself need not be owned by the current user or non-writable by
66    /// collaborators. Such collaborators may therefore deny service by removing
67    /// or replacing project entries, but cannot inject accepted storage objects:
68    /// every created/opened descendant remains current-user-owned mode 0700 and
69    /// all file operations retain the private, single-link, no-follow checks.
70    pub(crate) fn open_in_untrusted_project_anchor(
71        anchor: impl AsRef<Path>,
72        relative: impl AsRef<Path>,
73    ) -> Result<Self, SecureStoreError> {
74        #[cfg(unix)]
75        {
76            let anchor_path = anchor.as_ref().to_path_buf();
77            let anchor =
78                open_absolute_directory(&anchor_path, DirectoryPolicy::UntrustedProjectAnchor)?;
79            let mut directory = Self::open_private_in(&anchor, relative.as_ref())?;
80            directory.path = anchor_path.join(relative.as_ref());
81            Ok(directory)
82        }
83        #[cfg(not(unix))]
84        {
85            let _ = (anchor, relative);
86            Err(SecureStoreError::UnsupportedPlatform)
87        }
88    }

90    /// Opens or creates an absolute private directory.
91    pub fn open(path: impl AsRef<Path>) -> Result<Self, SecureStoreError> {
92        let path = path.as_ref();
93        let parent = path.parent().ok_or(SecureStoreError::UnsafePath)?;
94        let name = path.file_name().ok_or(SecureStoreError::UnsafePath)?;
95        Self::open_in(parent, Path::new(name))
96    }

98    /// Returns the diagnostic path. Security decisions never use this path.
99    #[must_use]
100    pub fn path(&self) -> &Path {
101        &self.path
102    }

104    /// Reads an optional private file from its held descriptor with a hard cap.
105    pub fn read(&self, name: &str, limit: u64) -> Result<Option<Vec<u8>>, SecureStoreError> {
106        validate_name(name)?;
107        read_file(&self.directory, name, limit)
108    }

110    /// Acquires and verifies a cross-process exclusive lock file.
111    pub fn lock(&self, name: &str) -> Result<SecureDirectoryLock<'_>, SecureStoreError> {
112        validate_name(name)?;
113        #[cfg(unix)]
114        {
115            let lock = open_or_create_file(&self.directory, name)?;
116            rustix::fs::flock(&lock, rustix::fs::FlockOperation::LockExclusive)
117                .map_err(io_error)?;
118            verify_current_entry(&self.directory, name, &lock)?;
119            rustix::fs::fsync(&self.directory).map_err(io_error)?;
120            Ok(SecureDirectoryLock {
121                directory: self,
122                lock_name: name.to_owned(),
123                _lock: lock,
124            })
125        }
126        #[cfg(not(unix))]
127        {
128            let _ = name;
129            Err(SecureStoreError::UnsupportedPlatform)
130        }
131    }
132
133    #[cfg(unix)]
134    fn open_private_in(parent: &fs::File, relative: &Path) -> Result<Self, SecureStoreError> {
135        let components = private_components(relative)?;
136        let mut current = parent.try_clone().map_err(SecureStoreError::Io)?;
137        for component in &components {
138            current =
139                open_or_create_directory(&current, Path::new(component), DirectoryPolicy::Private)?;
140        }
141        Ok(Self {
142            directory: current,
143            path: relative.to_path_buf(),
144        })
145    }

148/// A held lock authorizing lock/reread/replace transactions in one directory.
149#[derive(Debug)]
150pub struct SecureDirectoryLock<'a> {
151    directory: &'a SecureDirectory,
152    lock_name: String,
153    _lock: fs::File,
154}
155
156impl SecureDirectoryLock<'_> {
157    /// Reads bounded transaction-journal bytes stored in the held lock file.
158    pub fn read_journal(&self, limit: u64) -> Result<Vec<u8>, SecureStoreError> {
159        #[cfg(unix)]
160        {
161            use std::io::{Read as _, Seek as _, SeekFrom};
162
163            self.verify_lock()?;
164            let metadata = self._lock.metadata().map_err(SecureStoreError::Io)?;
165            if metadata.len() > limit {
166                return Err(SecureStoreError::TooLarge);
167            }
168            let mut file = self._lock.try_clone().map_err(SecureStoreError::Io)?;
169            file.seek(SeekFrom::Start(0))
170                .map_err(SecureStoreError::Io)?;
171            let mut bytes = Vec::with_capacity(
172                usize::try_from(metadata.len()).map_err(|_| SecureStoreError::TooLarge)?,
173            );
174            file.take(limit.saturating_add(1))
175                .read_to_end(&mut bytes)
176                .map_err(SecureStoreError::Io)?;
177            if bytes.len() as u64 > limit {
178                return Err(SecureStoreError::TooLarge);
179            }
180            self.verify_lock()?;
181            Ok(bytes)
182        }
183        #[cfg(not(unix))]
184        {
185            let _ = limit;
186            Err(SecureStoreError::UnsupportedPlatform)
187        }
188    }
189
190    /// Durably appends one bounded record to the held lock-file journal.
191    pub fn append_journal(&self, bytes: &[u8], limit: u64) -> Result<(), SecureStoreError> {
192        #[cfg(unix)]
193        {
194            use std::io::{Seek as _, SeekFrom, Write as _};
195
196            self.verify_lock()?;
197            let current = self._lock.metadata().map_err(SecureStoreError::Io)?.len();
198            if current.saturating_add(bytes.len() as u64) > limit {
199                return Err(SecureStoreError::TooLarge);
200            }
201            let mut file = self._lock.try_clone().map_err(SecureStoreError::Io)?;
202            file.seek(SeekFrom::End(0)).map_err(SecureStoreError::Io)?;
203            file.write_all(bytes).map_err(SecureStoreError::Io)?;
204            file.sync_all().map_err(SecureStoreError::Io)?;
205            self.verify_lock()?;
206            Ok(())
207        }
208        #[cfg(not(unix))]
209        {
210            let _ = (bytes, limit);
211            Err(SecureStoreError::UnsupportedPlatform)
212        }
213    }
214
215    /// Durably clears the held lock-file journal without replacing the lock inode.
216    pub fn clear_journal(&self) -> Result<(), SecureStoreError> {
217        #[cfg(unix)]
218        {
219            self.verify_lock()?;
220            self._lock.set_len(0).map_err(SecureStoreError::Io)?;
221            self._lock.sync_all().map_err(SecureStoreError::Io)?;
222            self.verify_lock()?;
223            Ok(())
224        }
225        #[cfg(not(unix))]
226        {
227            Err(SecureStoreError::UnsupportedPlatform)
228        }
229    }

231    /// Rereads an optional file while the cross-process lock is held.
232    pub fn read(&self, name: &str, limit: u64) -> Result<Option<Vec<u8>>, SecureStoreError> {
233        self.verify_lock()?;
234        let result = self.directory.read(name, limit)?;
235        self.verify_lock()?;
236        Ok(result)
237    }

239    /// Durably replaces one private file using an exclusive sibling temporary.
240    pub fn atomic_replace(&self, name: &str, bytes: &[u8]) -> Result<(), SecureStoreError> {
241        validate_name(name)?;
242        if name == self.lock_name {
243            return Err(SecureStoreError::UnsafePath);
244        }
245        #[cfg(unix)]
246        {
247            use std::io::Write as _;
248
249            self.verify_lock()?;
250
251            if let Some(destination) =
252                open_existing_file(&self.directory.directory, name, rustix::fs::OFlags::RDONLY)?
253            {
254                validate_private_file(&destination.metadata().map_err(SecureStoreError::Io)?)?;
255                verify_current_entry(&self.directory.directory, name, &destination)?;
256            }
257
258            let temporary = format!(".{name}.tmp-{}", Uuid::now_v7());
259            let mut file = create_file(&self.directory.directory, &temporary)?;
260            let result = (|| {
261                file.write_all(bytes).map_err(SecureStoreError::Io)?;
262                file.sync_all().map_err(SecureStoreError::Io)?;
263                verify_current_entry(&self.directory.directory, &temporary, &file)?;
264                self.verify_lock()?;
265                rustix::fs::renameat(
266                    &self.directory.directory,
267                    temporary.as_str(),
268                    &self.directory.directory,
269                    name,
270                )
271                .map_err(io_error)?;
272                let installed = open_existing_file(
273                    &self.directory.directory,
274                    name,
275                    rustix::fs::OFlags::RDONLY,
276                )?
277                .ok_or(SecureStoreError::UnsafePath)?;
278                verify_current_entry(&self.directory.directory, name, &installed)?;
279                rustix::fs::fsync(&self.directory.directory).map_err(io_error)?;
280                self.verify_lock()?;
281                Ok(())
282            })();
283            if result.is_err() {
284                let _ = rustix::fs::unlinkat(
285                    &self.directory.directory,
286                    temporary.as_str(),
287                    rustix::fs::AtFlags::empty(),
288                );
289            }
290            result
291        }
292        #[cfg(not(unix))]
293        {
294            let _ = (name, bytes);
295            Err(SecureStoreError::UnsupportedPlatform)
296        }
297    }

299    /// Removes an optional private file and fsyncs the directory.
300    pub fn remove(&self, name: &str) -> Result<(), SecureStoreError> {
301        validate_name(name)?;
302        if name == self.lock_name {
303            return Err(SecureStoreError::UnsafePath);
304        }
305        #[cfg(unix)]
306        {
307            self.verify_lock()?;
308            if let Some(file) =
309                open_existing_file(&self.directory.directory, name, rustix::fs::OFlags::RDONLY)?
310            {
311                verify_current_entry(&self.directory.directory, name, &file)?;
312                rustix::fs::unlinkat(
313                    &self.directory.directory,
314                    name,
315                    rustix::fs::AtFlags::empty(),
316                )
317                .map_err(io_error)?;
318                rustix::fs::fsync(&self.directory.directory).map_err(io_error)?;
319            }
320            self.verify_lock()?;
321            Ok(())
322        }
323        #[cfg(not(unix))]
324        {
325            let _ = name;
326            Err(SecureStoreError::UnsupportedPlatform)
327        }
328    }
329
330    #[cfg(unix)]
331    fn verify_lock(&self) -> Result<(), SecureStoreError> {
332        verify_current_entry(&self.directory.directory, &self.lock_name, &self._lock)
333    }
334}

336/// Fail-closed storage errors with no file contents or secret values.
337#[derive(Debug, Error)]
338pub enum SecureStoreError {
339    #[error("could not determine the home directory")]
340    HomeUnavailable,
341    #[error("secure storage is unsupported on this platform")]
342    UnsupportedPlatform,
343    #[error("secure storage path is unsafe")]
344    UnsafePath,
345    #[error("secure storage object exceeds its byte limit")]
346    TooLarge,
347    #[error("secure storage I/O failed")]
348    Io(#[source] io::Error),
349}
350
351#[derive(Clone, Copy)]
352enum DirectoryPolicy {
353    SafeAnchor,
354    UntrustedProjectAnchor,
355    Private,
356}

358#[cfg(unix)]
359fn open_absolute_directory(
360    path: &Path,
361    policy: DirectoryPolicy,
362) -> Result<fs::File, SecureStoreError> {
363    if !path.is_absolute() {
364        return Err(SecureStoreError::UnsafePath);
365    }
366    let flags = directory_flags();
367    let mut current = fs::File::from(
368        rustix::fs::open("/", flags, rustix::fs::Mode::empty()).map_err(path_error)?,
369    );
370    for component in path.components() {
371        match component {
372            Component::RootDir => {}
373            Component::Normal(name) => {
374                current = fs::File::from(
375                    rustix::fs::openat(&current, name, flags, rustix::fs::Mode::empty())
376                        .map_err(path_error)?,
377                );
378                validate_directory_type(&current.metadata().map_err(SecureStoreError::Io)?)?;
379            }
380            Component::CurDir | Component::ParentDir | Component::Prefix(_) => {
381                return Err(SecureStoreError::UnsafePath);
382            }
383        }
384    }
385    validate_directory(&current.metadata().map_err(SecureStoreError::Io)?, policy)?;
386    Ok(current)
387}

389#[cfg(unix)]
390fn private_components(path: &Path) -> Result<Vec<std::ffi::OsString>, SecureStoreError> {
391    let mut components = Vec::new();
392    for component in path.components() {
393        match component {
394            Component::Normal(name) => components.push(name.to_owned()),
395            Component::CurDir => {}
396            Component::RootDir | Component::ParentDir | Component::Prefix(_) => {
397                return Err(SecureStoreError::UnsafePath);
398            }
399        }
400    }
401    if components.is_empty() {
402        return Err(SecureStoreError::UnsafePath);
403    }
404    Ok(components)
405}

407#[cfg(unix)]
408fn open_or_create_directory(
409    parent: &fs::File,
410    relative: &Path,
411    policy: DirectoryPolicy,
412) -> Result<fs::File, SecureStoreError> {
413    let mut current = parent.try_clone().map_err(SecureStoreError::Io)?;
414    for component in private_components(relative)? {
415        let created = match rustix::fs::mkdirat(&current, &component, rustix::fs::Mode::RWXU) {
416            Ok(()) => true,
417            Err(error) if error == rustix::io::Errno::EXIST => false,
418            Err(error) => return Err(io_error(error)),
419        };
420        let next = fs::File::from(
421            rustix::fs::openat(
422                &current,
423                &component,
424                directory_flags(),
425                rustix::fs::Mode::empty(),
426            )
427            .map_err(path_error)?,
428        );
429        if created {
430            rustix::fs::fchmod(&next, rustix::fs::Mode::RWXU).map_err(io_error)?;
431        }
432        validate_directory(&next.metadata().map_err(SecureStoreError::Io)?, policy)?;
433        current = next;
434    }
435    Ok(current)
436}

438#[cfg(unix)]

443        | rustix::fs::OFlags::CLOEXEC

446#[cfg(unix)]
447fn read_file(
448    directory: &fs::File,
449    name: &str,
450    limit: u64,
451) -> Result<Option<Vec<u8>>, SecureStoreError> {
452    use std::io::Read as _;
453
454    let Some(file) = open_existing_file(directory, name, rustix::fs::OFlags::RDONLY)? else {
455        return Ok(None);
456    };
457    verify_current_entry(directory, name, &file)?;
458    let metadata = file.metadata().map_err(SecureStoreError::Io)?;
459    if metadata.len() > limit {
460        return Err(SecureStoreError::TooLarge);
461    }
462    let capacity = usize::try_from(metadata.len()).map_err(|_| SecureStoreError::TooLarge)?;
463    let mut bytes = Vec::with_capacity(capacity);
464    file.take(limit.saturating_add(1))
465        .read_to_end(&mut bytes)
466        .map_err(SecureStoreError::Io)?;
467    if bytes.len() as u64 > limit {
468        return Err(SecureStoreError::TooLarge);
469    }
470    Ok(Some(bytes))
471}

473#[cfg(unix)]
474fn open_or_create_file(directory: &fs::File, name: &str) -> Result<fs::File, SecureStoreError> {
475    if let Some(file) = open_existing_file(directory, name, rustix::fs::OFlags::RDWR)? {
476        return Ok(file);
477    }
478    match create_file(directory, name) {
479        Ok(file) => Ok(file),
480        Err(SecureStoreError::Io(error)) if error.kind() == io::ErrorKind::AlreadyExists => {
481            open_existing_file(directory, name, rustix::fs::OFlags::RDWR)?
482                .ok_or(SecureStoreError::UnsafePath)
483        }
484        Err(error) => Err(error),
485    }
486}

488#[cfg(unix)]
489fn open_existing_file(
490    directory: &fs::File,
491    name: &str,
492    access: rustix::fs::OFlags,
493) -> Result<Option<fs::File>, SecureStoreError> {
494    let flags = access
495        | rustix::fs::OFlags::NOFOLLOW
496        | rustix::fs::OFlags::NONBLOCK
497        | rustix::fs::OFlags::CLOEXEC;
498    match rustix::fs::openat(directory, name, flags, rustix::fs::Mode::empty()) {
499        Ok(fd) => {
500            let file = fs::File::from(fd);
501            validate_private_file(&file.metadata().map_err(SecureStoreError::Io)?)?;
502            Ok(Some(file))
503        }
504        Err(error) if error == rustix::io::Errno::NOENT => Ok(None),
505        Err(error) => Err(path_error(error)),
506    }
507}

509#[cfg(unix)]
510fn create_file(directory: &fs::File, name: &str) -> Result<fs::File, SecureStoreError> {
511    let flags = rustix::fs::OFlags::RDWR
512        | rustix::fs::OFlags::CREATE
513        | rustix::fs::OFlags::EXCL
514        | rustix::fs::OFlags::NOFOLLOW
515        | rustix::fs::OFlags::NONBLOCK
516        | rustix::fs::OFlags::CLOEXEC;
517    let mode = rustix::fs::Mode::RUSR | rustix::fs::Mode::WUSR;
518    let file =
519        fs::File::from(rustix::fs::openat(directory, name, flags, mode).map_err(path_error)?);
520    rustix::fs::fchmod(&file, mode).map_err(io_error)?;
521    validate_private_file(&file.metadata().map_err(SecureStoreError::Io)?)?;
522    Ok(file)
523}

525#[cfg(unix)]
526fn verify_current_entry(
527    directory: &fs::File,
528    name: &str,
529    held: &fs::File,
530) -> Result<(), SecureStoreError> {
531    use std::os::unix::fs::MetadataExt as _;
532
533    let current = open_existing_file(directory, name, rustix::fs::OFlags::RDONLY)?
534        .ok_or(SecureStoreError::UnsafePath)?;
535    let held_metadata = held.metadata().map_err(SecureStoreError::Io)?;
536    let current_metadata = current.metadata().map_err(SecureStoreError::Io)?;
537    if held_metadata.dev() == current_metadata.dev()
538        && held_metadata.ino() == current_metadata.ino()
539        && held_metadata.nlink() == 1
540        && current_metadata.nlink() == 1
541    {
542        Ok(())
543    } else {
544        Err(SecureStoreError::UnsafePath)
545    }
546}

548#[cfg(unix)]
549fn validate_directory_type(metadata: &fs::Metadata) -> Result<(), SecureStoreError> {
550    if metadata.file_type().is_dir() && !metadata.file_type().is_symlink() {
551        Ok(())
552    } else {
553        Err(SecureStoreError::UnsafePath)
554    }
555}

557#[cfg(unix)]
558fn validate_directory(
559    metadata: &fs::Metadata,
560    policy: DirectoryPolicy,
561) -> Result<(), SecureStoreError> {
562    use std::os::unix::fs::MetadataExt as _;
563
564    validate_directory_type(metadata)?;
565    let owner_ok = metadata.uid() == rustix::process::getuid().as_raw();
566    let mode = metadata.mode() & 0o777;
567    let mode_ok = match policy {
568        DirectoryPolicy::SafeAnchor => mode & 0o022 == 0,
569        DirectoryPolicy::UntrustedProjectAnchor => true,
570        DirectoryPolicy::Private => mode == PRIVATE_DIRECTORY_MODE,
571    };
572    let owner_ok = matches!(policy, DirectoryPolicy::UntrustedProjectAnchor) || owner_ok;
573    if owner_ok && mode_ok {
574        Ok(())
575    } else {
576        Err(SecureStoreError::UnsafePath)
577    }
578}

580#[cfg(unix)]
581fn validate_private_file(metadata: &fs::Metadata) -> Result<(), SecureStoreError> {
582    use std::os::unix::fs::MetadataExt as _;
583
584    if metadata.file_type().is_file()
585        && !metadata.file_type().is_symlink()
586        && metadata.uid() == rustix::process::getuid().as_raw()
587        && metadata.mode() & 0o777 == PRIVATE_FILE_MODE
588        && metadata.nlink() == 1
589    {
590        Ok(())
591    } else {
592        Err(SecureStoreError::UnsafePath)
593    }
594}
595
596fn validate_name(name: &str) -> Result<(), SecureStoreError> {
597    if name.is_empty()
598        || name == "."
599        || name == ".."
600        || name.as_bytes().contains(&b'/')
601        || name.as_bytes().contains(&0)
602    {
603        Err(SecureStoreError::UnsafePath)
604    } else {
605        Ok(())
606    }
607}

609#[cfg(unix)]
610fn path_error(error: rustix::io::Errno) -> SecureStoreError {
611    let error: io::Error = error.into();
612    if matches!(
613        error.raw_os_error(),
614        Some(code)
615            if matches!(
616                code,
617                libc::ELOOP
618                    | libc::ENOTDIR
619                    | libc::ENXIO
620                    | libc::ENODEV
621                    | libc::EOPNOTSUPP
622            )
623    ) {
624        SecureStoreError::UnsafePath
625    } else {
626        SecureStoreError::Io(error)
627    }
628}
