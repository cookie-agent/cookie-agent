190    #[must_use]
191    pub fn revision(self) -> RecipeRegistryRevision {
192        static REVISION: OnceLock<RecipeRegistryRevision> = OnceLock::new();
193        REVISION
194            .get_or_init(|| {
195                let bytes = serde_json::to_vec(&(
196                    RECIPES,
197                    crate::recipes::auth_methods().collect::<Vec<_>>(),
198                ))
199                .expect("static family registry serializes");
200                let mut digest = Sha256::new();
201                digest.update(b"cookie-agent/family-recipe-registry/v1\0");
202                digest.update(bytes);
203                RecipeRegistryRevision::new(format!("sha256:{:x}", digest.finalize()))
204                    .expect("registry digest is valid")
205            })
206            .clone()
207    }

210/// Maps one provider-level credential method onto the wire auth expected by an
211/// effective model family. Only equivalent credential semantics are mapped:
212/// single API keys may change header/bearer encoding, and an explicit access
213/// token may become bearer auth. AWS static credentials never cross families.
214#[must_use]
215pub fn compatible_auth_method(
216    source_method: &str,
217    effective: &FamilyRecipe,
218) -> Option<&'static str> {
219    if let Some(method) = effective
220        .allowed_auth_methods
221        .iter()
222        .copied()
223        .find(|method| *method == source_method)
224    {
225        return Some(method);
226    }
227    let source_is_api_key = matches!(
228        source_method,
229        "anthropic-api-key-v1"
230            | "azure-api-key-v1"
231            | "google-api-key-header-v1"
232            | "bearer-api-key-v1"
233    );
234    let source_is_access_token = source_method == "oauth-access-token-v1";
235    if !source_is_api_key && !source_is_access_token {
236        return None;
237    }
238    let preferred = match effective.family {
239        FamilyKind::Anthropic => {
240            // `@ai-sdk/anthropic` sends configured API keys as `x-api-key`, and
241            // Microsoft Foundry accepts `x-api-key` on its Anthropic endpoint.
242            if source_method == "bearer-api-key-v1" || source_is_access_token {
243                "bearer-api-key-v1"
244            } else {
245                "anthropic-api-key-v1"
246            }
247        }
248        FamilyKind::OpenAi | FamilyKind::OpenAiCompatibleChat | FamilyKind::Cohere => {
249            "bearer-api-key-v1"
250        }
251        FamilyKind::Google => "google-api-key-header-v1",
252        FamilyKind::Azure => {
253            if source_method == "bearer-api-key-v1" || source_is_access_token {
254                "bearer-api-key-v1"
255            } else {
256                "azure-api-key-v1"
257            }
258        }
259        FamilyKind::Vertex | FamilyKind::VertexAnthropic => return None,
260        FamilyKind::Bedrock => return None,
261    };
262    effective
263        .allowed_auth_methods
264        .contains(&preferred)
265        .then_some(preferred)
266}

268#[must_use]
269pub fn compatible_credential_field(
270    source_method: &str,
271    target_field: &str,
272) -> Option<&'static str> {
273    match target_field {
274        "api_key" if source_method == "oauth-access-token-v1" => Some("access_token"),
275        "api_key" => Some("api_key"),
276        "access_token" if source_method == "bearer-api-key-v1" => Some("api_key"),
277        "access_token" => Some("access_token"),
278        "access_key_id" if source_method == "aws-sigv4-credentials-v1" => Some("access_key_id"),
279        "secret_access_key" if source_method == "aws-sigv4-credentials-v1" => {
280            Some("secret_access_key")
281        }
282        "session_token" if source_method == "aws-sigv4-credentials-v1" => Some("session_token"),
283        _ => None,
284    }
285}

311pub fn resolve_model(
312    provider: &CatalogProviderRecord,
313    model: &CatalogModelRecord,
314    provider_shape: Option<ManagedModelShape>,
315    model_shape: Option<ManagedModelShape>,
316) -> Result<ResolvedFamilyModel, FamilyResolutionError> {
317    let override_provider = model.provider.as_ref();
318    let npm = override_provider
319        .and_then(|value| value.npm.as_deref())
320        .unwrap_or(&provider.npm);
321    let recipe = family_registry()
322        .by_npm(npm)
323        .ok_or(FamilyResolutionError::UnknownFamily)?;
324    let endpoint_template = override_provider
325        .and_then(|value| value.api.clone())
326        .or_else(|| provider.api.clone())
327        .or_else(|| recipe.default_endpoint.map(str::to_owned));
328    let catalog_shape = override_provider
329        .and_then(|value| value.shape.as_deref())
330        .or(model.shape.as_deref())
331        .or(provider.shape.as_deref());
332    let catalog_shape = catalog_shape
333        .map(|shape| match shape {
334            "responses" => Ok(ResolvedShape::Responses),
335            "completions" | "chat" => Ok(ResolvedShape::Chat),
336            _ => Err(FamilyResolutionError::UnsupportedShape),
337        })
338        .transpose()?;
339    let shape = model_shape
340        .or(provider_shape)
341        .map(|shape| match shape {
342            ManagedModelShape::Chat => ResolvedShape::Chat,
343            ManagedModelShape::Responses => ResolvedShape::Responses,
344        })
345        .or(catalog_shape)
346        .unwrap_or_else(|| {
347            if recipe.family == FamilyKind::OpenAi {
348                ResolvedShape::Responses
349            } else {
350                ResolvedShape::Chat
351            }
352        });
353    let adapter = adapter(recipe.family, shape)?;
354    Ok(ResolvedFamilyModel {
355        recipe,
356        npm: npm.to_owned(),
357        endpoint_template,
358        shape,
359        adapter,
360    })
361}

363fn adapter(
364    kind: FamilyKind,
365    shape: ResolvedShape,
366) -> Result<OvenAdapterFamily, FamilyResolutionError> {
367    Ok(match kind {
368        FamilyKind::OpenAiCompatibleChat => match shape {
369            ResolvedShape::Chat => OvenAdapterFamily::OpenaiCompatible,
370            ResolvedShape::Responses => OvenAdapterFamily::OpenaiResponses,
371        },
372        FamilyKind::Anthropic => OvenAdapterFamily::AnthropicCompatible,
373        FamilyKind::OpenAi => match shape {
374            ResolvedShape::Chat => OvenAdapterFamily::OpenaiChat,
375            ResolvedShape::Responses => OvenAdapterFamily::OpenaiResponses,
376        },
377        FamilyKind::Google => OvenAdapterFamily::GoogleGemini,
378        FamilyKind::Vertex | FamilyKind::VertexAnthropic => OvenAdapterFamily::GoogleVertexGemini,
379        FamilyKind::Bedrock => match shape {
380            ResolvedShape::Chat => OvenAdapterFamily::AwsBedrockConverse,
381            ResolvedShape::Responses => OvenAdapterFamily::OpenaiResponses,
382        },
383        FamilyKind::Azure => match shape {
384            ResolvedShape::Chat => OvenAdapterFamily::AzureOpenaiChat,
385            ResolvedShape::Responses => OvenAdapterFamily::AzureOpenaiResponses,
386        },
387        FamilyKind::Cohere => OvenAdapterFamily::CohereV2Chat,
388    })
389}

392pub fn environment_aliases<'a>(
393    provider: &'a CatalogProviderRecord,
394    credential: &str,
395) -> Vec<&'a str> {
396    match credential {

399            .first()
400            .map(String::as_str)

402            .collect(),
403        "access_key_id" => provider

407            .map(String::as_str)

409            .collect(),
410        "secret_access_key" => provider
411            .environment
412            .iter()
413            .find(|value| value.as_str() == "AWS_SECRET_ACCESS_KEY")
414            .map(String::as_str)
415            .into_iter()
416            .collect(),
417        "session_token" => provider
418            .environment
419            .iter()
420            .find(|value| value.as_str() == "AWS_SESSION_TOKEN")
421            .map(String::as_str)
422            .into_iter()
423            .collect(),
424        _ => Vec::new(),
425    }

428#[must_use]
429pub fn placeholders(template: &str) -> Vec<String> {
430    let mut result = Vec::new();
431    let mut rest = template;
432    while let Some(start) = rest.find("${") {
433        rest = &rest[start + 2..];
434        let Some(end) = rest.find('}') else { break };
435        let name = &rest[..end];
436        if !name.is_empty()
437            && name
438                .bytes()
439                .all(|byte| byte.is_ascii_uppercase() || byte.is_ascii_digit() || byte == b'_')
440            && !result.iter().any(|value| value == name)
441        {
442            result.push(name.to_owned());
443        }
444        rest = &rest[end + 1..];
445    }
446    result.sort();
447    result
448}
449
450pub fn substitute_placeholders(
451    template: &str,
452    values: &std::collections::BTreeMap<String, String>,
453) -> Option<String> {
454    let mut endpoint = template.to_owned();
455    for name in placeholders(template) {
456        let field = setup_field_name(&name);
457        let value = values.get(&name).or_else(|| values.get(&field))?;
458        endpoint = endpoint.replace(&format!("${{{name}}}"), value);
459    }
460    Some(endpoint)
461}

463#[must_use]
464pub fn setup_field_name(name: &str) -> String {
465    match name {
466        "AWS_REGION" => "region".to_owned(),
467        "AZURE_RESOURCE_NAME" => "resource_name".to_owned(),
468        "GOOGLE_VERTEX_PROJECT" => "project".to_owned(),
469        "GOOGLE_VERTEX_LOCATION" => "location".to_owned(),
470        "GOOGLE_VERTEX_ENDPOINT" => "endpoint".to_owned(),
471        _ => name.to_ascii_lowercase(),
472    }
