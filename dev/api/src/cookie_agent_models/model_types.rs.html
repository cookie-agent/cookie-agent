12/// Internal digest decoding remains permissive for existing compiled model state.
13#[derive(Clone, Debug, Deserialize, Eq, Hash, Ord, PartialEq, PartialOrd, Serialize)]
14#[serde(transparent)]
15pub struct Sha256Digest(String);
16
17impl Sha256Digest {
18    pub fn new(value: impl Into<String>) -> Result<Self, ModelTypeError> {
19        let value = value.into();
20        if value.len() == 64
21            && value
22                .bytes()
23                .all(|byte| byte.is_ascii_digit() || (b'a'..=b'f').contains(&byte))
24        {
25            Ok(Self(value))
26        } else {
27            Err(ModelTypeError::Fingerprint)
28        }
29    }
30
31    #[must_use]
32    pub fn as_str(&self) -> &str {
33        &self.0
34    }
35
36    pub(crate) fn hash(domain: &str, value: &impl Serialize) -> Result<Self, serde_json::Error> {
37        let mut hasher = Sha256::new();
38        hasher.update(domain.as_bytes());
39        hasher.update([0]);
40        hasher.update(serde_json::to_vec(value)?);
41        Ok(Self(format!("{:x}", hasher.finalize())))
42    }
43}

96#[derive(Clone, Debug, Default, Deserialize, PartialEq, Serialize)]
97#[serde(deny_unknown_fields)]
98/// Runtime-ready authoring defaults retain request application behavior and permissive decoding.
99pub struct ResolvedRequestDefaults {
100    pub request: RequestDefaults,
101    pub reasoning: Option<ReasoningBehavior>,
102}
103
104impl ResolvedRequestDefaults {
105    #[must_use]
106    pub fn apply(
107        &self,
108        _options: &ProviderOptions,
109        mut request: oven_sdk::Request,
110    ) -> oven_sdk::Request {
111        request.inference.max_output_tokens = request
112            .inference
113            .max_output_tokens
114            .or(self.request.max_output_tokens);
115        request.inference.temperature = request
116            .inference
117            .temperature
118            .or(self.request.temperature.map(|value| f64::from(value.get())));
119        request.inference.top_p = request
120            .inference
121            .top_p
122            .or(self.request.top_p.map(|value| f64::from(value.get())));
123        request.inference.reasoning_effort =
124            request
125                .inference
126                .reasoning_effort
127                .clone()
128                .or_else(|| match &self.reasoning {
129                    Some(ReasoningBehavior::Effort { value }) => Some(
130                        match value {
131                            ReasoningEffort::None => "none",
132                            ReasoningEffort::Minimal => "minimal",
133                            ReasoningEffort::Low => "low",
134                            ReasoningEffort::Medium => "medium",
135                            ReasoningEffort::High => "high",
136                            ReasoningEffort::Xhigh => "xhigh",
137                            ReasoningEffort::Max => "max",
138                            ReasoningEffort::Default => "default",
139                        }
140                        .to_owned(),
141                    ),
142                    Some(
143                        ReasoningBehavior::Toggle { .. } | ReasoningBehavior::BudgetTokens { .. },
144                    )
145                    | None => None,
146                });
147        request
148    }
149}
