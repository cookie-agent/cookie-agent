1use std::{fmt, io};
2
3use cookie_agent_identity::{AgentId, ModelKey, ProviderId};
4use thiserror::Error;
5
6#[derive(Error)]
7pub enum ConfigError {
8    #[error("configuration I/O failed")]
9    Io(#[source] io::Error),
10    #[error("configuration path is unsafe")]
11    UnsafePath,
12    #[error("required file was not found")]
13    NotFound,
14    #[error("{0} exceeds its byte limit")]
15    TooLarge(String),
16    #[error("{0} is not UTF-8")]
17    Utf8(&'static str),
18    #[error("invalid TOML: {0}")]
19    Toml(String),
20    #[error("TOML resource limit or unsupported datetime exceeded")]
21    TomlLimit,
22    #[error("provider `{provider}` is invalid")]
23    Provider {
24        provider: ProviderId,
25        #[source]
26        source: cookie_agent_models::authoring::AuthoringError,
27    },
28    #[error("runtime settings are invalid")]
29    InvalidRuntime,
30    #[error("MCP server `{server}` is invalid: {reason}")]
31    McpServer { server: String, reason: String },
32    #[error("environment interpolation is invalid at `{0}`")]
33    Interpolation(String),
34    #[error("environment variable `{variable}` is missing at `{path}`")]
35    MissingEnvironment { path: String, variable: String },
36    #[error("environment variable `{variable}` is not UTF-8 at `{path}`")]
37    NonUtf8Environment { path: String, variable: String },
38    #[error("invalid agent filename `{0}`")]
39    AgentFilename(String),
40    #[error("duplicate agent `{0}`")]
41    DuplicateAgent(AgentId),
42    #[error("agent ID `{0}` is reserved for a built-in agent")]
43    ReservedAgentId(AgentId),
44    #[error("invalid frontmatter for agent `{0}`")]
45    AgentFrontmatter(AgentId),
46    #[error(
47        "agent `{0}` permission resources support only `${{workspace_dir}}`, and only for read or write"
48    )]
49    AgentPermissionExpression(AgentId),
50    #[error("agent YAML resource limit exceeded")]
51    AgentYamlLimit,
52    #[error("agent `{0}` has an empty system prompt")]
53    EmptyPrompt(AgentId),
54    #[error("agent `{agent}` has invalid `{field}`")]
55    AgentField { agent: AgentId, field: &'static str },
56    #[error("agent `{0}` exceeds a list limit")]
57    AgentLimit(AgentId),
58    #[error("primary agent `{0}` requires a nonempty fallback")]
59    PrimaryFallback(AgentId),
60    #[error("agent `{0}` lists a tool more than once")]
61    DuplicateTool(AgentId),
62    #[error("agent `{0}` has invalid delegation")]
63    Delegation(AgentId),
64    #[error("agent `{agent}` delegates to unknown agent `{target}`")]
65    UnknownDelegationTarget { agent: AgentId, target: AgentId },
66    #[error("agent `{agent}` delegates to ineligible agent `{target}`")]
67    IneligibleDelegationTarget { agent: AgentId, target: AgentId },
68    #[error("agent `{agent}` repeats fallback model `{model}`")]
69    DuplicateFallbackModel { agent: AgentId, model: ModelKey },
70}
71
72impl fmt::Debug for ConfigError {
73    fn fmt(&self, formatter: &mut fmt::Formatter<'_>) -> fmt::Result {
74        formatter
75            .debug_tuple("ConfigError")
76            .field(&self.to_string())
77            .finish()
78    }
79}
