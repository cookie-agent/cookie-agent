153impl McpServerConfig {
154    pub fn validate(&self, name: &str) -> Result<(), ConfigError> {
155        let invalid_name =
156            name.is_empty() || name.len() > 128 || name.chars().any(char::is_control);
157        let transport_count = usize::from(self.command.is_some()) + usize::from(self.url.is_some());
158        let invalid_stdio = self.command.as_ref().is_some_and(String::is_empty)
159            || self.cwd.as_ref().is_some_and(String::is_empty);
160        let invalid_remote = self.url.as_ref().is_some_and(|url| {
161            url::Url::parse(url)
162                .map(|parsed| !matches!(parsed.scheme(), "http" | "https"))
163                .unwrap_or(true)
164        });
165        let mixed = self.command.is_some() && (!self.headers.is_empty() || !self.oauth.is_auto())
166            || self.url.is_some()
167                && (!self.args.is_empty() || !self.env.is_empty() || self.cwd.is_some());
168        let invalid_oauth = match &self.oauth {
169            McpOAuthConfig::Settings(settings) => {
170                settings.client_id.as_ref().is_some_and(String::is_empty)
171                    || settings
172                        .client_secret
173                        .as_ref()
174                        .is_some_and(String::is_empty)
175                    || settings.client_secret.is_some() && settings.client_id.is_none()
176                    || settings.scopes.iter().any(String::is_empty)
177                    || settings.client_metadata_url.as_ref().is_some_and(|value| {
178                        url::Url::parse(value)
179                            .map(|url| {
180                                url.scheme() != "https"
181                                    || url.host_str().is_none()
182                                    || url.path() == "/"
183                            })
184                            .unwrap_or(true)
185                    })
186            }
187            McpOAuthConfig::Auto | McpOAuthConfig::Enabled | McpOAuthConfig::Disabled => false,
188        };
189        if invalid_name
190            || transport_count != 1
191            || invalid_stdio
192            || invalid_remote
193            || mixed
194            || invalid_oauth
195            || self.timeout_ms == Some(0)
196        {
197            return Err(ConfigError::McpServer {
198                server: name.to_owned(),
199                reason: "configure exactly one of command or url; stdio-only and remote-only fields may not be mixed; OAuth client settings must be valid; and timeout_ms must be positive".into(),
200            });
201        }
202        Ok(())
203    }
204}

407fn deserialize_duration<'de, D>(deserializer: D) -> Result<Duration, D::Error>
408where
409    D: serde::Deserializer<'de>,
410{
411    let value = String::deserialize(deserializer)?;
412    parse_duration(&value).ok_or_else(|| {
413        serde::de::Error::custom("duration must be an integer followed by ms, s, m, h, or d")
414    })
415}
416
417fn parse_duration(value: &str) -> Option<Duration> {
418    let unit_start = value.find(|character: char| !character.is_ascii_digit())?;
419    let amount = value[..unit_start].parse::<u64>().ok()?;
420    let unit = &value[unit_start..];
421    let milliseconds = match unit {
422        "ms" => amount,
423        "s" => amount.checked_mul(1_000)?,
424        "m" => amount.checked_mul(60_000)?,
425        "h" => amount.checked_mul(3_600_000)?,
426        "d" => amount.checked_mul(86_400_000)?,
427        _ => return None,
428    };
429    Some(Duration::from_millis(milliseconds))
430}

456#[derive(Deserialize)]
457#[serde(deny_unknown_fields)]
458struct RawContextCompactionConfig {
459    #[serde(default = "yes", rename = "auto")]
460    auto_compaction: bool,
461    trigger: Option<ContextCompactionTrigger>,
462    buffer_tokens: Option<u64>,
463    #[serde(default = "default_summary")]
464    max_summary_bytes: usize,

467impl<'de> Deserialize<'de> for ContextCompactionConfig {
468    fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
469    where
470        D: serde::Deserializer<'de>,
471    {
472        let raw = RawContextCompactionConfig::deserialize(deserializer)?;
473        let trigger = match (raw.trigger, raw.buffer_tokens) {
474            (Some(_), Some(_)) => {
475                return Err(serde::de::Error::custom(
476                    "trigger and buffer_tokens cannot both be set",
477                ));
478            }
479            (Some(trigger), None) => trigger,
480            (None, Some(buffer_tokens)) => ContextCompactionTrigger::BufferTokens { buffer_tokens },
481            (None, None) => default_compaction_trigger(),
482        };
483        Ok(Self {
484            auto_compaction: raw.auto_compaction,
485            trigger,
486            max_summary_bytes: raw.max_summary_bytes,
487        })
488    }
489}

566pub(crate) fn validate_runtime(runtime: &RuntimeConfig) -> Result<(), ConfigError> {
567    if runtime.server.host.is_empty()
568        || runtime.server.host.len() > 255
569        || runtime.tool_output.max_lines == 0
570        || runtime.tool_output.max_bytes == 0
571        || runtime.approval.timeout_ms == 0
572        || runtime.delegation.max_depth == 0
573        || runtime.delegation.max_concurrency == Some(0)
574    {
575        return Err(ConfigError::InvalidRuntime);
576    }
577    let context = &runtime.context_compaction;
578    let invalid_trigger = match &context.trigger {
579        ContextCompactionTrigger::Percent { percent } => !(1..=99).contains(percent),
580        ContextCompactionTrigger::BufferTokens { buffer_tokens } => *buffer_tokens == 0,
581    };
582    if invalid_trigger
583        || context.max_summary_bytes == 0
584        || context.max_summary_bytes > 2 * 1024 * 1024
585    {
586        return Err(ConfigError::InvalidRuntime);
587    }
588    if runtime.session_title.max_chars == 0 || runtime.session_title.max_input_messages == 0 {
589        return Err(ConfigError::InvalidRuntime);
590    }
591    let caching = &runtime.prompt_caching;
592    if caching.enabled
593        && (caching.tools_ttl == AnthropicCacheTtlConfig::FiveMinutes
594            && (caching.system_ttl == AnthropicCacheTtlConfig::OneHour
595                || caching.rolling_ttl == AnthropicCacheTtlConfig::OneHour)
596            || caching.system_ttl == AnthropicCacheTtlConfig::FiveMinutes
597                && caching.rolling_ttl == AnthropicCacheTtlConfig::OneHour)
598    {
599        return Err(ConfigError::InvalidRuntime);
600    }
601    for (id, provider) in &runtime.providers {
602        provider
603            .validate_for(id)
604            .map_err(|source| ConfigError::Provider {
605                provider: id.clone(),
606                source,
607            })?;
608    }
609    Ok(())
