80/// Loads user config and exact-cwd `.cookie-agent` with no upward search.
81pub fn load(cwd: &Path) -> Result<LoadedConfiguration, ConfigError> {
82    let cwd = cwd.canonicalize().map_err(ConfigError::Io)?;
83    let user = user_root();
84    let mut loaded = load_from_roots(user.as_deref(), Some(&cwd.join(".cookie-agent")))?;
85    loaded.skills = crate::load_skills(&cwd)?;
86    Ok(loaded)
87}

89/// Loads explicit roots, primarily for composition and adversarial tests.
90pub fn load_from_roots(
91    user_root: Option<&Path>,
92    workspace_root: Option<&Path>,
93) -> Result<LoadedConfiguration, ConfigError> {
94    let user = user_root
95        .map(|path| open_layer_root(path, AgentDocumentSource::User))
96        .transpose()?
97        .flatten();
98    let workspace = workspace_root
99        .map(|path| open_layer_root(path, AgentDocumentSource::Workspace))
100        .transpose()?
101        .flatten();
102    let mut runtime = RuntimeConfig {
103        server: ServerConfig::default(),
104        tool_output: ToolOutputConfig::default(),
105        agent_md: AgentMdConfig::default(),
106        approval: ApprovalConfig::default(),
107        model_retry: ModelRetryConfig::default(),
108        context_compaction: ContextCompactionConfig::default(),
109        prompt_caching: PromptCachingConfig::default(),
110        session_title: SessionTitleConfig::default(),
111        delegation: DelegationConfig::default(),
112        pricing: Default::default(),
113        providers: BTreeMap::new(),
114    };
115    let mut agents = BTreeMap::new();
116    let mut preset_agents = BTreeMap::<String, BTreeMap<AgentId, AgentDocument>>::new();
117    let mut mcp_servers = BTreeMap::new();
118    let mut user_mcp_servers = BTreeMap::new();
119    let mut workspace_mcp_servers = BTreeMap::new();
120    let mut plugins = IndexMap::new();
121    for root in [user.as_ref(), workspace.as_ref()].into_iter().flatten() {
122        if let Some((mut layer, source_text)) = root.load_runtime()? {
123            apply_settings(&mut runtime, &layer);
124            if let Some(mcp) = layer.mcp.take() {
125                let source = match root.source {
126                    AgentDocumentSource::User => McpServerSource::UserFile,
127                    AgentDocumentSource::Workspace => McpServerSource::WorkspaceFile,
128                    AgentDocumentSource::BuiltIn => {
129                        unreachable!("configuration roots are authored")
130                    }
131                };
132                for (name, config) in mcp.servers {
133                    match source {
134                        McpServerSource::UserFile => {
135                            user_mcp_servers.insert(name.clone(), config.clone());
136                        }
137                        McpServerSource::WorkspaceFile => {
138                            workspace_mcp_servers.insert(name.clone(), config.clone());
139                        }
140                        McpServerSource::Runtime => unreachable!("file configuration layer"),
141                    }
142                    mcp_servers.insert(name, LoadedMcpServer { source, config });
143                }
144            }
145            if let Some(layer_plugins) = layer.plugins.take() {
146                plugins.extend(plugins_in_authored_order(layer_plugins, &source_text).map(
147                    |(name, config)| {
148                        (
149                            name,
150                            (
151                                config,
152                                root.path.join("config.toml"),
153                                Arc::clone(&source_text),
154                            ),
155                        )
156                    },
157                ));
158            }
159            for (id, mut value) in std::mem::take(&mut layer.providers) {
160                interpolate_provider_values(
161                    value.value_mut(),
162                    &mut vec!["providers".to_owned(), id.as_str().to_owned()],
163                )?;
164                let json = SensitiveJsonValue::from_toml(value.take());
165                let provider = ProviderDefinition::deserialize(json.value()).map_err(|error| {
166                    config_decode_error(
167                        &root.path.join("config.toml"),
168                        None,
169                        &format!("provider `{id}`: {error}"),
170                    )
171                })?;
172                runtime.providers.insert(id, provider);
173            }
174        }
175        let layer_agents = root.load_agents()?;
176        for (id, document) in layer_agents.shared {
177            agents.insert(id, document);
178        }
179        for (preset, documents) in layer_agents.presets {
180            preset_agents.entry(preset).or_default().extend(documents);
181        }
182    }
183    AgentRegistry::validate_ref(&agents)?;
184    let agent_presets = preset_agents
185        .into_iter()
186        .map(|(preset, documents)| {
187            let mut effective = agents.clone();
188            effective.extend(documents);
189            AgentRegistry::validate_ref(&effective)?;
190            Ok((preset, effective))
191        })
192        .collect::<Result<_, ConfigError>>()?;
193    validate_runtime(&runtime)?;
194    for (name, server) in &mcp_servers {
195        server.config.validate(name)?;
196    }
197    for (name, (plugin, path, text)) in &plugins {
198        if let Some(field) = plugin.invalid_field(name) {
199            return Err(config_decode_error(
200                path,
201                Some(text),
202                &format!("plugin `{name}`: invalid field `{field}`"),
203            ));
204        }
205    }
206    let plugins = plugins
207        .into_iter()
208        .map(|(name, (config, _, _))| (name, config))
209        .collect();
210    Ok(LoadedConfiguration {
211        runtime,
212        agents,
213        agent_presets,
214        mcp_servers,
215        user_mcp_servers,
216        workspace_mcp_servers,
217        plugins,
218        config_paths: ConfigLayerPaths {
219            user: user_root.map(|root| root.join("config.toml")),
220            workspace: workspace_root.map(|root| root.join("config.toml")),
221        },
222        skills: crate::SkillRegistry::default(),
223    })
224}

226fn plugins_in_authored_order(
227    mut plugins: crate::PluginsConfig,
228    source_text: &str,
229) -> impl Iterator<Item = (String, crate::PluginConfig)> {
230    let document = source_text
231        .parse::<toml_edit::DocumentMut>()
232        .expect("validated TOML reparses for plugin ordering");
233    let names: Vec<String> = document
234        .get("plugins")
235        .and_then(toml_edit::Item::as_table)
236        .map(|table| table.iter().map(|(name, _)| name.to_owned()).collect())
237        .unwrap_or_default();
238    names.into_iter().filter_map(move |name| {
239        plugins
240            .plugins
241            .shift_remove(&name)
242            .map(|config| (name, config))
243    })
244}

246impl LayerRoot {
247    fn load_runtime(&self) -> Result<Option<(RawRuntimeLayer, Arc<str>)>, ConfigError> {
248        let Some(bytes) = read_optional_file(&self.path, "config.toml", MAX_CONFIG_BYTES)? else {
249            return Ok(None);
250        };
251        let path = self.path.join("config.toml");
252        let text = std::str::from_utf8(&bytes)
253            .map_err(|_| ConfigError::Toml(format!("{}: content is not UTF-8", path.display())))?;
254        let value = text.parse::<toml::Value>().map_err(|error| {
255            ConfigError::Toml(format!(
256                "{}: {}",
257                path.display(),
258                safe_toml_error(text, &error)
259            ))
260        })?;
261        let mut value = SensitiveTomlValue::new(value);
262        validate_toml_value(value.value()).map_err(|_| {
263            ConfigError::Toml(format!(
264                "{}: TOML resource limit or unsupported datetime exceeded",
265                path.display()
266            ))
267        })?;
268        let layer = decode_runtime_layer(value.value_mut(), text, &path)?;
269        Ok(Some((layer, Arc::from(text))))
270    }
271
272    fn load_agents(&self) -> Result<AgentLayer, ConfigError> {
273        let Some(directory) = open_optional_directory(&self.path, "agents")? else {
274            return Ok(AgentLayer::default());
275        };
276        let names = sorted_entry_names(&directory)?;
277        let mut layer = AgentLayer::default();
278        for name in names {
279            if name.ends_with(".md") {
280                load_agent_document(&directory, &name, self.source, &mut layer.shared)?;
281                continue;
282            }
283            if regular_file_at(&directory, &name)? {
284                continue;
285            }
286            let preset_directory =
287                open_optional_directory(&directory, &name)?.ok_or(ConfigError::UnsafePath)?;
288            AgentId::new(&name).map_err(|_| ConfigError::AgentPresetName {
289                path: preset_directory.clone(),
290                name: name.clone(),
291            })?;
292            let mut documents = BTreeMap::new();
293            for document_name in sorted_entry_names(&preset_directory)? {
294                if !document_name.ends_with(".md") {
295                    return Err(ConfigError::UnsafePath);
296                }
297                load_agent_document(
298                    &preset_directory,
299                    &document_name,
300                    self.source,
301                    &mut documents,
302                )?;
303            }
304            layer.presets.insert(name, documents);
305        }
306        Ok(layer)
307    }
308}

316fn sorted_entry_names(directory: &Path) -> Result<Vec<String>, ConfigError> {
317    let mut names = std::fs::read_dir(directory)
318        .map_err(ConfigError::Io)?
319        .map(|entry| {
320            entry
321                .map_err(ConfigError::Io)?
322                .file_name()
323                .into_string()
324                .map_err(|_| ConfigError::UnsafePath)
325        })
326        .collect::<Result<Vec<_>, _>>()?;
327    names.sort_by(|left, right| left.as_bytes().cmp(right.as_bytes()));
328    Ok(names)
329}

331fn load_agent_document(
332    directory: &Path,
333    name: &str,
334    source: AgentDocumentSource,
335    documents: &mut BTreeMap<AgentId, AgentDocument>,
336) -> Result<(), ConfigError> {
337    if !regular_file_at(directory, name)? {
338        return Err(ConfigError::UnsafePath);
339    }
340    let stem = name.strip_suffix(".md").expect("suffix checked");
341    let id = AgentId::new(stem).map_err(|_| ConfigError::AgentFilename(name.to_owned()))?;
342    let bytes = read_required_file(directory, name, MAX_AGENT_BYTES)?;
343    let document = parse_agent(id.clone(), &bytes, source, &directory.join(name))?;
344    if documents.insert(id.clone(), document).is_some() {
345        return Err(ConfigError::DuplicateAgent(id));
346    }
347    Ok(())
348}

350fn decode_runtime_layer(
351    value: &mut toml::Value,
352    text: &str,
353    path: &Path,
354) -> Result<RawRuntimeLayer, ConfigError> {
355    const ALLOWED: &[&str] = &[
356        "server",
357        "tool_output",
358        "agent_md",
359        "approval",
360        "model_retry",
361        "context_compaction",
362        "prompt_caching",
363        "session_title",
364        "delegation",
365        "pricing",
366        "mcp",
367        "plugins",
368        "providers",
369    ];
370    let table = value.as_table_mut().ok_or_else(|| {
371        ConfigError::Toml(format!("{}: top level must be a table", path.display()))
372    })?;
373    if table.contains_key("schema_version") {
374        return Err(ConfigError::ConfigSchemaRemoved {
375            path: path.to_owned(),
376            line: key_line(text, "schema_version").unwrap_or(1),
377        });
378    }
379    if let Some(key) = table.keys().find(|key| !ALLOWED.contains(&key.as_str())) {
380        return Err(ConfigError::Toml(format!(
381            "{} line {}: unknown top-level field `{key}`; remove it",
382            path.display(),
383            key_line(text, key).unwrap_or(1)
384        )));
385    }
386    if table
387        .get("context_compaction")
388        .and_then(toml::Value::as_table)
389        .is_some_and(|context| {
390            context.contains_key("trigger") && context.contains_key("buffer_tokens")
391        })
392    {
393        return Err(ConfigError::Toml(format!(
394            "{}: context_compaction.trigger and context_compaction.buffer_tokens cannot both be set",
395            path.display()
396        )));
397    }
398    let providers = match table.remove("providers") {
399        None => SensitiveProviderValues::new(),
400        Some(toml::Value::Table(values)) => values
401            .into_iter()
402            .map(|(id, value)| {
403                let value = SensitiveTomlValue::new(value);
404                ProviderId::new(id).map(|id| (id, value)).map_err(|_| {
405                    ConfigError::Toml(format!("{}: invalid provider ID", path.display()))
406                })
407            })
408            .collect::<Result<_, _>>()?,
409        Some(mut other) => {
410            zeroize_toml_value(&mut other);
411            return Err(ConfigError::Toml(format!(
412                "{}: `providers` must be a table",
413                path.display()
414            )));
415        }
416    };
417    let owned = std::mem::replace(value, toml::Value::Table(Default::default()));
418    let json = SensitiveJsonValue::from_toml(owned);
419    let mut layer = RawRuntimeLayer::deserialize(json.value())
420        .map_err(|error| config_decode_error(path, Some(text), &error.to_string()))?;
421    layer.providers = providers;
422    Ok(layer)
423}

425fn config_decode_error(path: &Path, text: Option<&str>, message: &str) -> ConfigError {
426    let field = extract_serde_field(message, "unknown field")
427        .map(|field| (field, "unknown field"))
428        .or_else(|| {
429            extract_serde_field(message, "missing field").map(|field| (field, "missing field"))
430        })
431        .or_else(|| {
432            extract_serde_field(message, "invalid field").map(|field| (field, "invalid field"))
433        });
434    let detail = field.as_ref().map_or_else(
435        || "malformed configuration content".to_owned(),
436        |(field, kind)| {
437            let line = text.and_then(|text| key_line(text, field));
438            let advice = match *kind {
439                "unknown field" => "remove it",
440                "missing field" => "add it",
441                "invalid field" => "fix it",
442                _ => unreachable!("known configuration diagnostic kind"),
443            };
444            match line {
445                Some(line) => format!("line {line}: {kind} `{field}`; {advice}"),
446                None => format!("{kind} `{field}`; {advice}"),
447            }
448        },
449    );
450    ConfigError::Toml(format!("{}: {detail}", path.display()))
451}
452
453fn extract_serde_field(message: &str, marker: &str) -> Option<String> {
454    let rest = message.split_once(marker)?.1.trim_start();
455    let quote = rest.chars().next()?;
456    if !matches!(quote, '`' | '\'' | '"') {
457        return None;
458    }
459    rest[quote.len_utf8()..]
460        .split_once(quote)
461        .map(|(field, _)| field.to_owned())
462}
463
464fn key_line(text: &str, key: &str) -> Option<usize> {
465    text.lines()
466        .position(|line| {
467            line.split_once('=')
468                .is_some_and(|(candidate, _)| candidate.trim().trim_matches('"') == key)
469        })
470        .map(|index| index + 1)
471}
