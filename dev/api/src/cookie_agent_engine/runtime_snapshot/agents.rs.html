37impl AgentRegistry {
38    pub(crate) fn resolve(
39        authored: &ConfigAgentRegistry,
40        models: &CompiledModelRuntime,
41    ) -> Result<Self, EngineError> {
42        let mut documents = built_in_internal_documents()?;
43        documents.extend(
44            authored
45                .documents()
46                .iter()
47                .map(|(id, document)| (id.clone(), document.clone())),
48        );
49        let mut agents = BTreeMap::new();
50        for document in documents.into_values() {
51            let id = document.id.clone();
52            if id.as_str() == BUILT_IN_DEFAULT_AGENT_ID {
53                return Err(EngineError::RuntimeCompileFailed);
54            }
55            let mut resolved_fallback = Vec::new();
56            for fallback in &document.frontmatter.model_fallback {
57                match &fallback.model {
58                    AgentModelRef::ParentModel => {
59                        resolved_fallback.push(ResolvedAgentFallback::ParentModel);
60                    }
61                    AgentModelRef::Model(model_key) => {
62                        let variant = match (&fallback.variant, models.model(model_key)) {
63                            (None, Some(model)) => model.model.default_variant.clone(),
64                            (Some(cookie_agent_identity::ConfiguredVariantRef::Base), _) => None,
65                            (Some(cookie_agent_identity::ConfiguredVariantRef::Named(id)), _) => {
66                                Some(id.clone())
67                            }
68                            (None, None) => None,
69                        };
70                        resolved_fallback.push(ResolvedAgentFallback::Selection(ModelSelection {
71                            model: model_key.clone(),
72                            variant,
73                        }));
74                    }
75                }
76            }
77            let available = resolved_fallback
78                .iter()
79                .filter_map(|fallback| match fallback {
80                    ResolvedAgentFallback::Selection(selection) => Some(selection),
81                    ResolvedAgentFallback::ParentModel => None,
82                })
83                .any(|selection| selection_available(models, selection));
84            let runnable_as_root = document.frontmatter.enabled
85                && matches!(
86                    document.frontmatter.mode,
87                    AgentMode::Primary | AgentMode::All
88                )
89                && available;
90            agents.insert(
91                id,
92                ResolvedAgent {
93                    document,
94                    resolved_fallback,
95                    runnable_as_root,
96                },
97            );
98        }
99        if !agents.values().any(|agent| agent.runnable_as_root)
100            && let Some(selection) = first_available_selection(models)
101        {
102            let document = built_in_default_document(&selection)?;
103            agents.insert(
104                document.id.clone(),
105                ResolvedAgent {
106                    document,
107                    resolved_fallback: vec![ResolvedAgentFallback::Selection(selection)],
108                    runnable_as_root: true,
109                },
110            );
111        }
112        let descriptors = agents
113            .iter()
114            .map(|(id, agent)| AgentDescriptor {
115                id: id.clone(),
116                description: agent.document.frontmatter.description.clone(),
117                mode: agent.document.frontmatter.mode,
118                enabled: agent.document.frontmatter.enabled,
119                runnable_as_root: agent.runnable_as_root,
120                resolved_fallback: agent
121                    .resolved_fallback
122                    .iter()
123                    .filter_map(|fallback| match fallback {
124                        ResolvedAgentFallback::Selection(selection) => Some(selection.clone()),
125                        ResolvedAgentFallback::ParentModel => None,
126                    })
127                    .collect(),
128                delegation_targets: delegation_targets(&agent.document.frontmatter.permissions),
129            })
130            .collect();
131        Ok(Self {
132            agents,
133            descriptors,
134        })
135    }
136
137    pub(crate) fn get(&self, id: &AgentId) -> Option<&ResolvedAgent> {
138        self.agents.get(id)
139    }
140
141    pub(crate) fn descriptors(&self) -> &[AgentDescriptor] {
142        &self.descriptors
143    }
144}

146fn first_available_selection(models: &CompiledModelRuntime) -> Option<ModelSelection> {
147    models.models().values().find_map(|model| {
148        (model.model.status == CompiledModelStatus::Available).then(|| ModelSelection {
149            model: model.key.clone(),
150            variant: model.model.default_variant.clone(),
151        })
152    })
153}

155fn built_in_internal_documents() -> Result<BTreeMap<AgentId, AgentDocument>, EngineError> {
156    [
157        (
158            BUILT_IN_APPROVAL_AGENT_ID,
159            "Built-in approval evaluator",
160            "Evaluate the supplied approval request conservatively. Return only the requested structured decision.\n",
161            AgentLimits::default(),
162        ),
163        (
164            BUILT_IN_COMPACTION_AGENT_ID,
165            "Built-in context compaction agent",
166            "Summarize conversation context faithfully within the supplied bounds. Return summary text only.\n",
167            AgentLimits::default(),
168        ),
169        (
170            BUILT_IN_TITLE_AGENT_ID,
171            "Built-in session title agent",
172            "Generate a concise plain-text title from the supplied first user message. Return title text only.\n",
173            AgentLimits {
174                timeout_ms: 10_000,
175                max_input_tokens: 4_096,
176                max_output_tokens: 128,
177            },
178        ),
179    ]
180    .into_iter()
181    .map(|(id, description, body, limits)| {
182        let document = built_in_internal_document(id, description, body, limits)?;
183        Ok((document.id.clone(), document))
184    })
185    .collect()
186}

188fn built_in_internal_document(
189    id: &str,
190    description: &str,
191    body: &str,
192    limits: AgentLimits,
193) -> Result<AgentDocument, EngineError> {
194    let id = IdentityAgentId::new(id).map_err(|_| EngineError::RuntimeCompileFailed)?;
195    let body = body.to_owned();
196    let frontmatter = AgentFrontmatter {
197        description: description.to_owned(),
198        mode: AgentMode::Internal,
199        enabled: true,
200        model_fallback: vec![AgentModelFallback {
201            model: AgentModelRef::ParentModel,
202            variant: None,
203        }],
204        limits,
205        permissions: IndexMap::new(),
206    };
207    let document_fingerprint = fingerprint(
208        "cookie-agent/built-in-internal-agent-document/v1",
209        &(id.as_str(), description, &body, &frontmatter.limits),
210    )?;
211    let prompt_fingerprint = fingerprint("cookie-agent/system-prompt/v1", &body)?;
212    Ok(AgentDocument {
213        id,
214        frontmatter,
215        body,
216        source: AgentDocumentSource::BuiltIn,
217        document_fingerprint,
218        prompt_fingerprint,
219    })
220}

222fn built_in_default_document(selection: &ModelSelection) -> Result<AgentDocument, EngineError> {
223    let id = IdentityAgentId::new(BUILT_IN_DEFAULT_AGENT_ID)
224        .map_err(|_| EngineError::RuntimeCompileFailed)?;
225    let body = "You are Cookie Agent's built-in default coding agent. Help the user inspect, understand, and modify software safely and precisely.\n".to_owned();
226    let frontmatter = AgentFrontmatter {
227        description: "Built-in default coding agent".to_owned(),
228        mode: AgentMode::Primary,
229        enabled: true,
230        model_fallback: vec![AgentModelFallback {
231            model: AgentModelRef::Model(selection.model.clone()),
232            variant: selection
233                .variant
234                .clone()
235                .map(cookie_agent_identity::ConfiguredVariantRef::Named),
236        }],
237        limits: AgentLimits::default(),
238        permissions: built_in_default_permissions()?,
239    };
240    let document_fingerprint = fingerprint("cookie-agent/built-in-default-document/v1", selection)?;
241    let prompt_fingerprint = fingerprint("cookie-agent/system-prompt/v1", &body)?;
242    Ok(AgentDocument {
243        id,
244        frontmatter,
245        body,
246        source: AgentDocumentSource::BuiltIn,
247        document_fingerprint,
248        prompt_fingerprint,
249    })
250}

252fn fingerprint(
253    domain: &str,
254    value: &impl Serialize,
255) -> Result<cookie_agent_models::Sha256Digest, EngineError> {
256    let mut hasher = Sha256::new();
257    hasher.update(domain.as_bytes());
258    hasher.update([0]);
259    hasher.update(serde_json::to_vec(value).map_err(|_| EngineError::RuntimeCompileFailed)?);
260    cookie_agent_models::Sha256Digest::new(format!("{:x}", hasher.finalize()))
261        .map_err(|_| EngineError::RuntimeCompileFailed)
262}

264fn built_in_default_permissions() -> Result<IndexMap<PermissionAction, PermissionValue>, EngineError>
265{
266    let definitions = [
267        (PermissionAction::Read, "*?*", PermissionEffect::Allow),
268        (PermissionAction::Write, "*", PermissionEffect::Ask),
269        (PermissionAction::Bash, "*", PermissionEffect::Ask),
270        (PermissionAction::Delegate, "*", PermissionEffect::Ask),
271        (PermissionAction::Read, "/*", PermissionEffect::Ask),
272        (PermissionAction::Read, ".env", PermissionEffect::Deny),
273        (PermissionAction::Read, "*/.env", PermissionEffect::Deny),
274        (PermissionAction::Read, ".env.*", PermissionEffect::Deny),
275        (PermissionAction::Read, "*/.env.*", PermissionEffect::Deny),
276        (
277            PermissionAction::Read,
278            ".env.example",
279            PermissionEffect::Allow,
280        ),
281        (
282            PermissionAction::Read,
283            "*/.env.example",
284            PermissionEffect::Allow,
285        ),
286        (
287            PermissionAction::Read,
288            "store-v3.json",
289            PermissionEffect::Deny,
290        ),
291        (
292            PermissionAction::Read,
293            "*/store-v3.json",
294            PermissionEffect::Deny,
295        ),
296        (PermissionAction::Read, "token-v1", PermissionEffect::Deny),
297        (PermissionAction::Read, "*/token-v1", PermissionEffect::Deny),
298        (PermissionAction::Read, "id_*", PermissionEffect::Deny),
299        (PermissionAction::Read, "*/id_*", PermissionEffect::Deny),
300        (PermissionAction::Read, ".netrc", PermissionEffect::Deny),
301        (PermissionAction::Read, "*/.netrc", PermissionEffect::Deny),
302        (
303            PermissionAction::Read,
304            "application_default_credentials.json",
305            PermissionEffect::Deny,
306        ),
307        (
308            PermissionAction::Read,
309            "*/application_default_credentials.json",
310            PermissionEffect::Deny,
311        ),
312    ];
313    let mut permissions = IndexMap::new();
314    for (action, resource, effect) in definitions {
315        let entry = permissions
316            .entry(action)
317            .or_insert_with(|| PermissionValue::Resources(IndexMap::new()));
318        let PermissionValue::Resources(resources) = entry else {
319            unreachable!("synthetic permissions use map form")
320        };
321        resources.insert(
322            WildcardPattern::new(resource).map_err(|_| EngineError::RuntimeCompileFailed)?,
323            effect,
324        );
325    }
326    Ok(permissions)
327}

329pub(crate) fn delegation_targets(
330    permissions: &IndexMap<PermissionAction, PermissionValue>,
331) -> Vec<AgentId> {
332    let Some(PermissionValue::Resources(resources)) = permissions.get(&PermissionAction::Delegate)
333    else {
334        return Vec::new();
335    };
336    let mut targets = resources
337        .iter()
338        .filter(|(_, effect)| **effect != PermissionEffect::Deny)
339        .filter_map(|(resource, _)| AgentId::new(resource.as_str()).ok())
340        .collect::<Vec<_>>();
341    targets.sort();
342    targets
343}

345fn selection_available(models: &CompiledModelRuntime, selection: &ModelSelection) -> bool {
346    models.model(&selection.model).is_some_and(|model| {
347        model.model.status == CompiledModelStatus::Available
348            && selection
349                .variant
350                .as_ref()
351                .is_none_or(|variant| model.model.variants.contains_key(variant))
352    })
353}

358    use cookie_agent_identity::WildcardPattern;
359    use indexmap::IndexMap;
360
361    use super::delegation_targets;

363    #[test]
364    fn delegation_targets_require_named_non_deny_permissions() {
365        assert!(delegation_targets(&IndexMap::new()).is_empty());
366
367        let mut resources = IndexMap::new();
368        resources.insert(WildcardPattern::new("*").unwrap(), PermissionEffect::Deny);
369        resources.insert(
370            WildcardPattern::new("reviewer").unwrap(),
371            PermissionEffect::Ask,
372        );
373        let permissions = IndexMap::from([(
374            PermissionAction::Delegate,
375            PermissionValue::Resources(resources),
376        )]);
377        assert_eq!(
378            delegation_targets(&permissions),
379            [cookie_agent_protocol::AgentId::new("reviewer").unwrap()]
380        );
