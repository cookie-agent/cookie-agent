48    pub(crate) fn resolve(
49        authored: &ConfigAgentRegistry,
50        models: &CompiledModelRuntime,
51        preset: Option<String>,
52    ) -> Result<Self, EngineError> {
53        let mut documents = built_in_internal_documents()?;
54        documents.extend(
55            authored
56                .documents()
57                .iter()
58                .map(|(id, document)| (id.clone(), document.clone())),
59        );
60        let mut agents = BTreeMap::new();
61        for document in documents.into_values() {
62            let id = document.id.clone();
63            if id.as_str() == BUILT_IN_DEFAULT_AGENT_ID {
64                return Err(EngineError::RuntimeCompileFailed);
65            }
66            let mut resolved_fallback = Vec::new();
67            for fallback in &document.frontmatter.models {
68                match &fallback.model {
69                    AgentModelRef::ParentModel => {
70                        resolved_fallback.push(ResolvedAgentFallback::ParentModel {
71                            cache: fallback.cache.clone(),
72                        });
73                    }
74                    AgentModelRef::Model(model_key) => {
75                        let variant = match (&fallback.variant, models.model(model_key)) {
76                            (None, Some(model)) => model.model.default_variant.clone(),
77                            (Some(cookie_agent_identity::ConfiguredVariantRef::Base), _) => None,
78                            (Some(cookie_agent_identity::ConfiguredVariantRef::Named(id)), _) => {
79                                Some(id.clone())
80                            }
81                            (None, None) => None,
82                        };
83                        resolved_fallback.push(ResolvedAgentFallback::Selection {
84                            selection: ModelSelection {
85                                model: model_key.clone(),
86                                variant,
87                            },
88                            cache: fallback.cache.clone(),
89                        });
90                    }
91                }
92            }
93            let available = resolved_fallback
94                .iter()
95                .filter_map(|fallback| match fallback {
96                    ResolvedAgentFallback::Selection { selection, .. } => Some(selection),
97                    ResolvedAgentFallback::ParentModel { .. } => None,
98                })
99                .any(|selection| selection_available(models, selection));
100            let runnable_as_root = document.frontmatter.enabled
101                && matches!(
102                    document.frontmatter.mode,
103                    AgentMode::Primary | AgentMode::All
104                )
105                && available;
106            agents.insert(
107                id,
108                ResolvedAgent {
109                    document,
110                    resolved_fallback,
111                    runnable_as_root,
112                },
113            );
114        }
115        if !agents.values().any(|agent| agent.runnable_as_root)
116            && let Some(selection) = first_available_selection(models)
117        {
118            let document = built_in_default_document(&selection)?;
119            agents.insert(
120                document.id.clone(),
121                ResolvedAgent {
122                    document,
123                    resolved_fallback: vec![ResolvedAgentFallback::Selection {
124                        selection,
125                        cache: None,
126                    }],
127                    runnable_as_root: true,
128                },
129            );
130        }
131        let descriptors = agents
132            .iter()
133            .map(|(id, agent)| AgentDescriptor {
134                id: id.clone(),
135                preset: preset.clone(),
136                description: agent.document.frontmatter.description.clone(),
137                mode: agent.document.frontmatter.mode,
138                enabled: agent.document.frontmatter.enabled,
139                runnable_as_root: agent.runnable_as_root,
140                resolved_fallback: agent
141                    .resolved_fallback
142                    .iter()
143                    .filter_map(|fallback| match fallback {
144                        ResolvedAgentFallback::Selection { selection, .. } => {
145                            Some(selection.clone())
146                        }
147                        ResolvedAgentFallback::ParentModel { .. } => None,
148                    })
149                    .collect(),
150                delegation_targets: delegation_targets(
151                    &agent.document.frontmatter.permissions,
152                    &agents,
153                ),
154            })
155            .collect();
156        Ok(Self {
157            preset,
158            agents,
159            descriptors,
160        })
161    }

176fn first_available_selection(models: &CompiledModelRuntime) -> Option<ModelSelection> {
177    models.models().values().find_map(|model| {
178        (model.model.status == CompiledModelStatus::Available).then(|| ModelSelection {
179            model: model.key.clone(),
180            variant: model.model.default_variant.clone(),
181        })
182    })
183}

223fn built_in_internal_document(
224    id: &str,
225    description: &str,
226    body: &str,
227    limits: AgentLimits,
228) -> Result<AgentDocument, EngineError> {
229    let id = IdentityAgentId::new(id).map_err(|_| EngineError::RuntimeCompileFailed)?;
230    let body = body.to_owned();
231    let frontmatter = AgentFrontmatter {
232        description: description.to_owned(),
233        mode: AgentMode::Internal,
234        enabled: true,
235        models: vec![AgentModelFallback {
236            model: AgentModelRef::ParentModel,
237            variant: None,
238            cache: None,
239        }],
240        limits,
241        permissions: IndexMap::new(),
242    };
243    let document_fingerprint = fingerprint(
244        "cookie-agent/built-in-internal-agent-document/v2",
245        &(id.as_str(), &frontmatter, &body),
246    )?;
247    let prompt_fingerprint = fingerprint("cookie-agent/system-prompt/v1", &body)?;
248    Ok(AgentDocument {
249        id,
250        frontmatter,
251        body,
252        source: AgentDocumentSource::BuiltIn,
253        document_fingerprint,
254        prompt_fingerprint,
255    })
256}

292fn fingerprint(
293    domain: &str,
294    value: &impl Serialize,
295) -> Result<cookie_agent_models::Sha256Digest, EngineError> {
296    let mut hasher = Sha256::new();
297    hasher.update(domain.as_bytes());
298    hasher.update([0]);
299    hasher.update(serde_json::to_vec(value).map_err(|_| EngineError::RuntimeCompileFailed)?);
300    cookie_agent_models::Sha256Digest::new(format!("{:x}", hasher.finalize()))
301        .map_err(|_| EngineError::RuntimeCompileFailed)
302}

304fn built_in_default_permissions() -> Result<IndexMap<PermissionAction, PermissionValue>, EngineError>
305{
306    let definitions = [
307        (
308            PermissionAction::Read,
309            "tool_result:*",

311        ),

314        (PermissionAction::Bash, "*", PermissionEffect::Ask),
315        (PermissionAction::Delegate, "*", PermissionEffect::Ask),
316        (PermissionAction::Read, "/*", PermissionEffect::Ask),
317        (PermissionAction::Read, ".env", PermissionEffect::Deny),
318        (PermissionAction::Read, "*/.env", PermissionEffect::Deny),
319        (PermissionAction::Read, ".env.*", PermissionEffect::Deny),
320        (PermissionAction::Read, "*/.env.*", PermissionEffect::Deny),
321        (
322            PermissionAction::Read,
323            ".env.example",
324            PermissionEffect::Allow,
325        ),
326        (
327            PermissionAction::Read,
328            "*/.env.example",
329            PermissionEffect::Allow,
330        ),
331        (
332            PermissionAction::Read,
333            "store-v3.json",
334            PermissionEffect::Deny,
335        ),
336        (
337            PermissionAction::Read,
338            "*/store-v3.json",
339            PermissionEffect::Deny,
340        ),
341        (PermissionAction::Read, "token-v1", PermissionEffect::Deny),
342        (PermissionAction::Read, "*/token-v1", PermissionEffect::Deny),
343        (PermissionAction::Read, "id_*", PermissionEffect::Deny),
344        (PermissionAction::Read, "*/id_*", PermissionEffect::Deny),
345        (PermissionAction::Read, ".netrc", PermissionEffect::Deny),
346        (PermissionAction::Read, "*/.netrc", PermissionEffect::Deny),
347        (
348            PermissionAction::Read,
349            "application_default_credentials.json",
350            PermissionEffect::Deny,
351        ),
352        (
353            PermissionAction::Read,
354            "*/application_default_credentials.json",
355            PermissionEffect::Deny,
356        ),
357    ];
358    let mut permissions = IndexMap::new();
359    for (action, resource, effect) in definitions {
360        let entry = permissions
361            .entry(action)
362            .or_insert_with(|| PermissionValue::Resources(IndexMap::new()));
363        let PermissionValue::Resources(resources) = entry else {
364            unreachable!("synthetic permissions use map form")
365        };
366        resources.insert(
367            WildcardPattern::new(resource).map_err(|_| EngineError::RuntimeCompileFailed)?,
368            effect,
369        );
370    }
371    Ok(permissions)
372}

374pub(crate) fn delegation_targets(
375    permissions: &IndexMap<PermissionAction, PermissionValue>,
376    known_agents: &BTreeMap<AgentId, ResolvedAgent>,
377) -> Vec<AgentId> {
378    let eligible_matches = |pattern: &str| -> Vec<AgentId> {
379        known_agents
380            .iter()
381            .filter(move |(id, agent)| {
382                // Only eligible subagents become targets; runtime
383                // spawn enforces the same eligibility again.
384                agent.document.frontmatter.enabled
385                    && matches!(
386                        agent.document.frontmatter.mode,
387                        cookie_agent_protocol::AgentMode::Subagent
388                            | cookie_agent_protocol::AgentMode::All
389                    )
390                    && cookie_agent_config::simple_wildcard_match(pattern, id.as_str())
391            })
392            .map(|(id, _)| id.clone())
393            .collect()
394    };
395    let mut targets = Vec::new();
396    match permissions.get(&PermissionAction::Delegate) {
397        // Scalar effects are documented as equivalent to a `"*"` mapping:
398        // expand them over every eligible subagent like any wildcard rule.
399        Some(PermissionValue::Effect(effect)) if *effect != PermissionEffect::Deny => {
400            targets.extend(eligible_matches("*"));
401        }
402        Some(PermissionValue::Resources(resources)) => {
403            for (resource, effect) in resources {
404                if *effect == PermissionEffect::Deny {
405                    continue;
406                }
407                let pattern = resource.as_str();
408                if pattern.contains('*') || pattern.contains('?') {
409                    // Wildcard delegation uses the same matching as every
410                    // other permission: expand against the known agent set.
411                    targets.extend(eligible_matches(pattern));
412                } else if let Ok(target) = AgentId::new(pattern) {
413                    targets.push(target);
414                }
415            }
416        }
417        _ => {}
418    }
419    targets.sort();
420    targets.dedup();
421    targets
422}

424fn selection_available(models: &CompiledModelRuntime, selection: &ModelSelection) -> bool {
425    models.model(&selection.model).is_some_and(|model| {
426        model.model.status == CompiledModelStatus::Available
427            && selection
428                .variant
429                .as_ref()
430                .is_none_or(|variant| model.model.variants.contains_key(variant))
431    })
432}

509    #[test]
510    fn wildcard_delegation_expands_to_matching_eligible_subagents() {
511        let mut known = BTreeMap::new();
512        for (id, mode, enabled) in [
513            ("sub-worker", AgentMode::Subagent, true),
514            ("sub-disabled", AgentMode::Subagent, false),
515            ("sub-primary", AgentMode::Primary, true),
516            ("explore", AgentMode::Subagent, true),
517        ] {
518            let agent = test_agent(id, mode, enabled);
519            known.insert(agent.document.id.clone(), agent);
520        }
521        let mut resources = IndexMap::new();
522        resources.insert(
523            WildcardPattern::new("sub-*").unwrap(),
524            PermissionEffect::Allow,
525        );
526        resources.insert(
527            WildcardPattern::new("explore").unwrap(),
528            PermissionEffect::Allow,
529        );
530        let permissions = IndexMap::from([(
531            PermissionAction::Delegate,
532            PermissionValue::Resources(resources),
533        )]);
534        // Disabled and non-subagent matches are excluded; concrete and
535        // wildcard targets merge and dedupe.
536        assert_eq!(
537            delegation_targets(&permissions, &known),
538            [
539                cookie_agent_protocol::AgentId::new("explore").unwrap(),
540                cookie_agent_protocol::AgentId::new("sub-worker").unwrap(),
541            ]
542        );
543
544        // A scalar `delegate: allow` is equivalent to a `"*"` mapping and
545        // expands over every eligible subagent.
546        let scalar = IndexMap::from([(
547            PermissionAction::Delegate,
548            PermissionValue::Effect(PermissionEffect::Allow),
549        )]);
550        assert_eq!(
551            delegation_targets(&scalar, &known),
552            [
553                cookie_agent_protocol::AgentId::new("explore").unwrap(),
554                cookie_agent_protocol::AgentId::new("sub-worker").unwrap(),
555            ]
556        );
557        let scalar_deny = IndexMap::from([(
558            PermissionAction::Delegate,
559            PermissionValue::Effect(PermissionEffect::Deny),
560        )]);
561        assert!(delegation_targets(&scalar_deny, &known).is_empty());
562    }
