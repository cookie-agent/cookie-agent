43impl AgentRegistry {
44    pub(crate) fn resolve(
45        authored: &ConfigAgentRegistry,
46        models: &CompiledModelRuntime,
47        preset: Option<String>,
48    ) -> Result<Self, EngineError> {
49        let mut documents = built_in_internal_documents()?;
50        documents.extend(
51            authored
52                .documents()
53                .iter()
54                .map(|(id, document)| (id.clone(), document.clone())),
55        );
56        let mut agents = BTreeMap::new();
57        for document in documents.into_values() {
58            let id = document.id.clone();
59            if id.as_str() == BUILT_IN_DEFAULT_AGENT_ID {
60                return Err(EngineError::RuntimeCompileFailed);
61            }
62            let mut resolved_fallback = Vec::new();
63            for fallback in &document.frontmatter.models {
64                match &fallback.model {
65                    AgentModelRef::ParentModel => {
66                        resolved_fallback.push(ResolvedAgentFallback::ParentModel {
67                            cache: fallback.cache.clone(),
68                        });
69                    }
70                    AgentModelRef::Model(model_key) => {
71                        let variant = match (&fallback.variant, models.model(model_key)) {
72                            (None, Some(model)) => model.model.default_variant.clone(),
73                            (Some(cookie_agent_identity::ConfiguredVariantRef::Base), _) => None,
74                            (Some(cookie_agent_identity::ConfiguredVariantRef::Named(id)), _) => {
75                                Some(id.clone())
76                            }
77                            (None, None) => None,
78                        };
79                        resolved_fallback.push(ResolvedAgentFallback::Selection {
80                            selection: ModelSelection {
81                                model: model_key.clone(),
82                                variant,
83                            },
84                            cache: fallback.cache.clone(),
85                        });
86                    }
87                }
88            }
89            let available = resolved_fallback
90                .iter()
91                .filter_map(|fallback| match fallback {
92                    ResolvedAgentFallback::Selection { selection, .. } => Some(selection),
93                    ResolvedAgentFallback::ParentModel { .. } => None,
94                })
95                .any(|selection| selection_available(models, selection));
96            let runnable_as_root = document.frontmatter.enabled
97                && matches!(
98                    document.frontmatter.mode,
99                    AgentMode::Primary | AgentMode::All
100                )
101                && available;
102            agents.insert(
103                id,
104                ResolvedAgent {
105                    document,
106                    resolved_fallback,
107                    runnable_as_root,
108                },
109            );
110        }
111        if !agents.values().any(|agent| agent.runnable_as_root)
112            && let Some(selection) = first_available_selection(models)
113        {
114            let document = built_in_default_document(&selection)?;
115            agents.insert(
116                document.id.clone(),
117                ResolvedAgent {
118                    document,
119                    resolved_fallback: vec![ResolvedAgentFallback::Selection {
120                        selection,
121                        cache: None,
122                    }],
123                    runnable_as_root: true,
124                },
125            );
126        }
127        let descriptors = agents
128            .iter()
129            .map(|(id, agent)| AgentDescriptor {
130                id: id.clone(),
131                preset: preset.clone(),
132                description: agent.document.frontmatter.description.clone(),
133                mode: agent.document.frontmatter.mode,
134                enabled: agent.document.frontmatter.enabled,
135                runnable_as_root: agent.runnable_as_root,
136                resolved_fallback: agent
137                    .resolved_fallback
138                    .iter()
139                    .filter_map(|fallback| match fallback {
140                        ResolvedAgentFallback::Selection { selection, .. } => {
141                            Some(selection.clone())
142                        }
143                        ResolvedAgentFallback::ParentModel { .. } => None,
144                    })
145                    .collect(),
146                delegation_targets: delegation_targets(&agent.document.frontmatter.permissions),
147            })
148            .collect();
149        Ok(Self {
150            preset,
151            agents,
152            descriptors,
153        })
154    }

169fn first_available_selection(models: &CompiledModelRuntime) -> Option<ModelSelection> {
170    models.models().values().find_map(|model| {
171        (model.model.status == CompiledModelStatus::Available).then(|| ModelSelection {
172            model: model.key.clone(),
173            variant: model.model.default_variant.clone(),
174        })
175    })
176}

216fn built_in_internal_document(
217    id: &str,
218    description: &str,
219    body: &str,
220    limits: AgentLimits,
221) -> Result<AgentDocument, EngineError> {
222    let id = IdentityAgentId::new(id).map_err(|_| EngineError::RuntimeCompileFailed)?;
223    let body = body.to_owned();
224    let frontmatter = AgentFrontmatter {
225        description: description.to_owned(),
226        mode: AgentMode::Internal,
227        enabled: true,
228        models: vec![AgentModelFallback {
229            model: AgentModelRef::ParentModel,
230            variant: None,
231            cache: None,
232        }],
233        limits,
234        permissions: IndexMap::new(),
235    };
236    let document_fingerprint = fingerprint(
237        "cookie-agent/built-in-internal-agent-document/v2",
238        &(id.as_str(), &frontmatter, &body),
239    )?;
240    let prompt_fingerprint = fingerprint("cookie-agent/system-prompt/v1", &body)?;
241    Ok(AgentDocument {
242        id,
243        frontmatter,
244        body,
245        source: AgentDocumentSource::BuiltIn,
246        document_fingerprint,
247        prompt_fingerprint,
248    })
249}

251fn built_in_default_document(selection: &ModelSelection) -> Result<AgentDocument, EngineError> {
252    let id = IdentityAgentId::new(BUILT_IN_DEFAULT_AGENT_ID)
253        .map_err(|_| EngineError::RuntimeCompileFailed)?;
254    let body = "You are Cookie Agent's built-in default coding agent. Help the user inspect, understand, and modify software safely and precisely.\n".to_owned();
255    let frontmatter = AgentFrontmatter {
256        description: "Built-in default coding agent".to_owned(),
257        mode: AgentMode::Primary,
258        enabled: true,
259        models: vec![AgentModelFallback {
260            model: AgentModelRef::Model(selection.model.clone()),
261            variant: selection
262                .variant
263                .clone()
264                .map(cookie_agent_identity::ConfiguredVariantRef::Named),
265            cache: None,
266        }],
267        limits: AgentLimits::default(),
268        permissions: built_in_default_permissions()?,
269    };
270    let document_fingerprint = fingerprint(
271        "cookie-agent/built-in-default-document/v2",
272        &(id.as_str(), &frontmatter, &body),
273    )?;
274    let prompt_fingerprint = fingerprint("cookie-agent/system-prompt/v1", &body)?;
275    Ok(AgentDocument {
276        id,
277        frontmatter,
278        body,
279        source: AgentDocumentSource::BuiltIn,
280        document_fingerprint,
281        prompt_fingerprint,
282    })
283}

285fn fingerprint(
286    domain: &str,
287    value: &impl Serialize,
288) -> Result<cookie_agent_models::Sha256Digest, EngineError> {
289    let mut hasher = Sha256::new();
290    hasher.update(domain.as_bytes());
291    hasher.update([0]);
292    hasher.update(serde_json::to_vec(value).map_err(|_| EngineError::RuntimeCompileFailed)?);
293    cookie_agent_models::Sha256Digest::new(format!("{:x}", hasher.finalize()))
294        .map_err(|_| EngineError::RuntimeCompileFailed)
295}

297fn built_in_default_permissions() -> Result<IndexMap<PermissionAction, PermissionValue>, EngineError>
298{
299    let definitions = [
300        (
301            PermissionAction::Read,
302            "tool_result:*",
303            PermissionEffect::Allow,
304        ),
305        (PermissionAction::Read, "*?*", PermissionEffect::Allow),
306        (PermissionAction::Write, "*", PermissionEffect::Ask),
307        (PermissionAction::Bash, "*", PermissionEffect::Ask),
308        (PermissionAction::Delegate, "*", PermissionEffect::Ask),
309        (PermissionAction::Read, "/*", PermissionEffect::Ask),
310        (PermissionAction::Read, ".env", PermissionEffect::Deny),
311        (PermissionAction::Read, "*/.env", PermissionEffect::Deny),
312        (PermissionAction::Read, ".env.*", PermissionEffect::Deny),
313        (PermissionAction::Read, "*/.env.*", PermissionEffect::Deny),
314        (
315            PermissionAction::Read,
316            ".env.example",
317            PermissionEffect::Allow,
318        ),
319        (
320            PermissionAction::Read,
321            "*/.env.example",
322            PermissionEffect::Allow,
323        ),
324        (
325            PermissionAction::Read,
326            "store-v3.json",
327            PermissionEffect::Deny,
328        ),
329        (
330            PermissionAction::Read,
331            "*/store-v3.json",
332            PermissionEffect::Deny,
333        ),
334        (PermissionAction::Read, "token-v1", PermissionEffect::Deny),
335        (PermissionAction::Read, "*/token-v1", PermissionEffect::Deny),
336        (PermissionAction::Read, "id_*", PermissionEffect::Deny),
337        (PermissionAction::Read, "*/id_*", PermissionEffect::Deny),
338        (PermissionAction::Read, ".netrc", PermissionEffect::Deny),
339        (PermissionAction::Read, "*/.netrc", PermissionEffect::Deny),
340        (
341            PermissionAction::Read,
342            "application_default_credentials.json",
343            PermissionEffect::Deny,
344        ),
345        (
346            PermissionAction::Read,
347            "*/application_default_credentials.json",
348            PermissionEffect::Deny,
349        ),
350    ];
351    let mut permissions = IndexMap::new();
352    for (action, resource, effect) in definitions {
353        let entry = permissions
354            .entry(action)
355            .or_insert_with(|| PermissionValue::Resources(IndexMap::new()));
356        let PermissionValue::Resources(resources) = entry else {
357            unreachable!("synthetic permissions use map form")
358        };
359        resources.insert(
360            WildcardPattern::new(resource).map_err(|_| EngineError::RuntimeCompileFailed)?,
361            effect,
362        );
363    }
364    Ok(permissions)
365}

367pub(crate) fn delegation_targets(
368    permissions: &IndexMap<PermissionAction, PermissionValue>,
369) -> Vec<AgentId> {
370    let Some(PermissionValue::Resources(resources)) = permissions.get(&PermissionAction::Delegate)
371    else {
372        return Vec::new();
373    };
374    let mut targets = resources
375        .iter()
376        .filter(|(_, effect)| **effect != PermissionEffect::Deny)
377        .filter_map(|(resource, _)| AgentId::new(resource.as_str()).ok())
378        .collect::<Vec<_>>();
379    targets.sort();
380    targets
381}

383fn selection_available(models: &CompiledModelRuntime, selection: &ModelSelection) -> bool {
384    models.model(&selection.model).is_some_and(|model| {
385        model.model.status == CompiledModelStatus::Available
386            && selection
387                .variant
388                .as_ref()
389                .is_none_or(|variant| model.model.variants.contains_key(variant))
390    })
391}

395    use cookie_agent_config::{PermissionAction, PermissionEffect, PermissionValue};
396    use cookie_agent_identity::WildcardPattern;
397    use indexmap::IndexMap;
398
399    use super::delegation_targets;

401    #[test]
402    fn delegation_targets_require_named_non_deny_permissions() {
403        assert!(delegation_targets(&IndexMap::new()).is_empty());
404
405        let mut resources = IndexMap::new();
406        resources.insert(WildcardPattern::new("*").unwrap(), PermissionEffect::Deny);
407        resources.insert(
408            WildcardPattern::new("reviewer").unwrap(),
409            PermissionEffect::Ask,
410        );
411        let permissions = IndexMap::from([(
412            PermissionAction::Delegate,
413            PermissionValue::Resources(resources),
414        )]);
415        assert_eq!(
416            delegation_targets(&permissions),
417            [cookie_agent_protocol::AgentId::new("reviewer").unwrap()]
418        );
