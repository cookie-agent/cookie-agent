17pub(crate) fn build_runtime_snapshot(
18    models: &CompiledModelRuntime,
19    agents: &AgentRegistry,
20    agent_presets: &BTreeMap<String, std::sync::Arc<AgentRegistry>>,
21) -> Result<protocol::RuntimeSnapshotV1, EngineError> {
22    let providers = models
23        .providers()
24        .iter()
25        .map(|provider| provider_descriptor(models, provider))
26        .collect::<Result<Vec<_>, _>>()?;
27    let available_models = models
28        .models()
29        .values()
30        .filter(|model| model.model.status == CompiledModelStatus::Available)
31        .map(model_descriptor)
32        .collect::<Result<Vec<_>, _>>()?;
33    let agent_descriptors = agents
34        .descriptors()
35        .iter()
36        .cloned()
37        .chain(
38            agent_presets
39                .values()
40                .flat_map(|registry| registry.descriptors().iter().cloned()),
41        )
42        .collect::<Vec<_>>();
43    let agent_revision = revision::<protocol::AgentRevision, _>(
44        "cookie-agent/agent-runtime/v1",
45        &agent_descriptors,
46        protocol::AgentRevision::new,
47    )?;
48    let runtime_revision = runtime_revision(
49        &family_registry().revision(),
50        &models.catalog().revision,
51        &models.provider_state_revision(),
52        models.model_revision(),
53        &agent_revision,
54    )?;
55    let catalog = models.catalog();
56    let quarantine = quarantine_summary(models)?;
57    let last_error = catalog
58        .state
59        .last_error
60        .as_ref()
61        .map(|error| {
62            Ok::<_, EngineError>(protocol::CatalogSafeErrorMeta {
63                code: protocol::SafeCode::new(error.code.clone())
64                    .map_err(|_| EngineError::RuntimeCompileFailed)?,
65                message: protocol::SafeErrorMessage::new(error.safe_message.clone())
66                    .map_err(|_| EngineError::RuntimeCompileFailed)?,
67                time: error.occurred_at,
68            })
69        })
70        .transpose()?;
71    let snapshot = protocol::RuntimeSnapshotV1 {
72        snapshot_schema_version: protocol::RuntimeSnapshotSchemaVersion::current(),
73        recipe_registry_revision: family_registry().revision(),
74        catalog_revision: catalog.revision.clone(),
75        catalog_source: match catalog.source {
76            cookie_agent_models::catalog::CatalogSource::Network => {
77                protocol::CatalogSource::Network
78            }
79            cookie_agent_models::catalog::CatalogSource::Cache => protocol::CatalogSource::Cache,
80            cookie_agent_models::catalog::CatalogSource::Bootstrap => {
81                protocol::CatalogSource::Bootstrap
82            }
83        },
84        catalog_state: protocol::CatalogRuntimeState {
85            stale: !matches!(
86                catalog.state.availability,
87                cookie_agent_models::catalog::CatalogAvailability::Ready
88            ),
89            provider_quarantine_count: quarantine.provider_count,
90            model_quarantine_count: quarantine.model_count,
91            quarantine_digest: quarantine.digest,
92            last_error,
93        },
94        provider_state_revision: models.provider_state_revision(),
95        provider_store_generation: protocol::ProviderStoreGeneration::new(
96            models.store().generation().get(),
97        )
98        .map_err(|_| EngineError::RuntimeCompileFailed)?,
99        model_revision: models.model_revision().clone(),
100        agent_revision,
101        runtime_revision,
102        providers,
103        models: available_models,
104        agents: agent_descriptors,
105    };
106    snapshot
107        .validate()
108        .map_err(|_| EngineError::RuntimeCompileFailed)?;
109    Ok(snapshot)
110}

132fn quarantine_summary(
133    runtime: &CompiledModelRuntime,
134) -> Result<RuntimeQuarantineSummary, EngineError> {
135    let catalog = runtime.catalog();
136    let entries = catalog
137        .quarantine
138        .iter()
139        .map(|entry| RuntimeQuarantineEntry {
140            provider_id: entry.provider_id.clone(),
141            model_id: entry.model_id.clone(),
142            canonical_model_id: entry.canonical_model_id.clone(),
143            reason: RuntimeQuarantineReason::Parser(entry.reason.clone()),
144        })
145        .collect::<BTreeSet<_>>();
146    let provider_count = entries
147        .iter()
148        .filter(|entry| entry.model_id.is_none() && entry.canonical_model_id.is_none())
149        .count() as u32;
150    let model_count = entries.len() as u32 - provider_count;
151    let digest = protocol::Sha256Digest::new(hash_bytes(
152        "cookie-agent/runtime-quarantine/v1",
153        &serde_json::to_vec(&entries).map_err(|_| EngineError::RuntimeCompileFailed)?,
154    ))
155    .map_err(|_| EngineError::RuntimeCompileFailed)?;
156    Ok(RuntimeQuarantineSummary {
157        provider_count,
158        model_count,
159        digest,
160    })
161}

183fn provider_descriptor(
184    runtime: &CompiledModelRuntime,
185    provider: &cookie_agent_models::CompiledProviderState,
186) -> Result<protocol::ProviderDescriptor, EngineError> {
187    let catalog_entry = runtime.catalog().provider(&provider.id);
188    let quarantined = catalog_entry.is_some_and(|entry| entry.quarantine.is_some());
189    let quarantine_code = catalog_entry
190        .and_then(|entry| entry.quarantine.as_ref())
191        .map_or_else(
192            || "invalid_catalog_provider_record".to_owned(),
193            |reason| reason.code().to_owned(),
194        );
195    let support_reason = provider
196        .support_reason
197        .as_deref()
198        .map(safe_code)
199        .transpose()?;
200    let support = if quarantined {
201        protocol::ProviderSupport {
202            state: protocol::ProviderSupportState::Quarantined,
203            reason: Some(safe_code(&quarantine_code)?),
204        }
205    } else if provider.retained_family_match == Some(RetainedFamilyMatch::SupportedRemoved) {
206        protocol::ProviderSupport {
207            state: protocol::ProviderSupportState::Supported,
208            reason: None,
209        }
210    } else if provider.retained_family_match
211        == Some(RetainedFamilyMatch::RemovedWithoutRetainedFamilyMatch)
212    {
213        protocol::ProviderSupport {
214            state: protocol::ProviderSupportState::Unsupported,
215            reason: Some(safe_code("removed_without_retained_recipe_match")?),
216        }
217    } else if let Some(reason) = support_reason {
218        protocol::ProviderSupport {
219            state: protocol::ProviderSupportState::Unsupported,
220            reason: Some(reason),
221        }
222    } else {
223        protocol::ProviderSupport {
224            state: protocol::ProviderSupportState::Supported,
225            reason: None,
226        }
227    };
228    let record = catalog_entry.and_then(|entry| entry.record.as_ref());
229    let recipe = record
230        .and_then(|record| family_registry().classify(record))
231        .or_else(|| {
232            runtime
233                .store()
234                .provider(&provider.id)
235                .and_then(|connection| {
236                    family_registry().by_npm(connection.policy.package_claim.as_str())
237                })
238        });
239    let mut setup_ids = BTreeMap::<String, bool>::new();
240    let mut add_template = |template: &str| {
241        for name in placeholders(template) {
242            let secret = placeholder_is_secret(&name);
243            setup_ids
244                .entry(setup_field_name(&name))
245                .and_modify(|value| *value |= secret)
246                .or_insert(secret);
247        }
248    };
249    if let Some(record) = record {
250        if let Some(api) = record.api.as_deref() {
251            add_template(api);
252        }
253        for model in record
254            .models
255            .values()
256            .filter_map(|entry| entry.record.as_ref())
257        {
258            if let Some(api) = model
259                .provider
260                .as_ref()
261                .and_then(|provider| provider.api.as_deref())
262            {
263                add_template(api);
264            }
265        }
266    }
267    if let Some(recipe) = recipe {
268        match recipe.family {
269            cookie_agent_models::recipes::FamilyKind::Vertex
270            | cookie_agent_models::recipes::FamilyKind::VertexAnthropic => {
271                setup_ids.insert("project".to_owned(), false);
272                setup_ids.insert("location".to_owned(), false);
273            }
274            cookie_agent_models::recipes::FamilyKind::Bedrock => {
275                setup_ids.insert("region".to_owned(), false);
276            }
277            cookie_agent_models::recipes::FamilyKind::Azure => {
278                setup_ids.insert("resource_name".to_owned(), false);
279            }
280            _ => {}
281        }
282    }
283    let mut setup_fields = setup_ids
284        .iter()
285        .map(|(id, secret)| setup_descriptor(id, *secret))
286        .collect::<Result<Vec<_>, EngineError>>()?;
287    setup_fields.sort_by(|left, right| left.id.cmp(&right.id));
288    let mut auth_methods = if let Some(recipe) = recipe {
289        recipe
290            .allowed_auth_methods
291            .iter()
292            .filter_map(|id| auth_method(id))
293            .map(auth_descriptor)
294            .collect::<Result<Vec<_>, EngineError>>()?
295    } else {
296        Vec::new()
297    };
298    auth_methods.sort_by(|left, right| left.id.cmp(&right.id));
299    Ok(protocol::ProviderDescriptor {
300        id: provider.id.clone(),
301        display_name: protocol::SafeDisplayText::new(provider.display_name.clone())
302            .map_err(|_| EngineError::RuntimeCompileFailed)?,
303        presence: match provider.presence {
304            ModelProviderPresence::Current => protocol::ProviderPresence::Current,
305            ModelProviderPresence::Removed => protocol::ProviderPresence::Removed,
306        },
307        support,
308        setup_fields,
309        auth_methods,
310        configuration: match (provider.authored, provider.stored) {
311            (false, false) => protocol::ProviderConfigurationState::Unconfigured,
312            (true, false) => protocol::ProviderConfigurationState::Authored,
313            (false, true) => protocol::ProviderConfigurationState::Stored,
314            (true, true) => protocol::ProviderConfigurationState::AuthoredAndStored,
315        },
316        effective_auth_state: effective_auth_state(provider.effective_auth),
317        durable_connection: provider
318            .durable_connection
319            .as_ref()
320            .map(project_durable_connection)
321            .transpose()?,
322        quarantine: quarantined.then(|| protocol::QuarantineDiagnostic {
323            code: safe_code(&quarantine_code).expect("validated quarantine code is valid"),
324            message: protocol::SafeErrorMessage::new("catalog provider record is quarantined")
325                .expect("static quarantine message is valid"),
326        }),
327    })
328}

330fn placeholder_is_secret(name: &str) -> bool {
331    let name = name.to_ascii_uppercase();
332    name.contains("KEY") || name.contains("TOKEN") || name.contains("SECRET")
333}

335fn setup_descriptor(
336    field: &str,
337    secret: bool,
338) -> Result<protocol::SetupFieldDescriptor, EngineError> {
339    let id = protocol::SetupFieldId::new(field).map_err(|_| EngineError::RuntimeCompileFailed)?;
340    Ok(protocol::SetupFieldDescriptor {
341        id,
342        display_name: protocol::SafeDisplayText::new(field.replace('_', " "))
343            .map_err(|_| EngineError::RuntimeCompileFailed)?,
344        help: protocol::SafeDisplayText::new(format!("Provider setup field `{field}`"))
345            .map_err(|_| EngineError::RuntimeCompileFailed)?,
346        required: true,
347        default: None,
348        validation: protocol::SetupFieldValidation {
349            value_type: protocol::SetupFieldType::String,
350            min_length: Some(1),
351            max_length: Some(256),
352            minimum: None,
353            maximum: None,
354        },
355        safe_to_project: !secret,
356    })

359fn auth_descriptor(
360    method: &cookie_agent_models::recipes::AuthMethodRecipe,
361) -> Result<protocol::AuthMethodDescriptor, EngineError> {
362    let mut credentials = method
363        .credentials
364        .iter()
365        .map(|field| {
366            Ok(protocol::AuthCredentialDescriptor {
367                id: protocol::AuthFieldName::new(field.name)
368                    .map_err(|_| EngineError::RuntimeCompileFailed)?,
369                display_name: protocol::SafeDisplayText::new(field.name.replace('_', " "))
370                    .map_err(|_| EngineError::RuntimeCompileFailed)?,
371                help: protocol::SafeDisplayText::new(format!("Secret credential `{}`", field.name))
372                    .map_err(|_| EngineError::RuntimeCompileFailed)?,
373                required: field.required,
374                credential_type: match field.kind {
375                    CredentialKind::ApiKey => protocol::CredentialFieldType::ApiKey,
376                    CredentialKind::AccessToken => protocol::CredentialFieldType::AccessToken,
377                    CredentialKind::AccessKeyId => protocol::CredentialFieldType::AccessKeyId,
378                    CredentialKind::SecretAccessKey => {
379                        protocol::CredentialFieldType::SecretAccessKey
380                    }
381                    CredentialKind::SessionToken => protocol::CredentialFieldType::SessionToken,
382                },
383            })
384        })
385        .collect::<Result<Vec<_>, EngineError>>()?;
386    credentials.sort_by(|left, right| left.id.cmp(&right.id));
387    Ok(protocol::AuthMethodDescriptor {
388        id: protocol::AuthMethodId::new(method.id)
389            .map_err(|_| EngineError::RuntimeCompileFailed)?,
390        display_name: protocol::SafeDisplayText::new(method.id.replace('-', " "))
391            .map_err(|_| EngineError::RuntimeCompileFailed)?,
392        credentials,
393    })
394}

396fn model_descriptor(
397    model: &cookie_agent_models::CompiledRuntimeModel,
398) -> Result<protocol::AvailableModelDescriptor, EngineError> {
399    let mut capability_value = serde_json::to_value(&model.model.capabilities)
400        .map_err(|_| EngineError::RuntimeCompileFailed)?;
401    capability_value
402        .as_object_mut()
403        .ok_or(EngineError::RuntimeCompileFailed)?
404        .remove("compaction");
405    let capabilities =
406        serde_json::from_value(capability_value).map_err(|_| EngineError::RuntimeCompileFailed)?;
407    let variants = model
408        .model
409        .variants
410        .values()
411        .map(|variant| {
412            let fingerprint = protocol::Sha256Digest::new(hash(
413                "cookie-agent/model-variant/v1",
414                &(
415                    variant.id.clone(),
416                    &variant.defaults,
417                    &variant.options,
418                    &variant.reasoning,
419                ),
420            )?)
421            .map_err(|_| EngineError::RuntimeCompileFailed)?;
422            Ok(protocol::AvailableVariantDescriptor {
423                id: variant.id.clone(),
424                display_name: variant.display_name.clone(),
425                origin: match variant.origin {
426                    CompiledVariantOrigin::ModelsDevEffort => {
427                        protocol::VariantOrigin::ModelsDevEffort
428                    }
429                    CompiledVariantOrigin::ModelsDevToggle => {
430                        protocol::VariantOrigin::ModelsDevToggle
431                    }
432                    CompiledVariantOrigin::ModelsDevBudgetTokens => {
433                        protocol::VariantOrigin::ModelsDevBudgetTokens
434                    }
435                    CompiledVariantOrigin::Authored => protocol::VariantOrigin::Explicit,
436                },
437                behavior_fingerprint: fingerprint,
438            })
439        })
440        .collect::<Result<Vec<_>, EngineError>>()?;
441    Ok(protocol::AvailableModelDescriptor {
442        key: model.key.clone(),
443        display_name: model.model.display_name.clone(),
444        capabilities,
445        variants,
446        variant_order: model.model.variant_order.clone(),
447        default_variant: model.model.default_variant.clone(),
448        behavior_fingerprint: protocol::Sha256Digest::new(
449            model.model.behavior_fingerprint.as_str(),
450        )
451        .map_err(|_| EngineError::RuntimeCompileFailed)?,
452    })
453}

455pub(crate) fn project_durable_connection(
456    value: &cookie_agent_models::provider_store::DurableConnectionDescriptor,
457) -> Result<protocol::DurableConnectionDescriptor, EngineError> {
458    Ok(protocol::DurableConnectionDescriptor {
459        provider_id: value.provider_id.clone(),
460        setup_values: value
461            .setup_values
462            .iter()
463            .map(|(id, value)| {
464                let value = serde_json::from_value(
465                    serde_json::to_value(value).map_err(|_| EngineError::RuntimeCompileFailed)?,
466                )
467                .map_err(|_| EngineError::RuntimeCompileFailed)?;
468                Ok((id.clone(), value))
469            })
470            .collect::<Result<BTreeMap<_, _>, EngineError>>()?,
471        setup_fingerprint: protocol::Sha256Digest::new(value.setup_fingerprint.as_str())
472            .map_err(|_| EngineError::RuntimeCompileFailed)?,
473        recipe_fingerprint: protocol::Sha256Digest::new(value.recipe_fingerprint.as_str())
474            .map_err(|_| EngineError::RuntimeCompileFailed)?,
475        auth_method: value.auth_method.clone(),
476        credential_fields: value.credential_fields.clone(),
477        connection_generation: protocol::ProviderConnectionGeneration::new(
478            value.connection_generation.get(),
479        )
480        .map_err(|_| EngineError::RuntimeCompileFailed)?,
481        connected_at: value.connected_at,
482    })

517fn safe_code(value: &str) -> Result<protocol::SafeCode, EngineError> {
518    let normalized = value
519        .chars()
520        .map(|character| {
521            if character.is_ascii_alphanumeric() || character == '_' || character == '-' {
522                character.to_ascii_lowercase()
523            } else {
524                '_'
525            }
526        })
527        .collect::<String>();
528    protocol::SafeCode::new(normalized).map_err(|_| EngineError::RuntimeCompileFailed)
529}

531fn revision<T, E>(
532    domain: &str,
533    value: &impl Serialize,
534    constructor: impl FnOnce(String) -> Result<T, E>,
535) -> Result<T, EngineError> {
536    constructor(format!("sha256:{}", hash(domain, value)?))
537        .map_err(|_| EngineError::RuntimeCompileFailed)
538}
539
540fn hash(domain: &str, value: &impl Serialize) -> Result<String, EngineError> {
541    let bytes = serde_json::to_vec(value).map_err(|_| EngineError::RuntimeCompileFailed)?;
542    Ok(hash_bytes(domain, &bytes))
543}
544
545fn hash_bytes(domain: &str, bytes: &[u8]) -> String {
546    let mut hasher = Sha256::new();
547    hasher.update(domain.as_bytes());
548    hasher.update([0]);
549    hasher.update(bytes);
550    format!("{:x}", hasher.finalize())
551}
