37pub(crate) fn checkpoint_retained_history(
38    history: &[HistoryTurn],
39    events: &[StoredEvent],
40    summary: Option<&str>,
41) -> Vec<HistoryTurn> {
42    let pinned_skills = events
43        .iter()
44        .filter(|event| matches!(event.payload, EventPayload::SkillLoaded { .. }))
45        .count();
46    let pinned_agent_md = usize::from(latest_agent_md_event(events).is_some());
47    let pinned_count = pinned_agent_md.saturating_add(pinned_skills);
48    let mut history = history.iter();
49    let mut retained = history.next().cloned().into_iter().collect::<Vec<_>>();
50    retained.extend(
51        history
52            .filter(|turn| !is_framed_summary_turn(turn))
53            .take(pinned_count)
54            .cloned(),
55    );
56    if let Some(summary) = summary {
57        retained.push(HistoryTurn::user(user_text(&framed_compaction_summary(
58            summary,
59        ))));
60    }
61    retained
62}

64fn is_framed_summary_turn(turn: &HistoryTurn) -> bool {
65    matches!(
66        turn,
67        HistoryTurn::User(message)
68            if matches!(message.content.as_slice(), [InputPart::Text(text)]
69                if text.text.starts_with(COMPACTION_SUMMARY_PREFIX)
70                    && text.text.ends_with(COMPACTION_SUMMARY_SUFFIX))
71    )

148pub(crate) fn replay_decisions(
149    decisions: &[OvenReplayDecision],
150    binding: &FrozenModelBinding,
151) -> Vec<ReplayDecision> {
152    decisions
153        .iter()
154        .map(|decision| ReplayDecision {
155            history_index: decision.history_index as u64,
156            disposition: match &decision.disposition {
157                OvenReplayDisposition::Replayed => ReplayDisposition::Replayed,
158                OvenReplayDisposition::NoArtifact => ReplayDisposition::NoArtifact,
159                OvenReplayDisposition::DiscardedForeignAdapter { found, expected } => {
160                    ReplayDisposition::DiscardedForeignAdapter {
161                        found: exact_adapter_code(found.as_str()),
162                        expected: exact_adapter_code(expected.as_str()),
163                    }
164                }
165                OvenReplayDisposition::DiscardedForeignScope { found, expected } => {
166                    let found_selection = scope_selection(found);
167                    let expected_selection = scope_selection(expected);
168                    if found_selection.model == expected_selection.model {
169                        ReplayDisposition::DiscardedForeignVariant {
170                            found: found_selection.variant,
171                            expected: binding.selection.variant.clone(),
172                        }
173                    } else {
174                        ReplayDisposition::DiscardedForeignModelSelection {
175                            found: found_selection,
176                            expected: binding.selection.clone(),
177                        }
178                    }
179                }
180                OvenReplayDisposition::DiscardedInvalidPayload { reason } => {
181                    ReplayDisposition::DiscardedInvalidPayload {
182                        reason: SafeErrorMessage::new(sanitize_control_free(
183                            reason,
184                            SafeErrorMessage::MAX_BYTES,
185                        ))
186                        .expect("sanitized safe replay error"),
187                    }
188                }
189                OvenReplayDisposition::ReconstructedNormalized => {
190                    ReplayDisposition::ReconstructedNormalizedHistory
191                }
192            },
193        })
194        .collect()
195}

197pub(crate) fn replay_decisions_with_preflight(
198    decisions: &[OvenReplayDecision],
199    binding: &FrozenModelBinding,
200    preflight: &[ReplayDecision],
201) -> Vec<ReplayDecision> {
202    let preflight = preflight
203        .iter()
204        .map(|decision| (decision.history_index, decision.disposition.clone()))
205        .collect::<HashMap<_, _>>();
206    let mut emitted = HashSet::new();
207    let mut merged = Vec::new();
208    for decision in replay_decisions(decisions, binding) {
209        let Some(disposition) = preflight.get(&decision.history_index) else {
210            merged.push(decision);
211            continue;
212        };
213        if emitted.insert(decision.history_index) {
214            merged.push(ReplayDecision {
215                history_index: decision.history_index,
216                disposition: disposition.clone(),
217            });
218        }
219        if !matches!(decision.disposition, ReplayDisposition::NoArtifact) {
220            merged.push(decision);
221        }
222    }
223    for (history_index, disposition) in preflight {
224        if emitted.insert(history_index) {
225            merged.push(ReplayDecision {
226                history_index,
227                disposition,
228            });
229            merged.push(ReplayDecision {
230                history_index,
231                disposition: ReplayDisposition::ReconstructedNormalizedHistory,
232            });
233        } else if !merged.iter().any(|decision| {
234            decision.history_index == history_index
235                && matches!(
236                    decision.disposition,
237                    ReplayDisposition::ReconstructedNormalizedHistory
238                )
239        }) {
240            merged.push(ReplayDecision {
241                history_index,
242                disposition: ReplayDisposition::ReconstructedNormalizedHistory,
243            });
244        }
245    }
246    merged.sort_by_key(|decision| decision.history_index);
247    merged
248}

288fn native_replay_fingerprint(
289    adapter_id: &str,
290    provider_id: &str,
291    model_id: &str,
292    resource_id: &str,
293    payload: &serde_json::Value,
294) -> Sha256Digest {
295    let mut bytes = Vec::with_capacity(
296        adapter_id.len() + provider_id.len() + model_id.len() + resource_id.len() + 4,
297    );
298    for component in [adapter_id, provider_id, model_id, resource_id] {
299        bytes.extend_from_slice(component.as_bytes());
300        bytes.push(0);
301    }
302    serde_json::to_writer(&mut bytes, payload).expect("JSON value serializes");
303    Sha256Digest::of_bytes(&bytes)
304}

335fn rejected_unsigned_replay_fingerprints(events: &[StoredEvent]) -> HashSet<String> {
336    let abandoned = events
337        .iter()
338        .filter_map(|event| match event.payload {
339            EventPayload::AttemptAbandoned { attempt_id } => Some(attempt_id),
340            _ => None,
341        })
342        .collect::<HashSet<_>>();
343    events
344        .iter()
345        .filter_map(|event| match &event.payload {
346            EventPayload::ModelReplayEvaluated {
347                attempt_id,
348                ordered_decisions,
349                ..
350            } if abandoned.contains(attempt_id) => Some(ordered_decisions),
351            _ => None,
352        })
353        .flatten()
354        .filter_map(|decision| match &decision.disposition {
355            ReplayDisposition::DiscardedInvalidPayload { reason } => reason
356                .as_str()
357                .strip_prefix(REJECTED_UNSIGNED_REPLAY_PREFIX)
358                .and_then(|value| value.split_once(';'))
359                .map(|(fingerprint, _)| fingerprint.to_owned()),
360            _ => None,
361        })
362        .collect()
363}

365fn unsigned_anthropic_replay(artifact: &OvenReplayArtifact) -> bool {
366    if crate::policy::wire_adapter(artifact.adapter_id().as_str())
367        != cookie_agent_protocol::AdaptorId::Anthropic
368    {
369        return false;
370    }
371    artifact
372        .payload()
373        .pointer("/message/content")
374        .and_then(serde_json::Value::as_array)
375        .is_some_and(|content| {
376            content.iter().any(|block| {
377                block.get("type").and_then(serde_json::Value::as_str) == Some("thinking")
378                    && match block.get("signature") {
379                        None => true,
380                        Some(signature) => signature.as_str() == Some(""),
381                    }
382            })
383        })
384}

437fn selected_checkpoint_events(
438    events: &[StoredEvent],
439    source_through_seq: u64,
440    recent_from_seq: Option<u64>,
441    close_dependencies: bool,
442) -> Vec<StoredEvent> {
443    let latest_agent_md_seq = latest_agent_md_event(events).map(|event| event.seq);
444    let mut selected = events
445        .iter()
446        .filter(|event| {
447            is_pinned_event(event, latest_agent_md_seq, source_through_seq)
448                || recent_from_seq
449                    .is_some_and(|from| (from..=source_through_seq).contains(&event.seq))
450                || event.seq > source_through_seq
451        })
452        .cloned()
453        .collect::<Vec<_>>();
454    if close_dependencies {
455        close_event_dependencies(events, &mut selected);
456    }
457    selected.sort_by_key(|event| event.seq);
458    selected.dedup_by_key(|event| event.seq);
459    selected
460}

462fn close_event_dependencies(events: &[StoredEvent], selected: &mut Vec<StoredEvent>) {
463    // Applied inputs and late tool completions can refer to events before the retained range.
464    loop {
465        let selected_seqs = selected
466            .iter()
467            .map(|event| event.seq)
468            .collect::<HashSet<_>>();
469        let mut required_seqs = HashSet::new();
470        for event in selected.iter() {
471            match &event.payload {
472                EventPayload::UserInputApplied { user_input_seq } => {
473                    required_seqs.insert(*user_input_seq);
474                }
475                EventPayload::ToolCallStarted { start } => {
476                    if let Some(commit) = events.iter().rev().find(|candidate| {
477                        candidate.seq <= event.seq
478                            && candidate.run_id == event.run_id
479                            && matches!(&candidate.payload,
480                                EventPayload::ModelTurnCommitted { model_turn_seq, turn, .. }
481                                    if *model_turn_seq == start.owner.model_turn_seq
482                                        && turn.content.iter().any(|part| matches!(part,
483                                            PersistedAssistantPart::ToolCall { id, .. }
484                                                if id == &start.owner.model_call_id)))
485                    }) {
486                        required_seqs.insert(commit.seq);
487                    }
488                }
489                EventPayload::ToolCallTerminated { termination } => {
490                    if let Some(start) = events.iter().rev().find(|candidate| {
491                        candidate.seq <= event.seq
492                            && candidate.run_id == event.run_id
493                            && matches!(&candidate.payload,
494                                EventPayload::ToolCallStarted { start }
495                                    if start.tool_call_id == termination.tool_call_id)
496                    }) {
497                        required_seqs.insert(start.seq);
498                    }
499                }
500                _ => {}
501            }
502        }
503        let additions = events
504            .iter()
505            .filter(|event| {
506                required_seqs.contains(&event.seq) && !selected_seqs.contains(&event.seq)
507            })
508            .cloned()
509            .collect::<Vec<_>>();
510        if additions.is_empty() {
511            break;
512        }
513        selected.extend(additions);
514    }
515}

525fn insert_summary(assembled: &mut AssembledHistory, events: &[StoredEvent], summary: &str) {
526    let history_index = 1 + pinned_history_len(events);
527    assembled.history.insert(
528        history_index,
529        HistoryTurn::user(user_text(&framed_compaction_summary(summary))),
530    );
531    for decision in &mut assembled.replay_decisions {

533            decision.history_index += 1;
534        }

558pub(crate) fn compaction_prefix_history(
559    events: &[StoredEvent],
560    store: &ArtifactStore,
561    binding: &FrozenModelBinding,
562    composed_prompt: &str,
563    recent_from_seq: Option<u64>,
564) -> Result<Vec<HistoryTurn>, HistoryError> {
565    let Some(recent_from_seq) = recent_from_seq else {
566        return Ok(assemble_model_context(events, store, binding, composed_prompt)?.history);
567    };
568    let (mut visible, prior_summary) = if let Some(commit) = latest_checkpoint(events) {
569        let retained_from = match commit.checkpoint {
570            ContextCheckpoint::InternalSummary { .. } => commit.boundaries.recent_from_seq,
571            ContextCheckpoint::NativeWindow { .. } => None,
572        };
573        let summary = match &commit.checkpoint {
574            ContextCheckpoint::InternalSummary { checkpoint } => Some(checkpoint.summary()),
575            ContextCheckpoint::NativeWindow { .. } => None,
576        };
577        (
578            selected_checkpoint_events(
579                events,
580                commit.boundaries.source_through_seq,
581                retained_from,
582                true,
583            ),
584            summary,
585        )
586    } else {
587        (events.to_vec(), None)
588    };
589    let latest_agent_md_seq = latest_agent_md_event(events).map(|event| event.seq);
590    visible.retain(|event| {
591        event.seq < recent_from_seq || is_pinned_event(event, latest_agent_md_seq, u64::MAX)
592    });
593    close_event_dependencies(events, &mut visible);
594    visible.sort_by_key(|event| event.seq);
595    visible.dedup_by_key(|event| event.seq);
596    let mut assembled =
597        assemble_history_with_replay(&visible, events, store, binding, composed_prompt)?;
598    if let Some(summary) = prior_summary {
599        insert_summary(&mut assembled, events, summary);
600    }
601    Ok(assembled.history)
602}

604pub(crate) fn compaction_tail_candidates(events: &[StoredEvent]) -> Vec<u64> {
605    #[derive(Default)]
606    struct Group {
607        start: u64,
608        end: u64,
609        unresolved_calls: usize,
610    }
611
612    let visible = if let Some(commit) = latest_checkpoint(events) {
613        let recent_from = match &commit.checkpoint {
614            ContextCheckpoint::InternalSummary { .. } => commit.boundaries.recent_from_seq,
615            ContextCheckpoint::NativeWindow { .. } => None,
616        };
617        selected_checkpoint_events(
618            events,
619            commit.boundaries.source_through_seq,
620            recent_from,
621            true,
622        )
623    } else {
624        events.to_vec()
625    };
626    let visible_seqs = visible
627        .iter()
628        .map(|event| event.seq)
629        .collect::<HashSet<_>>();
630    let submitted = events
631        .iter()
632        .filter_map(|event| match event.payload {
633            EventPayload::UserInputSubmitted { .. } => Some(event.seq),
634            _ => None,
635        })
636        .collect::<HashSet<_>>();
637    let mut groups = Vec::<Group>::new();
638    let mut pending = HashMap::<
639        (
640            cookie_agent_protocol::RunId,
641            cookie_agent_protocol::ModelCallId,
642        ),
643        VecDeque<usize>,
644    >::new();
645    let mut started = HashMap::<(cookie_agent_protocol::RunId, ToolCallId), usize>::new();
646    let mut terminated = HashSet::new();
647
648    for event in &visible {
649        match &event.payload {
650            EventPayload::UserInputApplied { user_input_seq }
651                if submitted.contains(user_input_seq) =>
652            {
653                groups.push(Group {
654                    start: *user_input_seq,
655                    end: event.seq,
656                    ..Group::default()
657                });
658            }
659            EventPayload::MessageInjected { role, .. }
660                if *role != cookie_agent_protocol::ExtensionMessageRole::Tool =>
661            {
662                groups.push(Group {
663                    start: event.seq,
664                    end: event.seq,
665                    ..Group::default()
666                });
667            }
668            EventPayload::DelegatedContextSeeded { turns, .. } if !turns.is_empty() => {
669                groups.push(Group {
670                    start: event.seq,
671                    end: event.seq,
672                    ..Group::default()
673                });
674            }
675            EventPayload::ModelTurnCommitted { turn, .. } => {
676                let in_stream_results = turn
677                    .content
678                    .iter()
679                    .filter_map(|part| match part {
680                        PersistedAssistantPart::ToolResult { tool_call_id, .. } => {
681                            Some(tool_call_id.as_str())
682                        }
683                        _ => None,
684                    })
685                    .collect::<HashSet<_>>();
686                let index = groups.len();
687                let mut unresolved_calls = 0;
688                if let Some(run_id) = event.run_id {
689                    for id in turn.content.iter().filter_map(|part| match part {
690                        PersistedAssistantPart::ToolCall { id, .. }
691                            if !in_stream_results.contains(id.as_str()) =>
692                        {
693                            Some(id)
694                        }
695                        _ => None,
696                    }) {
697                        unresolved_calls += 1;
698                        pending
699                            .entry((run_id, id.clone()))
700                            .or_default()
701                            .push_back(index);
702                    }
703                }
704                groups.push(Group {
705                    start: event.seq,
706                    end: event.seq,
707                    unresolved_calls,
708                });
709            }
710            EventPayload::ToolCallStarted { start } => {
711                let Some(run_id) = event.run_id else { continue };
712                if let Some(index) = pending
713                    .get_mut(&(run_id, start.owner.model_call_id.clone()))
714                    .and_then(VecDeque::pop_front)
715                {
716                    started.insert((run_id, start.tool_call_id), index);
717                }
718            }
719            EventPayload::ToolCallTerminated { termination } => {
720                let Some(run_id) = event.run_id else { continue };
721                if terminated.insert((run_id, termination.tool_call_id))
722                    && let Some(index) = started.get(&(run_id, termination.tool_call_id))
723                {
724                    groups[*index].unresolved_calls =
725                        groups[*index].unresolved_calls.saturating_sub(1);
726                    groups[*index].end = groups[*index].end.max(event.seq);
727                }
728            }
729            EventPayload::ContextRehydrated { .. }
730            | EventPayload::PluginEventAdded { .. }
731            | EventPayload::DelegateFinished { .. }
732            | EventPayload::DelegateFinishedV2 { .. } => groups.push(Group {
733                start: event.seq,
734                end: event.seq,
735                ..Group::default()
736            }),
737            _ => {}
738        }
739    }
740
741    groups.sort_by_key(|group| group.start);
742    groups.dedup_by_key(|group| group.start);
743    let unresolved_from = groups
744        .iter()
745        .filter(|group| group.unresolved_calls != 0)
746        .map(|group| group.start)
747        .min();
748    let spans = groups
749        .iter()
750        .filter(|group| group.end > group.start)
751        .map(|group| (group.start, group.end))
752        .collect::<Vec<_>>();
753    groups
754        .into_iter()
755        .skip(1)
756        .filter(|group| unresolved_from.is_none_or(|start| group.start <= start))
757        .filter(|group| {
758            !spans
759                .iter()
760                .any(|(start, end)| *start < group.start && group.start <= *end)
761        })
762        .filter(|group| visible_seqs.contains(&group.start))
763        .map(|group| group.start)
764        .collect()
765}

772pub(crate) fn assemble_model_context(
773    events: &[StoredEvent],
774    store: &ArtifactStore,
775    binding: &FrozenModelBinding,
776    composed_prompt: &str,
777) -> Result<ModelContext, HistoryError> {
778    let checkpoint = latest_checkpoint(events);
779    let Some(commit) = checkpoint else {
780        let assembled =
781            assemble_history_with_replay(events, events, store, binding, composed_prompt)?;
782        return Ok(ModelContext {
783            history: assembled.history,
784            native_context: None,
785            replay_decisions: assembled.replay_decisions,
786        });
787    };
788    match &commit.checkpoint {
789        ContextCheckpoint::InternalSummary { checkpoint } => project_summary_context(
790            events,
791            store,
792            binding,
793            composed_prompt,
794            commit.boundaries.source_through_seq,
795            commit.boundaries.recent_from_seq,
796            checkpoint.summary(),
797        ),
798        ContextCheckpoint::NativeWindow { window } => {
799            commit
800                .validate_for_binding(binding)
801                .map_err(|error| HistoryError::Corrupt(error.to_string()))?;
802            let selected = selected_checkpoint_events(
803                events,
804                commit.boundaries.source_through_seq,
805                None,
806                false,
807            );
808            let assembled =
809                assemble_history_with_replay(&selected, events, store, binding, composed_prompt)?;
810            Ok(ModelContext {
811                history: assembled.history,
812                native_context: Some(restore_native_context(window)?),
813                replay_decisions: assembled.replay_decisions,
814            })
815        }
816    }
817}

828// Message selection must not roll back session-wide replay decisions or pinned context.
829// context_events is the full visible snapshot, before checkpoint or summary-prefix filtering.
830fn assemble_history_with_replay(
831    events: &[StoredEvent],
832    context_events: &[StoredEvent],
833    store: &ArtifactStore,
834    binding: &FrozenModelBinding,
835    composed_prompt: &str,
836) -> Result<AssembledHistory, HistoryError> {
837    let rejected_unsigned_replays = rejected_unsigned_replay_fingerprints(context_events);
838    let loaded_skills = context_events
839        .iter()
840        .filter_map(|event| match &event.payload {
841            EventPayload::SkillLoaded { rendered_body, .. } => Some(rendered_body.clone()),
842            _ => None,
843        })
844        .collect::<Vec<_>>();
845    let loaded_agent_md = latest_agent_md_event(context_events).and_then(|event| {
846        let EventPayload::AgentMdLoaded { entries } = &event.payload else {
847            return None;
848        };
849        Some(agent_md_turn(entries))
850    });
851    let mut logical = Vec::<LogicalTurn>::new();
852    let mut submitted = HashMap::<u64, String>::new();
853    let mut pending_model_calls = HashMap::<
854        (
855            cookie_agent_protocol::RunId,
856            cookie_agent_protocol::ModelCallId,
857        ),
858        VecDeque<(usize, usize)>,
859    >::new();
860    let mut engine_calls =
861        HashMap::<(cookie_agent_protocol::RunId, ToolCallId), (usize, usize)>::new();
862    let elisions = context_events
863        .iter()
864        .filter_map(|event| match &event.payload {
865            EventPayload::ToolOutputElided {
866                tool_call_id,
867                original_bytes,
868                retained,
869            } => Some((*tool_call_id, (*original_bytes, retained.clone()))),
870            _ => None,
871        })
872        .collect::<HashMap<_, _>>();
873
874    for envelope in events {
875        match &envelope.payload {
876            EventPayload::DelegatedContextSeeded { turns, .. } => {
877                logical.extend(turns.iter().map(|turn| match turn.role {
878                    DelegatedContextRole::User => LogicalTurn::User(user_text(&turn.text)),
879                    DelegatedContextRole::Assistant => {
880                        LogicalTurn::SeedAssistant(turn.text.clone())
881                    }
882                }));
883            }
884            EventPayload::MessageInjected { role, input } => match role {
885                cookie_agent_protocol::ExtensionMessageRole::System => {
886                    logical.push(LogicalTurn::System(input.clone()));
887                }
888                cookie_agent_protocol::ExtensionMessageRole::User => {
889                    logical.push(LogicalTurn::User(user_text(input)));
890                }
891                cookie_agent_protocol::ExtensionMessageRole::Assistant => {
892                    logical.push(LogicalTurn::SeedAssistant(input.clone()));
893                }
894                cookie_agent_protocol::ExtensionMessageRole::Tool => {}
895            },
896            EventPayload::UserInputSubmitted { input, .. } => {
897                submitted.insert(envelope.seq, input.clone());
898            }
899            EventPayload::UserInputApplied { user_input_seq } => {
900                if let Some(input) = submitted.remove(user_input_seq) {
901                    logical.push(LogicalTurn::User(user_text(&input)));
902                }
903            }
904            EventPayload::ModelTurnCommitted {
905                resolved_model,
906                turn,
907                ..
908            } => {
909                let logical_index = logical.len();
910                let mut calls = Vec::new();
911                let in_stream_results = turn
912                    .content
913                    .iter()
914                    .filter_map(|part| match part {
915                        PersistedAssistantPart::ToolResult { tool_call_id, .. } => {
916                            Some(tool_call_id.as_str())
917                        }
918                        _ => None,
919                    })
920                    .collect::<std::collections::HashSet<_>>();
921                for part in &turn.content {
922                    if let PersistedAssistantPart::ToolCall { id, .. } = part {
923                        let call_index = calls.len();
924                        let in_stream_result = in_stream_results.contains(id.as_str());
925                        calls.push(CallRecord {
926                            model_call_id: id.clone(),
927                            engine_call_id: None,
928                            result: None,
929                            additional_messages: Vec::new(),
930                            in_stream_result,
931                        });
932                        if !in_stream_result && let Some(run_id) = envelope.run_id {
933                            pending_model_calls
934                                .entry((run_id, id.clone()))
935                                .or_default()
936                                .push_back((logical_index, call_index));
937                        }
938                    }
939                }
940                logical.push(LogicalTurn::Assistant(Box::new(AssistantRecord {
941                    turn: turn.clone(),
942                    resolved_model: resolved_model.clone(),
943                    run_id: envelope.run_id,
944                    calls,
945                })));
946            }
947            EventPayload::ToolCallStarted { start } => {
948                let Some(run_id) = envelope.run_id else {
949                    continue;
950                };
951                let occurrence = pending_model_calls
952                    .get_mut(&(run_id, start.owner.model_call_id.clone()))
953                    .and_then(VecDeque::pop_front);
954                if let Some((logical_index, call_index)) = occurrence {
955                    let LogicalTurn::Assistant(assistant) = &mut logical[logical_index] else {
956                        return Err(HistoryError::Corrupt(
957                            "tool call mapped to a non-assistant turn".into(),
958                        ));
959                    };
960                    assistant.calls[call_index].engine_call_id = Some(start.tool_call_id);
961                    engine_calls.insert((run_id, start.tool_call_id), (logical_index, call_index));
962                }
963            }
964            EventPayload::ToolCallTerminated { termination }
965                if termination.outcome == ToolTerminationOutcome::Completed =>
966            {
967                if let Some(result) = &termination.result {
968                    let (result_part, additional_messages) =
969                        if let Some((original_bytes, retained)) =
970                            elisions.get(&termination.tool_call_id)
971                        {
972                            (
973                                ToolResultPart::new(
974                                    String::new(),
975                                    ToolContent::Text(tool_output_elision_marker(
976                                        retained,
977                                        *original_bytes,
978                                        result.additional_messages.len(),
979                                    )),
980                                ),
981                                Vec::new(),
982                            )
983                        } else {
984                            (
985                                tool_result_part(result, termination.tool_call_id, store)?,
986                                result.additional_messages.clone(),
987                            )
988                        };
989                    attach_result(
990                        &mut logical,
991                        &engine_calls,
992                        envelope.run_id,
993                        termination.tool_call_id,
994                        result_part,
995                        additional_messages,
996                    )?;
997                }
998            }
999            EventPayload::ToolCallTerminated { termination } => {
1000                let message = termination
1001                    .error
1002                    .as_ref()
1003                    .map_or("tool failed", |error| error.message.as_str());
1004                let (content, metadata) = if let Some(denied) = denied_failure(message) {
1005                    let visible_reason = denied.feedback.as_ref().map_or_else(
1006                        || denied.reason.clone(),
1007                        |feedback| format!("{}: {feedback}", denied.reason),
1008                    );
1009                    let mut metadata = BTreeMap::from([
1010                        (
1011                            "denial_source".into(),
1012                            serde_json::Value::String(match denied.source {
1013                                ApprovalDecisionSource::Policy => "policy".into(),
1014                                ApprovalDecisionSource::Model => "model".into(),
1015                                ApprovalDecisionSource::InternalAgent => "internal_agent".into(),
1016                                ApprovalDecisionSource::PermissionMode => "permission_mode".into(),
1017                                ApprovalDecisionSource::TreeGrant => "tree_grant".into(),
1018                                ApprovalDecisionSource::User => "user".into(),
1019                                ApprovalDecisionSource::DoomLoopGuard => "doom_loop_guard".into(),
1020                                ApprovalDecisionSource::System => "system".into(),
1021                            }),
1022                        ),
1023                        (
1024                            "denial_reason".into(),
1025                            serde_json::Value::String(denied.reason),
1026                        ),
1027                    ]);
1028                    if let Some(feedback) = denied.feedback {
1029                        metadata.insert("feedback".into(), serde_json::Value::String(feedback));
1030                    }
1031                    (
1032                        ToolContent::Denied {
1033                            reason: Some(visible_reason),
1034                        },
1035                        Some(metadata),
1036                    )
1037                } else {
1038                    (ToolContent::Text(message.to_owned()), None)
1039                };
1040                attach_result(
1041                    &mut logical,
1042                    &engine_calls,
1043                    envelope.run_id,
1044                    termination.tool_call_id,
1045                    ToolResultPart {
1046                        tool_call_id: String::new(),
1047                        content,
1048                        is_error: true,
1049                        metadata,
1050                    },
1051                    Vec::new(),
1052                )?;
1053            }
1054            EventPayload::ContextRehydrated { files } => {
1055                logical.push(LogicalTurn::Rehydration(files.clone()));
1056            }
1057            EventPayload::PluginEventAdded {
1058                plugin,
1059                name,
1060                payload,
1061            } => {
1062                logical.push(LogicalTurn::User(user_text(&format!(
1063                    "<plugin_event>{}</plugin_event>",
1064                    serde_json::json!({
1065                        "plugin": plugin,
1066                        "name": name,
1067                        "payload": payload,
1068                    })
1069                ))));
1070            }
1071            EventPayload::DelegateFinished {
1072                session_id,
1073                status,
1074                preview,
1075                total_lines,
1076            }
1077            | EventPayload::DelegateFinishedV2 {
1078                session_id,
1079                status,
1080                preview,
1081                total_lines,
1082                ..
1083            } => {
1084                logical.push(LogicalTurn::User(user_text(&format!(
1085                    "<subagent_notification>{}</subagent_notification>",
1086                    serde_json::json!({
1087                        "session_id": session_id,
1088                        "status": format!("{status:?}").to_ascii_lowercase(),
1089                        "preview": preview,
1090                        "total_lines": total_lines,
1091                    })
1092                ))));
1093            }
1094            _ => {}
1095        }
1096    }
1097
1098    let mut history = vec![HistoryTurn::system(SystemMessage::new(vec![
1099        SystemPart::Text(TextPart::new(composed_prompt)),
1100    ]))];
1101    if let Some(agent_md) = loaded_agent_md {
1102        history.push(HistoryTurn::user(user_text(&agent_md)));
1103    }
1104    history.extend(
1105        loaded_skills
1106            .iter()
1107            .map(|body| HistoryTurn::user(user_text(body))),
1108    );
1109    let mut replay_decisions = Vec::new();
1110    for turn in logical {
1111        match turn {
1112            LogicalTurn::System(text) => {
1113                let Some(HistoryTurn::System(message)) = history.first_mut() else {
1114                    unreachable!("assembled history starts with a system message")
1115                };
1116                message.content.push(SystemPart::Text(TextPart::new(text)));
1117            }
1118            LogicalTurn::User(user) => history.push(HistoryTurn::user(user)),
1119            LogicalTurn::SeedAssistant(text) => {
1120                history.push(HistoryTurn::assistant(CompletedTurn::new(
1121                    AssistantMessage::new(vec![AssistantPart::Text(TextPart::new(text))]),
1122                    Finish::new(oven_sdk::Usage::default(), FinishReason::Stop),
1123                )));
1124            }
1125            LogicalTurn::Assistant(mut assistant) => {
1126                let retained = assistant
1127                    .calls
1128                    .iter()
1129                    .filter(|call| call.in_stream_result || call.result.is_some())
1130                    .map(|call| call.model_call_id.as_str())
1131                    .collect::<std::collections::HashSet<_>>();
1132                let original_call_count = assistant.calls.len();
1133                assistant.turn.content.retain(|part| match part {
1134                    PersistedAssistantPart::ToolCall { id, .. } => retained.contains(id.as_str()),
1135                    PersistedAssistantPart::ToolApproval { tool_call_id, .. } => {
1136                        retained.contains(tool_call_id.as_str())
1137                    }
1138                    _ => true,
1139                });
1140                let forced_disposition = if retained.len() != original_call_count
1141                    && assistant.turn.native_replay.is_some()
1142                {
1143                    assistant.turn.native_replay = None;
1144                    Some(ReplayDisposition::DiscardedInvalidPayload {
1145                        reason: safe_replay_reason(
1146                            "native replay was discarded because normalized tool-call history changed",
1147                        ),
1148                    })
1149                } else if assistant
1150                    .turn
1151                    .native_replay
1152                    .as_ref()
1153                    .is_some_and(|artifact| {
1154                        rejected_unsigned_replays
1155                            .contains(persisted_native_replay_fingerprint(artifact).as_str())
1156                    })
1157                {
1158                    let artifact = assistant
1159                        .turn
1160                        .native_replay
1161                        .take()
1162                        .expect("checked artifact");
1163                    Some(ReplayDisposition::DiscardedInvalidPayload {
1164                        reason: rejected_persisted_replay_reason(&artifact),
1165                    })
1166                } else {
1167                    None
1168                };
1169                let has_content = !assistant.turn.content.is_empty();
1170                if has_content {
1171                    let history_index = history.len() as u64;
1172                    let (restored, disposition) = restore_turn_with_store(
1173                        &assistant.turn,
1174                        &assistant.resolved_model,
1175                        store,
1176                        binding,
1177                    )?;
1178                    if let Some(disposition) = forced_disposition.or(disposition) {
1179                        replay_decisions.push(ReplayDecision {
1180                            history_index,
1181                            disposition,
1182                        });
1183                    }
1184                    history.push(HistoryTurn::assistant(restored));
1185                }
1186                let mut results = assistant
1187                    .calls
1188                    .into_iter()
1189                    .filter_map(|call| call.result.map(|result| (result, call.additional_messages)))
1190                    .collect::<Vec<_>>();
1191                if !results.is_empty() {
1192                    if !has_content {
1193                        return Err(HistoryError::Corrupt(
1194                            "tool result has no retained assistant call".into(),
1195                        ));
1196                    }
1197                    let call_order = assistant
1198                        .turn
1199                        .content
1200                        .iter()
1201                        .enumerate()
1202                        .filter_map(|(index, part)| match part {
1203                            PersistedAssistantPart::ToolCall { id, .. } => {
1204                                Some((id.as_str(), index))
1205                            }
1206                            _ => None,
1207                        })
1208                        .collect::<HashMap<_, _>>();
1209                    results.sort_by_key(|(result, _)| {
1210                        call_order
1211                            .get(result.tool_call_id.as_str())
1212                            .copied()
1213                            .unwrap_or(usize::MAX)
1214                    });
1215                    let mut additional_messages = Vec::new();
1216                    let results = results
1217                        .into_iter()
1218                        .map(|(result, messages)| {
1219                            additional_messages.extend(messages);
1220                            result
1221                        })
1222                        .collect();
1223                    history.push(HistoryTurn::tool(ToolMessage::new(results)));
1224                    for message in additional_messages {
1225                        append_tool_emitted_message(&mut history, &message, store)?;
1226                    }
1227                }
1228                let _ = assistant.run_id;
1229            }
1230            LogicalTurn::Rehydration(files) => {
1231                let mut calls = Vec::with_capacity(files.len());
1232                let mut results = Vec::with_capacity(files.len());
1233                for (index, file) in files.into_iter().enumerate() {
1234                    let id = format!("context-rehydration-{index}");
1235                    calls.push(AssistantPart::ToolCall(ToolCallPart::new(
1236                        id.clone(),
1237                        "read",
1238                        serde_json::json!({"filePath": file.path.as_str()}),
1239                    )));
1240                    results.push(ToolResultPart::new(id, ToolContent::Text(file.content)));
1241                }
1242                history.push(HistoryTurn::assistant(CompletedTurn::new(
1243                    AssistantMessage::new(calls),
1244                    Finish::new(oven_sdk::Usage::default(), FinishReason::ToolCalls),
1245                )));
1246                history.push(HistoryTurn::tool(ToolMessage::new(results)));
1247            }
1248        }
1249    }
1250    Ok(AssembledHistory {
1251        history,
1252        replay_decisions,
1253    })
1254}

1256fn latest_agent_md_event(events: &[StoredEvent]) -> Option<&StoredEvent> {
1257    let latest_run = events.iter().rev().find_map(|event| {
1258        matches!(event.payload, EventPayload::RunStarted { .. }).then_some(event.run_id)
1259    });
1260    events.iter().rev().find(|event| {
1261        matches!(event.payload, EventPayload::AgentMdLoaded { .. })
1262            && latest_run.is_none_or(|run| event.run_id == run)
1263    })
1264}

1268        .iter()

1300fn attach_result(
1301    logical: &mut [LogicalTurn],
1302    engine_calls: &HashMap<(cookie_agent_protocol::RunId, ToolCallId), (usize, usize)>,
1303    run_id: Option<cookie_agent_protocol::RunId>,
1304    tool_call_id: ToolCallId,
1305    mut result: ToolResultPart,
1306    additional_messages: Vec<ToolEmittedMessage>,
1307) -> Result<(), HistoryError> {
1308    let Some(run_id) = run_id else {
1309        return Ok(());
1310    };
1311    let Some(&(logical_index, call_index)) = engine_calls.get(&(run_id, tool_call_id)) else {
1312        return Ok(());
1313    };
1314    let LogicalTurn::Assistant(assistant) = &mut logical[logical_index] else {
1315        return Err(HistoryError::Corrupt(
1316            "tool result mapped to a non-assistant turn".into(),
1317        ));
1318    };

1320    assistant.calls[call_index].result = Some(result);
1321    assistant.calls[call_index].additional_messages = additional_messages;
1322    Ok(())
1323}

1710fn restore_replay(
1711    artifact: &NativeReplayArtifact,
1712    resolved_model: &ResolvedModelRef,
1713    binding: &FrozenModelBinding,
1714) -> (Option<OvenReplayArtifact>, Option<ReplayDisposition>) {
1715    let expected_adapter = exact_adapter_code(binding.descriptor.adapter_id.as_str());
1716    if artifact.adapter_id() != &expected_adapter {
1717        return (
1718            None,
1719            Some(ReplayDisposition::DiscardedForeignAdapter {
1720                found: artifact.adapter_id().clone(),
1721                expected: expected_adapter,
1722            }),
1723        );
1724    }
1725    // Persisted-turn integrity: the artifact must have been recorded by the
1726    // adapter the turn itself resolved to, before any current-eligibility
1727    // checks expose its opaque payload. Artifacts record the Oven adapter ID
1728    // while the persisted turn carries the protocol adapter ID, so compare
1729    // through the shared family mapping.
1730    if crate::policy::wire_adapter(artifact.adapter_id().as_str()) != resolved_model.adapter_id {
1731        return (
1732            None,
1733            Some(ReplayDisposition::DiscardedInvalidPayload {
1734                reason: safe_replay_reason(
1735                    "native replay adapter does not match its persisted model turn",
1736                ),
1737            }),
1738        );
1739    }
1740    if artifact.selection_fingerprint() != &resolved_model.selection_fingerprint
1741        || artifact.scope().provider_id != resolved_model.provider_id
1742        || artifact.scope().model_id != resolved_model.model_id
1743    {
1744        return (
1745            None,
1746            Some(ReplayDisposition::DiscardedInvalidPayload {
1747                reason: safe_replay_reason(
1748                    "native replay identity does not match its persisted model turn",
1749                ),
1750            }),
1751        );
1752    }
1753    // Exact fingerprints validate the persisted turn above; current replay
1754    // eligibility is model-scoped so variants share their native history.
1755    if resolved_model.selection.model != binding.selection.model {
1756        return (
1757            None,
1758            Some(ReplayDisposition::DiscardedForeignModelSelection {
1759                found: resolved_model.selection.clone(),
1760                expected: binding.selection.clone(),
1761            }),
1762        );
1763    }
1764    let scope = match restore_scope(artifact.scope()) {
1765        Ok(scope) => scope,
1766        Err(error) => {
1767            return (
1768                None,
1769                Some(ReplayDisposition::DiscardedInvalidPayload {
1770                    reason: safe_replay_reason(&error.to_string()),
1771                }),
1772            );
1773        }
1774    };
1775    match OvenReplayArtifact::new(
1776        AdapterId::new(artifact.adapter_id().as_str()),
1777        scope,
1778        artifact.payload().clone(),
1779    ) {
1780        Ok(artifact) => (Some(artifact), None),
1781        Err(error) => (
1782            None,
1783            Some(ReplayDisposition::DiscardedInvalidPayload {
1784                reason: safe_replay_reason(&error.to_string()),
1785            }),
1786        ),
1787    }
1788}

1938fn restore_turn_with_store(
1939    turn: &PersistedModelTurn,
1940    resolved_model: &ResolvedModelRef,
1941    store: &ArtifactStore,
1942    binding: &FrozenModelBinding,
1943) -> Result<(CompletedTurn, Option<ReplayDisposition>), HistoryError> {
1944    let (native_replay, replay_disposition) = turn
1945        .native_replay
1946        .as_ref()
1947        .map_or((None, None), |artifact| {
1948            restore_replay(artifact, resolved_model, binding)
1949        });
1950    // A model switch may project normalized reasoning only when the target
1951    // declares support for replaying provider-authoritative reasoning.
1952    let preserve_reasoning = resolved_model.selection.model == binding.selection.model
1953        || binding.descriptor.capabilities.replay.reasoning;
1954    Ok((
1955        CompletedTurn {
1956            message: AssistantMessage {
1957                content: turn
1958                    .content
1959                    .iter()
1960                    .filter(|part| {
1961                        preserve_reasoning
1962                            || !matches!(part, PersistedAssistantPart::Reasoning { .. })
1963                    })
1964                    .map(|part| restore_assistant_part_with_store(part, store))
1965                    .collect::<Result<_, _>>()?,
1966                provider_options: turn.provider_options.clone(),
1967            },
1968            finish: Finish {
1969                usage: restore_usage(&turn.usage),
1970                finish_reason: restore_finish_reason(&turn.finish_reason),
1971                response_metadata: turn.response_metadata.clone(),
1972                provider_metadata: turn.provider_metadata.clone(),
1973                native_replay,
1974            },
1975            warnings: Vec::new(),
1976        },
1977        replay_disposition,
1978    ))
1979}

1992fn sanitize_control_free(value: &str, maximum: usize) -> String {
1993    let mut output = String::with_capacity(value.len().min(maximum));
1994    for character in value.chars() {
1995        if output.len() >= maximum {
1996            break;
1997        }
1998        let replacement = if character.is_control() {
1999            ' '
2000        } else {
2001            character
2002        };
2003        if output.len() + replacement.len_utf8() > maximum {
2004            break;
2005        }
2006        output.push(replacement);
2007    }
2008    if output.is_empty() {
2009        "unavailable".to_owned()
2010    } else {
2011        output
2012    }
2013}
