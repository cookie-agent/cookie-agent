232impl ContextTokenEstimator {
233    fn record_committed_turn(
234        &mut self,
235        serialized_context_bytes: usize,
236        input_tokens: Option<u64>,
237    ) {
238        self.last_committed_input_tokens = input_tokens.unwrap_or(0);
239        if serialized_context_bytes > 0
240            && let Some(input_tokens) = input_tokens.filter(|tokens| *tokens > 0)
241        {
242            self.tokens_per_byte = input_tokens as f64 / serialized_context_bytes as f64;
243        }
244    }
245
246    fn projected_tokens(self, serialized_message_bytes: usize) -> Option<u64> {
247        (self.tokens_per_byte > 0.0).then(|| {
248            self.last_committed_input_tokens
249                .saturating_add((serialized_message_bytes as f64 * self.tokens_per_byte) as u64)
250        })
251    }
252
253    fn estimated_context_tokens(self, serialized_context_bytes: usize) -> Option<u64> {
254        (self.tokens_per_byte > 0.0)
255            .then(|| (serialized_context_bytes as f64 * self.tokens_per_byte).ceil() as u64)
256    }
257
258    fn should_compact(self, serialized_message_bytes: usize, soft_tokens: u64) -> bool {
259        self.projected_tokens(serialized_message_bytes)
260            .is_some_and(|projected| projected >= soft_tokens)
261    }
262
263    fn record_compaction(&mut self, estimated_input_tokens: u64) {
264        self.last_committed_input_tokens = estimated_input_tokens;
265    }
266}
267
268fn should_run_predictive_compaction(
269    estimator: ContextTokenEstimator,
270    serialized_message_bytes: usize,
271    soft_tokens: u64,
272    session_persisted: bool,
273) -> bool {
274    session_persisted && estimator.should_compact(serialized_message_bytes, soft_tokens)
275}

456#[derive(Debug)]
457struct RuntimeRevisionIndex {
458    path: PathBuf,
459    mappings:
460        HashMap<cookie_agent_protocol::RuntimeRevision, cookie_agent_protocol::RuntimeRevision>,
461}
462
463impl RuntimeRevisionIndex {
464    fn open(path: PathBuf) -> Result<Self, EngineError> {
465        let mut mappings = HashMap::new();
466        for record in events::load_jsonl::<StoredRuntimeRevisionMapping>(&path)? {
467            match mappings.insert(
468                record.runtime_revision.clone(),
469                record.model_runtime_revision.clone(),
470            ) {
471                Some(existing) if existing != record.model_runtime_revision => {
472                    return Err(EngineError::RuntimeCompileFailed);
473                }
474                _ => {}
475            }
476        }
477        Ok(Self { path, mappings })
478    }
479
480    fn record(
481        &mut self,
482        runtime_revision: cookie_agent_protocol::RuntimeRevision,
483        model_runtime_revision: cookie_agent_protocol::RuntimeRevision,
484    ) -> Result<(), EngineError> {
485        if let Some(existing) = self.mappings.get(&runtime_revision) {
486            return if existing == &model_runtime_revision {
487                Ok(())
488            } else {
489                Err(EngineError::RuntimeCompileFailed)
490            };
491        }
492        events::append_jsonl(
493            &self.path,
494            &StoredRuntimeRevisionMapping {
495                protocol_version: cookie_agent_protocol::ProtocolVersion::current(),
496                runtime_revision: runtime_revision.clone(),
497                model_runtime_revision: model_runtime_revision.clone(),
498            },
499        )?;
500        self.mappings
501            .insert(runtime_revision, model_runtime_revision);
502        Ok(())
503    }
504
505    fn resolve(
506        &self,
507        runtime_revision: &cookie_agent_protocol::RuntimeRevision,
508    ) -> Option<cookie_agent_protocol::RuntimeRevision> {
509        self.mappings.get(runtime_revision).cloned()
510    }
511}

662impl SessionCommand {
663    fn compaction_deferred_kind(&self) -> Option<CompactionDeferredKind> {
664        match self {
665            Self::PromptSnapshot { .. } => Some(CompactionDeferredKind::PromptSnapshot),
666            Self::PromotePendingOrComplete { .. } => {
667                Some(CompactionDeferredKind::PromotePendingOrComplete)
668            }
669            Self::Resume { .. } => Some(CompactionDeferredKind::Resume),
670            _ => None,
671        }
672    }
673
674    fn reject_duplicate_during_compaction(self, session: SessionId) {
675        match self {
676            Self::PromptSnapshot { reply, .. } => {
677                let _ = reply.send(Err(EngineError::SessionRunning(session)));
678            }
679            Self::PromotePendingOrComplete { reply, .. } => {
680                let _ = reply.send(Ok(false));
681            }
682            Self::Resume { reply } => {
683                let _ = reply.send(Err(EngineError::SessionRunning(session)));
684            }
685            _ => unreachable!("only barrier-sensitive commands are superseded"),
686        }
687    }
688}

797            .map_err(|error| EngineError::MissingTool(error.to_string()))?,
798        );
799        for provider in &options.tools {
800            mcp.reserve_provider(provider.as_ref())
801                .map_err(|error| EngineError::MissingTool(error.to_string()))?;

804        tools.push(mcp.clone());
805        let journal = DelegationJournal::open(store.project_dir_path().join("delegations.jsonl"))?;
806        let grant_journal = GrantInvalidationJournal::open(
807            store.project_dir_path().join("grant-invalidations.jsonl"),
808        )?;
809        let (runtime_notifications, _) = broadcast::channel(64);
810        let mut runtime_revision_index = RuntimeRevisionIndex::open(
811            store.project_dir_path().join("runtime-revisions-v8.jsonl"),

823                store,
824                journal,
825                grant_journal,
826                model_manager: options.model_manager,
827                published_runtime: ArcSwap::from(published_runtime),
828                runtime_mutation: Mutex::new(()),
829                runtime_notifications,
830                runtime_revision_index: Mutex::new(runtime_revision_index),
831                manifest_store,
832                tools: Mutex::new(tools),
833                mcp,
834                mcp_mutation: tokio::sync::Mutex::new(()),
835                approvals: ApprovalStore::default(),
836                permissions: PermissionPipeline::default(),
837                active: Mutex::new(HashMap::new()),
838                inflight_delegations: Mutex::new(HashMap::new()),
839                delegations_by_session: Mutex::new(HashMap::new()),
840                delegation_queue: Mutex::new(VecDeque::new()),
841                delegation_admission: tokio::sync::Mutex::new(()),
842                next_admission_generation: AtomicU64::new(1),
843                subscribers: Mutex::new(HashMap::new()),
844                actors: Mutex::new(HashMap::new()),
845                output_hubs: Mutex::new(HashMap::new()),
846                finalized_output_hubs: Mutex::new(VecDeque::new()),
847                pending_approvals: Mutex::new(HashMap::new()),
848                permission_modes: Mutex::new(HashMap::new()),
849                permission_overlay_mutation: tokio::sync::Mutex::new(()),
850                compaction_in_progress: Mutex::new(HashSet::new()),
851                compaction_deferred: Mutex::new(HashMap::new()),
852                context_token_estimators: Mutex::new(HashMap::new()),
853                runtime: tokio::runtime::Handle::try_current().ok(),
854                admission_tasks: Mutex::new(Vec::new()),
855                admission_blocking_tasks: Mutex::new(Vec::new()),
856                admission_tasks_closing: AtomicBool::new(false),
857                recovery_waiters: Mutex::new(HashSet::new()),
858                #[cfg(test)]
859                prompt_snapshot_hook: Mutex::new(None),
860                #[cfg(test)]
861                approval_evaluation_hook: Mutex::new(None),
862                #[cfg(test)]
863                gap_send_hook: Mutex::new(None),
864                #[cfg(test)]
865                admission_confirmation_hook: Mutex::new(None),
866                #[cfg(test)]
867                resume_admission_hook: Mutex::new(None),
868                #[cfg(test)]
869                resume_attachment_hook: Mutex::new(None),
870                #[cfg(test)]
871                resume_rollback_hook: Mutex::new(None),
872                #[cfg(test)]
873                admission_blocking_hook: Mutex::new(None),
874                #[cfg(test)]
875                abandoned_sweep_hook: Mutex::new(None),
876                #[cfg(test)]
877                publication_failure: AtomicBool::new(false),
878                #[cfg(test)]
879                delegate_start_failures: AtomicU64::new(0),
880                #[cfg(test)]
881                delegate_terminal_append_failures: AtomicU64::new(0),
882                #[cfg(test)]
883                resume_monitor_failures: AtomicU64::new(0),
884            }),
885        };
886        engine.validate_referenced_manifests()?;
887        for session in engine.inner.store.all() {
888            engine.spawn_actor(session.meta.session_id);
889        }
890        engine.rebuild_approvals();
891        // Reconciliation uses the synchronous actor facade. When open is
892        // called by an async composition root, move that facade to a plain
893        // thread so `blocking_send` never runs on a Tokio worker.
894        if tokio::runtime::Handle::try_current().is_ok() {
895            let reconcile_engine = engine.clone();
896            std::thread::spawn(move || reconcile_engine.reconcile())
897                .join()
898                .map_err(|_| EngineError::ActorStopped)??;
899        } else {
900            engine.reconcile()?;
901        }
902        if let Some(runtime) = &engine.inner.runtime {
903            engine.inner.mcp.start_eager(runtime);
904        }
905        Ok(engine)
906    }

923    pub fn connect_provider(
924        &self,
925        params: ProviderConnectParams,
926    ) -> Result<ProviderConnectResult, EngineError> {
927        use cookie_agent_models::provider_store::{
928            ClientConnectId, ProviderAuthValues, ProviderStoreMutation,
929        };
930
931        let _mutation = self
932            .inner
933            .runtime_mutation
934            .lock()
935            .unwrap_or_else(|poisoned| poisoned.into_inner());
936        let provider_id = params.provider_id.clone();
937        let auth_values = params
938            .auth_values
939            .field_names()
940            .map(|name| {
941                let field = cookie_agent_protocol::AuthFieldName::new(name.to_owned())
942                    .map_err(|_| EngineError::RuntimeCompileFailed)?;
943                let value = params
944                    .auth_values
945                    .get(&field)
946                    .ok_or(EngineError::RuntimeCompileFailed)?
947                    .to_owned();
948                Ok((field, value))
949            })
950            .collect::<Result<_, EngineError>>()?;
951        let request = cookie_agent_models::ProviderConnectRequest {
952            provider_id,
953            expected_catalog_revision: params.expected_catalog_revision,
954            setup_values: params
955                .setup_values
956                .into_iter()
957                .map(|(id, value)| {
958                    let value = serde_json::from_value(
959                        serde_json::to_value(value)
960                            .map_err(|_| EngineError::RuntimeCompileFailed)?,
961                    )
962                    .map_err(|_| EngineError::RuntimeCompileFailed)?;
963                    Ok((id, value))
964                })
965                .collect::<Result<_, EngineError>>()?,
966            auth_method: params.auth_method,
967            auth_values: ProviderAuthValues::new(auth_values)
968                .map_err(cookie_agent_models::ModelManagerError::from)?,
969            client_connect_id: ClientConnectId::new(params.client_connect_id.as_str())
970                .map_err(cookie_agent_models::ModelManagerError::from)?,
971        };
972        let previous = self.current_runtime();
973        let result = self.inner.model_manager.connect(request, |candidate, _| {
974            self.prepare_publication(
975                candidate,
976                &previous,
977                vec![RuntimeChangeReason::ProviderConnected],
978            )
979            .map_err(|_| cookie_agent_models::ModelManagerError::RuntimeCompileFailed)
980        })?;
981        let runtime = result.publication.map_or_else(
982            || self.current_runtime(),
983            |publication| self.publish(publication),
984        );
985        let durable_connection = match &result.mutation {
986            ProviderStoreMutation::Connect {
987                durable_connection, ..
988            } => {
989                crate::runtime_snapshot::projection::project_durable_connection(durable_connection)?
990            }
991            ProviderStoreMutation::Disconnect { .. } => {
992                return Err(EngineError::RuntimeCompileFailed);
993            }
994        };
995        Ok(ProviderConnectResult {
996            durable_connection,
997            effective_auth_source: crate::runtime_snapshot::projection::effective_auth_source(
998                result.effective_auth,
999            )?,
1000            runtime: runtime.result.snapshot.clone(),
1001            replayed: result.replayed,
1002        })
1003    }

1005    pub fn disconnect_provider(
1006        &self,
1007        params: ProviderDisconnectParams,
1008    ) -> Result<ProviderDisconnectResult, EngineError> {
1009        use cookie_agent_models::provider_store::{ClientRequestId, ProviderStoreMutation};
1010
1011        let _mutation = self
1012            .inner
1013            .runtime_mutation
1014            .lock()
1015            .unwrap_or_else(|poisoned| poisoned.into_inner());
1016        let previous = self.current_runtime();
1017        let expected_model_runtime_revision = self
1018            .inner
1019            .runtime_revision_index
1020            .lock()
1021            .unwrap_or_else(|poisoned| poisoned.into_inner())
1022            .resolve(&params.expected_runtime_revision)
1023            .ok_or({
1024                EngineError::ModelManager(cookie_agent_models::ModelManagerError::ProviderStore(
1025                    cookie_agent_models::provider_store::ProviderStoreError::RuntimeRevisionConflict,
1026                ))
1027            })?;
1028        let request = cookie_agent_models::ProviderDisconnectRequest {
1029            provider_id: params.provider_id.clone(),
1030            expected_runtime_revision: expected_model_runtime_revision,
1031            expected_provider_state_revision: params.expected_provider_state_revision,
1032            expected_connection_generation: params
1033                .expected_connection_generation
1034                .map(|value| {
1035                    cookie_agent_models::provider_store::ProviderConnectionGeneration::new(
1036                        value.get(),
1037                    )
1038                })
1039                .transpose()
1040                .map_err(cookie_agent_models::ModelManagerError::from)?,
1041            client_request_id: ClientRequestId::new(params.client_request_id.as_str())
1042                .map_err(cookie_agent_models::ModelManagerError::from)?,
1043        };
1044        let result = self
1045            .inner
1046            .model_manager
1047            .disconnect(request, |candidate, _| {
1048                self.prepare_publication(
1049                    candidate,
1050                    &previous,
1051                    vec![RuntimeChangeReason::ProviderDisconnected],
1052                )
1053                .map_err(|_| cookie_agent_models::ModelManagerError::RuntimeCompileFailed)
1054            })?;
1055        let runtime = result.publication.map_or_else(
1056            || self.current_runtime(),
1057            |publication| self.publish(publication),
1058        );
1059        if !matches!(result.mutation, ProviderStoreMutation::Disconnect { .. }) {
1060            return Err(EngineError::RuntimeCompileFailed);
1061        }
1062        let receipt = result.mutation.durable_receipt();
1063        Ok(ProviderDisconnectResult {
1064            durable_receipt: cookie_agent_protocol::DurableProviderReceipt {
1065                receipt_id: receipt
1066                    .receipt_id
1067                    .to_string()
1068                    .parse()
1069                    .map_err(|_| EngineError::RuntimeCompileFailed)?,
1070                store_revision: receipt.store_revision.clone(),
1071                provider_state_revision: receipt.provider_state_revision.clone(),
1072                committed_at: receipt.committed_at,
1073            },
1074            provider_id: params.provider_id,
1075            disconnected: true,
1076            effective_auth_state: crate::runtime_snapshot::projection::effective_auth_state(
1077                result.effective_auth,
1078            ),
1079            runtime: runtime.result.clone(),
1080            replayed: result.replayed,
1081        })
1082    }

1084    pub fn reconcile_provider_store(&self) -> Result<bool, EngineError> {
1085        let _mutation = self
1086            .inner
1087            .runtime_mutation
1088            .lock()
1089            .unwrap_or_else(|poisoned| poisoned.into_inner());
1090        let previous = self.current_runtime();
1091        let reloaded = self
1092            .inner
1093            .model_manager
1094            .reload_store_if_changed(|candidate| {
1095                self.prepare_publication(
1096                    candidate,
1097                    &previous,
1098                    vec![
1099                        RuntimeChangeReason::ProviderStoreChanged,
1100                        RuntimeChangeReason::ProviderStoreReloaded,
1101                    ],
1102                )
1103                .map_err(|_| cookie_agent_models::ModelManagerError::RuntimeCompileFailed)
1104            })
1105            .map_err(|_| EngineError::ProviderStoreReloadFailed)?;
1106        if let Some((_, publication)) = reloaded {
1107            self.publish(publication);
1108            Ok(true)
1109        } else {
1110            Ok(false)
1111        }

1114    /// Atomically recompiles and publishes a newly acquired catalog snapshot.
1115    pub fn refresh_catalog(
1116        &self,
1117        catalog: Arc<cookie_agent_models::catalog::CatalogSnapshot>,
1118    ) -> Result<RuntimeSnapshotResult, EngineError> {
1119        let _mutation = self
1120            .inner
1121            .runtime_mutation
1122            .lock()
1123            .unwrap_or_else(|poisoned| poisoned.into_inner());
1124        let previous = self.current_runtime();
1125        let authored = previous.models.authored().clone();
1126        let reason = match catalog.source {
1127            cookie_agent_models::catalog::CatalogSource::Network => {
1128                RuntimeChangeReason::CatalogRefreshed
1129            }
1130            cookie_agent_models::catalog::CatalogSource::Cache
1131            | cookie_agent_models::catalog::CatalogSource::Bootstrap => {
1132                RuntimeChangeReason::CatalogFallback
1133            }
1134        };
1135        let (_, publication) =
1136            self.inner
1137                .model_manager
1138                .reload_inputs(authored, catalog, |candidate| {
1139                    self.prepare_publication(candidate, &previous, vec![reason])
1140                        .map_err(|_| cookie_agent_models::ModelManagerError::RuntimeCompileFailed)
1141                })?;
1142        Ok(self.publish(publication).result.clone())
1143    }

1145    fn prepare_publication(
1146        &self,
1147        models: &Arc<cookie_agent_models::CompiledModelRuntime>,
1148        previous: &Arc<PublishedRuntime>,
1149        mut reasons: Vec<RuntimeChangeReason>,
1150    ) -> Result<RuntimePublication, EngineError> {
1151        #[cfg(test)]
1152        if self.inner.publication_failure.swap(false, Ordering::AcqRel) {
1153            return Err(EngineError::RuntimeCompileFailed);
1154        }
1155        let authored_agents = self.inner.config.agent_registry();
1156        let agents = Arc::new(AgentRegistry::resolve(&authored_agents, models)?);
1157        let prepared = prepare_runtime_manifest(&self.inner.manifest_store, models)?;
1158        let snapshot = build_runtime_snapshot(models, &agents)?;
1159        self.inner
1160            .runtime_revision_index
1161            .lock()
1162            .unwrap_or_else(|poisoned| poisoned.into_inner())
1163            .record(
1164                snapshot.runtime_revision.clone(),
1165                models.runtime_revision().clone(),
1166            )?;
1167        reasons.sort();
1168        reasons.dedup();
1169        let notification = RuntimeChangedNotification {
1170            previous_revision: Some(previous.result.snapshot.runtime_revision.clone()),
1171            snapshot: snapshot.clone(),
1172            reasons,
1173        };
1174        Ok(RuntimePublication {
1175            runtime: Arc::new(PublishedRuntime {
1176                result: RuntimeSnapshotResult { snapshot },
1177                models: Arc::clone(models),
1178                agents,
1179                manifests: prepared.index,
1180                current_manifest: prepared.manifest,
1181            }),
1182            notification,
1183        })
1184    }

1197    fn validate_referenced_manifests(&self) -> Result<(), EngineError> {
1198        let runtime = self.current_runtime();
1199        for session in self.inner.store.all() {
1200            for event in session.log.events() {
1201                match event.payload {
1202                    Event::SessionCreated { creation_agent, .. } => {
1203                        for binding in &creation_agent.fallback_chain {
1204                            let validation = validate_referenced_binding(
1205                                &runtime.manifests,
1206                                &runtime.models,
1207                                binding,
1208                            );
1209                            if !matches!(
1210                                &validation,
1211                                Ok(())
1212                                    | Err(EngineError::SnapshotRehydration(
1213                                        RehydrationError::SnapshotConfigMismatch
1214                                            | RehydrationError::SnapshotCredentialsUnavailable
1215                                    ))
1216                            ) {
1217                                validation?;
1218                            }
1219                        }
1220                    }
1221                    Event::RunStarted {
1222                        selected_suffix, ..
1223                    } => {
1224                        for binding in &selected_suffix {
1225                            let validation = validate_referenced_binding(
1226                                &runtime.manifests,
1227                                &runtime.models,
1228                                binding,
1229                            );
1230                            if !matches!(
1231                                &validation,
1232                                Ok(())
1233                                    | Err(EngineError::SnapshotRehydration(
1234                                        RehydrationError::SnapshotConfigMismatch
1235                                            | RehydrationError::SnapshotCredentialsUnavailable
1236                                    ))
1237                            ) {
1238                                validation?;
1239                            }
1240                        }
1241                    }
1242                    _ => {}
1243                }
1244            }
1245        }
1246        for entry in self.inner.journal.entries() {
1247            let manifest = runtime
1248                .manifests
1249                .require(&entry.revisions.manifest_revision)?;
1250            if manifest.payload.catalog_revision != entry.revisions.catalog_revision
1251                || manifest.payload.provider_state_revision
1252                    != entry.revisions.provider_state_revision
1253                || manifest.payload.model_revision != entry.revisions.model_revision
1254                || manifest.payload.recipe_registry_revision
1255                    != entry.revisions.recipe_registry_revision
1256                || crate::runtime_snapshot::projection::runtime_revision(
1257                    &entry.revisions.recipe_registry_revision,
1258                    &entry.revisions.catalog_revision,
1259                    &entry.revisions.provider_state_revision,
1260                    &entry.revisions.model_revision,
1261                    &entry.revisions.agent_revision,
1262                )? != entry.revisions.runtime_revision
1263            {
1264                return Err(EngineError::RuntimeCompileFailed);
1265            }
1266            for binding in &entry.selected_suffix {
1267                let validation =
1268                    validate_referenced_binding(&runtime.manifests, &runtime.models, binding);
1269                if !matches!(
1270                    &validation,
1271                    Ok(())
1272                        | Err(EngineError::SnapshotRehydration(
1273                            RehydrationError::SnapshotConfigMismatch
1274                                | RehydrationError::SnapshotCredentialsUnavailable
1275                        ))
1276                ) {
1277                    validation?;
1278                }
1279            }
1280        }
1281        Ok(())
1282    }

1311            .map_err(|error| EngineError::MissingTool(error.to_string()))?;
1312        self.inner
1313            .tools
1314            .lock()
1315            .unwrap_or_else(|poisoned| poisoned.into_inner())
1316            .push(provider);
1317        Ok(())
1318    }
1319
1320    #[must_use]
1321    pub fn mcp_statuses(&self) -> Vec<crate::McpServerStatus> {
1322        self.inner.mcp.statuses()
1323    }
1324
1325    /// Stops new session mailbox traffic, cancels active work, and joins the
1326    /// journal worker. Existing client clones may keep a session mailbox alive.
1327    pub async fn shutdown(&self) {
1328        self.inner
1329            .admission_tasks_closing
1330            .store(true, Ordering::Release);
1331        let tasks = self
1332            .inner
1333            .admission_tasks
1334            .lock()
1335            .map(|mut tasks| tasks.drain(..).collect::<Vec<_>>())
1336            .unwrap_or_default();
1337        for task in &tasks {
1338            task.abort();
1339        }
1340        for task in tasks {
1341            if let Err(error) = task.await {
1342                eprintln!("admission task stopped during shutdown: {error}");
1343            }
1344        }
1345        let blocking_tasks = self
1346            .inner
1347            .admission_blocking_tasks
1348            .lock()
1349            .map(|mut tasks| tasks.drain(..).collect::<Vec<_>>())
1350            .unwrap_or_default();
1351        for task in blocking_tasks {
1352            if let Err(error) = task.await {
1353                eprintln!("admission blocking task stopped during shutdown: {error}");
1354            }
1355        }
1356        let active: Vec<_> = self
1357            .inner
1358            .active
1359            .lock()
1360            .unwrap_or_else(|poisoned| poisoned.into_inner())
1361            .values()
1362            .cloned()
1363            .collect();
1364        for run in active {
1365            run.cancellation.cancel();
1366        }
1367        self.inner.mcp.shutdown().await;
1368        self.inner
1369            .actors
1370            .lock()
1371            .unwrap_or_else(|poisoned| poisoned.into_inner())
1372            .clear();
1373        self.inner.journal.shutdown();
1374    }
1375}
