145    fn append_inner(
146        &self,
147        run_id: Option<RunId>,
148        payload: EventPayload,
149    ) -> Result<StoredEvent, EventLogError> {
150        let usage_cost_stamped = matches!(
151            payload,
152            EventPayload::ModelUsageRecorded { .. }
153                | EventPayload::InternalAgentUsageRecorded { .. }
154        );
155        let mut events = self
156            .events
157            .lock()
158            .unwrap_or_else(|poisoned| poisoned.into_inner());
159        let event = StoredEvent {
160            engine_version: Some(env!("CARGO_PKG_VERSION").to_owned()),
161            session_id: self.session_id,
162            run_id,
163            seq: self.next_seq.load(Ordering::Acquire),
164            timestamp: Timestamp::now(),
165            payload,
166        };
167        event.validate().map_err(|error| EventLogError::Corrupt {
168            path: self.path.clone(),
169            message: error.to_string(),
170        })?;
171        let mut candidate = events.clone();
172        candidate.push(event.clone());
173        let _ = validate_records(
174            &self.path,
175            self.session_id,
176            &candidate,
177            &self.validation_taint,
178            Some(event.seq),
179        )?;
180        if self.persisted.load(Ordering::Acquire) {
181            append_jsonl(&self.path, &event)?;
182        }
183        if usage_cost_stamped {
184            self.usage_cost_stamped
185                .lock()
186                .unwrap_or_else(|poisoned| poisoned.into_inner())
187                .insert(event.seq);
188        }
189        events.push(event.clone());
190        self.next_seq.store(event.seq + 1, Ordering::Release);
191        Ok(event)
192    }

471    fn mark_record(
472        &mut self,
473        seq: u64,
474        run_id: Option<RunId>,
475        payload: &Value,
476    ) -> Result<(), &'static str> {
477        match payload.get("type").and_then(Value::as_str) {
478            Some("run_started") => {
479                if let Some(run_id) = run_id {
480                    self.runs.entry(run_id).or_insert(seq);
481                    self.run_ordering.entry(run_id).or_insert(seq);
482                }
483                self.active_run_ordering.get_or_insert(seq);
484            }
485            Some("model_attempt_started") => {
486                if let Some(attempt_id) = json_field(payload, "attempt_id") {
487                    self.attempts.entry(attempt_id).or_insert(seq);
488                }
489                if let Some(run_id) = run_id {
490                    self.attempt_runs.entry(run_id).or_insert(seq);
491                    self.run_ordering.entry(run_id).or_insert(seq);
492                }
493            }
494            Some("attempt_abandoned") | Some("model_fallback") => {
495                if let Some(run_id) = run_id {
496                    self.run_ordering.entry(run_id).or_insert(seq);
497                }
498            }
499            Some("model_turn_committed") => {
500                if let Some(model_turn_seq) = payload.get("model_turn_seq").and_then(Value::as_u64)
501                {
502                    self.turns.entry(model_turn_seq).or_insert(seq);
503                }
504                self.turn_ordering.get_or_insert(seq);
505                if let Some(run_id) = run_id {
506                    self.run_ordering.entry(run_id).or_insert(seq);
507                }
508            }
509            Some("tool_call_started") => {
510                if let Some(tool_call_id) = json_field(payload, "tool_call_id") {
511                    self.tools.entry(tool_call_id).or_insert(seq);
512                }
513            }
514            Some("tool_call_terminated") => {
515                if let Some(tool_call_id) = json_field(payload, "tool_call_id") {
516                    self.tool_terminals.entry(tool_call_id).or_insert(seq);
517                }
518            }
519            Some("approval_requested") => {
520                if let Some(approval_id) = payload
521                    .get("request")
522                    .and_then(|request| json_field(request, "approval_id"))
523                {
524                    self.approvals.entry(approval_id).or_insert(seq);
525                }
526            }
527            Some("internal_agent_started") => {
528                if let Some(internal_run_id) = json_field(payload, "internal_run_id") {
529                    self.internal_runs.entry(internal_run_id).or_insert(seq);
530                }
531            }
532            Some("internal_agent_fallback") => {
533                if let Some(internal_run_id) = json_field(payload, "internal_run_id") {
534                    self.mark_internal_fallback(internal_run_id, seq)?;
535                }
536            }
537            Some("internal_agent_usage_recorded") => {
538                if let Some(internal_run_id) = json_field(payload, "internal_run_id") {
539                    self.mark_internal_usage(internal_run_id, seq)?;
540                }
541            }
542            Some("user_input_admitted") => {
543                self.admissions.entry(seq).or_insert(seq);
544            }
545            Some(
546                "delegation_reserved"
547                | "delegation_started"
548                | "delegation_run_started"
549                | "delegation_run_attached"
550                | "delegation_finished",
551            ) => {
552                if let Some(invocation_id) = delegation_invocation_from_value(payload) {
553                    self.delegations.entry(invocation_id).or_insert(seq);
554                }
555            }
556            Some("run_completed" | "run_failed" | "run_cancelled" | "run_interrupted") => {
557                self.active_run_ordering.get_or_insert(seq);
558            }
559            _ => {}
560        }
561        Ok(())
562    }

564    fn mark_event(&mut self, event: &StoredEvent) -> Result<(), &'static str> {
565        let seq = event.seq;
566        match &event.payload {
567            EventPayload::RunStarted { .. } => {
568                if let Some(run_id) = event.run_id {
569                    self.runs.entry(run_id).or_insert(seq);
570                    self.run_ordering.entry(run_id).or_insert(seq);
571                }
572                self.active_run_ordering.get_or_insert(seq);
573            }
574            EventPayload::ModelAttemptStarted { attempt_id, .. } => {
575                self.attempts.entry(*attempt_id).or_insert(seq);
576                if let Some(run_id) = event.run_id {
577                    self.attempt_runs.entry(run_id).or_insert(seq);
578                    self.run_ordering.entry(run_id).or_insert(seq);
579                }
580            }
581            EventPayload::AttemptAbandoned { .. } | EventPayload::ModelFallback { .. } => {
582                if let Some(run_id) = event.run_id {
583                    self.run_ordering.entry(run_id).or_insert(seq);
584                }
585            }
586            EventPayload::ModelTurnCommitted { model_turn_seq, .. } => {
587                self.turns.entry(*model_turn_seq).or_insert(seq);
588                self.turn_ordering.get_or_insert(seq);
589                if let Some(run_id) = event.run_id {
590                    self.run_ordering.entry(run_id).or_insert(seq);
591                }
592            }
593            EventPayload::ToolCallStarted { start } => {
594                self.tools.entry(start.tool_call_id).or_insert(seq);
595            }
596            EventPayload::ToolCallTerminated { termination } => {
597                self.tool_terminals
598                    .entry(termination.tool_call_id)
599                    .or_insert(seq);
600            }
601            EventPayload::ApprovalRequested { request } => {
602                self.approvals.entry(request.approval_id()).or_insert(seq);
603            }
604            EventPayload::InternalAgentStarted {
605                internal_run_id, ..
606            } => {
607                self.internal_runs.entry(*internal_run_id).or_insert(seq);
608            }
609            EventPayload::InternalAgentFallback {
610                internal_run_id, ..
611            } => {
612                self.mark_internal_fallback(*internal_run_id, seq)?;
613            }
614            EventPayload::InternalAgentUsageRecorded {
615                internal_run_id, ..
616            } => {
617                self.mark_internal_usage(*internal_run_id, seq)?;
618            }
619            EventPayload::UserInputAdmitted { .. } => {
620                self.admissions.entry(seq).or_insert(seq);
621            }
622            EventPayload::DelegationReserved { reservation, .. } => {
623                self.delegations
624                    .entry(reservation.invocation_id)
625                    .or_insert(seq);
626            }
627            EventPayload::DelegationStarted { invocation_id, .. }
628            | EventPayload::DelegationRunStarted { invocation_id, .. }
629            | EventPayload::DelegationRunAttached { invocation_id, .. }
630            | EventPayload::DelegationFinished { invocation_id, .. } => {
631                self.delegations.entry(*invocation_id).or_insert(seq);
632            }
633            EventPayload::RunCompleted { .. }
634            | EventPayload::RunFailed { .. }
635            | EventPayload::RunCancelled { .. }
636            | EventPayload::RunInterrupted { .. } => {
637                self.active_run_ordering.get_or_insert(seq);
638            }
639            _ => {}
640        }
641        Ok(())
642    }

644    fn mark_internal_fallback(
645        &mut self,
646        internal_run_id: cookie_agent_protocol::InternalAgentRunId,
647        seq: u64,
648    ) -> Result<(), &'static str> {
649        let model_len = self
650            .internal_model_phases
651            .get(&internal_run_id)
652            .map_or(0, Vec::len);
653        let usage_len = self
654            .internal_usage_phases
655            .get(&internal_run_id)
656            .map_or(0, Vec::len);
657        if model_len >= MAX_INTERNAL_PHASE_TAINTS_PER_RUN
658            || usage_len >= MAX_INTERNAL_PHASE_TAINTS_PER_RUN
659        {
660            return Err(INTERNAL_PHASE_TAINT_LIMIT_MESSAGE);
661        }
662        self.internal_model_phases
663            .entry(internal_run_id)
664            .or_default()
665            .push(seq);
666        self.internal_usage_phases
667            .entry(internal_run_id)
668            .or_default()
669            .push(seq);
670        Ok(())
671    }
672
673    fn mark_internal_usage(
674        &mut self,
675        internal_run_id: cookie_agent_protocol::InternalAgentRunId,
676        seq: u64,
677    ) -> Result<(), &'static str> {
678        let usage = self
679            .internal_usage_phases
680            .entry(internal_run_id)
681            .or_default();
682        if usage.len() >= MAX_INTERNAL_PHASE_TAINTS_PER_RUN {
683            return Err(INTERNAL_PHASE_TAINT_LIMIT_MESSAGE);
684        }
685        usage.push(seq);
686        Ok(())
687    }
688}

724fn validate_observed_duplicates(path: &Path, records: &[StoredEvent]) -> Result<(), EventLogError> {
725    let mut runs = HashSet::new();
726    let mut attempts = HashSet::new();
727    let mut turns = HashSet::new();
728    let mut usage_turns = HashSet::new();
729    let mut internal_runs = HashSet::new();
730    let mut tool_starts = HashSet::new();
731    let mut tool_terminations = HashSet::new();
732    let mut approvals = HashSet::new();
733    let mut model_calls = HashSet::new();
734    let mut provider_items = HashSet::new();
735    for record in records {
736        match &record.payload {
737            EventPayload::RunStarted { .. } => {
738                let Some(run_id) = record.run_id else {
739                    return corrupt(path, "RunStarted is missing run_id");
740                };
741                if !runs.insert(run_id) {
742                    return corrupt(path, "run_id has more than one RunStarted event");
743                }
744            }
745            EventPayload::ModelAttemptStarted { attempt_id, .. } => {
746                if !attempts.insert(*attempt_id) {
747                    return corrupt(path, "attempt_id has more than one ModelAttemptStarted");
748                }
749            }
750            EventPayload::ModelTurnCommitted {
751                model_turn_seq,
752                turn,
753                ..
754            } => {
755                if !turns.insert(*model_turn_seq) {
756                    return corrupt(path, "model turn sequence is duplicated");
757                }
758                if let Some(run_id) = record.run_id {
759                    for part in &turn.content {
760                        if let cookie_agent_protocol::PersistedAssistantPart::ToolCall {
761                            id,
762                            provider_item_id,
763                            ..
764                        } = part
765                        {
766                            if !model_calls.insert((run_id, id.clone())) {
767                                return corrupt(path, "model call id is reused within a run");
768                            }
769                            if let Some(provider_item_id) = provider_item_id
770                                && !provider_items.insert((run_id, provider_item_id.clone()))
771                            {
772                                return corrupt(path, "provider item id is reused within a run");
773                            }
774                        }
775                    }
776                }
777            }
778            EventPayload::ModelUsageRecorded { model_turn_seq, .. } => {
779                if !usage_turns.insert(*model_turn_seq) {
780                    return corrupt(path, "usage ownership does not match its model turn");
781                }
782            }
783            EventPayload::InternalAgentStarted {
784                internal_run_id, ..
785            } => {
786                if !internal_runs.insert(*internal_run_id) {
787                    return corrupt(path, "internal_run_id has more than one start");
788                }
789            }
790            EventPayload::ToolCallStarted { start } => {
791                if !tool_starts.insert(start.tool_call_id) {
792                    return corrupt(path, "tool_call_id has more than one start");
793                }
794            }
795            EventPayload::ToolCallTerminated { termination } => {
796                if !tool_terminations.insert(termination.tool_call_id) {
797                    return corrupt(path, "tool call has more than one terminal event");
798                }
799            }
800            EventPayload::ApprovalRequested { request }
801                if !approvals.insert(request.approval_id()) =>
802            {
803                return corrupt(
804                    path,
805                    "approval_id has more than one ApprovalRequested event",
806                );
807            }
808            _ => {}
809        }
810    }
811    Ok(())
812}

814fn validate_record_local(path: &Path, record: &StoredEvent) -> Result<(), EventLogError> {
815    match &record.payload {
816        EventPayload::SessionReverted { through_seq } => {
817            if record.run_id.is_some() || *through_seq == 0 || *through_seq >= record.seq {
818                return corrupt(
819                    path,
820                    "SessionReverted target is not an existing prior event",
821                );
822            }
823        }
824        EventPayload::SessionPermissionOverlaySet { .. } if record.run_id.is_some() => {
825            return corrupt(path, "SessionPermissionOverlaySet must not have run_id");
826        }
827        EventPayload::DelegateChildTerminated { .. } if record.run_id.is_some() => {
828            return corrupt(path, "DelegateChildTerminated must not have run_id");
829        }
830        EventPayload::DelegatedContextSeeded { .. } if record.run_id.is_some() => {
831            return corrupt(path, "DelegatedContextSeeded must be runless");
832        }
833        EventPayload::PluginEventAdded { .. } | EventPayload::PluginDiagnostic { .. }
834            if record.run_id.is_some() =>
835        {
836            return corrupt(path, "plugin events must be runless");
837        }
838        EventPayload::SessionTitleCommitted { change, .. } => {
839            let runless = matches!(
840                change,
841                cookie_agent_protocol::SessionTitleChange::UserSet { .. }
842                    | cookie_agent_protocol::SessionTitleChange::UserClear { .. }
843                    | cookie_agent_protocol::SessionTitleChange::UserReset { .. }
844                    | cookie_agent_protocol::SessionTitleChange::DelegatedSet { .. }
845            );
846            if runless != record.run_id.is_none() {
847                return corrupt(path, "SessionTitleCommitted has inconsistent run ownership");
848            }
849        }
850        _ => {}
851    }
852    Ok(())
853}

855fn validate_records(
856    path: &Path,
857    session_id: SessionId,
858    records: &[StoredEvent],
859    initial_taint: &ValidationTaint,
860    strict_from_seq: Option<u64>,
861) -> Result<ValidationTaint, EventLogError> {
862    validate_observed_duplicates(path, records)?;
863    let mut taint = initial_taint.clone();
864    let mut runs = HashMap::<RunId, RunAttribution>::new();
865    let mut approval_owners = HashMap::new();
866    let mut attempts = HashMap::<AttemptId, AttemptAttribution>::new();
867    let mut turns = HashMap::<u64, (RunId, cookie_agent_protocol::PersistedModelTurn)>::new();
868    let mut turn_models = HashMap::<u64, cookie_agent_protocol::ResolvedModelRef>::new();
869    let mut usage_turns = HashSet::<u64>::new();
870    let mut internal_runs =
871        HashMap::<cookie_agent_protocol::InternalAgentRunId, InternalRunAttribution>::new();
872    let mut delegations =
873        HashMap::<cookie_agent_protocol::InvocationId, DelegationAttribution>::new();
874    let mut model_call_owners = HashMap::<(RunId, ModelCallId), AssistantToolCallRef>::new();
875    let mut provider_item_owners = HashMap::<(RunId, ProviderItemId), AssistantToolCallRef>::new();
876    let mut tool_starts = HashMap::<ToolCallId, (RunId, ToolCallStart)>::new();
877    let mut terminated_tools = HashSet::<ToolCallId>::new();
878    let mut elided_tools = HashSet::<ToolCallId>::new();
879    let mut next_model_turn_seq = 1_u64;
880    let mut previous_timestamp = None;
881    let mut active_run = None;
882    for (index, record) in records.iter().enumerate() {
883        if index > 0 && record.seq <= records[index - 1].seq {
884            return corrupt(
885                path,
886                format!(
887                    "event sequence {} is not strictly greater than {}",
888                    record.seq,
889                    records[index - 1].seq
890                ),
891            );
892        }
893        if record.session_id != session_id {
894            return corrupt(
895                path,
896                "event envelope session ID does not match its directory",
897            );
898        }
899        if previous_timestamp.is_some_and(|timestamp| record.timestamp < timestamp) {
900            return corrupt(path, "event timestamps are not monotonic");
901        }
902        previous_timestamp = Some(record.timestamp);
903        if index == 0 {
904            let EventPayload::SessionCreated { .. } = &record.payload else {
905                return Err(EventLogError::MissingCreation(path.to_owned()));
906            };
907            if record.run_id.is_some() {
908                return corrupt(path, "invalid initial SessionCreated record");
909            }
910            continue;
911        }
912        if matches!(record.payload, EventPayload::SessionCreated { .. }) {
913            return corrupt(path, "SessionCreated appeared after sequence 1");
914        }
915        validate_record_local(path, record)?;
916        let strict = strict_from_seq.is_some_and(|from| record.seq >= from);
917        let missing_admission = |user_input_seq: u64, run_id: RunId, input: &str| {
918            !records[..index].iter().any(|prior| {
919                prior.seq == user_input_seq
920                    && prior.run_id == Some(run_id)
921                    && matches!(
922                        &prior.payload,
923                        EventPayload::UserInputAdmitted { input: admitted } if admitted == input
924                    )
925            })
926        };
927        let tainted_prerequisite = if strict {
928            false
929        } else {
930            match &record.payload {
931                EventPayload::SkillLoaded { .. } | EventPayload::SkillInvocationNoted { .. }
932                    if record.run_id.is_some() =>
933                {
934                    record.run_id.is_some_and(|run_id| {
935                        !runs.contains_key(&run_id) && taint.run_before(run_id, record.seq)
936                    })
937                }
938                EventPayload::SessionTitleCommitted { .. } if record.run_id.is_some() => {
939                    record.run_id.is_some_and(|run_id| {
940                        !runs.contains_key(&run_id) && taint.run_before(run_id, record.seq)
941                    })
942                }
943                EventPayload::UserInputRecalledV2 {
944                    user_input_seq,
945                    input,
946                } => record.run_id.is_some_and(|run_id| {
947                    missing_admission(*user_input_seq, run_id, input)
948                        && taint.admission_before(*user_input_seq, record.seq)
949                }),
950                EventPayload::ModelAttemptStarted { .. }
951                | EventPayload::InternalAgentStarted { .. }
952                | EventPayload::ModelFallback { .. }
953                | EventPayload::ApprovalRequested { .. } => record.run_id.is_some_and(|run_id| {
954                    !runs.contains_key(&run_id) && taint.run_before(run_id, record.seq)
955                }),
956                EventPayload::TextDelta { attempt_id, .. }
957                | EventPayload::ReasoningDelta { attempt_id, .. }
958                | EventPayload::ModelRequestPrepared { attempt_id, .. }
959                | EventPayload::AttemptAbandoned { attempt_id }
960                | EventPayload::ModelReplayEvaluated { attempt_id, .. }
961                | EventPayload::ModelTurnCommitted { attempt_id, .. } => {
962                    (!attempts.contains_key(attempt_id)
963                        && taint.attempt_before(*attempt_id, record.seq))
964                        || record.run_id.is_some_and(|run_id| {
965                            !runs.contains_key(&run_id) && taint.run_before(run_id, record.seq)
966                        })
967                }
968                EventPayload::ModelUsageRecorded { model_turn_seq, .. } => {
969                    (!turns.contains_key(model_turn_seq)
970                        && taint.turn_before(*model_turn_seq, record.seq))
971                        || record.run_id.is_some_and(|run_id| {
972                            !runs.contains_key(&run_id) && taint.run_before(run_id, record.seq)
973                        })
974                }
975                EventPayload::InternalAgentFallback {
976                    internal_run_id, ..
977                }
978                | EventPayload::InternalAgentUsageRecorded {
979                    internal_run_id, ..
980                } => {
981                    (!internal_runs.contains_key(internal_run_id)
982                        && taint.internal_run_before(*internal_run_id, record.seq))
983                        || record.run_id.is_some_and(|run_id| {
984                            !runs.contains_key(&run_id) && taint.run_before(run_id, record.seq)
985                        })
986                }
987                EventPayload::ToolCallStarted { start } => {
988                    (!turns.contains_key(&start.owner.model_turn_seq)
989                        && taint.turn_before(start.owner.model_turn_seq, record.seq))
990                        || record.run_id.is_some_and(|run_id| {
991                            !runs.contains_key(&run_id) && taint.run_before(run_id, record.seq)
992                        })
993                }
994                EventPayload::ToolCallTerminated { termination } => {
995                    !tool_starts.contains_key(&termination.tool_call_id)
996                        && taint.tool_before(termination.tool_call_id, record.seq)
997                }
998                EventPayload::ToolOutputElided { tool_call_id, .. }
999                | EventPayload::ToolCallProgress { tool_call_id, .. }
1000                | EventPayload::ToolStdinSubmitted { tool_call_id, .. }
1001                | EventPayload::ToolCallLinked { tool_call_id, .. } => {
1002                    !tool_starts.contains_key(tool_call_id)
1003                        && taint.tool_before(*tool_call_id, record.seq)
1004                }
1005                EventPayload::ApprovalEvaluated { approval_id, .. }
1006                | EventPayload::ApprovalEscalated { approval_id, .. }
1007                | EventPayload::ApprovalUserDecisionRecorded { approval_id, .. }
1008                | EventPayload::ApprovalFinalized { approval_id, .. }
1009                | EventPayload::ApprovalCancelled { approval_id, .. }
1010                | EventPayload::ApprovalDoomLoopDetected { approval_id, .. } => {
1011                    !approval_owners.contains_key(approval_id)
1012                        && taint.approval_before(*approval_id, record.seq)
1013                }
1014                EventPayload::TreeApprovalGrantCommitted { grant } => {
1015                    !approval_owners.contains_key(&grant.approval_id)
1016                        && taint.approval_before(grant.approval_id, record.seq)
1017                }
1018                EventPayload::DelegationReserved { reservation, .. } => {
1019                    taint.delegation_before(reservation.invocation_id, record.seq)
1020                        || record.run_id.is_some_and(|run_id| {
1021                            !runs.contains_key(&run_id) && taint.run_before(run_id, record.seq)
1022                        })
1023                }
1024                EventPayload::DelegationStarted { invocation_id, .. }
1025                | EventPayload::DelegationRunStarted { invocation_id, .. }
1026                | EventPayload::DelegationRunAttached { invocation_id, .. } => {
1027                    taint.delegation_before(*invocation_id, record.seq)
1028                        || record.run_id.is_some_and(|run_id| {
1029                            !runs.contains_key(&run_id) && taint.run_before(run_id, record.seq)
1030                        })
1031                }
1032                EventPayload::DelegationFinished {
1033                    invocation_id,
1034                    child_session_id,
1035                    child_run_id,
1036                    ..
1037                } => {
1038                    let repair_matches = delegations.get(invocation_id).is_some_and(|delegation| {
1039                        record.run_id == Some(delegation.parent_run_id)
1040                            && *child_session_id == delegation.child_session_id
1041                            && *child_run_id == delegation.child_run_id
1042                            && !delegation.finished
1043                    });
1044                    (taint.delegation_before(*invocation_id, record.seq) && !repair_matches)
1045                        || record.run_id.is_some_and(|run_id| {
1046                            !runs.contains_key(&run_id) && taint.run_before(run_id, record.seq)
1047                        })
1048                }
1049                _ => record.run_id.is_some_and(|run_id| {
1050                    !runs.contains_key(&run_id) && taint.run_before(run_id, record.seq)
1051                }),
1052            }
1053        };
1054        if tainted_prerequisite {
1055            taint
1056                .mark_event(record)
1057                .map_err(|message| EventLogError::Corrupt {
1058                    path: path.to_owned(),
1059                    message: message.into(),
1060                })?;
1061            continue;
1062        }
1063        match &record.payload {
1064            EventPayload::SessionReverted { through_seq } => {
1065                if record.run_id.is_some() || *through_seq == 0 || *through_seq >= record.seq {
1066                    return corrupt(
1067                        path,
1068                        "SessionReverted target is not an existing prior event",
1069                    );
1070                }
1071            }
1072            EventPayload::SessionPermissionOverlaySet { .. } => {
1073                if record.run_id.is_some() {
1074                    return corrupt(path, "SessionPermissionOverlaySet must not have run_id");
1075                }
1076            }
1077            EventPayload::DelegationReserved {
1078                reservation,
1079                request,
1080                ..
1081            } => {
1082                if record.run_id != Some(reservation.parent_run_id)
1083                    || record.session_id != reservation.parent_session_id
1084                    || !runs.contains_key(&reservation.parent_run_id)
1085                    || delegations
1086                        .insert(
1087                            reservation.invocation_id,
1088                            DelegationAttribution {
1089                                parent_run_id: reservation.parent_run_id,
1090                                child_session_id: reservation.child_session_id,
1091                                resume: request.resume_session_id.is_some(),
1092                                started: false,
1093                                child_run_id: None,
1094                                finished: false,
1095                            },
1096                        )
1097                        .is_some()
1098                {
1099                    return corrupt(path, "delegation reservation ownership is invalid");
1100                }
1101            }
1102            EventPayload::DelegationStarted {
1103                invocation_id,
1104                child_session_id,
1105            } => {
1106                let Some(delegation) = delegations.get_mut(invocation_id) else {
1107                    return corrupt(path, "delegation start appeared before its reservation");
1108                };
1109                if record.run_id != Some(delegation.parent_run_id)
1110                    || *child_session_id != delegation.child_session_id
1111                    || delegation.started
1112                    || delegation.finished
1113                {
1114                    return corrupt(path, "delegation start ownership is invalid");
1115                }
1116                delegation.started = true;
1117            }
1118            EventPayload::DelegationRunStarted {
1119                invocation_id,
1120                child_run_id,
1121            }
1122            | EventPayload::DelegationRunAttached {
1123                invocation_id,
1124                child_run_id,
1125            } => {
1126                let attached =
1127                    matches!(&record.payload, EventPayload::DelegationRunAttached { .. });
1128                let Some(delegation) = delegations.get_mut(invocation_id) else {
1129                    return corrupt(path, "delegation run appeared before its reservation");
1130                };
1131                if record.run_id != Some(delegation.parent_run_id)
1132                    || delegation.child_run_id.is_some()
1133                    || delegation.finished
1134                    || (attached && !delegation.resume)
1135                {
1136                    return corrupt(path, "delegation run ownership is invalid");
1137                }
1138                delegation.child_run_id = Some(*child_run_id);
1139            }
1140            EventPayload::DelegationFinished {
1141                invocation_id,
1142                child_session_id,
1143                child_run_id,
1144                ..
1145            } => {
1146                let Some(delegation) = delegations.get_mut(invocation_id) else {
1147                    return corrupt(path, "delegation finish appeared before its reservation");
1148                };
1149                if record.run_id != Some(delegation.parent_run_id)
1150                    || *child_session_id != delegation.child_session_id
1151                    || *child_run_id != delegation.child_run_id
1152                    || delegation.finished
1153                {
1154                    return corrupt(path, "delegation finish ownership is invalid");
1155                }
1156                if taint.delegation_unrepaired_before(*invocation_id, record.seq) {
1157                    taint.delegation_repairs.insert(*invocation_id, record.seq);
1158                }
1159                delegation.finished = true;
1160            }
1161            EventPayload::SkillLoaded { .. } | EventPayload::SkillInvocationNoted { .. } => {
1162                if record.run_id.is_some() {
1163                    require_started_run(path, &runs, record.run_id)?;
1164                }
1165            }
1166            EventPayload::DelegateChildTerminated { .. } => {
1167                if record.run_id.is_some() {
1168                    return corrupt(path, "DelegateChildTerminated must not have run_id");
1169                }
1170            }
1171            EventPayload::UserInputAdmitted { .. } | EventPayload::UserInputRecalled { .. }
1172                if record.run_id.is_none() =>
1173            {
1174                if active_run.is_some() && !taint.active_run_ordering_before(record.seq) {
1175                    return corrupt(path, "runless UserInputAdmitted requires no active run");
1176                }
1177                if taint.active_run_ordering_before(record.seq) {
1178                    active_run = None;
1179                }
1180            }
1181            EventPayload::UserInputRecalledV2 {
1182                user_input_seq,
1183                input,
1184            } => {
1185                let Some(run_id) = record.run_id else {
1186                    return corrupt(path, "UserInputRecalledV2 is missing run_id");
1187                };
1188                if !records[..index].iter().any(|prior| {
1189                    prior.seq == *user_input_seq
1190                        && prior.run_id == Some(run_id)
1191                        && matches!(
1192                            &prior.payload,
1193                            EventPayload::UserInputAdmitted { input: admitted }
1194                                if admitted == input
1195                        )
1196                }) {
1197                    return corrupt(path, "UserInputRecalledV2 target is not a prior admission");
1198                }
1199            }
1200            EventPayload::DelegatedContextSeeded { .. } => {
1201                if record.run_id.is_some() || !runs.is_empty() {
1202                    return corrupt(
1203                        path,
1204                        "DelegatedContextSeeded must be runless and precede the first run",
1205                    );
1206                }
1207            }
1208            EventPayload::RunStarted {
1209                agent,
1210                selected_suffix,
1211                ..
1212            } => {
1213                let Some(run_id) = record.run_id else {
1214                    return corrupt(path, "RunStarted is missing run_id");
1215                };
1216                let attribution = RunAttribution {
1217                    start_seq: record.seq,
1218                    agent_id: agent.agent.clone(),
1219                    prompt_fingerprint: agent.prompt_fingerprint.clone(),
1220                    selected_suffix: selected_suffix
1221                        .iter()
1222                        .map(crate::policy::wire_resolved)
1223                        .collect(),
1224                    active_fallback_index: 0,
1225                    next_attempt_ordinal: 1,
1226                    attempts_on_active: 0,
1227                    active_attempt: None,
1228                    ordering_tainted: false,
1229                };
1230                if runs.insert(run_id, attribution).is_some() {
1231                    return corrupt(path, "run_id has more than one RunStarted event");
1232                }
1233                active_run = Some(run_id);
1234            }
1235            EventPayload::SessionTitleCommitted { change, .. } => {
1236                let user = matches!(
1237                    change,
1238                    cookie_agent_protocol::SessionTitleChange::UserSet { .. }
1239                        | cookie_agent_protocol::SessionTitleChange::UserClear { .. }

1411                    .insert(*model_turn_seq, (run_id, turn.clone()))

1433                    return corrupt(path, "usage ownership does not match its model turn");
1434                }
1435            }
1436            EventPayload::InternalAgentStarted {
1437                invocation_id,
1438                internal_run_id,
1439                kind,
1440                backend,
1441                ..
1442            } => {

1466                    return corrupt(path, "internal_run_id has more than one start");

1470                invocation_id,
1471                internal_run_id,
1472                kind,
1473                from,
1474                to,
1475                ..
1476            } => {

1482                    cookie_agent_protocol::InternalAgentBackend::Model { resolved_model } => {

1516                    .internal_usage_phase_taint_before(*internal_run_id, record.seq)

1523                internal_run_id,
1524                kind,
1525                agent_id,

1527                ..
1528            } => {

1707                    return corrupt(
1708                        path,

1710                    );
1711                }

1721                    .insert(request.approval_id(), run_id)

1727                    );

1730            EventPayload::ApprovalEvaluated { approval_id, .. }
1731            | EventPayload::ApprovalEscalated { approval_id, .. }
1732            | EventPayload::ApprovalUserDecisionRecorded { approval_id, .. }
1733            | EventPayload::ApprovalFinalized { approval_id, .. }
1734            | EventPayload::ApprovalCancelled { approval_id, .. }
1735            | EventPayload::ApprovalDoomLoopDetected { approval_id, .. } => {
1736                validate_approval_owner(path, &approval_owners, *approval_id, record.run_id)?;
1737            }
1738            EventPayload::TreeApprovalGrantCommitted { grant } => {
1739                validate_approval_owner(path, &approval_owners, grant.approval_id, record.run_id)?;
1740            }
1741            EventPayload::PluginEventAdded { .. } | EventPayload::PluginDiagnostic { .. } => {
1742                if record.run_id.is_some() {
1743                    return corrupt(path, "plugin events must be runless");
1744                }
1745            }
1746            _ => {
1747                require_started_run(path, &runs, record.run_id)?;
1748            }
1749        }
1750        if matches!(
1751            record.payload,
1752            EventPayload::RunCompleted { .. }
1753                | EventPayload::RunFailed { .. }
1754                | EventPayload::RunCancelled { .. }
1755                | EventPayload::RunInterrupted { .. }
1756        ) && record.run_id == active_run
1757        {
1758            active_run = None;
1759        }
1760    }
1761    Ok(taint)
1762}

1764struct LoadedEvents {
1765    records: Vec<StoredEvent>,
1766    usage_cost_stamped: HashSet<u64>,
1767    diagnostics: Vec<EventLoadDiagnostic>,
1768    validation_taint: ValidationTaint,
1769    next_seq: u64,
1770}
1771
1772fn load_event_jsonl(path: &Path) -> Result<LoadedEvents, EventLogError> {
1773    let bytes = read_complete_jsonl(path)?;
1774    let mut records = Vec::new();
1775    let mut usage_cost_stamped = HashSet::new();
1776    let mut diagnostics = Vec::new();
1777    let mut validation_taint = ValidationTaint::default();
1778    let mut last_observed_seq = 0_u64;
1779    for (index, line) in bytes
1780        .split(|byte| *byte == b'\n')
1781        .filter(|line| !line.is_empty())
1782        .enumerate()
1783    {
1784        let line_number = index as u64 + 1;
1785        let mut value = match serde_json::from_slice::<serde_json::Value>(line) {
1786            Ok(value) => value,
1787            Err(error) => {
1788                if index == 0 {
1789                    return corrupt_value(
1790                        path,
1791                        format!("SessionCreated line is unreadable: {error}"),
1792                    );
1793                }
1794                diagnostics.push(EventLoadDiagnostic {
1795                    seq: line_number,
1796                    reason: format!("corrupt JSON line: {error}"),
1797                    engine_version: None,
1798                    skipped: true,
1799                });
1800                validation_taint.mark_broad(line_number, line_number);
1801                continue;
1802            }
1803        };
1804        let Some(object) = value.as_object_mut() else {
1805            if index == 0 {
1806                return corrupt_value(path, "SessionCreated line is not a JSON object");
1807            }
1808            diagnostics.push(EventLoadDiagnostic {
1809                seq: line_number,
1810                reason: "event envelope is not a JSON object".into(),
1811                engine_version: None,
1812                skipped: true,
1813            });
1814            validation_taint.mark_broad(line_number, line_number);
1815            continue;
1816        };
1817        let seq = object
1818            .get("seq")
1819            .and_then(Value::as_u64)
1820            .unwrap_or(line_number);
1821        let engine_version = object
1822            .get("engine_version")
1823            .and_then(Value::as_str)
1824            .map(str::to_owned);
1825        let envelope_run_id = object
1826            .get("run_id")
1827            .and_then(|value| serde_json::from_value(value.clone()).ok());
1828        if object.get("seq").and_then(Value::as_u64).is_some() {
1829            if seq <= last_observed_seq {
1830                return corrupt_value(path, "event sequences are not strictly increasing");
1831            }
1832            if last_observed_seq > 0 && seq > last_observed_seq + 1 {

1869                    format!(
1870                        "SessionCreated envelope has unknown fields: {}",
1871                        unknown.join(", ")
1872                    ),
1873                );
1874            }
1875            diagnostics.push(EventLoadDiagnostic {
1876                seq,
1877                reason: format!("unknown envelope fields: {}", unknown.join(", ")),
1878                engine_version,
1879                skipped: true,
1880            });

1882                validation_taint
1883                    .mark_record(seq, envelope_run_id, payload)
1884                    .map_err(|message| EventLogError::Corrupt {
1885                        path: path.to_owned(),
1886                        message: message.into(),
1887                    })?;
1888            } else {
1889                validation_taint.mark_broad(seq, seq);
1890            }
1891            continue;
1892        }
1893        let mut degraded = Vec::new();
1894        if object
1895            .get("engine_version")
1896            .is_some_and(|version| !version.is_null() && !version.is_string())
1897        {

1909                skipped: true,

1915            .as_object()
1916            .is_some_and(|payload| payload.contains_key("estimated_cost_pico_usd"));

1937                    seq,

1966            }
1967            Err(error) => {
1968                if index == 0 {
1969                    return corrupt_value(
1970                        path,
1971                        format!("SessionCreated is unsupported or corrupt: {error}"),
1972                    );
1973                }
1974                diagnostics.push(EventLoadDiagnostic {
1975                    seq,
1976                    reason: error.to_string(),
1977                    engine_version,
1978                    skipped: true,
1979                });

1981                    validation_taint
1982                        .mark_record(seq, envelope_run_id, payload)
1983                        .map_err(|message| EventLogError::Corrupt {
1984                            path: path.to_owned(),
1985                            message: message.into(),
1986                        })?;
1987                } else {
1988                    validation_taint.mark_broad(seq, seq);
1989                }
1990            }
1991        }
1992    }

1994        diagnostics
1995            .iter()
1996            .filter(|diagnostic| diagnostic.skipped)
1997            .map(|diagnostic| diagnostic.seq)

2000    );

2003        usage_cost_stamped,

2332impl OutputHub {
2333    #[must_use]
2334    pub fn new(call_id: ToolCallId, retention_bytes: usize) -> Self {
2335        Self {
2336            call_id,
2337            limit: retention_bytes,
2338            state: Arc::new(Mutex::new(HubState {
2339                streams: [StreamBuffer::default(), StreamBuffer::default()],

2346    pub fn emit(&self, stream: OutputStream, data: &[u8]) {
2347        if data.is_empty() {
2348            return;
2349        }
2350        let mut state = self
2351            .state
2352            .lock()
2353            .unwrap_or_else(|poisoned| poisoned.into_inner());
2354        if state.finalized {
2355            return;
2356        }
2357        let index = stream_index(stream);
2358        let (delta, end) = {
2359            let buffer = &mut state.streams[index];
2360            let delta = OutputDelta {
2361                call_id: self.call_id,
2362                stream,
2363                byte_offset: buffer.end,
2364                data: STANDARD.encode(data),
2365            };
2366            buffer.chunks.push_back(Chunk {
2367                offset: buffer.end,
2368                data: data.to_vec(),
2369            });
2370            buffer.end += data.len() as u64;
2371            while buffer
2372                .chunks
2373                .iter()
2374                .map(|chunk| chunk.data.len())
2375                .sum::<usize>()
2376                > self.limit
2377            {
2378                if let Some(chunk) = buffer.chunks.pop_front() {
2379                    buffer.start = chunk.offset + chunk.data.len() as u64;
2380                }
2381            }
2382            (delta, buffer.end)
2383        };
2384        let subscribers = &mut state.subscribers[index];
2385        subscribers.retain_mut(|subscriber| {
2386            if let Some(next_offset) = subscriber.gap {
2387                match subscriber.sender.try_send(OutputMessage::Gap(OutputGap {
2388                    call_id: self.call_id,
2389                    stream,
2390                    next_offset,
2391                })) {
2392                    Ok(()) => subscriber.gap = None,
2393                    Err(mpsc::error::TrySendError::Full(_)) => return true,
2394                    Err(mpsc::error::TrySendError::Closed(_)) => return false,
2395                }
2396            }
2397            match subscriber
2398                .sender
2399                .try_send(OutputMessage::Delta(delta.clone()))
2400            {
2401                Ok(()) => true,
2402                Err(mpsc::error::TrySendError::Full(_)) => {
2403                    subscriber.gap = Some(end);
2404                    true
2405                }
2406                Err(mpsc::error::TrySendError::Closed(_)) => false,
2407            }
2408        });
2409    }

2411    /// Holds the same lock while taking the snapshot and registering the
2412    /// receiver, so subsequent deltas cannot fall into a handoff gap.
2413    #[must_use]
2414    pub fn subscribe(
2415        &self,
2416        stream: OutputStream,
2417        queue: usize,
2418    ) -> (OutputSnapshot, mpsc::Receiver<OutputMessage>) {
2419        let mut state = self
2420            .state
2421            .lock()
2422            .unwrap_or_else(|poisoned| poisoned.into_inner());
2423        let index = stream_index(stream);
2424        let buffer = &mut state.streams[index];
2425        let snapshot = OutputSnapshot {
2426            call_id: self.call_id,
2427            start_offset: buffer.start,
2428            end_offset: buffer.end,
2429            chunks: buffer
2430                .chunks
2431                .iter()
2432                .map(|chunk| OutputDelta {
2433                    call_id: self.call_id,
2434                    stream,
2435                    byte_offset: chunk.offset,
2436                    data: STANDARD.encode(&chunk.data),
2437                })
2438                .collect(),
2439        };
2440        let (sender, receiver) = mpsc::channel(queue);
2441        // A nonzero retained start is an explicit loss boundary for a new
2442        // snapshot consumer. Queue the same marker used for lagging live
2443        // subscribers before any later deltas can be registered.
2444        let gap = (buffer.start > 0).then_some(buffer.start);
2445        let gap = match gap {
2446            Some(next_offset) => match sender.try_send(OutputMessage::Gap(OutputGap {
2447                call_id: self.call_id,
2448                stream,
2449                next_offset,
2450            })) {
2451                Ok(()) => None,
2452                Err(mpsc::error::TrySendError::Full(_)) => Some(next_offset),
2453                Err(mpsc::error::TrySendError::Closed(_)) => None,
2454            },
2455            None => None,
2456        };
2457        // A retained finalized hub is a snapshot-only resource. Do not retain a
2458        // sender for a receiver which can never receive another delta.
2459        if !state.finalized {
2460            state.subscribers[index].push(Subscriber { sender, gap });
2461        }
2462        (snapshot, receiver)
2463    }

2470            .lock()
2471            .unwrap_or_else(|poisoned| poisoned.into_inner());
2472        state.finalized = true;
2473        state.subscribers = [Vec::new(), Vec::new()];
2474    }
