7impl Engine {
8    pub(crate) fn load_agent_md(
9        &self,
10        preset: Option<&str>,
11    ) -> Result<Vec<AgentMdEntry>, EngineError> {
12        let config = &self.inner.config.runtime.agent_md;
13        if !config.enabled {
14            return Ok(Vec::new());
15        }
16        let cwd = self.inner.store.cwd();
17        let agents_dir = cwd.join(".cookie-agent").join("agents");
18        let mut entries = Vec::with_capacity(2);
19        let project_entry = if let Some(preset) = preset {
20            let source = format!(".cookie-agent/agents/{preset}/AGENTS.md");
21            match self.read_agent_md_file(
22                &agents_dir.join(preset).join("AGENTS.md"),
23                &source,
24                config.max_bytes,
25            )? {
26                Some(entry) => Some(entry),
27                None => self.read_agent_md_file(
28                    &agents_dir.join("AGENTS.md"),
29                    ".cookie-agent/agents/AGENTS.md",
30                    config.max_bytes,
31                )?,
32            }
33        } else {
34            self.read_agent_md_file(
35                &agents_dir.join("AGENTS.md"),
36                ".cookie-agent/agents/AGENTS.md",
37                config.max_bytes,
38            )?
39        };
40        if let Some(entry) = project_entry {
41            entries.push(entry);
42        }
43        if let Some(entry) =
44            self.read_agent_md_file(&cwd.join("AGENTS.md"), "AGENTS.md", config.max_bytes)?
45        {
46            entries.push(entry);
47        }
48        Ok(entries)
49    }
50
51    fn read_agent_md_file(
52        &self,
53        path: &Path,
54        source: &str,
55        max_bytes: usize,
56    ) -> Result<Option<AgentMdEntry>, EngineError> {
57        let mut file = match File::open(path) {
58            Ok(file) => file,
59            Err(error) if error.kind() == std::io::ErrorKind::NotFound => return Ok(None),
60            Err(source) => {
61                return Err(EngineError::AgentMdIo {
62                    path: path.to_owned(),
63                    source,
64                });
65            }
66        };
67        let metadata_len = file
68            .metadata()
69            .map_err(|source| EngineError::AgentMdIo {
70                path: path.to_owned(),
71                source,
72            })?
73            .len();
74        let mut bytes = Vec::with_capacity(max_bytes.saturating_add(1));
75        (&mut file)
76            .take(max_bytes.saturating_add(1) as u64)
77            .read_to_end(&mut bytes)
78            .map_err(|source| EngineError::AgentMdIo {
79                path: path.to_owned(),
80                source,
81            })?;
82        let final_metadata_len = file
83            .metadata()
84            .map_err(|source| EngineError::AgentMdIo {
85                path: path.to_owned(),
86                source,
87            })?
88            .len();
89        let observed_bytes = metadata_len.max(final_metadata_len).max(bytes.len() as u64);
90        let prefix_len = max_bytes.min(bytes.len());
91        let content = match std::str::from_utf8(&bytes[..prefix_len]) {
92            Ok(content) => content.to_owned(),
93            Err(error) if error.error_len().is_none() => {
94                std::str::from_utf8(&bytes[..error.valid_up_to()])
95                    .expect("valid_up_to is valid UTF-8")
96                    .to_owned()
97            }
98            Err(error) => {
99                return Err(EngineError::AgentMdIo {
100                    path: path.to_owned(),
101                    source: std::io::Error::new(std::io::ErrorKind::InvalidData, error),
102                });
103            }
104        };
105        let original_bytes = observed_bytes.max(content.len() as u64);
106        Ok(Some(AgentMdEntry {
107            source: SafeDisplayText::new(source).expect("AGENTS.md context source is bounded"),
108            truncated: original_bytes > content.len() as u64,
109            original_bytes,
110            content,
111        }))
112    }
