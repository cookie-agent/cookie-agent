64            .intercept_user_input(params.session_id, params.input)
65            .await?
66        {
67            UserInputInterception::Accepted {
68                input,
69                original_input,
70            } => {
71                params.input = input;
72                original_input

75                return Err(EngineError::InputHandled(reason));

77        };
78        let from_model = session.log.last_run_started().map(|(_, _, model)| model);
79        if from_model.as_ref() != Some(&params.selection.model) {
80            let context_id = crate::plugin::plugin_context_id();
81            for plugin in self.inner.plugins.interception_plugins(
82                cookie_agent_protocol::ExtensionInterceptionHook::ModelBeforeSelect,
83            ) {
84                let result = self
85                    .inner
86                    .plugins
87                    .intercept_named::<_, cookie_agent_protocol::ExtensionAllowBlockResult>(
88                        &plugin,
89                        cookie_agent_protocol::PLUGIN_INTERCEPT_MODEL_BEFORE_SELECT_METHOD,
90                        &cookie_agent_protocol::ExtensionModelBeforeSelectParams {
91                            session_id: params.session_id,
92                            context_id: context_id.clone(),
93                            from: from_model.clone(),
94                            to: params.selection.model.clone(),
95                            source: if from_model.is_some() {
96                                cookie_agent_protocol::ExtensionModelSelectSource::User
97                            } else {
98                                cookie_agent_protocol::ExtensionModelSelectSource::Config
99                            },
100                        },
101                        Some(params.session_id),
102                        Some(&context_id),
103                    )
104                    .await;
105                match result {
106                    Ok(result)
107                        if result.action
108                            == cookie_agent_protocol::ExtensionAllowBlockAction::Block =>
109                    {
110                        let reason = result
111                            .reason
112                            .unwrap_or_else(|| format!("model selection blocked by {plugin}"));
113                        self.record_plugin_diagnostic(
114                            params.session_id,
115                            plugin,
116                            PluginDiagnosticKind::HookBlocked,
117                            reason.clone(),
118                        );
119                        return Err(EngineError::ModelSelectionBlocked(reason));
120                    }
121                    Ok(_) => {}
122                    Err(error) => self.record_interception_error(params.session_id, plugin, error),
123                }

126        let staged_skill = self.pending_child_skill(params.session_id);
127        let direct_skill = (staged_skill.is_none())
128            .then(|| cookie_agent_protocol::decode_skill_submission(&params.input))
129            .flatten()
130            .map(|(name, args, prompt)| {
131                params.input = prompt
132                    .unwrap_or_else(|| "Apply the loaded skill to the current session.".into());
133                (name, args)
134            });
135        original_input = original_input.filter(|original| original != &params.input);
136        let direct_plan = direct_skill
137            .as_ref()
138            .map(|(name, args)| self.prepare_user_skill_invocation(name, args))
139            .transpose()?;

141        let prospective_grants = prospective_plan.and_then(|plan| plan.grants());
142        if let Some(run) = session
143            .runs
144            .values()
145            .find(|run| run.client_run_id == params.client_run_id)
146        {
147            if run.input != params.input || run.selection != params.selection {
148                return Err(EngineError::RunIdempotencyConflict);
149            }
150            return Ok(RunStartResult { run_id: run.id });
151        }
152        if session.status == SessionStatus::Running {
153            return Err(EngineError::SessionRunning(params.session_id));
154        }
155        self.resolve_interrupted_direct(params.session_id).await?;
156        let result_limits = policy::ResultLimits {
157            tool_output_max_lines: self.inner.config.runtime.tool_output.max_lines,
158            tool_output_max_bytes: self.inner.config.runtime.tool_output.max_bytes,
159        };
160        let is_root = matches!(session.meta.origin, SessionOrigin::Root);
161        let mut run_policy = match &session.meta.origin {
162            SessionOrigin::Root => {
163                self.reconcile_provider_store()?;
164                let runtime = self.current_runtime();
165                let agents = runtime.agents_for_preset(params.selection.preset.as_deref())?;
166                let agent = resolve_agent(&agents, &params.selection.agent)?;
167                if !agent.runnable_as_root {
168                    return Err(EngineError::NoRunnableModel);
169                }
170                freeze_root_agent_policy(
171                    agent,
172                    Arc::clone(&agents),
173                    runtime,
174                    &params.selection.model,
175                    self.inner.config.runtime.delegation.max_depth,
176                    result_limits,
177                    self.inner.config.runtime.model_retry,
178                    self.inner.config.runtime.prompt_caching.as_cache_config(),
179                )?
180            }
181            SessionOrigin::Delegated { invocation_id, .. } => {
182                if params.selection.preset != session.meta.creation_selection.preset {
183                    return Err(EngineError::NoRunnableModel);
184                }
185                let runtime = self.current_runtime();
186                let agents = runtime.agents_for_preset(params.selection.preset.as_deref())?;
187                let delegation = self
188                    .inner
189                    .delegation_events
190                    .get(*invocation_id)
191                    .filter(|entry| entry.reservation.child_session_id == params.session_id);
192                policy_for_session_selection(
193                    session.creation_agent.clone(),
194                    agents,
195                    runtime,
196                    &params.selection,
197                    result_limits.tool_output_max_lines,
198                    result_limits.tool_output_max_bytes,
199                    self.inner.config.runtime.model_retry,
200                    self.inner.config.runtime.prompt_caching.as_cache_config(),
201                    delegation
202                        .as_ref()
203                        .map(|entry| entry.cache_strategies.as_slice()),
204                )?
205            }
206        };
207        let agent_md = if is_root {
208            self.load_agent_md(params.selection.preset.as_deref())?
209        } else {
210            Vec::new()
211        };
212        self.compose_tool_prompt_sections(&mut run_policy, params.session_id)?;
213        self.compose_skill_listing(
214            params.session_id,
215            &mut run_policy,
216            prospective_grants.as_ref(),
217        )?;
218        let context_id = crate::plugin::plugin_context_id();
219        let mut injected_messages = Vec::new();
220        for plugin in self.inner.plugins.interception_plugins(
221            cookie_agent_protocol::ExtensionInterceptionHook::AgentBeforeStart,
222        ) {
223            let result = self
224                .inner
225                .plugins
226                .intercept_named::<_, cookie_agent_protocol::ExtensionAgentBeforeStartResult>(
227                    &plugin,
228                    cookie_agent_protocol::PLUGIN_INTERCEPT_AGENT_BEFORE_START_METHOD,
229                    &ExtensionAgentBeforeStartParams {
230                        session_id: params.session_id,
231                        context_id: context_id.clone(),
232                        agent_path: run_policy.agent.agent.to_string(),
233                        prompt_context: serde_json::json!({
234                            "session_id": params.session_id,
235                            "input": params.input,
236                            "system_prompt": run_policy.agent.composed_prompt,
237                            "injected_messages": injected_messages,
238                        }),
239                    },
240                    Some(params.session_id),
241                    Some(&context_id),
242                )
243                .await;
244            match result {
245                Ok(result) => {
246                    if let Some(replacement) = result
247                        .replace_system_prompt
248                        .filter(|value| !value.is_empty())
249                    {
250                        if replacement.len() > 128 * 1024 {
251                            self.record_plugin_diagnostic(
252                                params.session_id,
253                                plugin.clone(),
254                                PluginDiagnosticKind::InvalidModification,
255                                "plugin replacement system prompt exceeds the byte limit".into(),
256                            );
257                        } else {
258                            run_policy.agent.composed_prompt = replacement;
259                            run_policy.agent.prompt_fingerprint =
260                                Sha256Digest::of_bytes(run_policy.agent.composed_prompt.as_bytes());
261                        }
262                    }
263                    if let Some(addendum) = result
264                        .append_to_system_prompt
265                        .or(result.addendum)
266                        .filter(|value| !value.is_empty())
267                    {
268                        if run_policy.agent.composed_prompt.len() + 1 + addendum.len() > 128 * 1024
269                        {
270                            self.record_plugin_diagnostic(
271                                params.session_id,
272                                plugin.clone(),
273                                PluginDiagnosticKind::InvalidModification,
274                                "plugin agent addendum exceeds the system prompt byte limit".into(),
275                            );
276                        } else {
277                            run_policy.agent.composed_prompt.push('\n');
278                            run_policy.agent.composed_prompt.push_str(&addendum);
279                            run_policy.agent.prompt_fingerprint =
280                                Sha256Digest::of_bytes(run_policy.agent.composed_prompt.as_bytes());
281                        }
282                    }
283                    if let Some(message) = result.inject_message {
284                        if message.content.trim().is_empty()
285                            || message.role == cookie_agent_protocol::ExtensionMessageRole::Tool
286                        {
287                            self.record_plugin_diagnostic(
288                                params.session_id,
289                                plugin,
290                                PluginDiagnosticKind::InvalidModification,
291                                "plugin injected an empty or unsupported tool-role message".into(),
292                            );
293                        } else {
294                            injected_messages.push(message);
295                        }
296                    }
297                }
298                Err(error) => {
299                    let kind = if error.contains("crashed") || error.contains("not connected") {
300                        PluginDiagnosticKind::InterceptionCrash
301                    } else {
302                        PluginDiagnosticKind::InterceptionTimeout
303                    };
304                    self.record_plugin_diagnostic(params.session_id, plugin, kind, error);
305                }
306            }
307        }
308        run_policy.internal_agents = self.freeze_internal_agent_definitions(&run_policy)?;
309        let run_id = RunId::new_v7();
310        let input_through_seq = session.meta.last_event_seq;
311        self.append(
312            params.session_id,
313            Some(run_id),
314            origin.clone(),
315            Event::RunStarted {
316                client_run_id: params.client_run_id.clone(),
317                selection: params.selection.clone(),
318                agent: Box::new(run_policy.agent.clone()),
319                runtime_revision: run_policy.runtime.result.snapshot.runtime_revision.clone(),
320                catalog_revision: run_policy.runtime.result.snapshot.catalog_revision.clone(),
321                provider_state_revision: run_policy
322                    .runtime
323                    .result
324                    .snapshot
325                    .provider_state_revision
326                    .clone(),
327                model_revision: run_policy.runtime.result.snapshot.model_revision.clone(),
328                agent_revision: run_policy.runtime.result.snapshot.agent_revision.clone(),
329                recipe_registry_revision: run_policy
330                    .runtime
331                    .result
332                    .snapshot
333                    .recipe_registry_revision
334                    .clone(),
335                manifest_revision: run_policy.selected_suffix[0].manifest_revision.clone(),
336                selected_suffix: run_policy.selected_suffix.clone(),
337                internal_agents: run_policy.internal_agents.clone(),
338                input_through_seq,
339            },
340        )
341        .await?;
342        let active = Arc::new(ActiveRun {
343            session: params.session_id,
344            policy: Arc::new(run_policy),
345            cancellation: CancellationToken::new(),
346            cancelled_committed: Mutex::new(false),
347            stdin: Mutex::new(HashMap::new()),
348            fallback_index: AtomicU64::new(0),
349        });
350        self.inner
351            .active
352            .lock()
353            .unwrap_or_else(|poisoned| poisoned.into_inner())
354            .insert(run_id, active.clone());
355        if !agent_md.is_empty()
356            && let Err(error) = self
357                .append(
358                    params.session_id,
359                    Some(run_id),
360                    event_origin("engine:agent-md"),
361                    Event::AgentMdLoaded { entries: agent_md },
362                )
363                .await
364        {
365            return Err(self
366                .terminalize_run_setup_failure(&active, run_id, error)
367                .await);
368        }
369        for message in injected_messages {
370            if let Err(error) = self
371                .append(
372                    params.session_id,
373                    Some(run_id),
374                    event_origin("engine:model-loop"),
375                    Event::MessageInjected {
376                        role: message.role,
377                        input: message.content,
378                    },
379                )
380                .await
381            {
382                return Err(self
383                    .terminalize_run_setup_failure(&active, run_id, error)
384                    .await);
385            }
386        }
387        let serialized_message_bytes = match serialized_input_bytes(&params.input) {
388            Ok(bytes) => bytes,
389            Err(error) => {
390                return Err(self
391                    .terminalize_run_setup_failure(&active, run_id, error)
392                    .await);
393            }
394        };
395        let compacted = self
396            .maybe_predictive_compact_before_input(PredictiveCompactionInput {
397                session: params.session_id,
398                run: run_id,
399                serialized_message_bytes,
400                policy: &active.policy,
401                fallback_index: 0,
402                cancellation: &active.cancellation,
403                actor_direct: false,
404            })
405            .await;
406        if active.cancellation.is_cancelled() {
407            self.append_run_cancelled_once(&active, run_id, None)?;
408            self.inner
409                .active
410                .lock()
411                .unwrap_or_else(|poisoned| poisoned.into_inner())
412                .remove(&run_id);
413            return Ok(RunStartResult { run_id });
414        }
415        if let Err(error) = compacted {
416            return Err(self
417                .terminalize_run_setup_failure(&active, run_id, error)
418                .await);
419        }
420        let setup_result = async {
421            if let Some(original_input) = original_input {
422                self.append(
423                    params.session_id,
424                    Some(run_id),
425                    origin.clone(),
426                    Event::UserInputTransformed {
427                        original_input,
428                        input: params.input.clone(),
429                    },
430                )
431                .await?;
432            }
433            self.append(
434                params.session_id,
435                Some(run_id),
436                origin,
437                Event::UserInputSubmitted {
438                    input: params.input,
439                },
440            )
441            .await
442        }
443        .await;
444        if let Err(error) = setup_result {
445            return Err(self
446                .terminalize_run_setup_failure(&active, run_id, error)
447                .await);
448        }
449        self.clear_skill_turn_state(params.session_id);
450        if staged_skill.is_some()
451            && let Some(plan) = self.take_pending_child_skill(params.session_id)
452            && let Err(error) = self
453                .install_prepared_skill(params.session_id, Some(run_id), &plan)
454                .await
455        {
456            return Err(self
457                .terminalize_run_setup_failure(&active, run_id, error)
458                .await);
459        }
460        // A sweeper may have terminalized this durable run before active-run
461        // registration. Never resurrect a cancelled run with a live loop.
462        if self.run_cancelled_recorded(params.session_id, run_id)? {
463            self.inner
464                .active
465                .lock()
466                .unwrap_or_else(|poisoned| poisoned.into_inner())
467                .remove(&run_id);
468            return Ok(RunStartResult { run_id });
469        }
470        if let Some((invocation_id, generation)) = admission
471            && let Err(error) =
472                self.publish_admission_run(invocation_id, generation, params.session_id, run_id)
473        {
474            self.cancel_run_durably(run_id, Some("delegate admission publication failed".into()))?;
475            self.inner
476                .active
477                .lock()
478                .unwrap_or_else(|poisoned| poisoned.into_inner())
479                .remove(&run_id);
480            return Err(error);
481        }
482        if self.run_cancelled_recorded(params.session_id, run_id)? {
483            self.inner
484                .active
485                .lock()
486                .unwrap_or_else(|poisoned| poisoned.into_inner())
487                .remove(&run_id);
488            return Ok(RunStartResult { run_id });
489        }
490        let engine = self.clone();
491        tokio::spawn(async move {
492            if let Some((name, args)) = direct_skill
493                && let Err(error) = engine
494                    .execute_direct_skill(active.clone(), run_id, name, args)
495                    .await
496            {
497                if matches!(error, EngineError::Permission(_)) {
498                    tokio::select! {
499                                            () = active.cancellation.cancelled() => {
500                                                let _ = engine.append_run_cancelled_once(&active, run_id, None);
501                                            }
502                                            () = tokio::time::sleep(std::time::Duration::from_millis(250)) => {
503                                                let _ = engine
504                                                    .append(
505                                                        active.session,
506                                                        Some(run_id),
507                                                        event_origin("engine:model-loop"),
508                    Event::RunFailed {
509                                                            error: safe_error(&error.to_string()),
510                                                        },
511                                                    )
512                                                    .await;
513                                            }
514                                        }
515                } else if active.cancellation.is_cancelled() {
516                    let _ = engine.append_run_cancelled_once(&active, run_id, None);
517                } else {
518                    let _ = engine
519                        .append(
520                            active.session,
521                            Some(run_id),
522                            event_origin("engine:model-loop"),
523                            Event::RunFailed {
524                                error: safe_error(&error.to_string()),
525                            },
526                        )
527                        .await;
528                }
529                if let Ok(mut active_runs) = engine.inner.active.lock() {
530                    active_runs.remove(&run_id);
531                }
532                return;
533            }
534            if let Err(error) = engine.run_loop(run_id, active).await {
535                // A provider-attempt persistence error may also prevent the
536                // terminal append. Retain this active tombstone for reopen
537                // reconciliation rather than clearing a durably Running run.
538                eprintln!("run {run_id} terminalization failed: {error}");
539                return;
540            }
541            if let Ok(mut active_runs) = engine.inner.active.lock() {
542                active_runs.remove(&run_id);
543            }
544        });
545        Ok(RunStartResult { run_id })
546    }

548    pub(super) async fn maybe_predictive_compact_before_input(
549        &self,
550        input: PredictiveCompactionInput<'_>,
551    ) -> Result<bool, EngineError> {
552        let Some(binding) = input.policy.selected_suffix.get(input.fallback_index) else {
553            return Ok(false);
554        };
555        let Some(context_limit) = binding.descriptor.capabilities.limits.context else {
556            return Ok(false);
557        };
558        let config = &self.inner.config.runtime.context_compaction;
559        let trigger_tokens = resolve_compaction_trigger(context_limit, &config.trigger);
560        let estimator = self
561            .inner
562            .context_token_estimators
563            .lock()
564            .unwrap_or_else(|poisoned| poisoned.into_inner())
565            .get(&input.session)
566            .copied()
567            .unwrap_or_default();
568        if !config.auto_compaction
569            || trigger_tokens == 0
570            || !should_run_predictive_compaction(
571                estimator,
572                input.serialized_message_bytes,
573                trigger_tokens,
574                self.inner.store.is_persisted(input.session)?,
575            )
576        {
577            return Ok(false);
578        }
579        let projection = self.inner.store.get(input.session)?;
580        let events = projection.log.event_snapshot();
581        let before = projection.log.latest_checkpoint_seq();
582        let tools = self.tool_definitions(input.session, input.policy)?;
583        let internal_policy = self.internal_agent_policy(
584            InternalAgentKind::ContextCompaction,
585            input.policy,
586            Some(binding),
587        )?;
588        match self
589            .maybe_compact_context(CompactionInput {
590                session: input.session,
591                run: input.run,
592                cancellation: input.cancellation,
593                binding,
594                owner_policy: input.policy,
595                internal_policy: &internal_policy,
596                tools: &tools,
597                events,
598                force: true,
599                overflow_recovery: false,
600                focus: None,
601                actor_direct: input.actor_direct,
602                origin: event_origin("engine:auto-compact"),
603            })
604            .await
605        {
606            Ok(_) => {}
607            Err(EngineError::CompactionCancelled(_)) => return Ok(false),
608            Err(error) => return Err(error),
609        }
610        Ok(self
611            .inner
612            .store
613            .get(input.session)?
614            .log
615            .latest_checkpoint_seq()
616            > before)
617    }

619    pub(super) async fn intercept_user_input(
620        &self,
621        session_id: SessionId,
622        input: String,
623    ) -> Result<UserInputInterception, EngineError> {
624        let session = self.inner.store.get(session_id)?;
625        if !matches!(session.meta.origin, SessionOrigin::Root) {
626            return Ok(UserInputInterception::Accepted {
627                input,
628                original_input: None,
629            });
630        }
631        self.inner.plugins.await_eager_ready().await;
632        let original_input = input.clone();
633        let mut current = input;
634        let context_id = crate::plugin::plugin_context_id();
635        for plugin in self
636            .inner
637            .plugins
638            .interception_plugins(cookie_agent_protocol::ExtensionInterceptionHook::UserBeforeInput)
639        {
640            let result = self
641                .inner
642                .plugins
643                .intercept_named::<_, cookie_agent_protocol::ExtensionUserBeforeInputResult>(
644                    &plugin,
645                    cookie_agent_protocol::PLUGIN_INTERCEPT_USER_BEFORE_INPUT_METHOD,
646                    &ExtensionUserBeforeInputParams {
647                        session_id,
648                        context_id: context_id.clone(),
649                        text: current.clone(),
650                    },
651                    Some(session_id),
652                    Some(&context_id),
653                )
654                .await;
655            match result {
656                Ok(result) if result.action == ExtensionUserBeforeInputAction::Transform => {
657                    match result.new_text.filter(|text| !text.trim().is_empty()) {
658                        Some(text) => current = text,
659                        None => self.record_plugin_diagnostic(
660                            session_id,
661                            plugin,
662                            PluginDiagnosticKind::InvalidModification,
663                            "plugin transformed user input to empty text".into(),
664                        ),
665                    }
666                }
667                Ok(result) if result.action == ExtensionUserBeforeInputAction::Handled => {
668                    return Ok(UserInputInterception::Handled {
669                        reason: result
670                            .reason
671                            .unwrap_or_else(|| format!("handled by {plugin}")),
672                    });
673                }
674                Ok(_) => {}
675                Err(error) => self.record_interception_error(session_id, plugin, error),
676            }
677        }
678        let audit_original = (current != original_input).then_some(original_input);
679        Ok(UserInputInterception::Accepted {
680            input: current,
681            original_input: audit_original,
682        })
683    }

753    pub(super) async fn run_loop(
754        &self,
755        run_id: RunId,
756        active: Arc<ActiveRun>,
757    ) -> Result<(), EngineError> {
758        // Sticky chain position belongs to this run, not one agent-loop pass.
759        let mut fallback_entry = 0_usize;
760        loop {
761            if active.cancellation.is_cancelled() {
762                self.append_run_cancelled_once(&active, run_id, None)?;
763                return Ok(());
764            }
765            let override_model = self.take_skill_model_override(active.session);
766            let mut turn_policy = None;
767            if let Some(model) = override_model {
768                let descriptor = active
769                    .policy
770                    .runtime
771                    .result
772                    .snapshot
773                    .models
774                    .iter()
775                    .find(|descriptor| descriptor.key == model)
776                    .ok_or(EngineError::NoRunnableModel)?;
777                let selection = cookie_agent_protocol::ModelSelection {
778                    model,
779                    variant: descriptor.default_variant.clone(),
780                };
781                let binding = crate::model_snapshots::binding_for_selection(
782                    &active.policy.runtime.current_manifest,
783                    &active.policy.runtime.models,
784                    &selection,
785                )?;
786                let mut policy = active.policy.as_ref().clone();
787                policy.selected_suffix = vec![binding];
788                turn_policy = Some(policy);
789                fallback_entry = 0;
790            }
791            let policy = turn_policy.as_ref().unwrap_or(&active.policy);
792            let published_tools = self.published_tool_definitions(active.session, policy)?;
793            let prompt_events = self.prompt_events(active.session, run_id).await?;
794            let attempt = match self
795                .stream_attempt(
796                    active.session,
797                    run_id,
798                    &active.cancellation,
799                    policy,
800                    &mut fallback_entry,
801                    prompt_events,
802                    published_tools.definitions.clone(),
803                )
804                .await
805            {
806                Ok(events) => events,
807                Err(error) => {
808                    if active.cancellation.is_cancelled() {
809                        self.append_run_cancelled_once(&active, run_id, None)?;
810                        return Ok(());
811                    }
812                    self.append(
813                        active.session,
814                        Some(run_id),
815                        event_origin("engine:model-loop"),
816                        Event::RunFailed {
817                            error: safe_error(&error.to_string()),
818                        },
819                    )
820                    .await?;
821                    return Ok(());
822                }
823            };
824            let final_text = attempt
825                .turn
826                .content
827                .iter()
828                .filter_map(|part| match part {
829                    PersistedAssistantPart::Text { text, .. } => Some(text.as_str()),
830                    _ => None,
831                })
832                .collect::<String>();
833            if matches!(
834                attempt.turn.finish_reason,
835                cookie_agent_protocol::ModelFinishReason::Cancelled
836                    | cookie_agent_protocol::ModelFinishReason::Aborted
837            ) {
838                active.cancellation.cancel();
839            }
840            let in_stream_results = attempt
841                .turn
842                .content
843                .iter()
844                .filter_map(|part| match part {
845                    PersistedAssistantPart::ToolResult { tool_call_id, .. } => {
846                        Some(tool_call_id.as_str())
847                    }
848                    _ => None,
849                })
850                .collect::<HashSet<_>>();
851            let approvals = attempt
852                .turn
853                .content
854                .iter()
855                .filter_map(|part| match part {
856                    PersistedAssistantPart::ToolApproval {
857                        tool_call_id,
858                        message,
859                        ..
860                    } => Some((tool_call_id.as_str(), message.clone())),
861                    _ => None,
862                })
863                .collect::<HashMap<_, _>>();
864            let calls = attempt
865                .turn
866                .content
867                .iter()
868                .enumerate()
869                .filter_map(|(content_index, part)| match part {
870                    PersistedAssistantPart::ToolCall {
871                        id,
872                        provider_item_id,
873                        name,
874                        input,
875                        ..
876                    } if !in_stream_results.contains(id.as_str()) => Some((
877                        ToolCallId::new_v7(),
878                        content_index as u32,
879                        id.clone(),
880                        provider_item_id.clone(),
881                        name.clone(),
882                        input.clone(),
883                        approvals.get(id.as_str()).cloned().flatten(),
884                    )),
885                    _ => None,
886                })
887                .collect::<Vec<_>>();
888            if active.cancellation.is_cancelled() {
889                self.append_run_cancelled_once(&active, run_id, None)?;
890                return Ok(());
891            }
892            if calls.is_empty() {
893                let promoted = self
894                    .request(active.session, |reply| {
895                        SessionCommand::PromotePendingOrComplete {
896                            run: run_id,
897                            final_text: (!final_text.is_empty()).then_some(final_text.clone()),
898                            complete_if_empty: true,
899                            reply,
900                        }
901                    })
902                    .await?;
903                if promoted {
904                    continue;
905                }
906                return Ok(());
907            }
908            if calls.len() > MAX_PENDING_PREPARED_TOOLS {
909                return Err(ModelError::invalid_response(format!(
910                    "model requested {} prepared tools; the limit is {MAX_PENDING_PREPARED_TOOLS}",
911                    calls.len()
912                ))
913                .into());
914            }
915            let mut prepared = Vec::new();
916            for (id, content_index, model_call_id, provider_item_id, tool, arguments, approval) in
917                &calls
918            {
919                self.inner
920                    .output_hubs
921                    .lock()
922                    .unwrap_or_else(|poisoned| poisoned.into_inner())
923                    .entry(*id)
924                    .or_insert_with(|| OutputHub::new(*id, 64 * 1024));
925                let call = ToolCall {
926                    id: *id,
927                    name: tool.to_string(),
928                    arguments: arguments.clone(),
929                };
930                let prepared_call = self
931                    .prepare_published_tool_call(
932                        active.session,
933                        run_id,
934                        call,
935                        &active.policy,
936                        Arc::clone(&attempt.turn_context),
937                        published_tools.tools.get(tool.as_str()),
938                    )
939                    .await;
940                let operation_fingerprint = prepared_call.prepared.as_ref().map_or_else(
941                    |_| {
942                        fallback_operation_fingerprint(
943                            &prepared_call.call,
944                            prepared_call.permission_name.as_deref(),
945                        )
946                    },
947                    |prepared| OperationFingerprint::from_prepared_operation(prepared.operation()),
948                );
949                self.append(
950                    active.session,
951                    Some(run_id),
952                    event_origin("engine:model-loop"),
953                    Event::ToolCallStarted {
954                        start: ToolCallStart {
955                            tool_call_id: *id,
956                            owner: cookie_agent_protocol::AssistantToolCallRef {
957                                model_turn_seq: attempt.model_turn_seq,
958                                content_index: *content_index,
959                                model_call_id: model_call_id.clone(),
960                                provider_item_id: provider_item_id.clone(),
961                            },
962                            presentation: prepared_call.presentation.clone(),
963                            operation_fingerprint,
964                        },
965                    },
966                )
967                .await?;
968                prepared.push((prepared_call, approval.clone()));
969            }
970            let mut tasks = Vec::new();
971            for (prepared, approval) in prepared {
972                if prepared.prepared.is_err() {
973                    let Err(error) = prepared.prepared else {
974                        unreachable!()
975                    };
976                    tasks.push((PendingTool::ImmediateFailure(error), approval));
977                    continue;
978                }
979                let task = match self.decide_tool_permission(&active, &prepared) {
980                    Ok(permission) => PendingTool::Prepared {
981                        prepared: Box::new(prepared),
982                        permission,
983                    },
984                    Err(error) => PendingTool::ImmediateFailure(error),
985                };
986                tasks.push((task, approval));
987            }
988
989            // Resolve every interactive decision before dispatching any call. This keeps
990            // approval prompts serialized in model order and prevents auto-approved siblings
991            // from starting while an Ask sibling is pending.
992            let mut approved_tasks = Vec::with_capacity(tasks.len());
993            for (task, approval) in tasks {
994                let PendingTool::Prepared {
995                    prepared,
996                    permission,
997                } = task
998                else {
999                    approved_tasks.push(task);
1000                    continue;
1001                };
1002                if active.cancellation.is_cancelled() {
1003                    approved_tasks.push(PendingTool::Prepared {
1004                        prepared,
1005                        permission,
1006                    });
1007                    continue;
1008                }
1009                if let Some(message) = approval {
1010                    let outcome = self
1011                        .request_model_approval(
1012                            &active,
1013                            run_id,
1014                            ModelApprovalInput {
1015                                operation: prepared
1016                                    .prepared
1017                                    .as_ref()
1018                                    .expect("prepared operation")
1019                                    .operation(),
1020                                policy_labels: &prepared
1021                                    .prepared
1022                                    .as_ref()
1023                                    .expect("prepared operation")
1024                                    .policy_labels,
1025                                executor: prepared
1026                                    .prepared
1027                                    .as_ref()
1028                                    .expect("prepared operation")
1029                                    .executor
1030                                    .clone(),
1031                                message: Some(message),
1032                                tool: ApprovalToolInput {
1033                                    name: &prepared.call.name,
1034                                    normalized_parameters: prepared
1035                                        .prepared
1036                                        .as_ref()
1037                                        .expect("prepared operation")
1038                                        .normalized_arguments(),
1039                                },
1040                            },
1041                        )
1042                        .await;
1043                    match outcome {
1044                        Ok(outcome) if outcome.approved => {}
1045                        Ok(outcome) => {
1046                            approved_tasks.push(PendingTool::ImmediateFailure(ToolFailure {
1047                                code: ToolCallFailureCode::ExecutionFailed,
1048                                message: denied_tool_failure(
1049                                    ApprovalDecisionSource::Model,
1050                                    "model approval refused by user",
1051                                    outcome.feedback,
1052                                ),
1053                            }));
1054                            continue;
1055                        }
1056                        Err(error) => {
1057                            approved_tasks.push(PendingTool::ImmediateFailure(ToolFailure {
1058                                code: ToolCallFailureCode::ExecutionFailed,
1059                                message: error.to_string(),
1060                            }));
1061                            continue;
1062                        }
1063                    }
1064                }
1065                match self
1066                    .resolve_tool_permission(&active, run_id, &prepared, &permission)
1067                    .await
1068                {
1069                    Ok(()) => approved_tasks.push(PendingTool::Prepared {
1070                        prepared,
1071                        permission,
1072                    }),
1073                    Err(error) => approved_tasks.push(PendingTool::ImmediateFailure(error)),
1074                }
1075            }
1076
1077            let mut parallel = tokio::task::JoinSet::new();
1078            let mut exclusive = Vec::new();
1079            for (call, task) in calls.into_iter().zip(approved_tasks) {
1080                let concurrency = match &task {
1081                    PendingTool::Prepared { prepared, .. } => prepared.concurrency(),
1082                    PendingTool::ImmediateFailure(_) => ToolConcurrency::Parallel,
1083                };
1084                let execution = (
1085                    call.0,
1086                    call.4,
1087                    call.5,
1088                    task,
1089                    Arc::clone(&attempt.turn_context),
1090                );
1091                if concurrency == ToolConcurrency::Parallel {
1092                    let engine = self.clone();
1093                    let active = active.clone();
1094                    parallel.spawn(async move {
1095                        engine
1096                            .execute_and_commit_tool(
1097                                active,
1098                                run_id,
1099                                execution.0,
1100                                execution.1,
1101                                execution.2,
1102                                execution.3,
1103                                execution.4,
1104                            )
1105                            .await
1106                    });
1107                } else {
1108                    exclusive.push(execution);
1109                }
1110            }
1111
1112            let mut dispatch_error = None;
1113            while let Some(joined) = parallel.join_next().await {
1114                let result = joined.unwrap_or(Err(EngineError::ActorStopped));
1115                if let Err(error) = result
1116                    && dispatch_error.is_none()
1117                {
1118                    dispatch_error = Some(error);
1119                }
1120            }
1121            for (id, tool, arguments, task, turn_context) in exclusive {
1122                let result = self
1123                    .execute_and_commit_tool(
1124                        active.clone(),
1125                        run_id,
1126                        id,
1127                        tool,
1128                        arguments,
1129                        task,
1130                        turn_context,
1131                    )
1132                    .await;
1133                if let Err(error) = result
1134                    && dispatch_error.is_none()
1135                {
1136                    dispatch_error = Some(error);
1137                }
1138            }
1139            if let Some(error) = dispatch_error {
1140                return Err(error);
1141            }
1142            if active.cancellation.is_cancelled() {
1143                self.append_run_cancelled_once(&active, run_id, None)?;
1144                return Ok(());
1145            }
1146            self.request(active.session, |reply| {
1147                SessionCommand::PromotePendingOrComplete {
1148                    run: run_id,
1149                    final_text: None,
1150                    complete_if_empty: false,
1151                    reply,
1152                }
1153            })
1154            .await?;
1155        }
1156    }

1158    #[allow(clippy::too_many_arguments)]
1159    async fn execute_and_commit_tool(
1160        &self,
1161        active: Arc<ActiveRun>,
1162        run: RunId,
1163        id: ToolCallId,
1164        tool: cookie_agent_protocol::SafeCode,
1165        original_arguments: serde_json::Value,
1166        task: PendingTool,
1167        turn_context: Arc<TurnAgentContext>,
1168    ) -> Result<(), EngineError> {
1169        let (mut result, arguments) = if active.cancellation.is_cancelled() {
1170            (
1171                Err(ToolFailure {
1172                    code: ToolCallFailureCode::ExecutionFailed,
1173                    message: "tool call cancelled after it started".into(),
1174                }),
1175                original_arguments,
1176            )
1177        } else {
1178            match task {
1179                PendingTool::Prepared { prepared, .. } => {
1180                    let intercepted_arguments = prepared.intercepted_arguments.clone();
1181                    let result = self
1182                        .execute_approved_tool(active.clone(), run, *prepared, turn_context)
1183                        .await;
1184                    let arguments = intercepted_arguments
1185                        .lock()
1186                        .unwrap_or_else(|poisoned| poisoned.into_inner())
1187                        .clone();
1188                    (result, arguments)
1189                }
1190                PendingTool::ImmediateFailure(failure) => (Err(failure), original_arguments),
1191            }
1192        };
1193        let (mut result_content, is_error) = match &result {
1194            Ok(result) => (result.output.clone(), false),
1195            Err(failure) => (failure.message.clone(), true),
1196        };
1197        let context_id = crate::plugin::plugin_context_id();
1198        for plugin in self
1199            .inner
1200            .plugins
1201            .interception_plugins(cookie_agent_protocol::ExtensionInterceptionHook::ToolAfterResult)
1202        {
1203            let intercepted = self
1204                .inner
1205                .plugins
1206                .intercept_named::<_, cookie_agent_protocol::ExtensionToolAfterResultResult>(
1207                    &plugin,
1208                    cookie_agent_protocol::PLUGIN_INTERCEPT_TOOL_AFTER_RESULT_METHOD,
1209                    &ExtensionToolAfterResultParams {
1210                        session_id: active.session,
1211                        context_id: context_id.clone(),
1212                        tool: tool.to_string(),
1213                        arguments: arguments.clone(),
1214                        result_content: result_content.clone(),
1215                        is_error,
1216                    },
1217                    Some(active.session),
1218                    Some(&context_id),
1219                )
1220                .await;
1221            match intercepted {
1222                Ok(intercepted)
1223                    if intercepted.action == ExtensionToolAfterResultAction::Replace =>
1224                {
1225                    if let Some(replacement) = intercepted.replacement_content {
1226                        if replacement.len() > active.policy.result_limits.tool_output_max_bytes {
1227                            self.record_plugin_diagnostic(
1228                                active.session,
1229                                plugin,
1230                                PluginDiagnosticKind::InvalidModification,
1231                                "plugin replacement exceeds the tool output byte limit".into(),
1232                            );
1233                            continue;
1234                        }
1235                        result_content.clone_from(&replacement);
1236                        match &mut result {
1237                            Ok(result) => result.output = replacement,
1238                            Err(failure) => failure.message = replacement,
1239                        }
1240                    }
1241                }
1242                Ok(_) => {}
1243                Err(error) => {
1244                    let kind = if error.contains("crashed") || error.contains("not connected") {
1245                        PluginDiagnosticKind::InterceptionCrash
1246                    } else {
1247                        PluginDiagnosticKind::InterceptionTimeout
1248                    };
1249                    self.record_plugin_diagnostic(active.session, plugin, kind, error);
1250                }
1251            }
1252        }
1253        self.submit_tool_result_status(active.session, run, id, result)
1254            .await?;
1255        Ok(())
1256    }

1258    /// Streams one Oven attempt directly into the session actor and commits a
1259    /// complete turn only after strict lifecycle validation succeeds.
1260    #[allow(clippy::too_many_arguments)]
1261    pub(super) async fn stream_attempt(
1262        &self,
1263        session: SessionId,
1264        run: RunId,
1265        cancellation: &CancellationToken,
1266        policy: &FrozenRunPolicy,
1267        sticky_entry: &mut usize,
1268        prompt_events: Arc<[StoredEvent]>,
1269        tools: Vec<ToolDefinition>,
1270    ) -> Result<AttemptTurn, EngineError> {
1271        let chain = &policy.selected_suffix;
1272        let composed_prompt = &policy.agent.composed_prompt;
1273        let prompt_fingerprint = &policy.agent.prompt_fingerprint;
1274        let mut entry = *sticky_entry;
1275        let mut last_error = ModelError::invalid_request("model fallback chain is empty");
1276        let mut first_request = true;
1277        while entry < chain.len() {
1278            let binding = &chain[entry];
1279            let model = policy::resolve_model(binding, &policy.runtime)?;
1280            let mut attempts = 0_u32;
1281            let mut context_recovery_attempted = false;
1282            loop {
1283                attempts += 1;
1284                let attempt_id = cookie_agent_protocol::AttemptId::new_v7();
1285                let attempt_ordinal = self
1286                    .inner
1287                    .store
1288                    .get(session)?
1289                    .log
1290                    .events()
1291                    .iter()
1292                    .filter(|event| {
1293                        event.run_id == Some(run)
1294                            && matches!(event.payload, Event::ModelAttemptStarted { .. })
1295                    })
1296                    .count() as u32
1297                    + 1;
1298                self.append(
1299                    session,
1300                    Some(run),
1301                    event_origin("engine:model-loop"),
1302                    Event::ModelAttemptStarted {
1303                        attempt_id,
1304                        attempt_ordinal,
1305                        fallback_index: entry as u32,
1306                        retry_ordinal: attempts - 1,
1307                        resolved_model: wire_model(binding),
1308                        prompt_fingerprint: prompt_fingerprint.clone(),
1309                    },
1310                )
1311                .await?;
1312                let request_events = if first_request {
1313                    first_request = false;
1314                    Arc::clone(&prompt_events)
1315                } else {
1316                    self.prompt_events(session, run).await?
1317                };
1318                let compaction_policy = self.internal_agent_policy(
1319                    InternalAgentKind::ContextCompaction,
1320                    policy,
1321                    Some(binding),
1322                )?;
1323                let uncompacted_events = Arc::clone(&request_events);
1324                let request_events = match self
1325                    .maybe_compact_context(CompactionInput {
1326                        session,
1327                        run,
1328                        cancellation,
1329                        binding,
1330                        owner_policy: policy,
1331                        internal_policy: &compaction_policy,
1332                        tools: &tools,
1333                        events: request_events,
1334                        force: false,
1335                        overflow_recovery: false,
1336                        focus: None,
1337                        actor_direct: false,
1338                        origin: event_origin("engine:auto-compact"),
1339                    })
1340                    .await
1341                {
1342                    Ok(events) => events,
1343                    Err(EngineError::CompactionCancelled(_)) => uncompacted_events,
1344                    Err(error) => return Err(error),
1345                };
1346                let input_through_seq = request_events.last().map_or(0, |event| event.seq);
1347                let context = assemble_model_context(
1348                    &request_events,
1349                    &self.inner.artifacts,
1350                    binding,
1351                    composed_prompt,
1352                )?;
1353                let replay_preflight = context.replay_decisions;
1354                let turn_context = Arc::new(TurnAgentContext {
1355                    agent: policy.agent.agent.clone(),
1356                    model: binding.selection.model.clone(),
1357                    adapter: policy::wire_adapter(binding.protocol_recipe.as_str()),
1358                    adapter_family: policy::adapter_family(binding.protocol_recipe.as_str()),
1359                    capabilities: policy
1360                        .model_capabilities(binding)
1361                        .ok_or(EngineError::NoRunnableModel)?,
1362                });
1363                let mut request = ModelRequest::new(context.history).with_tools(tools.clone());
1364                request.inference.max_output_tokens = match (
1365                    binding.descriptor.capabilities.limits.output,
1366                    policy.agent.max_output_tokens,
1367                ) {
1368                    (Some(model), 0) => Some(model),
1369                    (Some(model), document) => Some(model.min(document)),
1370                    (None, 0) => None,

1957            }
1958            if selection_blocked {

1964                event_origin("engine:model-loop"),

1978                && let Some(active) = active_runs.get(&run)
1979            {
1980                active.fallback_index.store(entry as u64, Ordering::Release);
1981            }
1982        }
1983        Err(last_error.into())
1984    }

2018fn model_retry_delay(
2019    retry_ordinal: u32,
2020    retry_after: Option<std::time::Duration>,
2021    jitter_basis: u64,
2022    backoff_ceiling: std::time::Duration,
2023) -> std::time::Duration {
2024    let backoff = exponential_retry_delay(retry_ordinal, backoff_ceiling);
2025    let jitter_per_mille = 750 + jitter_basis % 501;
2026    let jittered_millis = backoff
2027        .as_millis()
2028        .saturating_mul(u128::from(jitter_per_mille))
2029        / 1000;
2030    let jittered_millis = jittered_millis.clamp(1, backoff_ceiling.as_millis());
2031    let jittered =
2032        std::time::Duration::from_millis(u64::try_from(jittered_millis).unwrap_or(u64::MAX));
2033    jittered.max(retry_after.unwrap_or_default())
2034}

2036fn exponential_retry_delay(
2037    retry_ordinal: u32,
2038    ceiling: std::time::Duration,
2039) -> std::time::Duration {
2040    let multiplier = 1_u64.checked_shl(retry_ordinal).unwrap_or(u64::MAX);
2041    let milliseconds = 1_000_u64
2042        .saturating_mul(multiplier)
2043        .min(u64::try_from(ceiling.as_millis()).unwrap_or(u64::MAX));
2044    std::time::Duration::from_millis(milliseconds)
2045}
