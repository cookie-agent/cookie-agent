43    pub(super) async fn start_run_direct(
44        &self,
45        params: RunStartParams,
46        admission: Option<(InvocationId, u64)>,
47    ) -> Result<RunStartResult, EngineError> {
48        self.inner.mcp.await_eager_ready().await;
49        if let Some((invocation_id, generation)) = admission
50            && !self.admission_generation_live(invocation_id, generation)
51        {
52            return Err(EngineError::MissingTool(
53                "delegate admission was abandoned or superseded".into(),
54            ));
55        }
56        let session = self.inner.store.get(params.session_id)?;
57        if let Some(run) = session
58            .runs
59            .values()
60            .find(|run| run.client_run_id == params.client_run_id)
61        {
62            if run.input != params.input || run.selection != params.selection {
63                return Err(EngineError::RunIdempotencyConflict);
64            }
65            return Ok(RunStartResult { run_id: run.id });
66        }
67        if session.status == SessionStatus::Running {
68            return Err(EngineError::SessionRunning(params.session_id));
69        }
70        self.resolve_interrupted_direct(params.session_id).await?;
71        let result_limits = policy::ResultLimits {
72            tool_output_max_lines: self.inner.config.runtime.tool_output.max_lines,
73            tool_output_max_bytes: self.inner.config.runtime.tool_output.max_bytes,
74        };
75        let run_policy = match &session.meta.origin {
76            SessionOrigin::Root => {
77                self.reconcile_provider_store()?;
78                let runtime = self.current_runtime();
79                let agents = Arc::clone(&runtime.agents);
80                let agent = resolve_agent(&agents, &params.selection.agent)?;
81                if !agent.runnable_as_root {
82                    return Err(EngineError::NoRunnableModel);
83                }
84                freeze_root_agent_policy(
85                    agent,
86                    Arc::clone(&agents),
87                    runtime,
88                    &params.selection.model,
89                    self.inner.config.runtime.delegation.max_depth,
90                    result_limits,
91                )?
92            }
93            SessionOrigin::Delegated { .. } => {
94                let runtime = self.current_runtime();
95                let agents = Arc::clone(&runtime.agents);
96                policy_for_session_selection(
97                    session.creation_agent.clone(),
98                    agents,
99                    runtime,
100                    &params.selection,
101                    result_limits.tool_output_max_lines,
102                    result_limits.tool_output_max_bytes,
103                )?
104            }
105        };
106        let run_id = RunId::new_v7();
107        let input_through_seq = session.meta.last_event_seq;
108        self.append(
109            params.session_id,
110            Some(run_id),
111            Event::RunStarted {
112                client_run_id: params.client_run_id.clone(),
113                selection: params.selection.clone(),
114                agent: Box::new(run_policy.agent.clone()),
115                runtime_revision: run_policy.runtime.result.snapshot.runtime_revision.clone(),
116                catalog_revision: run_policy.runtime.result.snapshot.catalog_revision.clone(),
117                provider_state_revision: run_policy
118                    .runtime
119                    .result
120                    .snapshot
121                    .provider_state_revision
122                    .clone(),
123                model_revision: run_policy.runtime.result.snapshot.model_revision.clone(),
124                agent_revision: run_policy.runtime.result.snapshot.agent_revision.clone(),
125                recipe_registry_revision: run_policy
126                    .runtime
127                    .result
128                    .snapshot
129                    .recipe_registry_revision
130                    .clone(),
131                manifest_revision: run_policy.selected_suffix[0].manifest_revision.clone(),
132                selected_suffix: run_policy.selected_suffix.clone(),
133                input_through_seq,
134            },
135        )
136        .await?;
137        let active = Arc::new(ActiveRun {
138            session: params.session_id,
139            policy: Arc::new(run_policy),
140            cancellation: CancellationToken::new(),
141            cancelled_committed: Mutex::new(false),
142            stdin: Mutex::new(HashMap::new()),
143            fallback_index: AtomicU64::new(0),
144        });
145        self.inner
146            .active
147            .lock()
148            .unwrap_or_else(|poisoned| poisoned.into_inner())
149            .insert(run_id, active.clone());
150        let serialized_message_bytes = serialized_input_bytes(&params.input)?;
151        let compacted = self
152            .maybe_predictive_compact_before_input(PredictiveCompactionInput {
153                session: params.session_id,
154                run: run_id,
155                serialized_message_bytes,
156                policy: &active.policy,
157                fallback_index: 0,
158                cancellation: &active.cancellation,
159                actor_direct: false,
160            })
161            .await;
162        if active.cancellation.is_cancelled() {
163            self.append_run_cancelled_once(&active, run_id, None)?;
164            self.inner
165                .active
166                .lock()
167                .unwrap_or_else(|poisoned| poisoned.into_inner())
168                .remove(&run_id);
169            return Ok(RunStartResult { run_id });
170        }
171        if let Err(error) = compacted {
172            self.inner
173                .active
174                .lock()
175                .unwrap_or_else(|poisoned| poisoned.into_inner())
176                .remove(&run_id);
177            return Err(error);
178        }
179        if let Err(error) = self
180            .append(
181                params.session_id,
182                Some(run_id),
183                Event::UserInputSubmitted {
184                    input: params.input,
185                },
186            )
187            .await
188        {
189            self.inner
190                .active
191                .lock()
192                .unwrap_or_else(|poisoned| poisoned.into_inner())
193                .remove(&run_id);
194            return Err(error);
195        }
196        // A sweeper may have terminalized this durable run before active-run
197        // registration. Never resurrect a cancelled run with a live loop.
198        if self.run_cancelled_recorded(params.session_id, run_id)? {
199            return Ok(RunStartResult { run_id });
200        }
201        if let Some((invocation_id, generation)) = admission
202            && let Err(error) =
203                self.publish_admission_run(invocation_id, generation, params.session_id, run_id)
204        {
205            let cancelled = self
206                .cancel_run_durably(run_id, Some("delegate admission publication failed".into()));
207            self.inner
208                .active
209                .lock()
210                .unwrap_or_else(|poisoned| poisoned.into_inner())
211                .remove(&run_id);
212            cancelled?;
213            return Err(error);
214        }
215        if self.run_cancelled_recorded(params.session_id, run_id)? {
216            self.inner
217                .active
218                .lock()
219                .unwrap_or_else(|poisoned| poisoned.into_inner())
220                .remove(&run_id);
221            return Ok(RunStartResult { run_id });
222        }
223        let engine = self.clone();
224        tokio::spawn(async move {
225            if let Err(error) = engine.run_loop(run_id, active).await {
226                // A provider-attempt persistence error may also prevent the
227                // terminal append. Retain this active tombstone for reopen
228                // reconciliation rather than clearing a durably Running run.
229                eprintln!("run {run_id} terminalization failed: {error}");
230                return;
231            }
232            if let Ok(mut active_runs) = engine.inner.active.lock() {
233                active_runs.remove(&run_id);
234            }
235        });
236        Ok(RunStartResult { run_id })
237    }

239    pub(super) async fn maybe_predictive_compact_before_input(
240        &self,
241        input: PredictiveCompactionInput<'_>,
242    ) -> Result<bool, EngineError> {
243        let Some(binding) = input.policy.selected_suffix.get(input.fallback_index) else {
244            return Ok(false);
245        };
246        let Some(context_limit) = binding.descriptor.capabilities.limits.context else {
247            return Ok(false);
248        };
249        let config = &self.inner.config.runtime.context_compaction;
250        let trigger_tokens = resolve_compaction_trigger(context_limit, &config.trigger);
251        let estimator = self
252            .inner
253            .context_token_estimators
254            .lock()
255            .unwrap_or_else(|poisoned| poisoned.into_inner())
256            .get(&input.session)
257            .copied()
258            .unwrap_or_default();
259        if !config.auto_compaction
260            || trigger_tokens == 0
261            || !should_run_predictive_compaction(
262                estimator,
263                input.serialized_message_bytes,
264                trigger_tokens,
265                self.inner.store.is_persisted(input.session)?,
266            )
267        {
268            return Ok(false);
269        }
270        let events = self.inner.store.get(input.session)?.log.events();
271        let before = latest_checkpoint_seq(&events);
272        let tools = self.tool_definitions(input.session, input.policy)?;
273        let internal_policy = self.internal_agent_policy(
274            InternalAgentKind::ContextCompaction,
275            input.policy,
276            Some(binding),
277        )?;
278        let compacted = self
279            .maybe_compact_context(CompactionInput {
280                session: input.session,
281                run: input.run,
282                cancellation: input.cancellation,
283                binding,
284                owner_policy: input.policy,
285                internal_policy: &internal_policy,
286                tools: &tools,
287                events,
288                force: true,
289                overflow_recovery: false,
290                focus: None,
291                actor_direct: input.actor_direct,
292            })
293            .await?;
294        Ok(latest_checkpoint_seq(&compacted) > before)
295    }

297    pub(super) async fn run_loop(
298        &self,
299        run_id: RunId,
300        active: Arc<ActiveRun>,
301    ) -> Result<(), EngineError> {
302        // Sticky chain position belongs to this run, not one agent-loop pass.
303        let mut fallback_entry = 0_usize;
304        loop {
305            if active.cancellation.is_cancelled() {
306                self.append_run_cancelled_once(&active, run_id, None)?;
307                return Ok(());
308            }
309            let tools = self.tool_definitions(active.session, &active.policy)?;
310            let prompt_events = self.prompt_events(active.session, run_id).await?;
311            let attempt = match self
312                .stream_attempt(
313                    active.session,
314                    run_id,
315                    &active.cancellation,
316                    &active.policy,
317                    &mut fallback_entry,
318                    prompt_events,
319                    tools,
320                )
321                .await
322            {
323                Ok(events) => events,
324                Err(error) => {
325                    if active.cancellation.is_cancelled() {
326                        self.append_run_cancelled_once(&active, run_id, None)?;
327                        return Ok(());
328                    }
329                    self.append(
330                        active.session,
331                        Some(run_id),
332                        Event::RunFailed {
333                            error: safe_error(&error.to_string()),
334                        },
335                    )
336                    .await?;
337                    return Ok(());
338                }
339            };
340            let final_text = attempt
341                .turn
342                .content
343                .iter()
344                .filter_map(|part| match part {
345                    PersistedAssistantPart::Text { text, .. } => Some(text.as_str()),
346                    _ => None,
347                })
348                .collect::<String>();
349            if matches!(
350                attempt.turn.finish_reason,
351                cookie_agent_protocol::ModelFinishReason::Cancelled
352                    | cookie_agent_protocol::ModelFinishReason::Aborted
353            ) {
354                active.cancellation.cancel();
355            }
356            let in_stream_results = attempt
357                .turn
358                .content
359                .iter()
360                .filter_map(|part| match part {
361                    PersistedAssistantPart::ToolResult { tool_call_id, .. } => {
362                        Some(tool_call_id.as_str())
363                    }
364                    _ => None,
365                })
366                .collect::<HashSet<_>>();
367            let approvals = attempt
368                .turn
369                .content
370                .iter()
371                .filter_map(|part| match part {
372                    PersistedAssistantPart::ToolApproval {
373                        tool_call_id,
374                        message,
375                        ..
376                    } => Some((tool_call_id.as_str(), message.clone())),
377                    _ => None,
378                })
379                .collect::<HashMap<_, _>>();
380            let calls = attempt
381                .turn
382                .content
383                .iter()
384                .enumerate()
385                .filter_map(|(content_index, part)| match part {
386                    PersistedAssistantPart::ToolCall {
387                        id,
388                        provider_item_id,
389                        name,
390                        input,
391                        ..
392                    } if !in_stream_results.contains(id.as_str()) => Some((
393                        ToolCallId::new_v7(),
394                        content_index as u32,
395                        id.clone(),
396                        provider_item_id.clone(),
397                        name.clone(),
398                        input.clone(),
399                        approvals.get(id.as_str()).cloned().flatten(),
400                    )),
401                    _ => None,
402                })
403                .collect::<Vec<_>>();
404            if active.cancellation.is_cancelled() {
405                self.append_run_cancelled_once(&active, run_id, None)?;
406                return Ok(());
407            }
408            if calls.is_empty() {
409                let promoted = self
410                    .request(active.session, |reply| {
411                        SessionCommand::PromotePendingOrComplete {
412                            run: run_id,
413                            final_text: (!final_text.is_empty()).then_some(final_text.clone()),
414                            complete_if_empty: true,
415                            reply,
416                        }
417                    })
418                    .await?;
419                if promoted {
420                    continue;
421                }
422                return Ok(());
423            }
424            if calls.len() > MAX_PENDING_PREPARED_TOOLS {
425                return Err(ModelError::invalid_response(format!(
426                    "model requested {} prepared tools; the limit is {MAX_PENDING_PREPARED_TOOLS}",
427                    calls.len()
428                ))
429                .into());
430            }
431            let mut prepared = Vec::new();
432            for (id, content_index, model_call_id, provider_item_id, tool, arguments, approval) in
433                &calls
434            {
435                self.inner
436                    .output_hubs
437                    .lock()
438                    .unwrap_or_else(|poisoned| poisoned.into_inner())
439                    .entry(*id)
440                    .or_insert_with(|| OutputHub::new(*id, 64 * 1024));
441                let call = ToolCall {
442                    id: *id,
443                    name: tool.to_string(),
444                    arguments: arguments.clone(),
445                };
446                let prepared_call = self
447                    .prepare_tool_call(
448                        active.session,
449                        run_id,
450                        call,
451                        &active.policy,
452                        Arc::clone(&attempt.turn_context),
453                    )
454                    .await;
455                let operation_fingerprint = prepared_call.prepared.as_ref().map_or_else(
456                    |_| {
457                        fallback_operation_fingerprint(
458                            &prepared_call.call,
459                            prepared_call.permission_name.as_deref(),
460                        )
461                    },
462                    |prepared| OperationFingerprint::from_prepared_operation(prepared.operation()),
463                );
464                self.append(
465                    active.session,
466                    Some(run_id),
467                    Event::ToolCallStarted {
468                        start: ToolCallStart {
469                            tool_call_id: *id,
470                            owner: cookie_agent_protocol::AssistantToolCallRef {
471                                model_turn_seq: attempt.model_turn_seq,
472                                content_index: *content_index,
473                                model_call_id: model_call_id.clone(),
474                                provider_item_id: provider_item_id.clone(),
475                            },
476                            presentation: prepared_call.presentation.clone(),
477                            operation_fingerprint,
478                        },
479                    },
480                )
481                .await?;
482                prepared.push((prepared_call, approval.clone()));
483            }
484            let mut tasks = Vec::new();
485            for (prepared, approval) in prepared {
486                if prepared.prepared.is_err() {
487                    let Err(error) = prepared.prepared else {
488                        unreachable!()
489                    };
490                    tasks.push(PendingTool::ImmediateFailure(error));
491                    continue;
492                }
493                if let Some(message) = approval {
494                    let outcome = self
495                        .request_model_approval(
496                            &active,
497                            run_id,
498                            ModelApprovalInput {
499                                operation: prepared
500                                    .prepared
501                                    .as_ref()
502                                    .expect("prepared operation")
503                                    .operation(),
504                                policy_labels: &prepared
505                                    .prepared
506                                    .as_ref()
507                                    .expect("prepared operation")
508                                    .policy_labels,
509                                executor: prepared
510                                    .prepared
511                                    .as_ref()
512                                    .expect("prepared operation")
513                                    .executor
514                                    .clone(),
515                                message: Some(message),
516                                tool: ApprovalToolInput {
517                                    name: &prepared.call.name,
518                                    normalized_parameters: prepared
519                                        .prepared
520                                        .as_ref()
521                                        .expect("prepared operation")
522                                        .normalized_arguments(),
523                                },
524                            },
525                        )
526                        .await?;
527                    if outcome.approved {
528                        tasks.push(PendingTool::Prepared(Box::new(prepared)));
529                    } else {
530                        tasks.push(PendingTool::ImmediateFailure(ToolFailure {
531                            code: ToolCallFailureCode::ExecutionFailed,
532                            message: denied_tool_failure(
533                                ApprovalDecisionSource::Model,
534                                "model approval refused by user",
535                                outcome.feedback,
536                            ),
537                        }));
538                    }
539                } else {
540                    tasks.push(PendingTool::Prepared(Box::new(prepared)));
541                }
542            }
543            // Awaiting task handles is outside any session actor. Results are
544            // committed in provider tool-call order, regardless of completion order.
545            for (id, task) in calls.iter().map(|call| call.0).zip(tasks) {
546                let result = if active.cancellation.is_cancelled() {
547                    Err(ToolFailure {
548                        code: ToolCallFailureCode::ExecutionFailed,
549                        message: "tool call cancelled after it started".into(),
550                    })
551                } else {
552                    match task {
553                        PendingTool::Prepared(prepared) => {
554                            self.execute_tool(
555                                active.clone(),
556                                run_id,
557                                *prepared,
558                                Arc::clone(&attempt.turn_context),
559                            )
560                            .await
561                        }
562                        PendingTool::ImmediateFailure(failure) => Err(failure),
563                    }
564                };
565                self.submit_tool_result_status(active.session, run_id, id, result)
566                    .await?;
567            }
568            if active.cancellation.is_cancelled() {
569                self.append_run_cancelled_once(&active, run_id, None)?;
570                return Ok(());
571            }
572            self.request(active.session, |reply| {
573                SessionCommand::PromotePendingOrComplete {
574                    run: run_id,
575                    final_text: None,
576                    complete_if_empty: false,
577                    reply,
578                }
579            })
580            .await?;
581        }
582    }

584    /// Streams one Oven attempt directly into the session actor and commits a
585    /// complete turn only after strict lifecycle validation succeeds.
586    #[allow(clippy::too_many_arguments)]
587    pub(super) async fn stream_attempt(
588        &self,
589        session: SessionId,
590        run: RunId,
591        cancellation: &CancellationToken,
592        policy: &FrozenRunPolicy,
593        sticky_entry: &mut usize,
594        prompt_events: Vec<StoredEvent>,
595        tools: Vec<ToolDefinition>,
596    ) -> Result<AttemptTurn, EngineError> {
597        let chain = &policy.selected_suffix;
598        let composed_prompt = &policy.agent.composed_prompt;
599        let prompt_fingerprint = &policy.agent.prompt_fingerprint;
600        let mut entry = *sticky_entry;
601        let mut last_error = ModelError::invalid_request("model fallback chain is empty");
602        let mut first_request = true;
603        while entry < chain.len() {
604            let binding = &chain[entry];
605            let model = policy::resolve_model(binding, &policy.runtime)?;
606            let mut attempts = 0_u32;
607            let mut context_recovery_attempted = false;
608            loop {
609                attempts += 1;
610                let attempt_id = cookie_agent_protocol::AttemptId::new_v7();
611                let attempt_ordinal = self
612                    .inner
613                    .store
614                    .get(session)?
615                    .log
616                    .events()
617                    .iter()
618                    .filter(|event| {
619                        event.run_id == Some(run)
620                            && matches!(event.payload, Event::ModelAttemptStarted { .. })
621                    })
622                    .count() as u32
623                    + 1;
624                self.append(
625                    session,
626                    Some(run),
627                    Event::ModelAttemptStarted {
628                        attempt_id,
629                        attempt_ordinal,
630                        fallback_index: entry as u32,
631                        retry_ordinal: attempts - 1,
632                        resolved_model: wire_model(binding),
633                        prompt_fingerprint: prompt_fingerprint.clone(),
634                    },
635                )
636                .await?;
637                let request_events = if first_request {
638                    first_request = false;
639                    prompt_events.clone()
640                } else {
641                    self.prompt_events(session, run).await?
642                };
643                let compaction_policy = self.internal_agent_policy(
644                    InternalAgentKind::ContextCompaction,
645                    policy,
646                    Some(binding),
647                )?;
648                let request_events = self
649                    .maybe_compact_context(CompactionInput {
650                        session,
651                        run,
652                        cancellation,
653                        binding,
654                        owner_policy: policy,
655                        internal_policy: &compaction_policy,
656                        tools: &tools,
657                        events: request_events,
658                        force: false,
659                        overflow_recovery: false,
660                        focus: None,
661                        actor_direct: false,
662                    })
663                    .await?;
664                let input_through_seq = request_events.last().map_or(0, |event| event.seq);
665                let context = assemble_model_context(
666                    &request_events,
667                    &self.inner.artifacts,
668                    binding,
669                    composed_prompt,
670                )?;
671                let serialized_context_bytes = serde_json::to_vec(&(&context.history, &tools))
672                    .map_err(|error| ModelError::invalid_request(error.to_string()))?
673                    .len();
674                let replay_preflight = context.replay_decisions;
675                let turn_context = Arc::new(TurnAgentContext {
676                    agent: policy.agent.agent.clone(),
677                    capabilities: policy
678                        .model_capabilities(binding)
679                        .ok_or(EngineError::NoRunnableModel)?,
680                });
681                let mut request = ModelRequest::new(context.history).with_tools(tools.clone());
682                if let Some(native_context) = context.native_context {
683                    request = request.with_native_context(native_context);
684                }
685                let request = model.prepare_request(request);
686                let abort = AbortBridge::new(cancellation.clone());
687                let response = tokio::select! {
688                    result = model.model().stream(request, abort.signal()) => result,
689                    _ = cancellation.cancelled() => {
690                        abort.abort();
691                        Err(ModelError::abort("model request was cancelled"))
692                    }
693                };
694                let (result, meaningful_output) = match response {
695                    Ok(response) => {
696                        let oven_sdk::StreamResponse {
697                            mut stream,
698                            request,
699                            response,
700                        } = response;
701                        self.append(
702                            session,
703                            Some(run),
704                            Event::ModelReplayEvaluated {
705                                attempt_id,
706                                resolved_model: wire_model(binding),
707                                ordered_decisions: replay_decisions_with_preflight(
708                                    &request.replay.decisions,
709                                    binding,
710                                    &replay_preflight,
711                                ),
712                            },
713                        )
714                        .await?;
715                        let mut accumulator = TurnAccumulator::default();
716                        let mut failure = None;
717                        let mut meaningful_output = false;
718                        loop {
719                            let item = tokio::select! {
720                                item = stream.next() => item,
721                                _ = cancellation.cancelled() => {
722                                    abort.abort();
723                                    failure = Some(Box::new(ModelError::abort("model stream was cancelled")));
724                                    break;
725                                }
726                            };
727                            let Some(item) = item else { break };
728                            match item {
729                                Ok(part) => match accumulator.push(part) {
730                                    Ok(effect) => {
731                                        meaningful_output |= effect.meaningful;
732                                        if let Some(text) = effect.text_delta {
733                                            self.append(
734                                                session,
735                                                Some(run),
736                                                Event::TextDelta { attempt_id, text },
737                                            )
738                                            .await?;
739                                        }
740                                        if let Some(text) = effect.reasoning_delta {
741                                            self.append(
742                                                session,
743                                                Some(run),
744                                                Event::ReasoningDelta { attempt_id, text },
745                                            )
746                                            .await?;
747                                        }
748                                    }
749                                    Err(error) => {
750                                        failure = Some(error);
751                                        break;
752                                    }
753                                },
754                                Err(error) => {
755                                    failure = Some(Box::new(error));
756                                    break;
757                                }
758                            }
759                        }
760                        let completed = match failure {
761                            Some(error) => Err(error),
762                            None => accumulator.finish(),
763                        };
764                        let completed = completed.map(|mut turn| {
765                            for (key, value) in response.response_metadata {
766                                turn.finish.response_metadata.entry(key).or_insert(value);
767                            }
768                            if let Some(status) = response.http_status {
769                                turn.finish
770                                    .response_metadata
771                                    .entry("oven.http_status".into())
772                                    .or_insert_with(|| serde_json::Value::from(status));
773                            }
774                            if let Some(request_id) = response.request_id {
775                                turn.finish
776                                    .response_metadata
777                                    .entry("oven.request_id".into())
778                                    .or_insert_with(|| serde_json::Value::from(request_id));
779                            }
780                            if !request.provider_metadata.is_empty() {
781                                turn.finish.provider_metadata.insert(
782                                    "oven.request".into(),
783                                    serde_json::to_value(request.provider_metadata)
784                                        .expect("safe request metadata serializes"),
785                                );
786                            }
787                            turn
788                        });
789                        (completed, meaningful_output)
790                    }
791                    Err(error) => (Err(Box::new(error)), false),
792                };
793                match result {
794                    Ok(turn) => {
795                        let (turn, warnings) = persist_turn(turn, &self.inner.artifacts, binding)?;
796                        let resolved_model = wire_model(binding);
797                        let model_turn_seq = self.next_model_turn_seq(session)?;
798                        self.append(
799                            session,
800                            Some(run),
801                            Event::ModelTurnCommitted {
802                                attempt_id,
803                                model_turn_seq,
804                                resolved_model,
805                                input_through_seq,
806                                turn: turn.clone(),
807                                warnings,
808                            },
809                        )
810                        .await?;
811                        self.inner
812                            .context_token_estimators
813                            .lock()
814                            .unwrap_or_else(|poisoned| poisoned.into_inner())
815                            .entry(session)
816                            .or_default()
817                            .record_committed_turn(
818                                serialized_context_bytes,
819                                turn.usage.input_tokens,
820                            );
821                        if !matches!(
822                            self.inner.store.get(session)?.meta.origin,
823                            cookie_agent_protocol::SessionOrigin::Delegated { .. }
824                        ) {
825                            let title_policy = self.internal_agent_policy(
826                                InternalAgentKind::SessionTitle,
827                                policy,
828                                Some(binding),
829                            )?;
830                            self.maybe_generate_session_title(
831                                session,
832                                run,
833                                input_through_seq,
834                                cancellation,
835                                &title_policy,
836                            )
837                            .await?;
838                        }
839                        return Ok(AttemptTurn {
840                            turn,
841                            model_turn_seq,
842                            turn_context,
843                        });
844                    }
845                    Err(error)
846                        if error.kind == oven_sdk::ModelErrorKind::ContextLength
847                            && !meaningful_output
848                            && !context_recovery_attempted =>
849                    {
850                        self.append(session, Some(run), Event::AttemptAbandoned { attempt_id })
851                            .await?;
852                        context_recovery_attempted = true;
853                        let before = self
854                            .inner
855                            .store
856                            .get(session)?
857                            .log
858                            .events()
859                            .iter()
860                            .rev()
861                            .find_map(|event| {
862                                matches!(event.payload, Event::ContextCheckpointCommitted { .. })
863                                    .then_some(event.seq)
864                            })
865                            .unwrap_or(0);
866                        let recovery_events = self.prompt_events(session, run).await?;
867                        let recovery_policy = self.internal_agent_policy(
868                            InternalAgentKind::ContextCompaction,
869                            policy,
870                            Some(binding),
871                        )?;
872                        let recovered = self
873                            .maybe_compact_context(CompactionInput {
874                                session,
875                                run,
876                                cancellation,
877                                binding,
878                                owner_policy: policy,
879                                internal_policy: &recovery_policy,
880                                tools: &tools,
881                                events: recovery_events,
882                                force: true,
883                                overflow_recovery: true,
884                                focus: None,
885                                actor_direct: false,
886                            })
887                            .await?;
888                        let after = recovered
889                            .iter()
890                            .rev()
891                            .find_map(|event| {
892                                matches!(event.payload, Event::ContextCheckpointCommitted { .. })
893                                    .then_some(event.seq)
894                            })
895                            .unwrap_or(0);
896                        if after > before {
897                            continue;
898                        }
899                        return Err(EngineError::Model(error));
900                    }
901                    Err(error) if classify_model_error(&error) == ErrorPolicy::FailRun => {
902                        self.append(session, Some(run), Event::AttemptAbandoned { attempt_id })
903                            .await?;
904                        return Err(EngineError::Model(error));
905                    }
906                    Err(error)
907                        if classify_model_error(&error) == ErrorPolicy::RetryEntry
908                            && attempts <= 2
909                            && !meaningful_output =>
910                    {
911                        self.append(session, Some(run), Event::AttemptAbandoned { attempt_id })
912                            .await?;
913                        tokio::select! {
914                            _ = tokio::time::sleep(std::time::Duration::from_millis(100_u64 << (attempts - 1))) => {}
915                            _ = cancellation.cancelled() => return Err(ModelError::abort("model retry was cancelled").into()),
916                        }
917                    }
918                    Err(error) => {
919                        self.append(session, Some(run), Event::AttemptAbandoned { attempt_id })
920                            .await?;
921                        last_error = *error;
922                        break;
923                    }
924                }
925            }
926            let Some(next) = chain.get(entry + 1) else {
927                return Err(last_error.into());
928            };
929            self.append(
930                session,
931                Some(run),
932                Event::ModelFallback {
933                    from: wire_model(binding),
934                    to: wire_model(next),
935                    from_fallback_index: entry as u32,
936                    to_fallback_index: entry as u32 + 1,
937                    error: model_error_summary(&last_error),
938                    attempts_on_from: attempts,
939                },
940            )
941            .await?;
942            entry += 1;
943            *sticky_entry = entry;
944            if let Ok(active_runs) = self.inner.active.lock()
945                && let Some(active) = active_runs.get(&run)
946            {
947                active.fallback_index.store(entry as u64, Ordering::Release);
948            }
949        }
950        Err(last_error.into())
951    }
952}

954pub(super) fn serialized_input_bytes(input: &str) -> Result<usize, EngineError> {
955    serde_json::to_vec(input)
956        .map(|bytes| bytes.len())
957        .map_err(|error| ModelError::invalid_request(error.to_string()).into())
958}
