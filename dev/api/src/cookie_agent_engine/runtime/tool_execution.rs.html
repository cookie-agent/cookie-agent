67    async fn prepare_tool_call_with_publication(
68        &self,
69        session_id: SessionId,
70        run: RunId,
71        call: ToolCall,
72        policy: &FrozenRunPolicy,
73        turn_context: Arc<crate::TurnAgentContext>,
74        published: Option<Option<&PublishedTool>>,
75    ) -> PreparedToolCall {
76        let fallback_presentation = tool_title_only(&call.name);
77        let session = match self.inner.store.get(session_id) {
78            Ok(session) => session,
79            Err(error) => {
80                return PreparedToolCall {
81                    intercepted_arguments: Arc::new(std::sync::Mutex::new(call.arguments.clone())),
82                    call,
83                    permission_name: None,
84                    presentation: fallback_presentation,
85                    prepared: Err(ToolFailure {
86                        code: ToolCallFailureCode::ExecutionFailed,
87                        message: error.to_string(),
88                    }),
89                    interception: None,
90                };
91            }
92        };
93        let depth = session_depth(&session.meta.origin);
94        let permission_overlay = &session.permission_overlay;
95        let grants = self.skill_grants_for_session(session_id);
96        let delegate_enabled = policy.agent.delegation.as_ref().is_some_and(|delegation| {
97            depth < delegation.effective_depth_ceiling
98                && delegation.targets.iter().any(|target| {
99                    policy.registry.get(target).is_some_and(|agent| {
100                        agent.document.frontmatter.enabled
101                            && matches!(
102                                agent.document.frontmatter.mode,
103                                cookie_agent_config::AgentMode::Subagent
104                                    | cookie_agent_config::AgentMode::All
105                            )
106                    })
107                })
108        });
109        let providers = self
110            .inner
111            .tools
112            .lock()
113            .unwrap_or_else(|poisoned| poisoned.into_inner())
114            .clone();
115        let current = providers.iter().find_map(|provider| {
116            provider
117                .tools_for_session(&SessionToolContext {
118                    session: session_id,
119                })
120                .ok()?
121                .into_iter()
122                .find(|tool| tool.name == call.name)
123                .map(|tool| (provider.clone(), tool))
124        });
125        let mut provider = match published {
126            Some(Some(published))
127                if current.as_ref().is_some_and(|(provider, spec)| {
128                    Arc::ptr_eq(provider, &published.provider) && spec == &published.spec
129                }) =>
130            {
131                Some((published.provider.clone(), published.spec.clone()))
132            }
133            Some(Some(_)) => {
134                return PreparedToolCall {
135                    intercepted_arguments: Arc::new(std::sync::Mutex::new(call.arguments.clone())),
136                    prepared: Err(ToolFailure {
137                        code: ToolCallFailureCode::OperationChanged,
138                        message: format!(
139                            "tool definition changed after `{}` was published to the model",
140                            call.name
141                        ),
142                    }),
143                    call,
144                    permission_name: None,
145                    presentation: fallback_presentation,
146                    interception: None,
147                };
148            }
149            Some(None) => {
150                return PreparedToolCall {
151                    intercepted_arguments: Arc::new(std::sync::Mutex::new(call.arguments.clone())),
152                    prepared: Err(ToolFailure {
153                        code: ToolCallFailureCode::ExecutionFailed,
154                        message: format!("tool `{}` was not published to the model", call.name),
155                    }),
156                    call,
157                    permission_name: None,
158                    presentation: fallback_presentation,
159                    interception: None,
160                };
161            }
162            None => current,
163        };
164        if published.is_none()
165            && provider.is_none()
166            && (call.name != "skill" || self.is_direct_skill_call(call.id))
167        {
168            provider = providers.iter().find_map(|candidate| {
169                candidate
170                    .permission_for_unlisted_tool(&call.name)
171                    .ok()
172                    .flatten()
173                    .map(|permission_name| {
174                        (
175                            candidate.clone(),
176                            crate::ToolSpec {
177                                name: call.name.clone(),
178                                permission_name: permission_name.to_owned(),
179                                description: String::new(),
180                                parameters: serde_json::json!(true),
181                            },
182                        )
183                    })
184            });
185        }
186        let Some((provider, tool_spec)) = provider else {
187            return PreparedToolCall {
188                intercepted_arguments: Arc::new(std::sync::Mutex::new(call.arguments.clone())),
189                prepared: Err(ToolFailure {
190                    code: ToolCallFailureCode::ExecutionFailed,
191                    message: format!("tool `{}` is unavailable", call.name),
192                }),
193                call,
194                permission_name: None,
195                presentation: fallback_presentation,
196                interception: None,
197            };
198        };
199        let permission_name = tool_spec.permission_name.clone();
200        let delegation_tool = permission_name == "delegate";
201        let enabled = !delegation_tool || delegate_enabled;
202        let direct_skill = call.name == "skill" && self.is_direct_skill_call(call.id);
203        if !enabled
204            || (!direct_skill
205                && !PermissionPipeline::tool_visible_with_grants(
206                    &policy.agent,
207                    Some(permission_overlay),
208                    grants.as_ref(),
209                    &permission_name,
210                    self.inner.store.cwd(),
211                ))
212        {
213            return PreparedToolCall {
214                intercepted_arguments: Arc::new(std::sync::Mutex::new(call.arguments.clone())),
215                prepared: Err(ToolFailure {
216                    code: ToolCallFailureCode::ExecutionFailed,
217                    message: format!("tool `{}` is not enabled for this session", call.name),
218                }),
219                call,
220                permission_name: Some(permission_name),
221                presentation: fallback_presentation,
222                interception: None,
223            };
224        }
225        let original_resource = provider
226            .get_permission_resource(&call.name, &call.arguments)
227            .ok()
228            .and_then(|(_, resource)| resource);
229        let presentation = provider.presentation(&call);
230        let context = ToolPreparationContext {
231            session: session_id,
232            run,
233            cwd: self.inner.store.cwd().to_owned(),
234            workspace_root: self.inner.store.cwd().to_owned(),
235            turn_context,
236        };
237        let prepared = match provider.prepare(context.clone(), call.clone()).await {
238            Ok(prepared) => {
239                apply_permission_resource(provider.as_ref(), &call.name, &permission_name, prepared)
240                    .map_err(Into::into)
241            }
242            Err(error) => Err(error.into()),
243        };
244        PreparedToolCall {
245            intercepted_arguments: Arc::new(std::sync::Mutex::new(call.arguments.clone())),
246            call,
247            permission_name: Some(permission_name.clone()),
248            presentation,
249            prepared,
250            interception: Some(ToolInterceptionContext {
251                provider,
252                spec: tool_spec,
253                preparation: context,
254                permission_name,
255                permission_resource: original_resource,
256            }),
257        }
258    }

260    pub(super) async fn execute_tool(
261        &self,
262        active: Arc<ActiveRun>,
263        run: RunId,
264        prepared: PreparedToolCall,
265        turn_context: Arc<crate::TurnAgentContext>,
266    ) -> Result<ToolResult, ToolFailure> {
267        let engine = self.clone();
268        {
269            let PreparedToolCall {
270                mut call,
271                prepared,
272                interception,
273                intercepted_arguments,
274                ..
275            } = prepared;
276            let mut prepared = prepared?;
277            let operation = prepared.operation.clone();
278            let policy_labels = prepared.policy_labels.clone();
279            let permission_overlay = engine
280                .inner
281                .store
282                .get(active.session)
283                .map_err(|error| ToolFailure {
284                    code: ToolCallFailureCode::ExecutionFailed,
285                    message: error.to_string(),
286                })?
287                .permission_overlay;
288            let grants = engine.skill_grants_for_session(active.session);
289            let permission = engine.inner.permissions.decide_operation_with_grants(
290                &active.policy.agent,
291                Some(&permission_overlay),
292                grants.as_ref(),
293                &operation,
294                &policy_labels,
295                engine.inner.store.cwd(),
296            );
297            if permission.effect != cookie_agent_protocol::PermissionEffect::Allow {
298                if permission.effect == cookie_agent_protocol::PermissionEffect::Ask {
299                    let allow_tree_grant = operation.resources().iter().all(|resource| {
300                        resource.binding_lifetime
301                            == cookie_agent_protocol::PreparedBindingLifetime::RestartStable
302                            && !matches!(
303                                resource.capability,
304                                cookie_agent_protocol::PermissionAction::Read
305                                    | cookie_agent_protocol::PermissionAction::Write
306                            )
307                    });
308                    let approval_policy = engine
309                        .active_internal_policy(&active, InternalAgentKind::Approval)
310                        .map_err(|error| ToolFailure {
311                            code: ToolCallFailureCode::ExecutionFailed,
312                            message: error.to_string(),
313                        })?;
314                    let request = ApprovalRequest::new(
315                        ApprovalId::new_v7(),
316                        1,
317                        ApprovalTrigger::PermissionPolicy,
318                        operation.clone(),
319                        permission.evaluations.clone(),
320                        ApprovalConstraints {
321                            allow_once: true,
322                            allow_tree_grant,
323                            cancellable: true,
324                            expires_at: approval_expiry(approval_policy.limits.timeout_ms),
325                        },
326                    )
327                    .map_err(|error| ToolFailure {
328                        code: ToolCallFailureCode::ExecutionFailed,
329                        message: error.to_string(),
330                    })?;
331                    let outcome = engine
332                        .await_user_approval(
333                            &active,
334                            run,
335                            request,
336                            prepared.executor.clone(),
337                            true,
338                            ApprovalToolInput {
339                                name: &call.name,
340                                normalized_parameters: prepared.normalized_arguments(),
341                            },
342                        )
343                        .await
344                        .map_err(|error| ToolFailure {
345                            code: ToolCallFailureCode::ExecutionFailed,
346                            message: error.to_string(),
347                        })?;
348                    if !outcome.approved {
349                        return Err(ToolFailure {
350                            code: ToolCallFailureCode::ExecutionFailed,
351                            message: denied_tool_failure(
352                                ApprovalDecisionSource::Policy,
353                                "permission refused by user",
354                                outcome.feedback,
355                            ),
356                        });
357                    }
358                } else {
359                    return Err(ToolFailure {
360                        code: ToolCallFailureCode::ExecutionFailed,
361                        message: denied_tool_failure(
362                            ApprovalDecisionSource::Policy,
363                            "permission denied",
364                            None,
365                        ),
366                    });
367                }
368            }
369            if let Some(interception) = interception {
370                let context_id = crate::plugin::plugin_context_id();
371                for plugin in engine.inner.plugins.interception_plugins(
372                    cookie_agent_protocol::ExtensionInterceptionHook::ToolBeforeCall,
373                ) {
374                    let result = engine
375                        .inner
376                        .plugins
377                        .intercept_named::<_, cookie_agent_protocol::ExtensionToolBeforeCallResult>(
378                            &plugin,
379                            cookie_agent_protocol::PLUGIN_INTERCEPT_TOOL_BEFORE_CALL_METHOD,
380                            &ExtensionToolBeforeCallParams {
381                                session_id: active.session,
382                                context_id: context_id.clone(),
383                                tool: call.name.clone(),
384                                arguments: call.arguments.clone(),
385                                permission_name: interception.permission_name.clone(),
386                                resource: interception.permission_resource.clone(),
387                            },
388                            Some(active.session),
389                            Some(&context_id),
390                        )
391                        .await;
392                    let result = match result {
393                        Ok(result) => result,
394                        Err(error) => {
395                            let kind =
396                                if error.contains("crashed") || error.contains("not connected") {
397                                    PluginDiagnosticKind::InterceptionCrash
398                                } else {
399                                    PluginDiagnosticKind::InterceptionTimeout
400                                };
401                            engine.record_plugin_diagnostic(active.session, plugin, kind, error);
402                            continue;
403                        }
404                    };
405                    if result.action == ExtensionToolBeforeCallAction::Block {
406                        let message = result
407                            .message_to_model
408                            .or(result.reason)
409                            .unwrap_or_else(|| format!("plugin `{plugin}` blocked the tool call"));
410                        engine.record_plugin_diagnostic(
411                            active.session,
412                            plugin,
413                            PluginDiagnosticKind::HookBlocked,
414                            message.clone(),
415                        );
416                        return Err(ToolFailure {
417                            code: ToolCallFailureCode::ExecutionFailed,
418                            message,
419                        });
420                    }
421                    let Some(arguments) = result.modified_arguments else {
422                        continue;
423                    };
424                    let valid_schema = jsonschema::validator_for(&interception.spec.parameters)
425                        .is_ok_and(|validator| validator.is_valid(&arguments));
426                    let modified_resource = interception
427                        .provider
428                        .get_permission_resource(&call.name, &arguments)
429                        .ok()
430                        .and_then(|(_, resource)| resource);
431                    if !arguments.is_object()
432                        || !valid_schema
433                        || modified_resource != interception.permission_resource
434                    {
435                        let message = format!(
436                            "plugin `{plugin}` produced invalid tool arguments or changed the permission resource"
437                        );
438                        engine.record_plugin_diagnostic(
439                            active.session,
440                            plugin,
441                            PluginDiagnosticKind::InvalidModification,
442                            message.clone(),
443                        );
444                        return Err(ToolFailure {
445                            code: ToolCallFailureCode::ExecutionFailed,
446                            message,
447                        });
448                    }
449                    let mut modified_call = call.clone();
450                    modified_call.arguments = arguments;
451                    let candidate = interception
452                        .provider
453                        .prepare(interception.preparation.clone(), modified_call.clone())
454                        .await
455                        .and_then(|prepared| {
456                            apply_permission_resource(
457                                interception.provider.as_ref(),
458                                &modified_call.name,
459                                &interception.permission_name,
460                                prepared,
461                            )
462                        })
463                        .map_err(ToolFailure::from)?;
464                    if candidate.operation.capabilities() != operation.capabilities()
465                        || candidate.operation.resources() != operation.resources()
466                        || candidate.policy_labels != policy_labels
467                    {
468                        let message = format!(
469                            "plugin `{plugin}` changed the approved permission capability or resource"
470                        );
471                        engine.record_plugin_diagnostic(
472                            active.session,
473                            plugin,
474                            PluginDiagnosticKind::InvalidModification,
475                            message.clone(),
476                        );
477                        return Err(ToolFailure {
478                            code: ToolCallFailureCode::ExecutionFailed,
479                            message,
480                        });
481                    }
482                    call = modified_call;
483                    prepared = candidate;
484                    *intercepted_arguments
485                        .lock()
486                        .unwrap_or_else(|poisoned| poisoned.into_inner()) = call.arguments.clone();
487                }
488            }
489            let _serialization_guard = if let Some(key) = &prepared.serialization_key {
490                Some(engine.mutation_lock(key).lock_owned().await)
491            } else {
492                None
493            };
494            let executor = prepared
495                .executor
496                .lock()
497                .await
498                .take()
499                .ok_or_else(|| ToolFailure {
500                    code: ToolCallFailureCode::PreparedCapabilityLost,
501                    message: "prepared executor capability was already consumed or lost".into(),
502                })?;
503            let (progress_tx, mut progress_rx) = mpsc::channel(64);
504            let hub = engine
505                .inner
506                .output_hubs
507                .lock()
508                .unwrap_or_else(|poisoned| poisoned.into_inner())
509                .entry(call.id)
510                .or_insert_with(|| OutputHub::new(call.id, 64 * 1024))
511                .clone();
512            let interactive = call.name == "bash"
513                && call
514                    .arguments
515                    .get("interactive")
516                    .and_then(Value::as_bool)
517                    .unwrap_or(false);
518            let capture = (call.name == "bash")
519                .then(|| OutputCapture::new(engine.inner.artifacts.clone()))
520                .transpose()
521                .map_err(|error| ToolFailure {
522                    code: ToolCallFailureCode::ExecutionFailed,
523                    message: format!("tool output capture setup failed: {error}"),
524                })?;
525            let (stdin_tx, stdin) = ToolStdin::channel(64);
526            if interactive {
527                active
528                    .stdin
529                    .lock()
530                    .unwrap_or_else(|poisoned| poisoned.into_inner())
531                    .insert(call.id, stdin_tx);
532            }
533            let invoke = executor.execute(ToolExecutionContext {
534                session: active.session,
535                run,
536                progress: capture.as_ref().map_or_else(
537                    || ProgressSink::new(progress_tx.clone(), hub.clone()),
538                    |capture| {
539                        ProgressSink::with_capture(
540                            progress_tx.clone(),
541                            hub.clone(),
542                            capture.clone(),
543                        )
544                    },
545                ),
546                cancellation: active.cancellation.child_token(),
547                stdin: interactive.then_some(stdin),
548                turn_context,
549                artifacts: engine.inner.artifacts.clone(),
550            });
551            tokio::pin!(invoke);
552            loop {
553                tokio::select! {
554                    result = &mut invoke => {
555                        active
556                            .stdin
557                            .lock()
558                            .unwrap_or_else(|poisoned| poisoned.into_inner())
559                            .remove(&call.id);
560                        // Tool implementations drain their producers before
561                        // resolving.  Finalizing here makes all emitted deltas
562                        // precede the completion notification committed by the
563                        // session actor.
564                        hub.finalize();
565                        engine.retain_finalized_output_hub(call.id);
566                        return match result {
567                            Ok(result) => {
568                                let bounded = if let Some(capture) = &capture {
569                                    capture.finish(
570                                        result,
571                                        active.policy.result_limits.tool_output_max_lines,
572                                        active.policy.result_limits.tool_output_max_bytes,
573                                    )
574                                } else {
575                                    bound_tool_result(
576                                        result,
577                                        &call.name,
578                                        call.id,
579                                        &engine.inner.artifacts,
580                                        active.policy.result_limits.tool_output_max_lines,
581                                        active.policy.result_limits.tool_output_max_bytes,
582                                    )
583                                };
584                                bounded.map_err(|error| ToolFailure {
585                                    code: ToolCallFailureCode::ExecutionFailed,
586                                    message: error.to_string(),
587                                })
588                            }
589                            Err(error) => {
590                                if let Some(capture) = &capture {
591                                    capture.discard();
592                                }
593                                Err(error.into())
594                            }
595                        };
596                    }
597                    Some(progress) = progress_rx.recv() => {
598                        let _ = engine.append(active.session, Some(run), Event::ToolCallProgress { tool_call_id: progress.tool_call_id, message: safe_display(&progress.message) }).await;
599                    }
600                    _ = active.cancellation.cancelled() => {
601                        active
602                            .stdin
603                            .lock()
604                            .unwrap_or_else(|poisoned| poisoned.into_inner())
605                            .remove(&call.id);
606                        if let Some(capture) = &capture {
607                            capture.discard();
608                        }
609                        hub.finalize();
610                        engine.retain_finalized_output_hub(call.id);
611                        return Err(ToolFailure {
612                            code: ToolCallFailureCode::ExecutionFailed,
613                            message: "tool call cancelled after it started".into(),
614                        });
615                    }
616                }
617            }
618        }
619    }

631    pub(super) fn published_tool_definitions(
632        &self,
633        session: SessionId,
634        policy: &FrozenRunPolicy,
635    ) -> Result<PublishedToolSet, EngineError> {
636        let session_projection = self.inner.store.get(session)?;
637        let grants = self.skill_grants_for_session(session);
638        let depth = session_depth(&session_projection.meta.origin);
639        let delegate_enabled = policy.agent.delegation.as_ref().is_some_and(|delegation| {
640            depth < delegation.effective_depth_ceiling
641                && delegation.targets.iter().any(|target| {
642                    policy.registry.get(target).is_some_and(|agent| {
643                        agent.document.frontmatter.enabled
644                            && matches!(
645                                agent.document.frontmatter.mode,
646                                cookie_agent_config::AgentMode::Subagent
647                                    | cookie_agent_config::AgentMode::All
648                            )
649                    })
650                })
651        });
652        let mut names = HashSet::new();
653        let mut output = Vec::new();
654        let mut published = std::collections::HashMap::new();
655        let providers = self
656            .inner
657            .tools
658            .lock()
659            .unwrap_or_else(|poisoned| poisoned.into_inner())
660            .clone();
661        for provider in &providers {
662            for tool in provider
663                .tools_for_session(&SessionToolContext { session })
664                .map_err(|error| EngineError::MissingTool(error.to_string()))?
665            {
666                let delegation_tool = tool.permission_name == "delegate";
667                let enabled = !delegation_tool || delegate_enabled;
668                if enabled
669                    && PermissionPipeline::tool_visible_with_grants(
670                        &policy.agent,
671                        Some(&session_projection.permission_overlay),
672                        grants.as_ref(),
673                        &tool.permission_name,
674                        self.inner.store.cwd(),
675                    )
676                {
677                    if !names.insert(tool.name.clone()) {
678                        return Err(EngineError::MissingTool(format!(
679                            "tool name `{}` is published by more than one provider",
680                            tool.name
681                        )));
682                    }
683                    let schema = JsonSchema::new(tool.parameters.clone()).map_err(|error| {
684                        EngineError::MissingTool(format!(
685                            "tool `{}` has invalid JSON Schema: {error}",
686                            tool.name
687                        ))
688                    })?;
689                    output.push(ToolDefinition::new(
690                        tool.name.clone(),
691                        tool.description.clone(),
692                        schema,
693                    ));
694                    published.insert(
695                        tool.name.clone(),
696                        PublishedTool {
697                            provider: provider.clone(),
698                            spec: tool,
699                        },
700                    );
701                }
702            }
703        }
704        output.sort_by(|left, right| left.name.cmp(&right.name));
705        Ok(PublishedToolSet {
706            definitions: output,
707            tools: published,
708        })
709    }

712pub(super) fn fallback_operation_fingerprint(
713    call: &ToolCall,
714    permission_name: Option<&str>,
715) -> OperationFingerprint {
716    let action = permission_name
717        .or_else(|| fallback_permission_name(&call.name))
718        .and_then(|name| PermissionPipeline::action_for_permission_name(name).ok())
719        .unwrap_or(cookie_agent_protocol::PermissionAction::Read);
720    let binding = serde_json::to_vec(&call.arguments).expect("tool arguments serialize");
721    let operation = PreparedOperationIdentity::new(
722        Sha256Digest::of_bytes(&binding),
723        vec![cookie_agent_protocol::ApprovalCapability {
724            action,
725            operation: cookie_agent_protocol::PreparedCapabilityOperation::new(format!(
726                "{}:prepare",
727                call.name
728                    .replace(|character: char| !character.is_ascii_alphanumeric(), "_")
729            ))
730            .unwrap_or_else(|_| {
731                cookie_agent_protocol::PreparedCapabilityOperation::new("tool:prepare")
732                    .expect("static operation")
733            }),
734        }],
735        vec![cookie_agent_protocol::PreparedApprovalResource {
736            capability: action,
737            canonical: cookie_agent_protocol::PreparedResourceIdentity::new("tool:prepare-failed")
738                .expect("static resource identity"),
739            binding_digest:
740                cookie_agent_protocol::PreparedResourceDigest::from_canonical_binding_bytes(
741                    &binding,
742                ),
743            binding_lifetime: cookie_agent_protocol::PreparedBindingLifetime::ProcessLocal,
744            boundary: cookie_agent_protocol::ApprovalBoundary::Exact,
745            source: cookie_agent_protocol::ApprovalResourceSource::PrimaryOperation,
746        }],
747        Sha256Digest::of_bytes(b"prepare-failed"),
748    )
749    .expect("fallback prepared identity");
750    OperationFingerprint::from_prepared_operation(&operation)
751}
752
753fn fallback_permission_name(tool_name: &str) -> Option<&'static str> {
754    match tool_name {
755        "read" => Some("read"),
756        "write" | "edit" => Some("write"),
757        "bash" => Some("bash"),
758        "delegate_subagent" | "get_subagent_result" | "steer_subagent" | "cancel_subagent" => {
759            Some("delegate")
760        }
761        "skill" => Some("skill"),
762        _ => None,
763    }
764}

766pub(crate) fn apply_permission_resource(
767    provider: &dyn ToolProvider,
768    name: &str,
769    expected_permission_name: &str,
770    prepared: PreparedTool,
771) -> Result<PreparedTool, ToolError> {
772    let (permission_name, resource) =
773        provider.get_permission_resource(name, prepared.normalized_arguments())?;
774    if permission_name != expected_permission_name
775        && !(permission_name == "plugin" && expected_permission_name.starts_with("plugin:"))
776    {
777        return Err(ToolError::execution(
778            "tool permission metadata changed between discovery and preparation",
779        ));
780    }
781    let action = PermissionPipeline::action_for_permission_name(permission_name)
782        .map_err(|error| ToolError::execution(error.to_string()))?;
783    if prepared
784        .operation()
785        .resources()
786        .iter()
787        .any(|prepared| prepared.capability != action)
788    {
789        return Err(ToolError::execution(
790            "prepared permission capability does not match provider metadata",
791        ));
792    }
793    prepared.with_permission_resource(resource)
794}

810pub(super) fn redact_presentation(value: &str) -> String {
811    const SECRET_MARKERS: [&str; 6] = [
812        "token",
813        "secret",
814        "password",
815        "api_key",
816        "apikey",
817        "authorization",
818    ];
819    let mut sanitized = sanitize_safe_text(value, 512);
820    let lowercase = sanitized.to_ascii_lowercase();
821    if SECRET_MARKERS
822        .iter()
823        .any(|marker| lowercase.contains(marker))
824    {
825        sanitized = "<redacted>".into();
826    }
827    sanitized
828}
829pub(super) struct TruncatedToolOutput {
830    pub(super) content: String,
831}
832
833pub(super) fn truncate_tool_output(
834    output: &str,
835    max_lines: usize,
836    max_bytes: usize,
837) -> Option<TruncatedToolOutput> {
838    let lines = output.split('\n').collect::<Vec<_>>();
839    let line_truncated = lines.len() > max_lines;
840    let mut preview = lines
841        .iter()
842        .take(max_lines)
843        .copied()
844        .collect::<Vec<_>>()
845        .join("\n");
846    let byte_truncated = output.len() > max_bytes || preview.len() > max_bytes;
847    if !line_truncated && !byte_truncated {
848        return None;
849    }
850    if preview.len() > max_bytes {
851        let mut boundary = max_bytes;
852        while boundary > 0 && !preview.is_char_boundary(boundary) {
853            boundary -= 1;
854        }
855        preview.truncate(boundary);
856    }
857    Some(TruncatedToolOutput { content: preview })
858}

860pub(super) fn bound_tool_result(
861    mut result: ToolResult,
862    _tool_name: &str,
863    _call_id: ToolCallId,
864    artifacts: &ArtifactStore,
865    max_lines: usize,
866    max_bytes: usize,
867) -> std::io::Result<ToolResult> {
868    let Some(preview) = truncate_tool_output(&result.output, max_lines, max_bytes) else {
869        return Ok(result);
870    };
871    let original_bytes = result.output.len() as u64;
872    let original_lines = result.output.split('\n').count() as u64;
873    let (retained, _) = artifacts.retain(result.output.as_bytes())?;
874    result.output = preview.content;
875    result.truncation = Some(ToolOutputTruncation {
876        original_bytes,
877        original_lines,
878        retained,
879    });
880    Ok(result)
881}

883pub(crate) fn validate_attachment(
884    mime_type: &str,
885    path: &Path,
886    bytes: &[u8],
887) -> Result<(), ToolError> {
888    if bytes.len() as u64 > MAX_ATTACHMENT_BYTES {
889        return Err(ToolError::resource_limit(format!(
890            "attachment is {} bytes; the limit is {MAX_ATTACHMENT_BYTES} bytes",
891            bytes.len()
892        )));
893    }
894    let validated = approved_media_type(path, bytes)?
895        .ok_or_else(|| ToolError::execution("attachment is not a supported image or PDF"))?;
896    if validated != mime_type {
897        return Err(ToolError::execution(format!(
898            "attachment MIME mismatch: declared {mime_type}, validated {validated}"
899        )));
900    }
901    Ok(())
902}
