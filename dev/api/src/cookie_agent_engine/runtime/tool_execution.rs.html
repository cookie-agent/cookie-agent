34    pub(super) async fn prepare_tool_call(
35        &self,
36        session_id: SessionId,
37        run: RunId,
38        call: ToolCall,
39        policy: &FrozenRunPolicy,
40        turn_context: Arc<crate::TurnAgentContext>,
41    ) -> PreparedToolCall {
42        let fallback_presentation = tool_title_only(&call.name);
43        let session = match self.inner.store.get(session_id) {
44            Ok(session) => session,
45            Err(error) => {
46                return PreparedToolCall {
47                    call,
48                    presentation: fallback_presentation,
49                    prepared: Err(ToolFailure {
50                        code: ToolCallFailureCode::ExecutionFailed,
51                        message: error.to_string(),
52                    }),
53                };
54            }
55        };
56        let depth = session_depth(&session.meta.origin);
57        let delegate_enabled = policy.agent.delegation.as_ref().is_some_and(|delegation| {
58            depth < delegation.effective_depth_ceiling
59                && delegation.targets.iter().any(|target| {
60                    policy.registry.get(target).is_some_and(|agent| {
61                        agent.document.frontmatter.enabled
62                            && matches!(
63                                agent.document.frontmatter.mode,
64                                cookie_agent_config::AgentMode::Subagent
65                                    | cookie_agent_config::AgentMode::All
66                            )
67                    })
68                })
69        });
70        let enabled_tools = policy.tools();
71        if (call.name == "delegate" && !delegate_enabled)
72            || (call.name != "delegate"
73                && (!enabled_tools.contains(&call.name)
74                    || !PermissionPipeline::tool_visible(&policy.agent, &call.name)))
75        {
76            return PreparedToolCall {
77                prepared: Err(ToolFailure {
78                    code: ToolCallFailureCode::ExecutionFailed,
79                    message: format!("tool `{}` is not enabled for this session", call.name),
80                }),
81                call,
82                presentation: fallback_presentation,
83            };
84        }
85        let providers = self
86            .inner
87            .tools
88            .lock()
89            .unwrap_or_else(|poisoned| poisoned.into_inner())
90            .clone();
91        let provider = providers.into_iter().find(|provider| {
92            provider
93                .tools_for_session(&SessionToolContext {
94                    session: session_id,
95                })
96                .ok()
97                .is_some_and(|tools| tools.iter().any(|tool| tool.name == call.name))
98        });
99        let Some(provider) = provider else {
100            return PreparedToolCall {
101                prepared: Err(ToolFailure {
102                    code: ToolCallFailureCode::ExecutionFailed,
103                    message: format!("tool `{}` is unavailable", call.name),
104                }),
105                call,
106                presentation: fallback_presentation,
107            };
108        };
109        let presentation = provider.presentation(&call);
110        let context = ToolPreparationContext {
111            session: session_id,
112            run,
113            cwd: self.inner.store.cwd().to_owned(),
114            workspace_root: self.inner.store.cwd().to_owned(),
115            turn_context,
116        };
117        let prepared = match provider.prepare(context, call.clone()).await {
118            Ok(prepared) => apply_primary_argument_labels(provider.as_ref(), &call.name, prepared)
119                .map_err(Into::into),
120            Err(error) => Err(error.into()),
121        };
122        PreparedToolCall {
123            call,
124            presentation,
125            prepared,
126        }
127    }

129    pub(super) async fn execute_tool(
130        &self,
131        active: Arc<ActiveRun>,
132        run: RunId,
133        prepared: PreparedToolCall,
134        turn_context: Arc<crate::TurnAgentContext>,
135    ) -> Result<ToolResult, ToolFailure> {
136        let engine = self.clone();
137        {
138            let PreparedToolCall { call, prepared, .. } = prepared;
139            let prepared = prepared?;
140            let operation = prepared.operation.clone();
141            let policy_labels = prepared.policy_labels.clone();
142            let _serialization_guard = if let Some(key) = &prepared.serialization_key {
143                Some(engine.mutation_lock(key).lock_owned().await)
144            } else {
145                None
146            };
147            let permission = engine.inner.permissions.decide_operation(
148                &active.policy.agent,
149                &operation,
150                &policy_labels,
151                engine.inner.store.cwd(),
152            );
153            if permission.effect != cookie_agent_protocol::PermissionEffect::Allow {
154                if permission.effect == cookie_agent_protocol::PermissionEffect::Ask {
155                    let allow_tree_grant = operation.resources().iter().all(|resource| {
156                        resource.binding_lifetime
157                            == cookie_agent_protocol::PreparedBindingLifetime::RestartStable
158                            && !matches!(
159                                resource.capability,
160                                cookie_agent_protocol::PermissionAction::Read
161                                    | cookie_agent_protocol::PermissionAction::Write
162                            )
163                    });
164                    let approval_policy = engine
165                        .active_internal_policy(&active, InternalAgentKind::Approval)
166                        .map_err(|error| ToolFailure {
167                            code: ToolCallFailureCode::ExecutionFailed,
168                            message: error.to_string(),
169                        })?;
170                    let request = ApprovalRequest::new(
171                        ApprovalId::new_v7(),
172                        1,
173                        ApprovalTrigger::PermissionPolicy,
174                        operation.clone(),
175                        permission.evaluations.clone(),
176                        ApprovalConstraints {
177                            allow_once: true,
178                            allow_tree_grant,
179                            cancellable: true,
180                            expires_at: approval_expiry(approval_policy.limits.timeout_ms),
181                        },
182                    )
183                    .map_err(|error| ToolFailure {
184                        code: ToolCallFailureCode::ExecutionFailed,
185                        message: error.to_string(),
186                    })?;
187                    let outcome = engine
188                        .await_user_approval(
189                            &active,
190                            run,
191                            request,
192                            prepared.executor.clone(),
193                            true,
194                            ApprovalToolInput {
195                                name: &call.name,
196                                normalized_parameters: prepared.normalized_arguments(),
197                            },
198                        )
199                        .await
200                        .map_err(|error| ToolFailure {
201                            code: ToolCallFailureCode::ExecutionFailed,
202                            message: error.to_string(),
203                        })?;
204                    if !outcome.approved {
205                        return Err(ToolFailure {
206                            code: ToolCallFailureCode::ExecutionFailed,
207                            message: denied_tool_failure(
208                                ApprovalDecisionSource::Policy,
209                                "permission refused by user",
210                                outcome.feedback,
211                            ),
212                        });
213                    }
214                } else {
215                    return Err(ToolFailure {
216                        code: ToolCallFailureCode::ExecutionFailed,
217                        message: denied_tool_failure(
218                            ApprovalDecisionSource::Policy,
219                            "permission denied",
220                            None,
221                        ),
222                    });
223                }
224            }
225            let executor = prepared
226                .executor
227                .lock()
228                .await
229                .take()
230                .ok_or_else(|| ToolFailure {
231                    code: ToolCallFailureCode::PreparedCapabilityLost,
232                    message: "prepared executor capability was already consumed or lost".into(),
233                })?;
234            let (progress_tx, mut progress_rx) = mpsc::channel(64);
235            let hub = engine
236                .inner
237                .output_hubs
238                .lock()
239                .unwrap_or_else(|poisoned| poisoned.into_inner())
240                .entry(call.id)
241                .or_insert_with(|| OutputHub::new(call.id, 64 * 1024))
242                .clone();
243            let interactive = call.name == "bash"
244                && call
245                    .arguments
246                    .get("interactive")
247                    .and_then(Value::as_bool)
248                    .unwrap_or(false);
249            let capture = (call.name == "bash")
250                .then(|| OutputCapture::new(engine.inner.artifacts.clone()))
251                .transpose()
252                .map_err(|error| ToolFailure {
253                    code: ToolCallFailureCode::ExecutionFailed,
254                    message: format!("tool output capture setup failed: {error}"),
255                })?;
256            let (stdin_tx, stdin) = ToolStdin::channel(64);
257            if interactive {
258                active
259                    .stdin
260                    .lock()
261                    .unwrap_or_else(|poisoned| poisoned.into_inner())
262                    .insert(call.id, stdin_tx);
263            }
264            let invoke = executor.execute(ToolExecutionContext {
265                session: active.session,
266                run,
267                progress: capture.as_ref().map_or_else(
268                    || ProgressSink::new(progress_tx.clone(), hub.clone()),
269                    |capture| {
270                        ProgressSink::with_capture(
271                            progress_tx.clone(),
272                            hub.clone(),
273                            capture.clone(),
274                        )
275                    },
276                ),
277                cancellation: active.cancellation.child_token(),
278                stdin: interactive.then_some(stdin),
279                turn_context,
280                artifacts: engine.inner.artifacts.clone(),
281            });
282            tokio::pin!(invoke);
283            loop {
284                tokio::select! {
285                    result = &mut invoke => {
286                        active
287                            .stdin
288                            .lock()
289                            .unwrap_or_else(|poisoned| poisoned.into_inner())
290                            .remove(&call.id);
291                        // Tool implementations drain their producers before
292                        // resolving.  Finalizing here makes all emitted deltas
293                        // precede the completion notification committed by the
294                        // session actor.
295                        hub.finalize();
296                        engine.retain_finalized_output_hub(call.id);
297                        return match result {
298                            Ok(result) => {
299                                let bounded = if let Some(capture) = &capture {
300                                    capture.finish(
301                                        result,
302                                        active.policy.result_limits.tool_output_max_lines,
303                                        active.policy.result_limits.tool_output_max_bytes,
304                                    )
305                                } else {
306                                    bound_tool_result(
307                                        result,
308                                        &call.name,
309                                        call.id,
310                                        &engine.inner.artifacts,
311                                        active.policy.result_limits.tool_output_max_lines,
312                                        active.policy.result_limits.tool_output_max_bytes,
313                                    )
314                                };
315                                bounded.map_err(|error| ToolFailure {
316                                    code: ToolCallFailureCode::ExecutionFailed,
317                                    message: error.to_string(),
318                                })
319                            }
320                            Err(error) => {
321                                if let Some(capture) = &capture {
322                                    capture.discard();
323                                }
324                                Err(error.into())
325                            }
326                        };
327                    }
328                    Some(progress) = progress_rx.recv() => {
329                        let _ = engine.append(active.session, Some(run), Event::ToolCallProgress { tool_call_id: progress.tool_call_id, message: safe_display(&progress.message) }).await;
330                    }
331                    _ = active.cancellation.cancelled() => {
332                        active
333                            .stdin
334                            .lock()
335                            .unwrap_or_else(|poisoned| poisoned.into_inner())
336                            .remove(&call.id);
337                        if let Some(capture) = &capture {
338                            capture.discard();
339                        }
340                        hub.finalize();
341                        engine.retain_finalized_output_hub(call.id);
342                        return Err(ToolFailure {
343                            code: ToolCallFailureCode::ExecutionFailed,
344                            message: "tool call cancelled after it started".into(),
345                        });
346                    }
347                }
348            }
349        }
350    }

352    pub(super) fn tool_definitions(
353        &self,
354        session: SessionId,
355        policy: &FrozenRunPolicy,
356    ) -> Result<Vec<ToolDefinition>, EngineError> {
357        let depth = session_depth(&self.inner.store.get(session)?.meta.origin);
358        let delegate_enabled = policy.agent.delegation.as_ref().is_some_and(|delegation| {
359            depth < delegation.effective_depth_ceiling
360                && delegation.targets.iter().any(|target| {
361                    policy.registry.get(target).is_some_and(|agent| {
362                        agent.document.frontmatter.enabled
363                            && matches!(
364                                agent.document.frontmatter.mode,
365                                cookie_agent_config::AgentMode::Subagent
366                                    | cookie_agent_config::AgentMode::All
367                            )
368                    })
369                })
370        });
371        let enabled_tools = policy.tools();
372        let mut names = HashSet::new();
373        let mut output = Vec::new();
374        let providers = self
375            .inner
376            .tools
377            .lock()
378            .unwrap_or_else(|poisoned| poisoned.into_inner())
379            .clone();
380        for provider in &providers {
381            for tool in provider
382                .tools_for_session(&SessionToolContext { session })
383                .map_err(|error| EngineError::MissingTool(error.to_string()))?
384            {
385                if ((tool.name != "delegate"
386                    && enabled_tools.contains(&tool.name)
387                    && PermissionPipeline::tool_visible(&policy.agent, &tool.name))
388                    || (tool.name == "delegate" && delegate_enabled))
389                    && names.insert(tool.name.clone())
390                {
391                    let schema = JsonSchema::new(tool.parameters).map_err(|error| {
392                        EngineError::MissingTool(format!(
393                            "tool `{}` has invalid JSON Schema: {error}",
394                            tool.name
395                        ))
396                    })?;
397                    output.push(ToolDefinition::new(tool.name, tool.description, schema));
398                }
399            }
400        }
401        output.sort_by(|left, right| left.name.cmp(&right.name));
402        Ok(output)
403    }

406pub(super) fn fallback_operation_fingerprint(call: &ToolCall) -> OperationFingerprint {
407    let action = PermissionPipeline::action_for_tool(&call.name)
408        .unwrap_or(cookie_agent_protocol::PermissionAction::Read);
409    let binding = serde_json::to_vec(&call.arguments).expect("tool arguments serialize");
410    let operation = PreparedOperationIdentity::new(
411        Sha256Digest::of_bytes(&binding),
412        vec![cookie_agent_protocol::ApprovalCapability {
413            action,
414            operation: cookie_agent_protocol::PreparedCapabilityOperation::new(format!(
415                "{}:prepare",
416                call.name
417                    .replace(|character: char| !character.is_ascii_alphanumeric(), "_")
418            ))
419            .unwrap_or_else(|_| {
420                cookie_agent_protocol::PreparedCapabilityOperation::new("tool:prepare")
421                    .expect("static operation")
422            }),
423        }],
424        vec![cookie_agent_protocol::PreparedApprovalResource {
425            capability: action,
426            canonical: cookie_agent_protocol::PreparedResourceIdentity::new("tool:prepare-failed")
427                .expect("static resource identity"),
428            binding_digest:
429                cookie_agent_protocol::PreparedResourceDigest::from_canonical_binding_bytes(
430                    &binding,
431                ),
432            binding_lifetime: cookie_agent_protocol::PreparedBindingLifetime::ProcessLocal,
433            boundary: cookie_agent_protocol::ApprovalBoundary::Exact,
434            source: cookie_agent_protocol::ApprovalResourceSource::PrimaryOperation,
435        }],
436        Sha256Digest::of_bytes(b"prepare-failed"),
437    )
438    .expect("fallback prepared identity");
439    OperationFingerprint::from_prepared_operation(&operation)
440}

465pub(super) fn redact_presentation(value: &str) -> String {
466    const SECRET_MARKERS: [&str; 6] = [
467        "token",
468        "secret",
469        "password",
470        "api_key",
471        "apikey",
472        "authorization",
473    ];
474    let mut sanitized = sanitize_safe_text(value, 512);
475    let lowercase = sanitized.to_ascii_lowercase();
476    if SECRET_MARKERS
477        .iter()
478        .any(|marker| lowercase.contains(marker))
479    {
480        sanitized = "<redacted>".into();
481    }
482    sanitized
483}

488pub(super) fn truncate_tool_output(
489    output: &str,
490    max_lines: usize,
491    max_bytes: usize,
492) -> Option<TruncatedToolOutput> {
493    let lines = output.split('\n').collect::<Vec<_>>();
494    let line_truncated = lines.len() > max_lines;
495    let mut preview = lines
496        .iter()
497        .take(max_lines)
498        .copied()
499        .collect::<Vec<_>>()
500        .join("\n");
501    let byte_truncated = output.len() > max_bytes || preview.len() > max_bytes;
502    if !line_truncated && !byte_truncated {
503        return None;
504    }
505    if preview.len() > max_bytes {
506        let mut boundary = max_bytes;
507        while boundary > 0 && !preview.is_char_boundary(boundary) {
508            boundary -= 1;
509        }
510        preview.truncate(boundary);
511    }
512    Some(TruncatedToolOutput { content: preview })
513}

515pub(super) fn bound_tool_result(
516    mut result: ToolResult,
517    _tool_name: &str,
518    _call_id: ToolCallId,
519    artifacts: &ArtifactStore,
520    max_lines: usize,
521    max_bytes: usize,
522) -> std::io::Result<ToolResult> {
523    let Some(preview) = truncate_tool_output(&result.output, max_lines, max_bytes) else {
524        return Ok(result);
525    };
526    let original_bytes = result.output.len() as u64;
527    let original_lines = result.output.split('\n').count() as u64;
528    let (retained, _) = artifacts.retain(result.output.as_bytes())?;
529    result.output = preview.content;
530    result.truncation = Some(ToolOutputTruncation {
531        original_bytes,
532        original_lines,
533        retained,
534    });
535    Ok(result)
536}

538pub(crate) fn validate_attachment(
539    mime_type: &str,
540    path: &Path,
541    bytes: &[u8],
542) -> Result<(), ToolError> {
543    if bytes.len() as u64 > MAX_ATTACHMENT_BYTES {
544        return Err(ToolError::resource_limit(format!(
545            "attachment is {} bytes; the limit is {MAX_ATTACHMENT_BYTES} bytes",
546            bytes.len()
547        )));
548    }
549    let validated = approved_media_type(path, bytes)?
550        .ok_or_else(|| ToolError::execution("attachment is not a supported image or PDF"))?;
551    if validated != mime_type {
552        return Err(ToolError::execution(format!(
553            "attachment MIME mismatch: declared {mime_type}, validated {validated}"
554        )));
555    }
556    Ok(())
