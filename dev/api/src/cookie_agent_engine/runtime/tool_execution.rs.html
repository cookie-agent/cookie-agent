5    pub(super) async fn prepare_tool_call(
6        &self,
7        session_id: SessionId,
8        run: RunId,
9        call: ToolCall,
10        policy: &FrozenRunPolicy,
11    ) -> PreparedToolCall {
12        let fallback_presentation = tool_title_only(&call.name);
13        let session = match self.inner.store.get(session_id) {
14            Ok(session) => session,
15            Err(error) => {
16                return PreparedToolCall {
17                    call,
18                    presentation: fallback_presentation,
19                    prepared: Err(ToolFailure {
20                        code: ToolCallFailureCode::ExecutionFailed,
21                        message: error.to_string(),
22                    }),
23                };
24            }
25        };
26        let depth = session_depth(&session.meta.origin);
27        let delegate_enabled = policy.agent.delegation.as_ref().is_some_and(|delegation| {
28            depth < delegation.effective_depth_ceiling
29                && delegation.targets.iter().any(|target| {
30                    policy.registry.get(target).is_some_and(|agent| {
31                        agent.document.frontmatter.enabled
32                            && matches!(
33                                agent.document.frontmatter.mode,
34                                cookie_agent_config::AgentMode::Subagent
35                                    | cookie_agent_config::AgentMode::All
36                            )
37                    })
38                })
39        });
40        let enabled_tools = policy.tools();
41        if (call.name == "delegate" && !delegate_enabled)
42            || (call.name != "delegate"
43                && (!enabled_tools.contains(&call.name)
44                    || !PermissionPipeline::tool_visible(&policy.agent, &call.name)))
45        {
46            return PreparedToolCall {
47                prepared: Err(ToolFailure {
48                    code: ToolCallFailureCode::ExecutionFailed,
49                    message: format!("tool `{}` is not enabled for this session", call.name),
50                }),
51                call,
52                presentation: fallback_presentation,
53            };
54        }
55        let providers = self
56            .inner
57            .tools
58            .lock()
59            .unwrap_or_else(|poisoned| poisoned.into_inner())
60            .clone();
61        let provider = providers.into_iter().find(|provider| {
62            provider
63                .tools_for_session(&SessionToolContext {
64                    session: session_id,
65                })
66                .ok()
67                .is_some_and(|tools| tools.iter().any(|tool| tool.name == call.name))
68        });
69        let Some(provider) = provider else {
70            return PreparedToolCall {
71                prepared: Err(ToolFailure {
72                    code: ToolCallFailureCode::ExecutionFailed,
73                    message: format!("tool `{}` is unavailable", call.name),
74                }),
75                call,
76                presentation: fallback_presentation,
77            };
78        };
79        let presentation = provider.presentation(&call);
80        let context = ToolPreparationContext {
81            session: session_id,
82            run,
83            cwd: self.inner.store.cwd().to_owned(),
84            workspace_root: self.inner.store.cwd().to_owned(),
85        };
86        let prepared = match provider.prepare(context, call.clone()).await {
87            Ok(prepared) => apply_primary_argument_labels(provider.as_ref(), &call.name, prepared)
88                .map_err(Into::into),
89            Err(error) => Err(error.into()),
90        };
91        PreparedToolCall {
92            call,
93            presentation,
94            prepared,
95        }
96    }

98    pub(super) async fn execute_tool(
99        &self,
100        active: Arc<ActiveRun>,
101        run: RunId,
102        prepared: PreparedToolCall,
103    ) -> Result<ToolResult, ToolFailure> {
104        let engine = self.clone();
105        {
106            let PreparedToolCall { call, prepared, .. } = prepared;
107            let prepared = prepared?;
108            let operation = prepared.operation.clone();
109            let policy_labels = prepared.policy_labels.clone();
110            let _serialization_guard = if let Some(key) = &prepared.serialization_key {
111                Some(engine.mutation_lock(key).lock_owned().await)
112            } else {
113                None
114            };
115            let permission = engine.inner.permissions.decide_operation(
116                &active.policy.agent,
117                &operation,
118                &policy_labels,
119                engine.inner.store.cwd(),
120            );
121            if permission.effect != cookie_agent_protocol::PermissionEffect::Allow {
122                if permission.effect == cookie_agent_protocol::PermissionEffect::Ask {
123                    let allow_tree_grant = operation.resources().iter().all(|resource| {
124                        resource.binding_lifetime
125                            == cookie_agent_protocol::PreparedBindingLifetime::RestartStable
126                            && !matches!(
127                                resource.capability,
128                                cookie_agent_protocol::PermissionAction::Read
129                                    | cookie_agent_protocol::PermissionAction::Write
130                            )
131                    });
132                    let approval_policy = engine
133                        .active_internal_policy(&active, InternalAgentKind::Approval)
134                        .map_err(|error| ToolFailure {
135                            code: ToolCallFailureCode::ExecutionFailed,
136                            message: error.to_string(),
137                        })?;
138                    let request = ApprovalRequest::new(
139                        ApprovalId::new_v7(),
140                        1,
141                        ApprovalTrigger::PermissionPolicy,
142                        operation.clone(),
143                        permission.evaluations.clone(),
144                        ApprovalConstraints {
145                            allow_once: true,
146                            allow_tree_grant,
147                            cancellable: true,
148                            expires_at: approval_expiry(approval_policy.limits.timeout_ms),
149                        },
150                    )
151                    .map_err(|error| ToolFailure {
152                        code: ToolCallFailureCode::ExecutionFailed,
153                        message: error.to_string(),
154                    })?;
155                    let outcome = engine
156                        .await_user_approval(
157                            &active,
158                            run,
159                            request,
160                            prepared.executor.clone(),
161                            true,
162                            ApprovalToolInput {
163                                name: &call.name,
164                                normalized_parameters: prepared.normalized_arguments(),
165                            },
166                        )
167                        .await
168                        .map_err(|error| ToolFailure {
169                            code: ToolCallFailureCode::ExecutionFailed,
170                            message: error.to_string(),
171                        })?;
172                    if !outcome.approved {
173                        return Err(ToolFailure {
174                            code: ToolCallFailureCode::ExecutionFailed,
175                            message: denied_tool_failure(
176                                ApprovalDecisionSource::Policy,
177                                "permission refused by user",
178                                outcome.feedback,
179                            ),
180                        });
181                    }
182                } else {
183                    return Err(ToolFailure {
184                        code: ToolCallFailureCode::ExecutionFailed,
185                        message: denied_tool_failure(
186                            ApprovalDecisionSource::Policy,
187                            "permission denied",
188                            None,
189                        ),
190                    });
191                }
192            }
193            let executor = prepared
194                .executor
195                .lock()
196                .await
197                .take()
198                .ok_or_else(|| ToolFailure {
199                    code: ToolCallFailureCode::PreparedCapabilityLost,
200                    message: "prepared executor capability was already consumed or lost".into(),
201                })?;
202            let (progress_tx, mut progress_rx) = mpsc::channel(64);
203            let hub = engine
204                .inner
205                .output_hubs
206                .lock()
207                .unwrap_or_else(|poisoned| poisoned.into_inner())
208                .entry(call.id)
209                .or_insert_with(|| OutputHub::new(call.id, 64 * 1024))
210                .clone();
211            let interactive = call.name == "bash"
212                && call
213                    .arguments
214                    .get("interactive")
215                    .and_then(Value::as_bool)
216                    .unwrap_or(false);
217            let capture = (call.name == "bash")
218                .then(|| OutputCapture::new(engine.inner.artifacts.clone()))
219                .transpose()
220                .map_err(|error| ToolFailure {
221                    code: ToolCallFailureCode::ExecutionFailed,
222                    message: format!("tool output capture setup failed: {error}"),
223                })?;
224            let (stdin_tx, stdin) = ToolStdin::channel(64);
225            if interactive {
226                active
227                    .stdin
228                    .lock()
229                    .unwrap_or_else(|poisoned| poisoned.into_inner())
230                    .insert(call.id, stdin_tx);
231            }
232            let invoke = executor.execute(ToolExecutionContext {
233                session: active.session,
234                run,
235                progress: capture.as_ref().map_or_else(
236                    || ProgressSink::new(progress_tx.clone(), hub.clone()),
237                    |capture| {
238                        ProgressSink::with_capture(
239                            progress_tx.clone(),
240                            hub.clone(),
241                            capture.clone(),
242                        )
243                    },
244                ),
245                cancellation: active.cancellation.child_token(),
246                stdin: interactive.then_some(stdin),
247                artifacts: engine.inner.artifacts.clone(),
248            });
249            tokio::pin!(invoke);
250            loop {
251                tokio::select! {
252                    result = &mut invoke => {
253                        active
254                            .stdin
255                            .lock()
256                            .unwrap_or_else(|poisoned| poisoned.into_inner())
257                            .remove(&call.id);
258                        // Tool implementations drain their producers before
259                        // resolving.  Finalizing here makes all emitted deltas
260                        // precede the completion notification committed by the
261                        // session actor.
262                        hub.finalize();
263                        engine.retain_finalized_output_hub(call.id);
264                        return match result {
265                            Ok(result) => {
266                                let bounded = if let Some(capture) = &capture {
267                                    capture.finish(
268                                        result,
269                                        active.policy.result_limits.tool_output_max_lines,
270                                        active.policy.result_limits.tool_output_max_bytes,
271                                    )
272                                } else {
273                                    bound_tool_result(
274                                        result,
275                                        &call.name,
276                                        call.id,
277                                        &engine.inner.artifacts,
278                                        active.policy.result_limits.tool_output_max_lines,
279                                        active.policy.result_limits.tool_output_max_bytes,
280                                    )
281                                };
282                                bounded.map_err(|error| ToolFailure {
283                                    code: ToolCallFailureCode::ExecutionFailed,
284                                    message: error.to_string(),
285                                })
286                            }
287                            Err(error) => {
288                                if let Some(capture) = &capture {
289                                    capture.discard();
290                                }
291                                Err(error.into())
292                            }
293                        };
294                    }
295                    Some(progress) = progress_rx.recv() => {
296                        let _ = engine.append(active.session, Some(run), Event::ToolCallProgress { tool_call_id: progress.tool_call_id, message: safe_display(&progress.message) }).await;
297                    }
298                    _ = active.cancellation.cancelled() => {
299                        active
300                            .stdin
301                            .lock()
302                            .unwrap_or_else(|poisoned| poisoned.into_inner())
303                            .remove(&call.id);
304                        if let Some(capture) = &capture {
305                            capture.discard();
306                        }
307                        hub.finalize();
308                        engine.retain_finalized_output_hub(call.id);
309                        return Err(ToolFailure {
310                            code: ToolCallFailureCode::ExecutionFailed,
311                            message: "tool call cancelled after it started".into(),
312                        });
313                    }
314                }
315            }
316        }
317    }

319    pub(super) fn tool_definitions(
320        &self,
321        session: SessionId,
322        policy: &FrozenRunPolicy,
323    ) -> Result<Vec<ToolDefinition>, EngineError> {
324        let depth = session_depth(&self.inner.store.get(session)?.meta.origin);
325        let delegate_enabled = policy.agent.delegation.as_ref().is_some_and(|delegation| {
326            depth < delegation.effective_depth_ceiling
327                && delegation.targets.iter().any(|target| {
328                    policy.registry.get(target).is_some_and(|agent| {
329                        agent.document.frontmatter.enabled
330                            && matches!(
331                                agent.document.frontmatter.mode,
332                                cookie_agent_config::AgentMode::Subagent
333                                    | cookie_agent_config::AgentMode::All
334                            )
335                    })
336                })
337        });
338        let enabled_tools = policy.tools();
339        let mut names = HashSet::new();
340        let mut output = Vec::new();
341        let providers = self
342            .inner
343            .tools
344            .lock()
345            .unwrap_or_else(|poisoned| poisoned.into_inner())
346            .clone();
347        for provider in &providers {
348            for tool in provider
349                .tools_for_session(&SessionToolContext { session })
350                .map_err(|error| EngineError::MissingTool(error.to_string()))?
351            {
352                if ((tool.name != "delegate"
353                    && enabled_tools.contains(&tool.name)
354                    && PermissionPipeline::tool_visible(&policy.agent, &tool.name))
355                    || (tool.name == "delegate" && delegate_enabled))
356                    && names.insert(tool.name.clone())
357                {
358                    let schema = JsonSchema::new(tool.parameters).map_err(|error| {
359                        EngineError::MissingTool(format!(
360                            "tool `{}` has invalid JSON Schema: {error}",
361                            tool.name
362                        ))
363                    })?;
364                    output.push(ToolDefinition::new(tool.name, tool.description, schema));
365                }
366            }
367        }
368        output.sort_by(|left, right| left.name.cmp(&right.name));
369        Ok(output)
370    }

373pub(super) fn fallback_operation_fingerprint(call: &ToolCall) -> OperationFingerprint {
374    let action = PermissionPipeline::action_for_tool(&call.name)
375        .unwrap_or(cookie_agent_protocol::PermissionAction::Read);
376    let binding = serde_json::to_vec(&call.arguments).expect("tool arguments serialize");
377    let operation = PreparedOperationIdentity::new(
378        Sha256Digest::of_bytes(&binding),
379        vec![cookie_agent_protocol::ApprovalCapability {
380            action,
381            operation: cookie_agent_protocol::PreparedCapabilityOperation::new(format!(
382                "{}:prepare",
383                call.name
384                    .replace(|character: char| !character.is_ascii_alphanumeric(), "_")
385            ))
386            .unwrap_or_else(|_| {
387                cookie_agent_protocol::PreparedCapabilityOperation::new("tool:prepare")
388                    .expect("static operation")
389            }),
390        }],
391        vec![cookie_agent_protocol::PreparedApprovalResource {
392            capability: action,
393            canonical: cookie_agent_protocol::PreparedResourceIdentity::new("tool:prepare-failed")
394                .expect("static resource identity"),
395            binding_digest:
396                cookie_agent_protocol::PreparedResourceDigest::from_canonical_binding_bytes(
397                    &binding,
398                ),
399            binding_lifetime: cookie_agent_protocol::PreparedBindingLifetime::ProcessLocal,
400            boundary: cookie_agent_protocol::ApprovalBoundary::Exact,
401            source: cookie_agent_protocol::ApprovalResourceSource::PrimaryOperation,
402        }],
403        Sha256Digest::of_bytes(b"prepare-failed"),
404    )
405    .expect("fallback prepared identity");
406    OperationFingerprint::from_prepared_operation(&operation)
407}

432pub(super) fn redact_presentation(value: &str) -> String {
433    const SECRET_MARKERS: [&str; 6] = [
434        "token",
435        "secret",
436        "password",
437        "api_key",
438        "apikey",
439        "authorization",
440    ];
441    let mut sanitized = sanitize_safe_text(value, 512);
442    let lowercase = sanitized.to_ascii_lowercase();
443    if SECRET_MARKERS
444        .iter()
445        .any(|marker| lowercase.contains(marker))
446    {
447        sanitized = "<redacted>".into();
448    }
449    sanitized
450}

455pub(super) fn truncate_tool_output(
456    output: &str,
457    max_lines: usize,
458    max_bytes: usize,
459) -> Option<TruncatedToolOutput> {
460    let lines = output.split('\n').collect::<Vec<_>>();
461    let line_truncated = lines.len() > max_lines;
462    let mut preview = lines
463        .iter()
464        .take(max_lines)
465        .copied()
466        .collect::<Vec<_>>()
467        .join("\n");
468    let byte_truncated = output.len() > max_bytes || preview.len() > max_bytes;
469    if !line_truncated && !byte_truncated {
470        return None;
471    }
472    if preview.len() > max_bytes {
473        let mut boundary = max_bytes;
474        while boundary > 0 && !preview.is_char_boundary(boundary) {
475            boundary -= 1;
476        }
477        preview.truncate(boundary);
478    }
479    Some(TruncatedToolOutput { content: preview })
480}

482pub(super) fn bound_tool_result(
483    mut result: ToolResult,
484    _tool_name: &str,
485    _call_id: ToolCallId,
486    artifacts: &ArtifactStore,
487    max_lines: usize,
488    max_bytes: usize,
489) -> std::io::Result<ToolResult> {
490    let Some(preview) = truncate_tool_output(&result.output, max_lines, max_bytes) else {
491        return Ok(result);
492    };
493    let original_bytes = result.output.len() as u64;
494    let original_lines = result.output.split('\n').count() as u64;
495    let (retained, _) = artifacts.retain(result.output.as_bytes())?;
496    result.output = preview.content;
497    result.truncation = Some(ToolOutputTruncation {
498        original_bytes,
499        original_lines,
500        retained,
501    });
502    Ok(result)
503}

505pub(crate) fn validate_attachment(
506    mime_type: &str,
507    path: &Path,
508    bytes: &[u8],
509) -> Result<(), ToolError> {
510    if bytes.len() as u64 > MAX_ATTACHMENT_BYTES {
511        return Err(ToolError::resource_limit(format!(
512            "attachment is {} bytes; the limit is {MAX_ATTACHMENT_BYTES} bytes",
513            bytes.len()
514        )));
515    }
516    let validated = approved_media_type(path, bytes)?
517        .ok_or_else(|| ToolError::execution("attachment is not a supported image or PDF"))?;
518    if validated != mime_type {
519        return Err(ToolError::execution(format!(
520            "attachment MIME mismatch: declared {mime_type}, validated {validated}"
521        )));
522    }
523    Ok(())
