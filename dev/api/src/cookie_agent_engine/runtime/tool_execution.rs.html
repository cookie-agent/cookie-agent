38fn tool_progress_event(progress: &ToolProgress) -> Event {
39    Event::ToolCallProgress {
40        tool_call_id: progress.tool_call_id,
41        message: safe_display(&progress.message),
42        output_chunk: progress.output_chunk.as_deref().map(safe_display),
43    }
44}

47    engine: &Engine,

62async fn enqueue_cleanup_tool_progress(
63    engine: &Engine,
64    session: SessionId,
65    run: RunId,
66    progress: ToolProgress,
67) -> Result<(), EngineError> {
68    #[cfg(test)]
69    let block = engine
70        .inner
71        .tool_progress_append_block
72        .lock()
73        .unwrap_or_else(|poisoned| poisoned.into_inner())
74        .clone();
75    #[cfg(test)]
76    if let Some(block) = block {
77        block.notified().await;
78    }
79    let completion = engine
80        .enqueue_append(
81            session,
82            Some(run),
83            super::event_origin("engine:tool-execution"),
84            tool_progress_event(&progress),
85        )
86        .await?;
87    drop(completion);
88    Ok(())
89}

91fn close_and_discard_progress(progress_rx: &mut mpsc::Receiver<ToolProgress>) -> usize {
92    progress_rx.close();
93    let mut discarded = 0;
94    while progress_rx.try_recv().is_ok() {
95        discarded += 1;
96    }
97    discarded
98}

133    async fn prepare_tool_call_with_publication(
134        &self,
135        session_id: SessionId,
136        run: RunId,
137        call: ToolCall,
138        policy: &FrozenRunPolicy,
139        turn_context: Arc<crate::TurnAgentContext>,
140        published: Option<Option<&PublishedTool>>,
141    ) -> PreparedToolCall {
142        let fallback_presentation = tool_title_only(&call.name);
143        let session = match self.inner.store.get(session_id) {
144            Ok(session) => session,
145            Err(error) => {
146                return PreparedToolCall {
147                    intercepted_arguments: Arc::new(std::sync::Mutex::new(call.arguments.clone())),
148                    call,
149                    permission_name: None,
150                    presentation: fallback_presentation,
151                    prepared: Err(ToolFailure {
152                        code: ToolCallFailureCode::ExecutionFailed,
153                        message: error.to_string(),
154                    }),
155                    interception: None,
156                };
157            }
158        };
159        let depth = session_depth(&session.meta.origin);
160        let permission_overlay = &session.permission_overlay;
161        let grants = self.skill_grants_for_session(session_id);
162        let delegate_enabled = policy.agent.delegation.as_ref().is_some_and(|delegation| {
163            depth < delegation.effective_depth_ceiling
164                && delegation
165                    .targets
166                    .iter()
167                    .any(|target| policy.delegation_target_available(target))
168        });
169        let providers = self
170            .inner
171            .tools
172            .lock()
173            .unwrap_or_else(|poisoned| poisoned.into_inner())
174            .clone();
175        let current = providers.iter().find_map(|provider| {
176            provider
177                .tools_for_session(&SessionToolContext {
178                    session: session_id,
179                })
180                .ok()?
181                .into_iter()
182                .find(|tool| tool.name == call.name)
183                .map(|tool| (provider.clone(), tool))
184        });
185        let mut provider = match published {
186            Some(Some(published))
187                if current.as_ref().is_some_and(|(provider, spec)| {
188                    Arc::ptr_eq(provider, &published.provider) && spec == &published.spec
189                }) =>
190            {
191                Some((published.provider.clone(), published.spec.clone()))
192            }
193            Some(Some(_)) => {
194                return PreparedToolCall {
195                    intercepted_arguments: Arc::new(std::sync::Mutex::new(call.arguments.clone())),
196                    prepared: Err(ToolFailure {
197                        code: ToolCallFailureCode::OperationChanged,
198                        message: format!(
199                            "tool definition changed after `{}` was published to the model",
200                            call.name
201                        ),
202                    }),
203                    call,
204                    permission_name: None,
205                    presentation: fallback_presentation,
206                    interception: None,
207                };
208            }
209            Some(None) => {
210                return PreparedToolCall {
211                    intercepted_arguments: Arc::new(std::sync::Mutex::new(call.arguments.clone())),
212                    prepared: Err(ToolFailure {
213                        code: ToolCallFailureCode::ExecutionFailed,
214                        message: format!("tool `{}` was not published to the model", call.name),
215                    }),
216                    call,
217                    permission_name: None,
218                    presentation: fallback_presentation,
219                    interception: None,
220                };
221            }
222            None => current,
223        };
224        if published.is_none()
225            && provider.is_none()
226            && (call.name != "skill" || self.is_direct_skill_call(call.id))
227        {
228            provider = providers.iter().find_map(|candidate| {
229                candidate
230                    .permission_for_unlisted_tool(&call.name)
231                    .ok()
232                    .flatten()
233                    .map(|permission_name| {
234                        (
235                            candidate.clone(),
236                            crate::ToolSpec {
237                                name: call.name.clone(),
238                                permission_name: permission_name.to_owned(),
239                                description: String::new(),
240                                parameters: serde_json::json!(true),
241                                result_truncation: Default::default(),
242                            },
243                        )
244                    })
245            });
246        }
247        let Some((provider, tool_spec)) = provider else {
248            return PreparedToolCall {
249                intercepted_arguments: Arc::new(std::sync::Mutex::new(call.arguments.clone())),
250                prepared: Err(ToolFailure {
251                    code: ToolCallFailureCode::ExecutionFailed,
252                    message: format!("tool `{}` is unavailable", call.name),
253                }),
254                call,
255                permission_name: None,
256                presentation: fallback_presentation,
257                interception: None,
258            };
259        };
260        let permission_name = tool_spec.permission_name.clone();
261        let delegation_tool = permission_name == "delegate";
262        let enabled = !delegation_tool || delegate_enabled;
263        let direct_skill = call.name == "skill" && self.is_direct_skill_call(call.id);
264        if !enabled
265            || (!direct_skill
266                && !PermissionPipeline::tool_visible_with_grants(
267                    &policy.agent,
268                    Some(permission_overlay),
269                    grants.as_ref(),
270                    &permission_name,
271                    self.inner.store.cwd(),
272                ))
273        {
274            return PreparedToolCall {
275                intercepted_arguments: Arc::new(std::sync::Mutex::new(call.arguments.clone())),
276                prepared: Err(ToolFailure {
277                    code: ToolCallFailureCode::ExecutionFailed,
278                    message: format!("tool `{}` is not enabled for this session", call.name),
279                }),
280                call,
281                permission_name: Some(permission_name),
282                presentation: fallback_presentation,
283                interception: None,
284            };
285        }
286        let original_resource = provider
287            .get_permission_resource(&call.name, &call.arguments)
288            .ok()
289            .and_then(|(_, resource)| resource);
290        let presentation = provider.presentation(&call);
291        let context = ToolPreparationContext {
292            session: session_id,
293            run,
294            cwd: self.inner.store.cwd().to_owned(),
295            workspace_root: self.inner.store.cwd().to_owned(),
296            turn_context,
297        };
298        let prepared = match provider.prepare(context.clone(), call.clone()).await {
299            Ok(prepared) => {
300                apply_permission_resource(provider.as_ref(), &call.name, &permission_name, prepared)
301                    .map_err(Into::into)
302            }
303            Err(error) => Err(error.into()),
304        };
305        PreparedToolCall {
306            intercepted_arguments: Arc::new(std::sync::Mutex::new(call.arguments.clone())),
307            call,
308            permission_name: Some(permission_name.clone()),
309            presentation,
310            prepared,
311            interception: Some(ToolInterceptionContext {
312                provider,
313                spec: tool_spec,
314                preparation: context,
315                permission_name,
316                permission_resource: original_resource,
317            }),
318        }
319    }

321    pub(super) async fn execute_tool(
322        &self,
323        active: Arc<ActiveRun>,
324        run: RunId,
325        prepared: PreparedToolCall,
326        turn_context: Arc<crate::TurnAgentContext>,
327    ) -> Result<ToolResult, ToolFailure> {
328        let engine = self.clone();
329        {
330            let PreparedToolCall {
331                mut call,
332                prepared,
333                interception,
334                intercepted_arguments,
335                ..
336            } = prepared;
337            let mut prepared = prepared?;
338            let result_truncation = interception
339                .as_ref()
340                .map(|interception| interception.spec.result_truncation)
341                .unwrap_or_default();
342            let operation = prepared.operation.clone();
343            let policy_labels = prepared.policy_labels.clone();
344            let permission_overlay = engine
345                .inner
346                .store
347                .get(active.session)
348                .map_err(|error| ToolFailure {
349                    code: ToolCallFailureCode::ExecutionFailed,
350                    message: error.to_string(),
351                })?
352                .permission_overlay;
353            let grants = engine.skill_grants_for_session(active.session);
354            let permission = engine.inner.permissions.decide_operation_with_grants(
355                &active.policy.agent,
356                Some(&permission_overlay),
357                grants.as_ref(),
358                &operation,
359                &policy_labels,
360                engine.inner.store.cwd(),
361            );
362            if permission.effect != cookie_agent_protocol::PermissionEffect::Allow {
363                if permission.effect == cookie_agent_protocol::PermissionEffect::Ask {
364                    let allow_tree_grant = operation.resources().iter().all(|resource| {
365                        resource.binding_lifetime
366                            == cookie_agent_protocol::PreparedBindingLifetime::RestartStable
367                            && !matches!(
368                                resource.capability,
369                                cookie_agent_protocol::PermissionAction::Read
370                                    | cookie_agent_protocol::PermissionAction::Write
371                            )
372                    });
373                    let approval_policy = engine
374                        .active_internal_policy(&active, InternalAgentKind::Approval)
375                        .map_err(|error| ToolFailure {
376                            code: ToolCallFailureCode::ExecutionFailed,
377                            message: error.to_string(),
378                        })?;
379                    let request = ApprovalRequest::new(
380                        ApprovalId::new_v7(),
381                        1,
382                        ApprovalTrigger::PermissionPolicy,
383                        operation.clone(),
384                        permission.evaluations.clone(),
385                        ApprovalConstraints {
386                            allow_once: true,
387                            allow_tree_grant,
388                            cancellable: true,
389                            expires_at: approval_expiry(approval_policy.limits.timeout_ms),
390                        },
391                    )
392                    .map_err(|error| ToolFailure {
393                        code: ToolCallFailureCode::ExecutionFailed,
394                        message: error.to_string(),
395                    })?;
396                    let outcome = engine
397                        .await_user_approval(
398                            &active,
399                            run,
400                            request,
401                            prepared.executor.clone(),
402                            true,
403                            ApprovalToolInput {
404                                name: &call.name,
405                                normalized_parameters: prepared.normalized_arguments(),
406                            },
407                        )
408                        .await
409                        .map_err(|error| ToolFailure {
410                            code: ToolCallFailureCode::ExecutionFailed,
411                            message: error.to_string(),
412                        })?;
413                    if !outcome.approved {
414                        return Err(ToolFailure {
415                            code: ToolCallFailureCode::ExecutionFailed,
416                            message: denied_tool_failure(
417                                ApprovalDecisionSource::Policy,
418                                "permission refused by user",
419                                outcome.feedback,
420                            ),
421                        });
422                    }
423                } else {
424                    return Err(ToolFailure {
425                        code: ToolCallFailureCode::ExecutionFailed,
426                        message: denied_tool_failure(
427                            ApprovalDecisionSource::Policy,
428                            "permission denied",
429                            None,
430                        ),
431                    });
432                }
433            }
434            if let Some(interception) = interception {
435                let context_id = crate::plugin::plugin_context_id();
436                for plugin in engine.inner.plugins.interception_plugins(
437                    cookie_agent_protocol::ExtensionInterceptionHook::ToolBeforeCall,
438                ) {
439                    let result = engine
440                        .inner
441                        .plugins
442                        .intercept_named::<_, cookie_agent_protocol::ExtensionToolBeforeCallResult>(
443                            &plugin,
444                            cookie_agent_protocol::PLUGIN_INTERCEPT_TOOL_BEFORE_CALL_METHOD,
445                            &ExtensionToolBeforeCallParams {
446                                session_id: active.session,
447                                context_id: context_id.clone(),
448                                tool: call.name.clone(),
449                                arguments: call.arguments.clone(),
450                                permission_name: interception.permission_name.clone(),
451                                resource: interception.permission_resource.clone(),
452                            },
453                            Some(active.session),
454                            Some(&context_id),
455                        )
456                        .await;
457                    let result = match result {
458                        Ok(result) => result,
459                        Err(error) => {
460                            let kind =
461                                if error.contains("crashed") || error.contains("not connected") {
462                                    PluginDiagnosticKind::InterceptionCrash
463                                } else {
464                                    PluginDiagnosticKind::InterceptionTimeout
465                                };
466                            engine.record_plugin_diagnostic(active.session, plugin, kind, error);
467                            continue;
468                        }
469                    };
470                    if result.action == ExtensionToolBeforeCallAction::Block {
471                        let message = result
472                            .message_to_model
473                            .or(result.reason)
474                            .unwrap_or_else(|| format!("plugin `{plugin}` blocked the tool call"));
475                        engine.record_plugin_diagnostic(
476                            active.session,
477                            plugin,
478                            PluginDiagnosticKind::HookBlocked,
479                            message.clone(),
480                        );
481                        return Err(ToolFailure {
482                            code: ToolCallFailureCode::ExecutionFailed,
483                            message,
484                        });
485                    }
486                    let Some(arguments) = result.modified_arguments else {
487                        continue;
488                    };
489                    let valid_schema = jsonschema::validator_for(&interception.spec.parameters)
490                        .is_ok_and(|validator| validator.is_valid(&arguments));
491                    let modified_resource = interception
492                        .provider
493                        .get_permission_resource(&call.name, &arguments)
494                        .ok()
495                        .and_then(|(_, resource)| resource);
496                    if !arguments.is_object()
497                        || !valid_schema
498                        || modified_resource != interception.permission_resource
499                    {
500                        let message = format!(
501                            "plugin `{plugin}` produced invalid tool arguments or changed the permission resource"
502                        );
503                        engine.record_plugin_diagnostic(
504                            active.session,
505                            plugin,
506                            PluginDiagnosticKind::InvalidModification,
507                            message.clone(),
508                        );
509                        return Err(ToolFailure {
510                            code: ToolCallFailureCode::ExecutionFailed,
511                            message,
512                        });
513                    }
514                    let mut modified_call = call.clone();
515                    modified_call.arguments = arguments;
516                    let candidate = interception
517                        .provider
518                        .prepare(interception.preparation.clone(), modified_call.clone())
519                        .await
520                        .and_then(|prepared| {
521                            apply_permission_resource(
522                                interception.provider.as_ref(),
523                                &modified_call.name,
524                                &interception.permission_name,
525                                prepared,
526                            )
527                        })
528                        .map_err(ToolFailure::from)?;
529                    if candidate.operation.capabilities() != operation.capabilities()
530                        || candidate.operation.resources() != operation.resources()
531                        || candidate.policy_labels != policy_labels
532                    {
533                        let message = format!(
534                            "plugin `{plugin}` changed the approved permission capability or resource"
535                        );
536                        engine.record_plugin_diagnostic(
537                            active.session,
538                            plugin,
539                            PluginDiagnosticKind::InvalidModification,
540                            message.clone(),
541                        );
542                        return Err(ToolFailure {
543                            code: ToolCallFailureCode::ExecutionFailed,
544                            message,
545                        });
546                    }
547                    call = modified_call;
548                    prepared = candidate;
549                    *intercepted_arguments
550                        .lock()
551                        .unwrap_or_else(|poisoned| poisoned.into_inner()) = call.arguments.clone();
552                }
553            }
554            let _serialization_guard = if let Some(key) = &prepared.serialization_key {
555                Some(engine.mutation_lock(key).lock_owned().await)
556            } else {
557                None
558            };
559            let executor = prepared
560                .executor
561                .lock()
562                .await
563                .take()
564                .ok_or_else(|| ToolFailure {
565                    code: ToolCallFailureCode::PreparedCapabilityLost,
566                    message: "prepared executor capability was already consumed or lost".into(),
567                })?;
568            let (progress_tx, mut progress_rx) = mpsc::channel(64);
569            let hub = engine
570                .inner
571                .output_hubs
572                .lock()
573                .unwrap_or_else(|poisoned| poisoned.into_inner())
574                .entry(call.id)
575                .or_insert_with(|| OutputHub::new(call.id, 64 * 1024))
576                .clone();
577            let interactive = call.name == "bash"
578                && call
579                    .arguments
580                    .get("interactive")
581                    .and_then(Value::as_bool)
582                    .unwrap_or(false);
583            let capture = (call.name == "bash")
584                .then(|| OutputCapture::new(engine.inner.artifacts.clone()))
585                .transpose()
586                .map_err(|error| ToolFailure {
587                    code: ToolCallFailureCode::ExecutionFailed,
588                    message: format!("tool output capture setup failed: {error}"),
589                })?;
590            let (stdin_tx, stdin) = ToolStdin::channel(64);
591            if interactive {
592                active
593                    .stdin
594                    .lock()
595                    .unwrap_or_else(|poisoned| poisoned.into_inner())
596                    .insert(call.id, stdin_tx);
597            }
598            let tool_cancellation = active.cancellation.child_token();
599            let invoke = executor.execute(ToolExecutionContext {
600                session: active.session,
601                run,
602                progress: capture.as_ref().map_or_else(
603                    || ProgressSink::new(progress_tx.clone(), hub.clone()),
604                    |capture| {
605                        ProgressSink::with_capture(
606                            progress_tx.clone(),
607                            hub.clone(),
608                            capture.clone(),
609                        )
610                    },
611                ),
612                cancellation: tool_cancellation.clone(),
613                stdin: interactive.then_some(stdin),
614                turn_context,
615                artifacts: engine.inner.artifacts.clone(),
616            });
617            tokio::pin!(invoke);
618            loop {
619                tokio::select! {
620                    result = &mut invoke => {
621                        active
622                            .stdin
623                            .lock()
624                            .unwrap_or_else(|poisoned| poisoned.into_inner())
625                            .remove(&call.id);
626                        while let Ok(progress) = progress_rx.try_recv() {
627                            let _ = append_tool_progress(
628                                &engine,
629                                active.session,
630                                run,
631                                progress,
632                            )
633                            .await;
634                        }
635                        // Tool implementations drain their producers before
636                        // resolving.  Finalizing here makes all emitted deltas
637                        // precede the completion notification committed by the
638                        // session actor.
639                        hub.finalize();
640                        engine.retain_finalized_output_hub(call.id);
641                        return match result {
642                            Ok(result) => {
643                                if let Some(capture) = &capture {
644                                    return capture
645                                        .finish(
646                                        result,
647                                        active.policy.result_limits.tool_output_max_lines,
648                                        active.policy.result_limits.tool_output_max_bytes,
649                                    )
650                                        .map_err(|error| ToolFailure {
651                                            code: ToolCallFailureCode::ExecutionFailed,
652                                            message: error.to_string(),
653                                        });
654                                }
655                                bound_tool_result(
656                                    result,
657                                    result_truncation,
658                                    &engine.inner.artifacts,
659                                    active.policy.result_limits.tool_output_max_lines,
660                                    active.policy.result_limits.tool_output_max_bytes,
661                                )
662                                .map_err(ToolFailure::from)
663                            }
664                            Err(error) => {
665                                if let Some(capture) = &capture {
666                                    capture.discard();
667                                }
668                                Err(error.into())
669                            }
670                        };
671                    }
672                    Some(progress) = progress_rx.recv() => {
673                        let _ = append_tool_progress(
674                            &engine,
675                            active.session,
676                            run,
677                            progress,
678                        )
679                        .await;
680                    }
681                    _ = active.cancellation.cancelled() => {
682                        tool_cancellation.cancel();
683                        active
684                            .stdin
685                            .lock()
686                            .unwrap_or_else(|poisoned| poisoned.into_inner())
687                            .remove(&call.id);
688                        let cleanup_deadline =
689                            tokio::time::Instant::now() + TOOL_CANCELLATION_CLEANUP_TIMEOUT;
690                        let cleanup = tokio::time::sleep_until(cleanup_deadline);
691                        tokio::pin!(cleanup);
692                        let mut discarded_progress = 0;
693                        let mut cleanup_timed_out = false;
694                        'cleanup: loop {
695                            tokio::select! {
696                                _ = &mut invoke => {
697                                    while let Ok(progress) = progress_rx.try_recv() {
698                                        if tokio::time::timeout_at(
699                                            cleanup_deadline,
700                                            enqueue_cleanup_tool_progress(&engine, active.session, run, progress),
701                                        )
702                                        .await
703                                        .is_err()
704                                        {
705                                            cleanup_timed_out = true;
706                                            discarded_progress = 1 + close_and_discard_progress(&mut progress_rx);
707                                            break 'cleanup;
708                                        }
709                                    }
710                                    break;
711                                }
712                                Some(progress) = progress_rx.recv() => {
713                                    if tokio::time::timeout_at(
714                                        cleanup_deadline,
715                                        enqueue_cleanup_tool_progress(&engine, active.session, run, progress),
716                                    )
717                                    .await
718                                    .is_err()
719                                    {
720                                        cleanup_timed_out = true;
721                                        discarded_progress = 1 + close_and_discard_progress(&mut progress_rx);
722                                        break;
723                                    }
724                                }
725                                () = &mut cleanup => {
726                                    cleanup_timed_out = true;
727                                    discarded_progress = close_and_discard_progress(&mut progress_rx);
728                                    break;
729                                }
730                            }
731                        }
732                        if let Some(capture) = &capture {
733                            capture.discard();
734                        }
735                        hub.finalize();
736                        engine.retain_finalized_output_hub(call.id);
737                        let message = if cleanup_timed_out {
738                            format!(
739                                "tool call cancelled after it started; cleanup deadline elapsed and {discarded_progress} progress record(s) never entered the session mailbox and were discarded"
740                            )
741                        } else {
742                            "tool call cancelled after it started".into()
743                        };
744                        return Err(ToolFailure {
745                            code: ToolCallFailureCode::ExecutionFailed,
746                            message,
747                        });
748                    }
749                }
750            }
751        }
752    }

764    pub(super) fn published_tool_definitions(
765        &self,
766        session: SessionId,
767        policy: &FrozenRunPolicy,
768    ) -> Result<PublishedToolSet, EngineError> {
769        let session_projection = self.inner.store.get(session)?;
770        let grants = self.skill_grants_for_session(session);
771        let depth = session_depth(&session_projection.meta.origin);
772        let delegate_enabled = policy.agent.delegation.as_ref().is_some_and(|delegation| {
773            depth < delegation.effective_depth_ceiling
774                && delegation
775                    .targets
776                    .iter()
777                    .any(|target| policy.delegation_target_available(target))
778        });
779        let mut names = HashSet::new();
780        let mut output = Vec::new();
781        let mut published = std::collections::HashMap::new();
782        let providers = self
783            .inner
784            .tools
785            .lock()
786            .unwrap_or_else(|poisoned| poisoned.into_inner())
787            .clone();
788        for provider in &providers {
789            for tool in provider
790                .tools_for_session(&SessionToolContext { session })
791                .map_err(|error| EngineError::MissingTool(error.to_string()))?
792            {
793                let delegation_tool = tool.permission_name == "delegate";
794                let enabled = !delegation_tool || delegate_enabled;
795                if enabled
796                    && PermissionPipeline::tool_visible_with_grants(
797                        &policy.agent,
798                        Some(&session_projection.permission_overlay),
799                        grants.as_ref(),
800                        &tool.permission_name,
801                        self.inner.store.cwd(),
802                    )
803                {
804                    if !names.insert(tool.name.clone()) {
805                        return Err(EngineError::MissingTool(format!(
806                            "tool name `{}` is published by more than one provider",
807                            tool.name
808                        )));
809                    }
810                    let schema = JsonSchema::new(tool.parameters.clone()).map_err(|error| {
811                        EngineError::MissingTool(format!(
812                            "tool `{}` has invalid JSON Schema: {error}",
813                            tool.name
814                        ))
815                    })?;
816                    output.push(ToolDefinition::new(
817                        tool.name.clone(),
818                        tool.description.clone(),
819                        schema,
820                    ));
821                    published.insert(
822                        tool.name.clone(),
823                        PublishedTool {
824                            provider: provider.clone(),
825                            spec: tool,
826                        },
827                    );
828                }
829            }
830        }
831        output.sort_by(|left, right| left.name.cmp(&right.name));
832        Ok(PublishedToolSet {
833            definitions: output,
834            tools: published,
835        })
836    }

839pub(super) fn fallback_operation_fingerprint(
840    call: &ToolCall,
841    permission_name: Option<&str>,
842) -> OperationFingerprint {
843    let action = permission_name
844        .or_else(|| fallback_permission_name(&call.name))
845        .and_then(|name| PermissionPipeline::action_for_permission_name(name).ok())
846        .unwrap_or(cookie_agent_protocol::PermissionAction::Read);
847    let binding = serde_json::to_vec(&call.arguments).expect("tool arguments serialize");
848    let operation = PreparedOperationIdentity::new(
849        Sha256Digest::of_bytes(&binding),
850        vec![cookie_agent_protocol::ApprovalCapability {
851            action,
852            operation: cookie_agent_protocol::PreparedCapabilityOperation::new(format!(
853                "{}:prepare",
854                call.name
855                    .replace(|character: char| !character.is_ascii_alphanumeric(), "_")
856            ))
857            .unwrap_or_else(|_| {
858                cookie_agent_protocol::PreparedCapabilityOperation::new("tool:prepare")
859                    .expect("static operation")
860            }),
861        }],
862        vec![cookie_agent_protocol::PreparedApprovalResource {
863            capability: action,
864            canonical: cookie_agent_protocol::PreparedResourceIdentity::new("tool:prepare-failed")
865                .expect("static resource identity"),
866            binding_digest:
867                cookie_agent_protocol::PreparedResourceDigest::from_canonical_binding_bytes(
868                    &binding,
869                ),
870            binding_lifetime: cookie_agent_protocol::PreparedBindingLifetime::ProcessLocal,
871            boundary: cookie_agent_protocol::ApprovalBoundary::Exact,
872            source: cookie_agent_protocol::ApprovalResourceSource::PrimaryOperation,
873        }],
874        Sha256Digest::of_bytes(b"prepare-failed"),
875    )
876    .expect("fallback prepared identity");
877    OperationFingerprint::from_prepared_operation(&operation)
878}
879
880fn fallback_permission_name(tool_name: &str) -> Option<&'static str> {
881    match tool_name {
882        "read" => Some("read"),
883        "write" | "edit" => Some("write"),
884        "bash" => Some("bash"),
885        "delegate_subagent" | "get_subagent_result" | "steer_subagent" | "cancel_subagent" => {
886            Some("delegate")
887        }
888        "skill" => Some("skill"),
889        _ => None,
890    }
891}

893pub(crate) fn apply_permission_resource(
894    provider: &dyn ToolProvider,
895    name: &str,
896    expected_permission_name: &str,
897    prepared: PreparedTool,
898) -> Result<PreparedTool, ToolError> {
899    let (permission_name, resource) =
900        provider.get_permission_resource(name, prepared.normalized_arguments())?;
901    if permission_name != expected_permission_name
902        && !(permission_name == "plugin" && expected_permission_name.starts_with("plugin:"))
903    {
904        return Err(ToolError::execution(
905            "tool permission metadata changed between discovery and preparation",
906        ));
907    }
908    let action = PermissionPipeline::action_for_permission_name(permission_name)
909        .map_err(|error| ToolError::execution(error.to_string()))?;
910    if prepared
911        .operation()
912        .resources()
913        .iter()
914        .any(|prepared| prepared.capability != action)
915    {
916        return Err(ToolError::execution(
917            "prepared permission capability does not match provider metadata",
918        ));
919    }
920    prepared.with_permission_resource(resource)
921}

937pub(super) fn redact_presentation(value: &str) -> String {
938    const SECRET_MARKERS: [&str; 6] = [
939        "token",
940        "secret",
941        "password",
942        "api_key",
943        "apikey",
944        "authorization",
945    ];
946    let mut sanitized = sanitize_safe_text(value, 512);
947    let lowercase = sanitized.to_ascii_lowercase();
948    if SECRET_MARKERS
949        .iter()
950        .any(|marker| lowercase.contains(marker))
951    {
952        sanitized = "<redacted>".into();
953    }
954    sanitized

957    pub(super) content: String,

960pub(super) fn truncate_tool_output(
961    output: &str,
962    max_lines: usize,
963    max_bytes: usize,
964) -> Option<TruncatedToolOutput> {
965    let lines = output.split('\n').collect::<Vec<_>>();
966    let line_truncated = lines.len() > max_lines;
967    let mut preview = lines
968        .iter()
969        .take(max_lines)
970        .copied()
971        .collect::<Vec<_>>()
972        .join("\n");
973    let byte_truncated = output.len() > max_bytes || preview.len() > max_bytes;
974    if !line_truncated && !byte_truncated {
975        return None;
976    }
977    if preview.len() > max_bytes {
978        let mut boundary = max_bytes;
979        while boundary > 0 && !preview.is_char_boundary(boundary) {
980            boundary -= 1;
981        }
982        preview.truncate(boundary);
983    }
984    Some(TruncatedToolOutput { content: preview })
985}

987pub(super) fn bound_tool_result(
988    mut result: ToolResult,
989    policy: crate::ToolResultTruncationPolicy,
990    artifacts: &ArtifactStore,
991    max_lines: usize,
992    max_bytes: usize,
993) -> Result<ToolResult, ToolError> {
994    if policy == crate::ToolResultTruncationPolicy::OptOut {
995        result.truncation = None;
996        if result.output.len() > ToolResult::MAX_OUTPUT_BYTES {
997            return Err(ToolError::resource_limit(format!(
998                "tool output is {} bytes; the event limit is {} bytes",
999                result.output.len(),
1000                ToolResult::MAX_OUTPUT_BYTES
1001            )));
1002        }
1003        return Ok(result);
1004    }
1005    let Some(preview) = truncate_tool_output(&result.output, max_lines, max_bytes) else {
1006        return Ok(result);
1007    };
1008    let original_bytes = result.output.len() as u64;
1009    let original_lines = result.output.split('\n').count() as u64;
1010    let (retained, _) = artifacts
1011        .retain(result.output.as_bytes())
1012        .map_err(|error| ToolError::execution(error.to_string()))?;
1013    result.output = preview.content;
1014    result.truncation = Some(ToolOutputTruncation {
1015        original_bytes,
1016        original_lines,
1017        retained,
1018    });
1019    Ok(result)
1020}

1022pub(crate) fn validate_attachment(
1023    mime_type: &str,
1024    path: &Path,
1025    bytes: &[u8],
1026) -> Result<(), ToolError> {
1027    let max_attachment_bytes = if mime_type.starts_with("video/") {
1028        MAX_VIDEO_ATTACHMENT_BYTES
1029    } else {
1030        MAX_ATTACHMENT_BYTES
1031    };
1032    if bytes.len() as u64 > max_attachment_bytes {
1033        return Err(ToolError::resource_limit(format!(
1034            "attachment is {} bytes; the limit is {max_attachment_bytes} bytes",
1035            bytes.len()
1036        )));
1037    }
1038    let validated = approved_media_type(path, bytes)?.ok_or_else(|| {
1039        ToolError::execution("attachment is not a supported image, PDF, or video")
1040    })?;
1041    if canonical_video_mime_type(validated) != canonical_video_mime_type(mime_type) {
1042        return Err(ToolError::execution(format!(
1043            "attachment MIME mismatch: declared {mime_type}, validated {validated}"
1044        )));
1045    }
1046    Ok(())
1047}
