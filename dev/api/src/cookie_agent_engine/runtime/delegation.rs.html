15                run: parent_run_id,
16                tool_call_id: parent_tool_call_id,
17                child_session_id,
18                reply,
19            }
20        })
21        .await

24    pub(crate) fn ensure_parent_link_blocking(
25        &self,
26        parent_session_id: SessionId,
27        parent_run_id: RunId,
28        parent_tool_call_id: ToolCallId,
29        child_session_id: SessionId,
30    ) -> Result<(), EngineError> {
31        let ensure = || {
32            self.request_blocking(parent_session_id, |reply| {
33                SessionCommand::EnsureToolCallLinked {
34                    run: parent_run_id,
35                    tool_call_id: parent_tool_call_id,
36                    child_session_id,
37                    reply,
38                }
39            })
40        };
41        if tokio::runtime::Handle::try_current().is_ok() {
42            std::thread::scope(|scope| {
43                scope
44                    .spawn(ensure)
45                    .join()
46                    .expect("ensure-link helper thread panicked")
47            })
48        } else {
49            ensure()
50        }
51    }

53    pub(super) async fn terminal_parent_delegate(
54        &self,
55        parent_session_id: SessionId,
56        parent_run_id: RunId,
57        parent_tool_call_id: ToolCallId,
58    ) -> Result<bool, EngineError> {
59        let parent = self.inner.store.get(parent_session_id)?;
60        let Some(run) = parent.runs.get(&parent_run_id) else {
61            return Ok(true);
62        };
63        if !matches!(
64            run.status,
65            SessionStatus::Cancelled | SessionStatus::Failed | SessionStatus::Completed
66        ) {
67            return Ok(false);
68        }
69        if run
70            .pending_calls
71            .get(&parent_tool_call_id)
72            .is_some_and(|tool| tool == "delegate")
73        {
74            let result =
75                cancelled_delegate_result_with_reason(None, "parent run was already terminal");
76            self.append(
77                parent_session_id,
78                Some(parent_run_id),
79                Event::ToolCallTerminated {
80                    termination: ToolCallTermination {
81                        tool_call_id: parent_tool_call_id,
82                        owner: self.tool_call_owner(
83                            parent_session_id,
84                            parent_run_id,
85                            parent_tool_call_id,
86                        )?,
87                        outcome: ToolTerminationOutcome::Cancelled,
88                        result: Some(result),
89                        error: Some(SafeToolError {
90                            code: safe_code("parent_terminal"),
91                            message: safe_error("parent run was already terminal"),
92                        }),
93                    },
94                },
95            )
96            .await?;
97        }
98        Ok(true)
99    }

101    /// Admits a delegate invocation, creates/attaches its child, and starts the
102    /// invocation-derived child run exactly once.
103    pub async fn delegate_invoke(
104        &self,
105        invocation: DelegateInvocation,
106    ) -> Result<DelegateHandle, EngineError> {
107        let invocation_id = invocation_id(
108            invocation.parent_session_id,
109            invocation.parent_run_id,
110            invocation.parent_tool_call_id,
111        );
112        let mut admission =
113            AdmissionGuard::begin(self.inner.clone(), invocation_id, invocation.parent_run_id);
114        admission.set_parent(invocation.parent_session_id, invocation.parent_tool_call_id);
115        // The admission task, rather than this observer, owns the durable start
116        // confirmation and therefore survives a dropped caller future.
117        let (reply, receiver) = oneshot::channel();
118        let engine = self.clone();
119        let generation = admission.generation;
120        let Some(runtime) = tokio::runtime::Handle::try_current().ok() else {
121            return Err(EngineError::ActorStopped);
122        };
123        if !self.spawn_admission_task(&runtime, async move {
124            let result = engine
125                .delegate_invoke_admitted(invocation, invocation_id, generation)
126                .await;
127            let _ = reply.send(result);
128        }) {
129            return Err(EngineError::ActorStopped);
130        }
131        let result = receiver.await.map_err(|_| EngineError::ActorStopped)?;
132        if result.is_ok() {
133            admission.handoff();
134        } else {
135            admission.complete();
136        }
137        result
138    }

140    pub(super) async fn delegate_invoke_admitted(
141        &self,
142        invocation: DelegateInvocation,
143        invocation_id: InvocationId,
144        generation: u64,
145    ) -> Result<DelegateHandle, EngineError> {
146        let _admission = self.inner.delegation_admission.lock().await;
147        let parent = self.inner.store.get(invocation.parent_session_id)?;
148        if parent
149            .runs
150            .get(&invocation.parent_run_id)
151            .is_some_and(|run| run.status == SessionStatus::Interrupted)
152            && self.journal_get(invocation_id).await?.is_none()
153        {
154            return Err(EngineError::MissingTool(
155                "delegate parent run is interrupted; use recovery".into(),
156            ));
157        }
158        let active_parent = self
159            .inner
160            .active
161            .lock()
162            .unwrap_or_else(|poisoned| poisoned.into_inner())
163            .get(&invocation.parent_run_id)
164            .cloned()
165            .ok_or_else(|| EngineError::MissingTool("delegate parent run is not active".into()))?;
166        let parent_policy = Arc::clone(&active_parent.policy);
167        let parent_delegation = parent_policy
168            .agent
169            .delegation
170            .as_ref()
171            .ok_or_else(|| EngineError::MissingTool("delegation is disabled".into()))?;
172        enforce_delegation_concurrency(self, &parent)?;
173        if !parent_delegation.targets.contains(&invocation.agent)
174            || session_depth(&parent.meta.origin) >= parent_delegation.effective_depth_ceiling
175        {
176            return Err(EngineError::MissingTool(
177                "delegate target or depth is not allowed".into(),
178            ));
179        }
180        let child_agent = resolve_agent(&parent_policy.registry, &invocation.agent)?;
181        if !child_agent.document.frontmatter.enabled
182            || !matches!(
183                child_agent.document.frontmatter.mode,
184                cookie_agent_config::AgentMode::Subagent | cookie_agent_config::AgentMode::All
185            )
186        {
187            return Err(EngineError::DisabledAgent(invocation.agent));
188        }
189        let fallback_index = active_parent.fallback_index.load(Ordering::Acquire) as usize;
190        let inherited = parent_policy.active_suffix(fallback_index);
191        let child_selection = child_agent
192            .resolved_fallback
193            .iter()
194            .filter_map(|fallback| match fallback {
195                crate::runtime_snapshot::ResolvedAgentFallback::Selection(selection) => {
196                    Some(selection)
197                }
198                crate::runtime_snapshot::ResolvedAgentFallback::ParentModel => None,
199            })
200            .find(|selection| {
201                parent_policy
202                    .runtime
203                    .models
204                    .model(&selection.model)
205                    .is_some_and(|model| {
206                        model.model.status
207                            == cookie_agent_models::compiler::CompiledModelStatus::Available
208                    })
209            })
210            .cloned()
211            .or_else(|| {
212                if child_agent.resolved_fallback.is_empty() {
213                    inherited.first().map(|binding| binding.selection.clone())
214                } else {
215                    None
216                }
217            })
218            .ok_or_else(|| EngineError::MissingTool("delegate child has no model suffix".into()))?;
219        let child_policy = freeze_delegated_child_policy(
220            child_agent,
221            &parent_policy,
222            &child_selection,
223            inherited,
224            parent_delegation.effective_depth_ceiling,
225            policy::ResultLimits {
226                tool_output_max_lines: self.inner.config.runtime.tool_output.max_lines,
227                tool_output_max_bytes: self.inner.config.runtime.tool_output.max_bytes,
228            },
229        )?;
230        let fingerprint_payload = serde_json::to_vec(&(
231            &invocation.agent,
232            &invocation.task,
233            &invocation.context,
234            &invocation.success_criteria,
235            &invocation.expected_output,
236            &child_policy.agent,
237            &child_policy.selected_suffix_wire,
238        ))
239        .map_err(|_| EngineError::RuntimeCompileFailed)?;
240        let fingerprint = Sha256Digest::new(format!("{:x}", Sha256::digest(&fingerprint_payload)))
241            .map_err(|_| EngineError::RuntimeCompileFailed)?;
242        let request = journal::DelegateRequestPayload {
243            task: invocation.task,
244            context: invocation.context,
245            success_criteria: invocation.success_criteria,
246            expected_output: invocation.expected_output,
247        };
248        let child = match self
249            .create_child(
250                invocation.parent_session_id,
251                invocation.parent_run_id,
252                invocation.parent_tool_call_id,
253                &invocation.agent,
254                child_policy,
255                fingerprint,
256                request,
257                Some((invocation_id, generation)),
258            )
259            .await
260        {
261            Ok(child) => child,
262            Err(error) => {
263                if is_journal_append_failure(&error) {
264                    let result = delegate_failure_result(None, "delegate journal append failed");
265                    self.resolve_delegate_failure_if_pending(
266                        invocation.parent_session_id,
267                        invocation.parent_run_id,
268                        invocation.parent_tool_call_id,
269                        result,
270                    )
271                    .await?;
272                }
273                return Err(error);
274            }
275        };
276        self.publish_admission_child(invocation_id, generation, child.session_id)?;
277        let entry = self
278            .journal_get(invocation_id)
279            .await?
280            .ok_or_else(|| EngineError::MissingTool("delegate reservation disappeared".into()))?;
281        let child_run_id = match self
282            .ensure_delegate_run(&entry, Some((invocation_id, generation)))
283            .await
284        {
285            Ok(run_id) => run_id,
286            Err(error) => {
287                if is_journal_append_failure(&error) {
288                    let result = delegate_failure_result(
289                        Some(entry.reservation.child_session_id),
290                        "delegate journal run confirmation failed",
291                    );
292                    self.resolve_delegate_failure_if_pending(
293                        entry.reservation.parent_session_id,
294                        entry.reservation.parent_run_id,
295                        entry.reservation.parent_tool_call_id,
296                        result,
297                    )
298                    .await?;
299                }
300                return Err(error);
301            }
302        };
303        Ok(DelegateHandle {
304            invocation_id,
305            child_session_id: child.session_id,
306            child_run_id,
307        })
308    }

323    pub(super) async fn await_delegate_inner(
324        &self,
325        handle: DelegateHandle,
326    ) -> Result<ToolResult, EngineError> {
327        loop {
328            let child = match self.inner.store.get(handle.child_session_id) {
329                Ok(child) => child,
330                Err(_) => {
331                    return Ok(delegate_failure_result(
332                        Some(handle.child_session_id),
333                        "child session is missing",
334                    ));
335                }
336            };
337            match child.status {
338                SessionStatus::Running | SessionStatus::Idle => {
339                    let active = {
340                        self.inner
341                            .active
342                            .lock()
343                            .unwrap_or_else(|poisoned| poisoned.into_inner())
344                            .get(&handle.child_run_id)
345                            .cloned()
346                    };
347                    if let Some(active) = active {
348                        // Event-driven terminal-state wakeups are post-MVP; this
349                        // bounded cancellation-aware wait keeps the MVP responsive.
350                        tokio::select! {
351                            () = active.cancellation.cancelled() => {},
352                            () = tokio::time::sleep(std::time::Duration::from_millis(25)) => {},
353                        }
354                    } else {
355                        tokio::time::sleep(std::time::Duration::from_millis(25)).await;
356                    }
357                }
358                SessionStatus::Completed => {
359                    let result = completed_delegate_result(
360                        &child,
361                        Some(handle.child_run_id),
362                        &self.inner.artifacts,
363                    )?;
364                    self.clear_delegate_admissions(handle.invocation_id);
365                    return Ok(result);
366                }
367                SessionStatus::Cancelled => {
368                    let result = cancelled_delegate_result(handle.child_session_id, None);
369                    self.clear_delegate_admissions(handle.invocation_id);
370                    return Ok(result);
371                }
372                SessionStatus::Failed | SessionStatus::Interrupted => {
373                    let result = delegate_failure_result(
374                        Some(handle.child_session_id),
375                        child
376                            .runs
377                            .get(&handle.child_run_id)
378                            .and_then(|run| run.final_text.as_deref())
379                            .unwrap_or("child run failed or was interrupted"),
380                    );
381                    self.clear_delegate_admissions(handle.invocation_id);
382                    return Ok(result);
383                }
384            }
385        }
386    }

388    /// Cancels the child run and returns the structured delegate cancellation
389    /// result used by the parent tool invocation.
390    pub async fn cancel_delegate(&self, handle: DelegateHandle) -> Result<ToolResult, EngineError> {
391        let child = match self.inner.store.get(handle.child_session_id) {
392            Ok(child) => child,
393            Err(_) => {
394                return Ok(delegate_failure_result(
395                    Some(handle.child_session_id),
396                    "child session is missing",
397                ));
398            }
399        };
400        if !matches!(child.status, SessionStatus::Running | SessionStatus::Idle) {
401            return self.await_delegate(handle).await;
402        }
403        let _ = self.cancel_run(handle.child_run_id).await;
404        self.await_delegate(handle).await
405    }

442    pub(super) fn resolve_delegate_failure_if_pending_direct(
443        &self,
444        session_id: SessionId,
445        run_id: RunId,
446        tool_call_id: ToolCallId,
447        result: ToolResult,
448    ) -> Result<bool, EngineError> {
449        let pending = self
450            .inner
451            .store
452            .get(session_id)?
453            .runs
454            .get(&run_id)
455            .is_some_and(|run| {
456                run.pending_calls
457                    .get(&tool_call_id)
458                    .is_some_and(|tool| tool == "delegate")
459            });
460        if !pending {
461            return Ok(false);
462        }
463        self.append_direct(
464            session_id,
465            Some(run_id),
466            Event::ToolCallTerminated {
467                termination: ToolCallTermination {
468                    tool_call_id,
469                    owner: self.tool_call_owner(session_id, run_id, tool_call_id)?,
470                    outcome: ToolTerminationOutcome::Failed,
471                    result: Some(result),
472                    error: Some(SafeToolError {
473                        code: safe_code("delegate_failed"),
474                        message: safe_error("delegated run failed"),
475                    }),
476                },
477            },
478        )?;
479        Ok(true)
480    }

482    pub(super) fn resolve_abandoned_delegate_failure_if_pending_direct(
483        &self,
484        invocation_id: InvocationId,
485        generation: u64,
486        session_id: SessionId,
487        run_id: RunId,
488        tool_call_id: ToolCallId,
489        result: ToolResult,
490    ) -> Result<bool, EngineError> {
491        let mut admissions = self
492            .inner
493            .inflight_delegations
494            .lock()
495            .map_err(|_| EngineError::ActorStopped)?;
496        let still_abandoned = admissions.get(&invocation_id).is_some_and(|entries| {
497            entries
498                .get(&generation)
499                .is_some_and(|entry| entry.cancelled)
500                && entries.values().all(|entry| entry.cancelled)
501        });
502        if !still_abandoned {
503            return Ok(false);
504        }
505        let resolved = self.resolve_delegate_failure_if_pending_direct(
506            session_id,
507            run_id,
508            tool_call_id,
509            result,
510        )?;
511        admissions.remove(&invocation_id);
512        Ok(resolved)
513    }

515    pub(super) async fn ensure_delegate_run(
516        &self,
517        entry: &journal::JournalEntry,
518        admission: Option<(InvocationId, u64)>,
519    ) -> Result<RunId, EngineError> {
520        if admission.is_some_and(|(invocation_id, generation)| {
521            !self.admission_generation_live(invocation_id, generation)
522        }) {
523            return Err(EngineError::MissingTool(
524                "delegate cancelled before child start".into(),
525            ));
526        }
527        let child = self.inner.store.get(entry.reservation.child_session_id)?;
528        if let Some((invocation_id, generation)) = admission {
529            self.mark_admission_starting(invocation_id, generation)?;
530        }
531        if let Some(run_id) = entry.child_run_id
532            && child.runs.contains_key(&run_id)
533        {
534            if let Some((invocation_id, generation)) = admission {
535                self.publish_admission_run(
536                    invocation_id,
537                    generation,
538                    entry.reservation.child_session_id,
539                    run_id,
540                )?;
541                if !self.admission_generation_live(invocation_id, generation) {
542                    self.sweep_abandoned_admission(invocation_id, generation)
543                        .await?;
544                    return Err(EngineError::MissingTool(
545                        "delegate cancelled before child start".into(),
546                    ));
547                }
548            }
549            return Ok(run_id);
550        }
551        let client_run_id = delegate_client_run_id(entry.reservation.invocation_id);
552        let existing_run = child
553            .runs
554            .values()
555            .find(|run| run.client_run_id == client_run_id)
556            .map(|run| run.id);
557        let run_id = match existing_run {
558            Some(run_id) => run_id,
559            None => match self
560                .request(entry.reservation.child_session_id, |reply| {
561                    SessionCommand::Start {
562                        params: RunStartParams {
563                            session_id: entry.reservation.child_session_id,
564                            client_run_id: client_run_id.clone(),
565                            selection: child.meta.creation_selection.clone(),
566                            input: render_delegate_input(&entry.request),
567                        },
568                        admission,
569                        reply,
570                    }
571                })
572                .await
573            {
574                Ok(started) => started.run_id,
575                Err(error) => {
576                    if let Some((invocation_id, generation)) = admission {
577                        self.clear_admission_starting(invocation_id, generation);
578                        if !self.admission_generation_live(invocation_id, generation) {
579                            self.sweep_abandoned_admission(invocation_id, generation)
580                                .await?;
581                        }
582                    }
583                    return Err(error);
584                }
585            },
586        };
587        let cancelled = if let Some((invocation_id, generation)) = admission {
588            // The Start actor published a newly-created run before delivering
589            // its reply. Existing local runs reach the same state here.
590            self.publish_admission_run(
591                invocation_id,
592                generation,
593                entry.reservation.child_session_id,
594                run_id,
595            )?;
596            !self.admission_generation_live(invocation_id, generation)
597        } else {
598            false
599        };
600        // The actor-owned confirmation must run even after its observer drops:
601        // the start reply may already have created the child run.
602        #[cfg(test)]
603        let confirmation_hook = self
604            .inner
605            .admission_confirmation_hook
606            .lock()
607            .expect("admission confirmation hook lock poisoned")
608            .clone();
609        #[cfg(test)]
610        if let Some(hook) = confirmation_hook {
611            let _ = hook.reached.send(());
612            hook.release.wait().await;
613        }
614        let journal = self.inner.journal.clone();
615        let invocation_id = entry.reservation.invocation_id;
616        let confirmation = self
617            .spawn_admission_blocking(move || journal.mark_run_started(invocation_id, run_id))
618            .await;
619        if let Err(error) = confirmation {
620            // A failed confirmation may have poisoned the sole journal writer.
621            // The child already has an active run, so terminally cancel it before
622            // the caller resolves the parent through its actor.
623            let _ = self.cancel_run_durably(
624                run_id,
625                Some("delegate journal run confirmation failed".into()),
626            );
627            return Err(error);
628        }
629        if cancelled {
630            if let Some((invocation_id, generation)) = admission {
631                self.sweep_abandoned_admission(invocation_id, generation)
632                    .await?;
633            }
634            return Err(EngineError::MissingTool(
635                "delegate cancelled during child start".into(),
636            ));
637        }
638        Ok(run_id)
639    }

661pub(super) fn enforce_delegation_concurrency(
662    engine: &Engine,
663    parent: &session::SessionProjection,
664) -> Result<(), EngineError> {
665    let Some(limit) = engine.inner.config.runtime.delegation.max_concurrency else {
666        return Ok(());
667    };
668    let root = match parent.meta.origin {
669        SessionOrigin::Delegated {
670            root_session_id, ..
671        } => root_session_id,
672        _ => parent.meta.session_id,
673    };
674    let active_sessions = engine
675        .inner
676        .active
677        .lock()
678        .unwrap_or_else(|poisoned| poisoned.into_inner())
679        .values()
680        .map(|active| active.session)
681        .collect::<HashSet<_>>();
682    let active_delegated = active_sessions
683        .into_iter()
684        .filter(|session_id| {
685            engine.inner.store.get(*session_id).is_ok_and(|session| {
686                matches!(
687                    session.meta.origin,
688                    SessionOrigin::Delegated { root_session_id, .. } if root_session_id == root
689                )
690            })
691        })
692        .count() as u32;
693    if delegation_concurrency_limit_reached(limit, active_delegated) {
694        return Err(EngineError::MissingTool(format!(
695            "delegate admission denied: active delegated-session concurrency limit {limit} reached"
696        )));
697    }
698    Ok(())
699}
700
701const fn delegation_concurrency_limit_reached(limit: u32, active_delegated: u32) -> bool {
702    active_delegated >= limit
703}

725    child: &session::SessionProjection,

731        .and_then(|run| run.final_text.clone())

741            truncation: None,
742            attachments: Vec::new(),
743        },
744        "delegate",
745        ToolCallId::new_v7(),
746        artifacts,
747        usize::MAX,
748        32 * 1024,
749    )
750}
