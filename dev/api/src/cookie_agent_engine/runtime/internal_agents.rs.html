28    pub(super) async fn run_internal_text_agent(
29        &self,
30        session: SessionId,
31        parent_run: Option<RunId>,
32        kind: InternalAgentKind,
33        policy: &FrozenInternalAgentPolicy,
34        input: String,
35        execution: InternalAgentExecution<'_>,
36    ) -> Result<InternalAgentTextResult, EngineError> {
37        let max_input_bytes = usize::try_from(internal_agent_max_input_limit(policy))
38            .unwrap_or(usize::MAX)
39            .saturating_mul(4);
40        let input = truncate_utf8(&input, max_input_bytes);
41        let history = vec![
42            oven_sdk::HistoryTurn::system(oven_sdk::SystemMessage::new(vec![
43                oven_sdk::SystemPart::Text(oven_sdk::TextPart::new(
44                    policy.agent.composed_prompt.clone(),
45                )),
46            ])),
47            oven_sdk::HistoryTurn::user(oven_sdk::UserMessage::new(vec![
48                oven_sdk::InputPart::Text(oven_sdk::TextPart::new(input.clone())),
49            ])),
50        ];
51        self.run_internal_history_agent(
52            session,
53            parent_run,
54            kind,
55            policy,
56            InternalAgentHistoryInput {
57                history,
58                summary_source: input,
59                tools: Vec::new(),
60                reject_non_text: false,
61            },
62            execution,
63        )
64        .await
65    }

67    pub(super) async fn run_internal_history_agent(
68        &self,
69        session: SessionId,
70        parent_run: Option<RunId>,
71        kind: InternalAgentKind,
72        policy: &FrozenInternalAgentPolicy,
73        input: InternalAgentHistoryInput,
74        execution: InternalAgentExecution<'_>,
75    ) -> Result<InternalAgentTextResult, EngineError> {
76        let name = match kind {
77            InternalAgentKind::Approval => "approval",
78            InternalAgentKind::ContextCompaction => "context_compaction",
79            InternalAgentKind::SessionTitle => "session_title",
80        };
81        let policy = policy.clone();
82        let input_tokens = internal_history_tokens(&input.history, &input.tools)?;
83        let invocation_id = InternalAgentInvocationId::new_v7();
84        let internal_run_id = InternalAgentRunId::new_v7();
85        let call = SafeInternalAgentCall {
86            name: safe_code(name),
87            input_summary: safe_display(&format!(
88                "bounded {name} input ({} bytes)",
89                input.summary_source.len()
90            )),
91            input_digest: Sha256Digest::of_bytes(input.summary_source.as_bytes()),
92        };
93        let mut previous_backend = None;
94        let mut last_failure = InternalAgentFailure {
95            code: safe_code("agent_unavailable"),
96            message: safe_error("no frozen internal model is available"),
97            retryable: false,
98            model_error: None,
99        };
100        for (index, binding) in policy.models.iter().enumerate() {
101            let backend = InternalAgentBackend::Model {
102                resolved_model: wire_model(binding),
103            };
104            if index == 0 {
105                self.append_internal_agent_event(
106                    session,
107                    parent_run,
108                    Event::InternalAgentStarted {
109                        invocation_id,
110                        internal_run_id,
111                        kind,
112                        backend: backend.clone(),
113                        call: call.clone(),
114                    },
115                    execution.actor_direct,
116                )
117                .await?;
118            } else if let Some(from) = previous_backend.take() {
119                self.append_internal_agent_event(
120                    session,
121                    parent_run,
122                    Event::InternalAgentFallback {
123                        invocation_id,
124                        internal_run_id,
125                        kind,
126                        from,
127                        to: backend.clone(),
128                        failure: last_failure.clone(),
129                        attempts: index as u32,
130                    },
131                    execution.actor_direct,
132                )
133                .await?;
134            }
135            let max_input_tokens = internal_agent_input_limit(binding, &policy);
136            if !internal_agent_input_fits(input_tokens, binding, &policy) {
137                last_failure = InternalAgentFailure {
138                    code: safe_code("input_too_large"),
139                    message: safe_error(&format!(
140                        "internal agent input is {input_tokens} estimated tokens, exceeding this model's frozen {max_input_tokens}-token limit"
141                    )),
142                    retryable: false,
143                    model_error: None,
144                };
145                previous_backend = Some(backend);
146                continue;
147            }
148            let runtime = policy
149                .runtime
150                .as_ref()
151                .ok_or(EngineError::NoRunnableModel)?;
152            let model = policy::resolve_model(binding, runtime)?;
153            let max_output_tokens = internal_agent_output_limit(binding, &policy);
154            let request = internal_model_request(
155                input.history.clone(),
156                input.tools.clone(),
157                max_output_tokens,
158            );
159            let request = model.prepare_request_with_cache_strategy(
160                request,
161                policy.prompt_cache_strategy.as_ref(),
162            );
163            let abort = AbortBridge::new(execution.cancellation.child_token());
164            let call_future = model.model().complete(request, abort.signal());
165            let result = tokio::select! {
166                result = tokio::time::timeout(
167                    std::time::Duration::from_millis(match policy.limits.timeout_ms {
168                        0 => DEFAULT_INTERNAL_TIMEOUT_MS,
169                        configured => configured,
170                    }),
171                    call_future,
172                ) => match result {
173                    Ok(result) => result,
174                    Err(_) => {
175                        abort.abort();
176                        Err(ModelError::timeout("internal agent timed out"))
177                    },
178                },
179                _ = execution.cancellation.cancelled() => {
180                    abort.abort();
181                    Err(ModelError::abort("internal agent was cancelled"))
182                }
183            };
184            if execution.cancellation.is_cancelled() {
185                abort.abort();
186                self.append_internal_agent_event(
187                    session,
188                    parent_run,
189                    Event::InternalAgentCancelled {
190                        invocation_id,
191                        internal_run_id,
192                        kind,
193                        reason: Some(safe_error("parent run cancelled")),
194                    },
195                    execution.actor_direct,
196                )
197                .await?;
198                return Err(ModelError::abort("internal agent was cancelled").into());
199            }
200            match result {
201                Ok(completed) => {
202                    let usage =
203                        crate::model_history::persist_usage(completed.turn.finish.usage.clone());
204                    let resolved_model = wire_model(binding);
205                    let estimated_cost_pico_usd = crate::usage::estimated_cost_pico_usd(
206                        &resolved_model,
207                        &usage,
208                        &self.inner.config.runtime.pricing,
209                        &self.catalog_pricing(),
210                    );
211                    self.append_internal_agent_event(
212                        session,
213                        parent_run,
214                        Event::InternalAgentUsageRecorded {
215                            internal_run_id,
216                            kind,
217                            agent_id: policy.agent.agent.clone(),
218                            resolved_model,
219                            usage,
220                            estimated_cost_pico_usd,
221                        },
222                        execution.actor_direct,
223                    )
224                    .await?;
225                    if invalid_internal_output(
226                        &completed.turn.message.content,
227                        input.reject_non_text,
228                    ) {
229                        last_failure = InternalAgentFailure {
230                            code: safe_code("invalid_non_text_output"),
231                            message: safe_error(
232                                "internal agent returned non-text output that cannot be executed",
233                            ),
234                            retryable: false,
235                            model_error: None,
236                        };
237                        previous_backend = Some(backend);
238                        continue;
239                    }
240                    let output = completed
241                        .turn
242                        .message
243                        .content
244                        .iter()
245                        .filter_map(|part| match part {
246                            oven_sdk::AssistantPart::Text(text) => Some(text.text.as_str()),
247                            _ => None,
248                        })
249                        .collect::<String>();
250                    let output_exceeds_document_limit = (policy.limits.max_output_tokens != 0)
251                        && output.len()
252                            > usize::try_from(policy.limits.max_output_tokens)
253                                .unwrap_or(usize::MAX)
254                                .saturating_mul(4);
255                    if output_exceeds_document_limit {
256                        last_failure = InternalAgentFailure {
257                            code: safe_code("output_too_large"),
258                            message: safe_error("internal agent output exceeded its hard bound"),
259                            retryable: false,
260                            model_error: None,
261                        };
262                        previous_backend = Some(backend);
263                        continue;
264                    }
265                    self.append_internal_agent_event(
266                        session,
267                        parent_run,
268                        Event::InternalAgentCompleted {
269                            invocation_id,
270                            internal_run_id,
271                            kind,
272                            result: SafeInternalAgentResult {
273                                output_summary: safe_display(&format!(
274                                    "validated {name} output ({} bytes)",
275                                    output.len()
276                                )),
277                                output_digest: Sha256Digest::of_bytes(output.as_bytes()),
278                            },
279                        },
280                        execution.actor_direct,
281                    )
282                    .await?;
283                    return Ok(InternalAgentTextResult {
284                        invocation_id,
285                        internal_run_id,
286                        text: output,
287                    });
288                }
289                Err(error) => {
290                    last_failure = InternalAgentFailure {
291                        code: safe_code("model_failure"),
292                        message: safe_error(&error.message),
293                        retryable: error.retryable,
294                        model_error: Some(model_error_summary(&error)),
295                    };
296                    previous_backend = Some(backend);
297                }
298            }
299        }
300        if policy.models.is_empty() {
301            self.append_internal_agent_event(
302                session,
303                parent_run,
304                Event::InternalAgentStarted {
305                    invocation_id,
306                    internal_run_id,
307                    kind,
308                    backend: InternalAgentBackend::Builtin {
309                        name: safe_code("unavailable"),
310                        revision: safe_display(UNAVAILABLE_BUILTIN_REVISION),
311                    },
312                    call,
313                },
314                execution.actor_direct,
315            )
316            .await?;
317        }
318        self.append_internal_agent_event(
319            session,
320            parent_run,
321            Event::InternalAgentFailed {
322                invocation_id,
323                internal_run_id,
324                kind,
325                failure: last_failure,
326            },
327            execution.actor_direct,
328        )
329        .await?;
330        Err(ModelError::invalid_response("internal agent failed safely").into())
331    }

360    pub(crate) fn internal_agent_policy(
361        &self,
362        kind: InternalAgentKind,
363        owner: &FrozenRunPolicy,
364        parent_binding: Option<&cookie_agent_protocol::FrozenModelBinding>,
365    ) -> Result<FrozenInternalAgentPolicy, EngineError> {
366        let id = AgentId::new(match kind {
367            InternalAgentKind::Approval => cookie_agent_config::BUILT_IN_APPROVAL_AGENT_ID,
368            InternalAgentKind::ContextCompaction => {
369                cookie_agent_config::BUILT_IN_COMPACTION_AGENT_ID
370            }
371            InternalAgentKind::SessionTitle => cookie_agent_config::BUILT_IN_TITLE_AGENT_ID,
372        })
373        .map_err(|_| EngineError::RuntimeCompileFailed)?;
374        let resolved = owner
375            .registry
376            .get(&id)
377            .ok_or_else(|| EngineError::InvalidRuntimeAgent(id.clone()))?;
378        if resolved.document.frontmatter.mode != cookie_agent_config::AgentMode::Internal {
379            return Err(EngineError::InvalidRuntimeAgent(id));
380        }
381        let mut models = Vec::new();
382        let fallbacks = if resolved.document.frontmatter.enabled {
383            resolved.resolved_fallback.as_slice()
384        } else {
385            &[]
386        };
387        for fallback in fallbacks {
388            match fallback {
389                crate::runtime_snapshot::ResolvedAgentFallback::ParentModel => {
390                    if let Some(binding) = parent_binding {
391                        models.push(binding.clone());
392                    }
393                }
394                crate::runtime_snapshot::ResolvedAgentFallback::Selection(selection) => {
395                    if owner
396                        .runtime
397                        .models
398                        .model(&selection.model)
399                        .is_some_and(|model| {
400                            model.model.status
401                                == cookie_agent_models::compiler::CompiledModelStatus::Available
402                        })
403                    {
404                        models.push(crate::model_snapshots::binding_for_selection(
405                            &owner.runtime.current_manifest,
406                            &owner.runtime.models,
407                            selection,
408                        )?);
409                    }
410                }
411            }
412        }
413        let document = &resolved.document;
414        let snapshot = cookie_agent_protocol::AgentSnapshot {
415            agent: document.id.clone(),
416            schema: cookie_agent_protocol::AgentSchemaVersion::current(),
417            mode: AgentMode::Internal,
418            description: document.frontmatter.description.clone(),
419            document_source: match document.source {
420                cookie_agent_config::AgentDocumentSource::BuiltIn => {
421                    cookie_agent_protocol::AgentDocumentSource::BuiltIn
422                }
423                cookie_agent_config::AgentDocumentSource::User => {
424                    cookie_agent_protocol::AgentDocumentSource::User
425                }
426                cookie_agent_config::AgentDocumentSource::Workspace => {
427                    cookie_agent_protocol::AgentDocumentSource::Workspace
428                }
429            },
430            document_fingerprint: Sha256Digest::new(document.document_fingerprint.as_str())
431                .map_err(|_| EngineError::RuntimeCompileFailed)?,
432            composed_prompt: document.body.clone(),
433            prompt_fingerprint: Sha256Digest::new(document.prompt_fingerprint.as_str())
434                .map_err(|_| EngineError::RuntimeCompileFailed)?,
435            max_output_tokens: document.frontmatter.limits.max_output_tokens,
436            permissions: Vec::new(),
437            delegation: None,
438            fallback_chain: models.clone(),
439            selected_suffix_start: 0,
440        };
441        let limits = &document.frontmatter.limits;
442        let mut prompt_cache_strategy = owner.prompt_cache_strategy.clone();
443        if matches!(
444            kind,
445            InternalAgentKind::Approval | InternalAgentKind::SessionTitle
446        ) && let Some(strategy) = &mut prompt_cache_strategy
447        {
448            strategy.rolling = None;
449        }
450        Ok(FrozenInternalAgentPolicy {
451            agent: snapshot,
452            models,
453            runtime: Some(Arc::clone(&owner.runtime)),
454            limits: InternalAgentLimits {
455                max_output_tokens: limits.max_output_tokens,
456                timeout_ms: limits.timeout_ms,
457            },
458            prompt_cache_strategy,
459        })
460    }

463fn internal_agent_max_input_limit(policy: &FrozenInternalAgentPolicy) -> u64 {
464    policy
465        .models
466        .iter()
467        .map(|binding| internal_agent_input_limit(binding, policy))
468        .max()
469        .unwrap_or(UNKNOWN_INTERNAL_CONTEXT_LIMIT)
470}

472pub(super) fn internal_agent_input_limit(
473    binding: &cookie_agent_protocol::FrozenModelBinding,
474    policy: &FrozenInternalAgentPolicy,
475) -> u64 {
476    binding.descriptor.capabilities.limits.context.map_or(
477        UNKNOWN_INTERNAL_CONTEXT_LIMIT,
478        |context| {
479            context
480                .saturating_sub(internal_agent_output_limit(binding, policy).unwrap_or(0))
481                .max(1)
482        },
483    )
484}

486pub(super) fn internal_agent_input_fits(
487    input_tokens: u64,
488    binding: &cookie_agent_protocol::FrozenModelBinding,
489    policy: &FrozenInternalAgentPolicy,
490) -> bool {
491    input_tokens <= internal_agent_input_limit(binding, policy)
492}

494pub(super) fn internal_agent_output_limit(
495    binding: &cookie_agent_protocol::FrozenModelBinding,
496    policy: &FrozenInternalAgentPolicy,
497) -> Option<u64> {
498    match (
499        binding.descriptor.capabilities.limits.output,
500        policy.limits.max_output_tokens,
501    ) {
502        (Some(model), 0) => Some(model),
503        (Some(model), document) => Some(model.min(document)),
504        (None, 0) => None,
505        (None, document) => Some(document),
506    }
507}

519pub(super) fn internal_history_tokens(
520    history: &[oven_sdk::HistoryTurn],
521    tools: &[ToolDefinition],
522) -> Result<u64, EngineError> {
523    let bytes = serde_json::to_vec(&(history, tools))
524        .map_err(|error| EngineError::from(ModelError::invalid_request(error.to_string())))?
525        .len();
526    Ok((bytes as u64).div_ceil(4))
527}

529fn invalid_internal_output(parts: &[oven_sdk::AssistantPart], reject_non_text: bool) -> bool {
530    // Reasoning parts are visible assistant text, not executable output; only
531    // parts that would need execution or attachment handling are invalid.
532    reject_non_text
533        && parts.iter().any(|part| {
534            !matches!(
535                part,
536                oven_sdk::AssistantPart::Text(_) | oven_sdk::AssistantPart::Reasoning(_)
537            )
538        })
539}

541pub(super) fn parse_internal_approval(value: &str) -> Option<ApprovalInternalDecisionKind> {
542    #[derive(Deserialize)]
543    #[serde(deny_unknown_fields)]
544    struct Decision {
545        decision: String,
546    }
547    match serde_json::from_str::<Decision>(value.trim())
548        .ok()?
549        .decision
550        .as_str()
551    {
552        "allow" => Some(ApprovalInternalDecisionKind::Allow),
553        "deny" => Some(ApprovalInternalDecisionKind::Deny),
554        "ask" => Some(ApprovalInternalDecisionKind::Ask),
555        _ => None,
556    }
557}
