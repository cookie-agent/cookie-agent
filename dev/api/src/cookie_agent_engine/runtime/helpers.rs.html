13impl Engine {
14    pub(super) fn tool_call_owner(
15        &self,
16        session_id: SessionId,
17        run_id: RunId,
18        tool_call_id: ToolCallId,
19    ) -> Result<cookie_agent_protocol::AssistantToolCallRef, EngineError> {
20        self.inner
21            .store
22            .get(session_id)?
23            .log
24            .events()
25            .into_iter()
26            .find_map(|event| match event.payload {
27                Event::ToolCallStarted { start }
28                    if event.run_id == Some(run_id) && start.tool_call_id == tool_call_id =>
29                {
30                    Some(start.owner)
31                }
32                _ => None,
33            })
34            .ok_or_else(|| EngineError::MissingTool("tool ownership is missing".into()))
35    }

37    pub(super) fn terminate_tool_direct(
38        &self,
39        session_id: SessionId,
40        run_id: RunId,
41        tool_call_id: ToolCallId,
42        outcome: ToolTerminationOutcome,
43        result: Option<ToolResult>,
44        error: Option<SafeToolError>,
45    ) -> Result<(), EngineError> {
46        self.append_direct(
47            session_id,
48            Some(run_id),
49            Event::ToolCallTerminated {
50                termination: ToolCallTermination {
51                    tool_call_id,
52                    owner: self.tool_call_owner(session_id, run_id, tool_call_id)?,
53                    outcome,
54                    result,
55                    error,
56                },
57            },
58        )
59    }

127pub(super) fn hash_parts(parts: &[&str]) -> u128 {
128    use std::hash::{Hash, Hasher};
129    let mut first = std::collections::hash_map::DefaultHasher::new();
130    parts.hash(&mut first);
131    let high = first.finish() as u128;
132    let mut second = std::collections::hash_map::DefaultHasher::new();
133    "cookie_agent".hash(&mut second);
134    parts.hash(&mut second);
135    (high << 64) | second.finish() as u128
136}

152pub(super) fn sanitize_safe_text(value: &str, maximum: usize) -> String {
153    let mut output = String::new();
154    for character in value.chars() {
155        let character = if character.is_control() {
156            ' '
157        } else {
158            character
159        };
160        if output.len() + character.len_utf8() > maximum {
161            break;
162        }
163        output.push(character);
164    }
165    if output.is_empty() {
166        "unavailable".into()
167    } else {
168        output
169    }
170}

172pub(super) fn truncate_utf8(value: &str, maximum: usize) -> String {
173    if value.len() <= maximum {
174        return value.to_owned();
175    }
176    let mut end = maximum;
177    while end > 0 && !value.is_char_boundary(end) {
178        end -= 1;
179    }
180    value[..end].to_owned()
181}
