26    pub(super) async fn request_model_approval(
27        &self,
28        active: &ActiveRun,
29        run: RunId,
30        input: ModelApprovalInput<'_>,
31    ) -> Result<ApprovalOutcome, EngineError> {
32        let approval_policy = self.active_internal_policy(active, InternalAgentKind::Approval)?;
33        let request = approval_request_for_operation(
34            ApprovalTrigger::ModelToolApproval,
35            input.operation.clone(),
36            input
37                .operation
38                .resources()
39                .iter()
40                .zip(input.policy_labels)
41                .map(|(resource, label)| cookie_agent_protocol::DecisionTrace {
42                    action: resource.capability,
43                    normalized_resource: label
44                        .clone()
45                        .unwrap_or_else(|| UNSCOPED_PERMISSION_RESOURCE_DISPLAY.to_owned()),
46                    candidates: Vec::new(),
47                    effect: cookie_agent_protocol::PermissionEffect::Ask,
48                    precedence_reason: input
49                        .message
50                        .clone()
51                        .unwrap_or_else(|| "model requested tool approval".into()),
52                })
53                .collect(),
54            false,
55            approval_expiry(approval_policy.limits.timeout_ms),
56        );
57        self.await_user_approval(active, run, request, input.executor, false, input.tool)
58            .await
59    }

61    pub(super) async fn await_user_approval(
62        &self,
63        active: &ActiveRun,
64        run: RunId,
65        request: ApprovalRequest,
66        executor: PreparedExecutorCell,
67        allow_prior_grant: bool,
68        tool: ApprovalToolInput<'_>,
69    ) -> Result<ApprovalOutcome, EngineError> {
70        let approval_id = request.approval_id();
71        let session = self.inner.store.get(active.session)?;
72        let root = root_id(&session.meta.origin, active.session);
73        self.append(
74            active.session,
75            Some(run),
76            super::event_origin("engine:approvals"),
77            Event::ApprovalRequested {
78                request: request.clone(),
79            },
80        )
81        .await?;
82
83        let repetitions = doom_loop_repetitions(
84            &self.inner.store.get(active.session)?.log.events(),
85            run,
86            request.operation_fingerprint(),
87        );
88        if repetitions >= 4 {
89            self.append(
90                active.session,
91                Some(run),
92                super::event_origin("engine:approvals"),
93                Event::ApprovalDoomLoopDetected {
94                    approval_id,
95                    operation_fingerprint: request.operation_fingerprint().clone(),
96                    repetitions,
97                },
98            )
99            .await?;
100            self.append(
101                active.session,
102                Some(run),
103                super::event_origin("engine:approvals"),
104                Event::ApprovalFinalized {
105                    approval_id,
106                    decision: ApprovalFinalDecision {
107                        outcome: ApprovalFinalOutcome::Rejected,
108                        source: ApprovalDecisionSource::DoomLoopGuard,
109                        reason_code: ApprovalReasonCode::DoomLoopDetected,
110                        feedback: None,
111                        tree_grant_id: None,
112                    },
113                },
114            )
115            .await?;
116            return Ok(ApprovalOutcome {
117                approved: false,
118                feedback: None,
119            });
120        }
121
122        if allow_prior_grant
123            && let Some(grant) = self.inner.approvals.matching(root, request.operation())
124        {
125            let decision = ApprovalInternalDecision {
126                decision: ApprovalInternalDecisionKind::Allow,
127                source: ApprovalDecisionSource::TreeGrant,
128                reason_code: ApprovalReasonCode::TreeGrantMatched,
129                evaluations: approval_evaluations(&request),
130            };
131            self.append(
132                active.session,
133                Some(run),
134                super::event_origin("engine:approvals"),
135                Event::ApprovalEvaluated {
136                    approval_id,
137                    decision,
138                },
139            )
140            .await?;
141            self.append(
142                active.session,
143                Some(run),
144                super::event_origin("engine:approvals"),
145                Event::ApprovalFinalized {
146                    approval_id,
147                    decision: ApprovalFinalDecision {
148                        outcome: ApprovalFinalOutcome::Approved,
149                        source: ApprovalDecisionSource::TreeGrant,
150                        reason_code: ApprovalReasonCode::TreeGrantMatched,
151                        feedback: None,
152                        tree_grant_id: Some(grant.grant_id),
153                    },
154                },
155            )
156            .await?;
157            return Ok(ApprovalOutcome {
158                approved: true,
159                feedback: None,
160            });
161        }
162
163        if approval_evaluations(&request)
164            .iter()
165            .any(|evaluation| evaluation.effect == cookie_agent_protocol::PermissionEffect::Deny)
166        {
167            let decision = ApprovalInternalDecision {
168                decision: ApprovalInternalDecisionKind::Deny,
169                source: ApprovalDecisionSource::Policy,
170                reason_code: ApprovalReasonCode::PolicyDenied,
171                evaluations: approval_evaluations(&request),
172            };
173            self.append(
174                active.session,
175                Some(run),
176                super::event_origin("engine:approvals"),
177                Event::ApprovalEvaluated {
178                    approval_id,
179                    decision,
180                },
181            )
182            .await?;
183            self.append(
184                active.session,
185                Some(run),
186                super::event_origin("engine:approvals"),
187                Event::ApprovalFinalized {
188                    approval_id,
189                    decision: ApprovalFinalDecision {
190                        outcome: ApprovalFinalOutcome::Rejected,
191                        source: ApprovalDecisionSource::Policy,
192                        reason_code: ApprovalReasonCode::PolicyDenied,
193                        feedback: None,
194                        tree_grant_id: None,
195                    },
196                },
197            )
198            .await?;
199            return Ok(ApprovalOutcome {
200                approved: false,
201                feedback: None,
202            });
203        }
204
205        let permission_mode = self
206            .inner
207            .permission_modes
208            .lock()
209            .unwrap_or_else(|poisoned| poisoned.into_inner())
210            .get(&active.session)
211            .copied()
212            .unwrap_or_default();
213        if permission_mode == PermissionMode::Yolo {
214            self.append(
215                active.session,
216                Some(run),
217                super::event_origin("engine:approvals"),
218                Event::ApprovalEvaluated {
219                    approval_id,
220                    decision: ApprovalInternalDecision {
221                        decision: ApprovalInternalDecisionKind::Allow,
222                        source: ApprovalDecisionSource::Policy,
223                        reason_code: ApprovalReasonCode::YoloApproved,
224                        evaluations: approval_evaluations(&request),
225                    },
226                },
227            )
228            .await?;
229            self.append(
230                active.session,
231                Some(run),
232                super::event_origin("engine:approvals"),
233                Event::ApprovalFinalized {
234                    approval_id,
235                    decision: ApprovalFinalDecision {
236                        outcome: ApprovalFinalOutcome::Approved,
237                        source: ApprovalDecisionSource::Policy,
238                        reason_code: ApprovalReasonCode::YoloApproved,
239                        feedback: None,
240                        tree_grant_id: None,
241                    },
242                },
243            )
244            .await?;
245            return Ok(ApprovalOutcome {
246                approved: true,
247                feedback: None,
248            });
249        }
250
251        let internal_kind = match permission_mode {
252            PermissionMode::Ask => ApprovalInternalDecisionKind::Ask,
253            PermissionMode::AutoApprove
254            | PermissionMode::AutoApproveN
255            | PermissionMode::AutoApproveY => {
256                #[cfg(test)]
257                let hook = {
258                    self.inner
259                        .approval_evaluation_hook
260                        .lock()
261                        .expect("approval evaluation hook lock poisoned")
262                        .take()
263                };
264                #[cfg(test)]
265                if let Some(hook) = hook {
266                    if let Some(reached) = hook
267                        .reached
268                        .lock()
269                        .expect("approval evaluation reached lock poisoned")
270                        .take()
271                    {
272                        let _ = reached.send(());
273                    }
274                    hook.release.notified().await;
275                }
276                let approval_policy =
277                    self.active_internal_policy(active, InternalAgentKind::Approval)?;
278                self.evaluate_stateless_approval(active, run, &session, &approval_policy, tool)
279                    .await
280            }
281            PermissionMode::Yolo => unreachable!("yolo approvals resolve before prompting"),
282        };
283        let transition = self
284            .request(active.session, |reply| {
285                SessionCommand::ApprovalEvaluationComplete {
286                    run,
287                    request: request.clone(),
288                    executor: executor.clone(),
289                    decision: internal_kind,
290                    permission_mode,
291                    cancelled: active.cancellation.is_cancelled(),
292                    reply,
293                }
294            })
295            .await?;
296        let mut receiver = match transition {
297            ApprovalEvaluationTransition::Resolved(outcome) => return Ok(outcome),
298            ApprovalEvaluationTransition::Escalated(receiver) => receiver,
299        };
300        let expiry_wait = approval_expiry_wait(approval_constraints(&request).expires_at);
301        tokio::select! {
302            decision = &mut receiver => decision.map_err(|_| EngineError::ActorStopped),
303            _ = active.cancellation.cancelled() => {
304                let finalized = self.request(active.session, |reply| {
305                    SessionCommand::ApprovalTerminal {
306                        run,
307                        approval_id,
308                        terminal: ApprovalTerminal::Cancelled,
309                        reply,
310                    }
311                }).await?;
312                if finalized {
313                    Ok(ApprovalOutcome {
314                        approved: false,
315                        feedback: Some("cancelled".into()),
316                    })
317                } else {
318                    receiver.await.map_err(|_| EngineError::ActorStopped)
319                }
320            },
321            _ = tokio::time::sleep(expiry_wait) => {
322                let finalized = self.request(active.session, |reply| {
323                    SessionCommand::ApprovalTerminal {
324                        run,
325                        approval_id,
326                        terminal: ApprovalTerminal::Expired,
327                        reply,
328                    }
329                }).await?;
330                if finalized {
331                    Ok(ApprovalOutcome {
332                        approved: false,
333                        feedback: Some("approval expired unattended".into()),
334                    })
335                } else {
336                    receiver.await.map_err(|_| EngineError::ActorStopped)
337                }
338            }
339        }
340    }

342    async fn evaluate_stateless_approval(
343        &self,
344        active: &ActiveRun,
345        run: RunId,
346        session: &crate::session::SessionProjection,
347        policy: &FrozenInternalAgentPolicy,
348        tool: ApprovalToolInput<'_>,
349    ) -> ApprovalInternalDecisionKind {
350        let events = session.log.events();
351        let prompt = approval_stateless_input(tool, latest_user_message(&events, run));
352        let history =
353            approval_stateless_history(policy.agent.composed_prompt.clone(), prompt.clone());
354        let result = self
355            .run_internal_history_agent(
356                active.session,
357                Some(run),
358                InternalAgentKind::Approval,
359                policy,
360                InternalAgentHistoryInput {
361                    history,
362                    summary_source: prompt,
363                    tools: Vec::new(),
364                    reject_non_text: true,
365                },
366                InternalAgentExecution {
367                    cancellation: &active.cancellation,
368                    actor_direct: false,
369                },
370            )
371            .await;
372        match result {
373            Ok(result) => {
374                parse_internal_approval(&result.text).unwrap_or(ApprovalInternalDecisionKind::Ask)
375            }
376            Err(_) => ApprovalInternalDecisionKind::Ask,
377        }
378    }

381fn approval_stateless_input(tool: ApprovalToolInput<'_>, latest_user: Option<&str>) -> String {
382    let tool_call = serde_json::json!({
383        "name": tool.name,
384        "normalized_parameters": canonical_approval_parameters(tool.normalized_parameters),
385    });
386    format!(
387        "{APPROVAL_USER_REQUEST_PREFIX}{}{APPROVAL_USER_REQUEST_SUFFIX}{APPROVAL_TOOL_CALL_PREFIX}{}{APPROVAL_TOOL_CALL_SUFFIX}",
388        latest_user.unwrap_or(APPROVAL_NO_USER_MESSAGE),
389        serde_json::to_string(&tool_call).expect("safe approval tool call serializes")
390    )
391}
392
393fn canonical_approval_parameters(value: &Value) -> Value {
394    match value {
395        Value::Array(values) => {
396            Value::Array(values.iter().map(canonical_approval_parameters).collect())
397        }
398        Value::Object(values) => {
399            let mut keys = values.keys().collect::<Vec<_>>();
400            keys.sort_unstable();
401            Value::Object(
402                keys.into_iter()
403                    .map(|key| (key.clone(), canonical_approval_parameters(&values[key])))
404                    .collect(),
405            )
406        }
407        _ => value.clone(),
408    }
409}

411fn latest_user_message(events: &[StoredEvent], run: RunId) -> Option<&str> {
412    events
413        .iter()
414        .rev()
415        .find_map(|event| match &event.payload {
416            Event::UserInputSubmitted { input } if event.run_id == Some(run) => {
417                Some(input.as_str())
418            }
419            _ => None,
420        })
421        .or_else(|| {
422            events.iter().rev().find_map(|event| match &event.payload {
423                Event::UserInputSubmitted { input } => Some(input.as_str()),
424                _ => None,
425            })
426        })
427}

456pub(super) fn approval_request_for_operation(
457    trigger: ApprovalTrigger,
458    operation: PreparedOperationIdentity,
459    traces: Vec<cookie_agent_protocol::DecisionTrace>,
460    allow_tree_grant: bool,
461    expires_at: Option<jiff::Timestamp>,
462) -> ApprovalRequest {
463    let evaluations = operation
464        .resources()
465        .iter()
466        .zip(traces)
467        .map(|(resource, trace)| ApprovalEvaluation {
468            resource_digest: resource.binding_digest.clone(),
469            effect: trace.effect,
470            trace,
471        })
472        .collect::<Vec<_>>();
473    ApprovalRequest::new(
474        ApprovalId::new_v7(),
475        1,
476        trigger,
477        operation,
478        evaluations,
479        ApprovalConstraints {
480            allow_once: true,
481            allow_tree_grant,
482            cancellable: true,
483            expires_at,
484        },
485    )
486    .expect("prepared approval request is complete")
487}

489pub(super) fn approval_expiry(timeout_ms: u64) -> Option<jiff::Timestamp> {
490    jiff::Timestamp::now()
491        .checked_add(std::time::Duration::from_millis(timeout_ms))
492        .ok()
493}
494
495pub(super) fn approval_deadline_exhausted(expires_at: Option<jiff::Timestamp>) -> bool {
496    expires_at.is_some_and(|expires_at| expires_at <= jiff::Timestamp::now())
497}
498
499pub(super) fn approval_expiry_wait(expires_at: Option<jiff::Timestamp>) -> std::time::Duration {
500    let Some(expires_at) = expires_at else {
501        return std::time::Duration::from_secs(100 * 365 * 24 * 60 * 60);
502    };
503    let now = jiff::Timestamp::now();
504    if expires_at <= now {
505        std::time::Duration::ZERO
506    } else {
507        expires_at.duration_since(now).unsigned_abs()
508    }
509}
