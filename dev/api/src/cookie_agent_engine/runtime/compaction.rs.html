102    pub(super) async fn compact_session_direct(
103        &self,
104        session: SessionId,
105        focus: Option<&str>,
106        origin: cookie_agent_protocol::EventOrigin,
107    ) -> Result<SessionCompactResult, EngineError> {
108        let projection = self.inner.store.get(session)?;
109        if projection.status == SessionStatus::Running {
110            return Err(EngineError::SessionRunning(session));
111        }
112        let events = projection.log.event_snapshot();
113        let run = projection
114            .log
115            .last_run_started()
116            .map(|(_, run, _)| run)
117            .ok_or(EngineError::NoRunnableModel)?;
118        let policy = self.historical_title_policy(&events, run)?;
119        let binding = active_compaction_binding(&policy, &events, run)?;
120        let internal_policy = self.internal_agent_policy(
121            InternalAgentKind::ContextCompaction,
122            &policy,
123            Some(binding),
124        )?;
125        let tools = self.tool_definitions(session, &policy)?;
126        let before = projection.log.latest_checkpoint_seq();
127        match self
128            .maybe_compact_context(CompactionInput {
129                session,
130                run,
131                cancellation: &CancellationToken::new(),
132                binding,
133                owner_policy: &policy,
134                internal_policy: &internal_policy,
135                tools: &tools,
136                events,
137                force: true,
138                overflow_recovery: false,
139                focus,
140                actor_direct: false,
141                origin,
142            })
143            .await
144        {
145            Ok(_) => {}
146            Err(EngineError::CompactionCancelled(reason)) => {
147                return Ok(SessionCompactResult {
148                    compacted: false,
149                    cancellation_reason: Some(reason),
150                });
151            }
152            Err(error) => return Err(error),
153        }
154        Ok(SessionCompactResult {
155            compacted: projection.log.latest_checkpoint_seq() > before,
156            cancellation_reason: None,
157        })
158    }

160    pub(super) async fn maybe_compact_context(
161        &self,
162        input: CompactionInput<'_>,
163    ) -> Result<Arc<[StoredEvent]>, EngineError> {
164        let Some(context_limit) = input.binding.descriptor.capabilities.limits.context else {
165            return Ok(input.events);
166        };
167        let config = &self.inner.config.runtime.context_compaction;
168        let trigger_tokens = resolve_compaction_trigger(context_limit, &config.trigger);
169        if !compaction_gate(input.force, config.auto_compaction, trigger_tokens) {
170            return Ok(input.events);
171        }
172        let projection = self.inner.store.get(input.session)?;
173        if !input.force {
174            let log = &projection.log;
175            let Some((usage_seq, observed_tokens)) = log.latest_real_usage() else {
176                return Ok(input.events);
177            };
178            let last_checkpoint_seq = log.latest_checkpoint_seq();
179            if usage_seq < last_checkpoint_seq {
180                return Ok(input.events);
181            }
182            if !usage_reaches_compaction_trigger(observed_tokens, trigger_tokens) {
183                return Ok(input.events);
184            }
185        }
186
187        let requested_input_through_seq = input.events.last().map_or(0, |event| event.seq);
188        let current_events = projection.log.event_snapshot();
189        if projection
190            .log
191            .checkpoint_covers_input(requested_input_through_seq)
192        {
193            return Ok(current_events);
194        }
195
196        let mut compaction_focus = input.focus.map(str::to_owned);
197        let mut additions = Vec::new();
198        let context_id = crate::plugin::plugin_context_id();
199        for plugin in self.inner.plugins.interception_plugins(
200            cookie_agent_protocol::ExtensionInterceptionHook::SessionBeforeCompact,
201        ) {
202            let result = self
203                .inner
204                .plugins
205                .intercept_named::<_, cookie_agent_protocol::ExtensionSessionBeforeCompactResult>(
206                    &plugin,
207                    cookie_agent_protocol::PLUGIN_INTERCEPT_SESSION_BEFORE_COMPACT_METHOD,
208                    &ExtensionSessionBeforeCompactParams {
209                        session_id: input.session,
210                        context_id: context_id.clone(),
211                        checkpoint_id: format!("{}:{requested_input_through_seq}", input.session),
212                        additions: additions.clone(),
213                        instructions: compaction_focus.clone(),
214                    },
215                    Some(input.session),
216                    Some(&context_id),
217                )
218                .await;
219            match result {
220                Ok(result) => {
221                    if result.cancel {
222                        let reason = result
223                            .reason
224                            .unwrap_or_else(|| "compaction cancelled by plugin".into());
225                        self.record_plugin_diagnostic(
226                            input.session,
227                            plugin,
228                            PluginDiagnosticKind::HookBlocked,
229                            reason.clone(),
230                        );
231                        return Err(EngineError::CompactionCancelled(reason));
232                    }
233                    if let Some(instructions) = result.instructions_override {
234                        if instructions.len() > 64 * 1024 {
235                            self.record_plugin_diagnostic(
236                                input.session,
237                                plugin.clone(),
238                                PluginDiagnosticKind::InvalidModification,
239                                "plugin compaction instruction override exceeds the 64 KiB limit"
240                                    .into(),
241                            );
242                        } else {
243                            compaction_focus = Some(instructions);
244                        }
245                    }
246                    if let Some(addendum) = result.addendum.filter(|value| !value.is_empty()) {
247                        let addition_bytes = additions.iter().map(String::len).sum::<usize>();
248                        if addition_bytes.saturating_add(addendum.len()) > 64 * 1024 {
249                            self.record_plugin_diagnostic(
250                                input.session,
251                                plugin,
252                                PluginDiagnosticKind::InvalidModification,
253                                "plugin compaction additions exceed the 64 KiB limit".into(),
254                            );
255                            continue;
256                        }
257                        let focus = compaction_focus.get_or_insert_with(String::new);
258                        if !focus.is_empty() {
259                            focus.push('\n');
260                        }
261                        focus.push_str(&addendum);
262                        additions.push(addendum);
263                    }
264                }
265                Err(error) => {
266                    let kind = if error.contains("crashed") || error.contains("not connected") {
267                        PluginDiagnosticKind::InterceptionCrash
268                    } else {
269                        PluginDiagnosticKind::InterceptionTimeout
270                    };
271                    self.record_plugin_diagnostic(input.session, plugin, kind, error);
272                }
273            }
274        }
275
276        let composed_prompt = self.run_agent_prompt(input.session, input.run)?;
277        let mut events = input.events.to_vec();
278        let mut context = assemble_model_context(
279            &events,
280            &self.inner.artifacts,
281            input.binding,
282            &composed_prompt,
283        )?;
284        let raw_fits = if let Some(raw_fits) = raw_fit_from_real_usage(
285            input.overflow_recovery,
286            projection
287                .log
288                .latest_real_usage()
289                .map(|(_, observed_tokens)| observed_tokens),
290            |tokens| compaction_input_fits(input.binding, input.internal_policy, tokens),
291        ) {
292            raw_fits
293        } else {
294            let raw_fit_tokens = if input.binding.descriptor.capabilities.compaction
295                == CompactionCapability::Native
296            {
297                self.estimated_request_tokens(input.session, &context.history, input.tools)?
298            } else {
299                let (history, _) = compaction_history(
300                    context.history.clone(),
301                    compaction_focus.as_deref(),
302                    &input.internal_policy.agent.composed_prompt,
303                );
304                self.estimated_request_tokens(input.session, &history, input.tools)?
305            };
306            compaction_input_fits(input.binding, input.internal_policy, raw_fit_tokens)
307        };
308        let context_tokens_before = if raw_fits {
309            self.estimated_request_tokens(input.session, &context.history, input.tools)?
310        } else {
311            events = self
312                .stage_tool_output_elision(
313                    input.session,
314                    events,
315                    input.actor_direct,
316                    input.origin.clone(),
317                )
318                .await?;
319            context = assemble_model_context(
320                &events,
321                &self.inner.artifacts,
322                input.binding,
323                &composed_prompt,
324            )?;
325            self.estimated_request_tokens(input.session, &context.history, input.tools)?
326        };
327        if !input.force && context_tokens_before < trigger_tokens {
328            return Ok(Arc::from(events));
329        }
330
331        // Rehydration runs outside a normal turn, so it uses the owner and binding that triggered
332        // this checkpoint.
333        let rehydration_turn_context = Arc::new(TurnAgentContext {
334            agent: input.owner_policy.agent.agent.clone(),
335            model: input.binding.selection.model.clone(),
336            adapter: policy::wire_adapter(input.binding.protocol_recipe.as_str()),
337            capabilities: input
338                .owner_policy
339                .model_capabilities(input.binding)
340                .ok_or(EngineError::NoRunnableModel)?,
341        });
342
343        let input_through_seq = events.last().map_or(0, |event| event.seq);
344        let previous = projection.log.latest_checkpoint_seq();
345        let source_from_seq = if previous == 0 {
346            1
347        } else {
348            previous.saturating_add(1)
349        };
350        let boundaries = ContextCheckpointBoundaries {
351            source_from_seq,
352            source_through_seq: input_through_seq,
353            input_through_seq,
354            prior_checkpoint_seq: (previous > 0).then_some(previous),
355        };
356        let summary_limit = SummaryByteLimit::new(config.max_summary_bytes as u64)
357            .map_err(|error| EngineError::from(ModelError::invalid_request(error.to_string())))?;
358        let native_checkpoint = if input.binding.descriptor.capabilities.compaction
359            == CompactionCapability::Native
360        {
361            if let Ok(model) = policy::resolve_model(input.binding, &input.owner_policy.runtime) {
362                let mut request =
363                    ModelRequest::new(context.history.clone()).with_tools(input.tools.to_vec());
364                if let Some(native_context) = context.native_context.clone() {
365                    request = request.with_native_context(native_context);
366                }
367                let cache_strategy = input
368                    .owner_policy
369                    .cache_strategy(input.binding, input.session);
370                let request =
371                    model.prepare_request_with_cache_strategy(request, cache_strategy.as_ref());
372                let mut compact_request = CompactionRequest::new(request);
373                let instructions = compaction_focus.clone();
374                compact_request = match input.binding.protocol_recipe.as_str() {
375                    "oven.openai.responses" => compact_request
376                        .with_openai_responses_compaction_options(
377                            OpenAiResponsesCompactionOptions {
378                                instructions,
379                                ..OpenAiResponsesCompactionOptions::default()
380                            },
381                        ),
382                    "oven.azure.openai.responses" => compact_request
383                        .with_azure_openai_compaction_options(AzureOpenAiCompactionOptions {
384                            instructions,
385                            ..AzureOpenAiCompactionOptions::default()
386                        }),
387                    _ => compact_request,
388                };
389                if model.model().supports_compaction(&compact_request) {
390                    let abort = AbortBridge::new(input.cancellation.child_token());
391                    match model.model().compact(compact_request, abort.signal()).await {
392                        Ok(result) => model_history::persist_native_context(
393                            result.native_context,
394                            input.binding,
395                        )
396                        .ok(),
397                        Err(_) => None,
398                    }
399                } else {
400                    None
401                }
402            } else {
403                None
404            }
405        } else {
406            None
407        };
408        let checkpoint_prefix =
409            model_history::checkpoint_retained_history(&context.history, &events, None);
410        if let Some(window) = native_checkpoint {
411            let input_tokens_after = estimated_request_tokens(&checkpoint_prefix, input.tools)?;
412            let budgets = ContextCheckpointBudgets {
413                context_limit_tokens: context_limit,
414                trigger_tokens: trigger_tokens.max(1).min(context_limit),
415                input_tokens_before: context_tokens_before,
416                input_tokens_after,
417                max_summary_bytes: summary_limit,
418            };
419            let commit = ContextCheckpointCommit {
420                checkpoint: ContextCheckpoint::NativeWindow { window },
421                boundaries: boundaries.clone(),
422                budgets,
423            };
424            if commit.validate_for_binding(input.binding).is_ok() {
425                self.append_compaction_event(
426                    input.session,
427                    Some(input.run),
428                    Event::ContextCheckpointCommitted { commit },
429                    input.actor_direct,
430                    input.origin.clone(),
431                )
432                .await?;
433                return self
434                    .finalize_context_checkpoint(
435                        input,
436                        events,
437                        input_tokens_after,
438                        rehydration_turn_context,
439                    )
440                    .await;
441            }
442        }
443
444        let (history, instruction) = compaction_history(
445            context.history,
446            compaction_focus.as_deref(),
447            &input.internal_policy.agent.composed_prompt,
448        );
449        let input_tokens_before =
450            self.estimated_request_tokens(input.session, &history, input.tools)?;
451        let summary = self
452            .run_internal_history_agent(
453                input.session,
454                Some(input.run),
455                InternalAgentKind::ContextCompaction,
456                input.internal_policy,
457                InternalAgentHistoryInput {
458                    history,
459                    summary_source: instruction,
460                    tools: input.tools.to_vec(),
461                    reject_non_text: true,
462                },
463                InternalAgentExecution {
464                    cancellation: input.cancellation,
465                    actor_direct: input.actor_direct,
466                },
467            )
468            .await;
469        let Ok(summary) = summary else {
470            // Deliberately leave history fully intact when the raw context fit. Failed
471            // compaction no longer performs consolation elision on that path.
472            return Ok(Arc::from(events));
473        };
474        if summary.text.trim().is_empty() {
475            return Ok(Arc::from(events));
476        }
477        let checkpoint = InternalSummaryCheckpoint::new(
478            summary.text,
479            summary.invocation_id,
480            summary.internal_run_id,
481            summary_limit,
482        )
483        .map_err(|error| EngineError::from(ModelError::invalid_response(error.to_string())))?;
484        let retained_history = model_history::checkpoint_retained_history(
485            &checkpoint_prefix,
486            &events,
487            Some(checkpoint.summary()),
488        );
489        let input_tokens_after = estimated_request_tokens(&retained_history, input.tools)?;
490        let budgets = ContextCheckpointBudgets {
491            context_limit_tokens: context_limit,
492            trigger_tokens: trigger_tokens.max(1).min(context_limit),
493            input_tokens_before,
494            input_tokens_after,
495            max_summary_bytes: summary_limit,
496        };
497        let commit = ContextCheckpointCommit {
498            checkpoint: ContextCheckpoint::InternalSummary { checkpoint },
499            boundaries,
500            budgets,
501        };
502        if commit.validate().is_err() {
503            return Ok(Arc::from(events));
504        }
505        self.append_compaction_event(
506            input.session,
507            Some(input.run),
508            Event::ContextCheckpointCommitted { commit },
509            input.actor_direct,
510            input.origin.clone(),
511        )
512        .await?;
513        self.finalize_context_checkpoint(
514            input,
515            events,
516            input_tokens_after,
517            rehydration_turn_context,
518        )
519        .await
520    }

522    async fn finalize_context_checkpoint(
523        &self,
524        input: CompactionInput<'_>,
525        events: Vec<StoredEvent>,
526        input_tokens_after: u64,
527        turn_context: Arc<TurnAgentContext>,
528    ) -> Result<Arc<[StoredEvent]>, EngineError> {
529        self.inner
530            .context_token_estimators
531            .lock()
532            .unwrap_or_else(|poisoned| poisoned.into_inner())
533            .entry(input.session)
534            .or_default()
535            .record_compaction(input_tokens_after);
536        let files = self
537            .rehydrated_files(RehydrationInput {
538                session: input.session,
539                run: input.run,
540                owner_policy: input.owner_policy,
541                cancellation: input.cancellation,
542                events: &events,
543                turn_context,
544            })
545            .await;
546        if !files.is_empty() {
547            self.append_compaction_event(
548                input.session,
549                Some(input.run),
550                Event::ContextRehydrated { files },
551                input.actor_direct,
552                input.origin.clone(),
553            )
554            .await?;
555        }
556        Ok(self.inner.store.get(input.session)?.log.event_snapshot())
557    }

559    async fn stage_tool_output_elision(
560        &self,
561        session: SessionId,
562        events: Vec<StoredEvent>,
563        actor_direct: bool,
564        origin: cookie_agent_protocol::EventOrigin,
565    ) -> Result<Vec<StoredEvent>, EngineError> {
566        let protected_turns = events
567            .iter()
568            .rev()
569            .filter_map(|event| match &event.payload {
570                Event::ModelTurnCommitted { model_turn_seq, .. } => Some(*model_turn_seq),
571                _ => None,
572            })
573            .take(2)
574            .collect::<HashSet<_>>();
575        let starts = events
576            .iter()
577            .filter_map(|event| match &event.payload {
578                Event::ToolCallStarted { start } => {
579                    Some((start.tool_call_id, start.owner.model_turn_seq))
580                }
581                _ => None,
582            })
583            .collect::<HashMap<_, _>>();
584        let already_elided = events
585            .iter()
586            .filter_map(|event| match event.payload {
587                Event::ToolOutputElided { tool_call_id, .. } => Some(tool_call_id),
588                _ => None,
589            })
590            .collect::<HashSet<_>>();
591        for event in &events {
592            let Event::ToolCallTerminated { termination } = &event.payload else {
593                continue;
594            };
595            let Some(result) = &termination.result else {
596                continue;
597            };
598            let Some(model_turn_seq) = starts.get(&termination.tool_call_id) else {
599                continue;
600            };
601            if !should_elide_tool_output(
602                *model_turn_seq,
603                &protected_turns,
604                already_elided.contains(&termination.tool_call_id),
605                elidable_bytes(result),
606            ) {
607                continue;
608            }
609            let (retained, _) = self.inner.artifacts.retain(result.output.as_bytes())?;
610            self.append_compaction_event(
611                session,
612                event.run_id,
613                Event::ToolOutputElided {
614                    tool_call_id: termination.tool_call_id,
615                    original_bytes: result.output.len() as u64,
616                    retained,
617                },
618                actor_direct,
619                origin.clone(),
620            )
621            .await?;
622        }
623        Ok(self.inner.store.get(session)?.log.events())
624    }

626    fn estimated_request_tokens(
627        &self,
628        session: SessionId,
629        history: &[oven_sdk::HistoryTurn],
630        tools: &[ToolDefinition],
631    ) -> Result<u64, EngineError> {
632        let bytes = serialized_fit_request_bytes(history, tools)?;
633        let calibrated = self
634            .inner
635            .context_token_estimators
636            .lock()
637            .unwrap_or_else(|poisoned| poisoned.into_inner())
638            .get(&session)
639            .copied()
640            .and_then(|estimator| estimator.estimated_context_tokens(bytes));
641        if let Some(calibrated) = calibrated {
642            return Ok(calibrated);
643        }
644        Ok(estimated_tokens_for_bytes(bytes))
645    }

647    async fn rehydrated_files(&self, input: RehydrationInput<'_>) -> Vec<ContextRehydratedFile> {
648        let mut files = Vec::new();
649        let mut total = 0_usize;
650        for display_path in recent_read_candidates(input.events) {
651            if total >= REHYDRATION_MAX_TOTAL_BYTES {
652                break;
653            }
654            let call_id = ToolCallId::new_v7();
655            let prepared = self
656                .prepare_tool_call(
657                    input.session,
658                    input.run,
659                    ToolCall {
660                        id: call_id,
661                        name: "read".into(),
662                        arguments: serde_json::json!({
663                            "filePath": display_path,
664                            "limit": null,
665                            "offset": null
666                        }),
667                    },
668                    input.owner_policy,
669                    Arc::clone(&input.turn_context),
670                )
671                .await;
672            let Ok(prepared) = prepared.prepared else {
673                continue;
674            };
675            let Ok(session) = self.inner.store.get(input.session) else {
676                continue;
677            };
678            let permission_overlay = session.permission_overlay;
679            let grants = self.skill_grants_for_session(input.session);
680            let permission = self.inner.permissions.decide_operation_with_grants(
681                &input.owner_policy.agent,
682                Some(&permission_overlay),
683                grants.as_ref(),
684                &prepared.operation,
685                &prepared.policy_labels,
686                self.inner.store.cwd(),
687            );
688            if permission.effect != cookie_agent_protocol::PermissionEffect::Allow {
689                continue;
690            }
691            let Some(executor) = prepared.executor.lock().await.take() else {
692                continue;
693            };
694            let (progress_tx, _progress_rx) = mpsc::channel(1);
695            let result = executor
696                .execute(ToolExecutionContext {
697                    session: input.session,
698                    run: input.run,
699                    progress: ProgressSink::new(progress_tx, OutputHub::new(call_id, 1024)),
700                    cancellation: input.cancellation.child_token(),
701                    stdin: None,
702                    turn_context: Arc::clone(&input.turn_context),
703                    artifacts: self.inner.artifacts.clone(),
704                })
705                .await;
706            let Ok(result) = result else {
707                continue;
708            };
709            let remaining = REHYDRATION_MAX_TOTAL_BYTES.saturating_sub(total);
710            let content = truncate_utf8(&result.output, REHYDRATION_MAX_FILE_BYTES.min(remaining));
711            if content.is_empty() {
712                continue;
713            }
714            total = total.saturating_add(content.len());
715            files.push(ContextRehydratedFile {
716                path: safe_display(&display_path),
717                byte_length: content.len() as u64,
718                sha256: Sha256Digest::of_bytes(content.as_bytes()),
719                content,
720            });
721        }
722        files
723    }

815pub(crate) fn serialized_fit_request_bytes(
816    history: &[oven_sdk::HistoryTurn],
817    tools: &[ToolDefinition],
818) -> Result<usize, EngineError> {
819    let (history, attachment_surrogate_bytes) = fit_history(history);
820    let mut writer = CountingWriter::default();
821    serde_json::to_writer(&mut writer, &(history, tools))
822        .map_err(|error| EngineError::from(ModelError::invalid_request(error.to_string())))?;
823    Ok(writer.bytes.saturating_add(attachment_surrogate_bytes))
824}

826fn fit_history(history: &[oven_sdk::HistoryTurn]) -> (Vec<oven_sdk::HistoryTurn>, usize) {
827    // Media costs are flat-ish, usually hundreds to low-thousands of tokens per part, and do not
828    // track serialized byte size. Exclude image/PDF/audio files; video alone gets a flat surrogate
829    // because frame sampling can be materially expensive. Overestimation can spuriously compact
830    // and drop attachments during rehydration, while real usage calibrates residual error.
831    let mut history = history.to_vec();
832    let mut part_bytes = 0_usize;
833    for turn in &mut history {
834        match turn {
835            oven_sdk::HistoryTurn::System(message) => {
836                message.content.retain(|part| match part {
837                    oven_sdk::SystemPart::Text(text) => {
838                        part_bytes =
839                            part_bytes.saturating_add(fit_part_bytes(FitPart::Text(&text.text)));
840                        false
841                    }
842                    oven_sdk::SystemPart::Custom(_) => true,
843                });
844            }
845            oven_sdk::HistoryTurn::User(message) => {
846                message.content.retain(|part| match part {
847                    oven_sdk::InputPart::Text(text) => {
848                        part_bytes =
849                            part_bytes.saturating_add(fit_part_bytes(FitPart::Text(&text.text)));
850                        false
851                    }
852                    oven_sdk::InputPart::File(file) => {
853                        part_bytes = part_bytes
854                            .saturating_add(fit_part_bytes(FitPart::File(&file.media_type)));
855                        false
856                    }
857                    oven_sdk::InputPart::Custom(_) => true,
858                });
859            }
860            oven_sdk::HistoryTurn::Assistant(turn) => {
861                for part in &mut turn.message.content {
862                    if let oven_sdk::AssistantPart::ToolResult(result) = part {
863                        part_bytes = part_bytes
864                            .saturating_add(remove_tool_content_parts(&mut result.content));
865                    }
866                }
867                turn.message.content.retain(|part| match part {
868                    oven_sdk::AssistantPart::Text(text) => {
869                        part_bytes =
870                            part_bytes.saturating_add(fit_part_bytes(FitPart::Text(&text.text)));
871                        false
872                    }
873                    oven_sdk::AssistantPart::File(file) => {
874                        part_bytes = part_bytes
875                            .saturating_add(fit_part_bytes(FitPart::File(&file.media_type)));
876                        false
877                    }
878                    _ => true,
879                });
880            }
881            oven_sdk::HistoryTurn::Tool(message) => {
882                for result in &mut message.results {
883                    part_bytes =
884                        part_bytes.saturating_add(remove_tool_content_parts(&mut result.content));
885                }
886            }
887        }
888    }
889    (history, part_bytes)
890}
891
892fn remove_tool_content_parts(content: &mut oven_sdk::ToolContent) -> usize {
893    match content {
894        oven_sdk::ToolContent::Text(text) => {
895            let bytes = fit_part_bytes(FitPart::Text(text));
896            text.clear();
897            bytes
898        }
899        oven_sdk::ToolContent::Mixed(values) => {
900            let mut part_bytes = 0_usize;
901            values.retain(|value| match value {
902                oven_sdk::ContentValue::Text(text) => {
903                    part_bytes = part_bytes.saturating_add(fit_part_bytes(FitPart::Text(text)));
904                    false
905                }
906                oven_sdk::ContentValue::File(file) => {
907                    part_bytes =
908                        part_bytes.saturating_add(fit_part_bytes(FitPart::File(&file.media_type)));
909                    false
910                }
911                oven_sdk::ContentValue::Json(_) => true,
912            });
913            part_bytes
914        }
915        oven_sdk::ToolContent::Json(_) | oven_sdk::ToolContent::Denied { .. } => 0,
916    }
917}

925fn fit_part_bytes(part: FitPart<'_>) -> usize {
926    match part {
927        FitPart::Text(text) => text.len(),
928        FitPart::File(media_type) if media_type.starts_with("video/") => {
929            VIDEO_FILE_FIT_SURROGATE_BYTES

935fn elidable_bytes(result: &PersistedToolResult) -> usize {
936    let emitted_bytes = result
937        .additional_messages
938        .iter()
939        .map(|message| {
940            let marker_bytes =
941                usize::from(message.role == cookie_agent_protocol::ToolEmittedMessageRole::System)
942                    .saturating_mul(fit_part_bytes(FitPart::Text(
943                        model_history::TOOL_EMITTED_SYSTEM_USER_MARKER,
944                    )));
945            message.content.iter().fold(marker_bytes, |total, part| {
946                total.saturating_add(match part {
947                    ToolEmittedContent::Text(text) => fit_part_bytes(FitPart::Text(text)),
948                    ToolEmittedContent::File(attachment) => {
949                        fit_part_bytes(FitPart::File(attachment.mime_type.as_str()))
950                    }
951                })
952            })
953        })
954        .fold(0_usize, usize::saturating_add);
955    result.attachments.iter().fold(
956        fit_part_bytes(FitPart::Text(&result.output)).saturating_add(emitted_bytes),
957        |total, attachment| {
958            total.saturating_add(fit_part_bytes(FitPart::File(attachment.mime_type.as_str())))
959        },
960    )
961}

1074fn recent_read_candidates(events: &[StoredEvent]) -> Vec<String> {
1075    let turns = events
1076        .iter()
1077        .filter_map(|event| match &event.payload {
1078            Event::ModelTurnCommitted {
1079                model_turn_seq,
1080                turn,
1081                ..
1082            } => Some((*model_turn_seq, (event.run_id, turn))),
1083            _ => None,
1084        })
1085        .collect::<HashMap<_, _>>();
1086    let starts = events
1087        .iter()
1088        .filter_map(|event| match &event.payload {
1089            Event::ToolCallStarted { start } => Some((
1090                start.tool_call_id,
1091                (
1092                    event.run_id,
1093                    start.owner.model_turn_seq,
1094                    start.owner.content_index,
1095                ),
1096            )),
1097            _ => None,
1098        })
1099        .collect::<HashMap<_, _>>();
1100    let mut paths = Vec::new();
1101    let mut seen = HashSet::new();
1102    for event in events.iter().rev() {
1103        let Event::ToolCallTerminated { termination } = &event.payload else {
1104            continue;
1105        };
1106        if termination.result.is_none() {
1107            continue;
1108        }
1109        let Some((start_run, model_turn_seq, content_index)) =
1110            starts.get(&termination.tool_call_id)
1111        else {
1112            continue;
1113        };
1114        if event.run_id != *start_run {
1115            continue;
1116        }
1117        let Some((turn_run, turn)) = turns.get(model_turn_seq) else {
1118            continue;
1119        };
1120        if turn_run != start_run {
1121            continue;
1122        }
1123        let Some(PersistedAssistantPart::ToolCall { name, input, .. }) =
1124            turn.content.get(*content_index as usize)
1125        else {
1126            continue;
1127        };
1128        if name.as_str() != "read" {
1129            continue;
1130        }
1131        let Some(path) = input.get("filePath").and_then(Value::as_str) else {
1132            continue;
1133        };
1134        if seen.insert(path.to_owned()) {
1135            paths.push(path.to_owned());
1136        }
1137        if paths.len() == REHYDRATION_MAX_FILES {
1138            break;
1139        }
1140    }
1141    paths.reverse();
1142    paths
1143}
