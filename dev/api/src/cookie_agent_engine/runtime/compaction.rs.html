81    pub(super) async fn compact_session_direct(
82        &self,
83        session: SessionId,
84        focus: Option<&str>,
85    ) -> Result<bool, EngineError> {
86        let projection = self.inner.store.get(session)?;
87        if projection.status == SessionStatus::Running {
88            return Err(EngineError::SessionRunning(session));
89        }
90        let events = projection.log.events();
91        let run = events
92            .iter()
93            .rev()
94            .find_map(|event| {
95                matches!(event.payload, Event::RunStarted { .. }).then_some(event.run_id)
96            })
97            .flatten()
98            .ok_or(EngineError::NoRunnableModel)?;
99        let policy = self.historical_title_policy(&events, run)?;
100        let binding = active_compaction_binding(&policy, &events, run)?;
101        let internal_policy = self.internal_agent_policy(
102            InternalAgentKind::ContextCompaction,
103            &policy,
104            Some(binding),
105        )?;
106        let tools = self.tool_definitions(session, &policy)?;
107        let before = latest_checkpoint_seq(&events);
108        let compacted = self
109            .maybe_compact_context(CompactionInput {
110                session,
111                run,
112                cancellation: &CancellationToken::new(),
113                binding,
114                owner_policy: &policy,
115                internal_policy: &internal_policy,
116                tools: &tools,
117                events,
118                force: true,
119                overflow_recovery: false,
120                focus,
121                actor_direct: false,
122            })
123            .await?;
124        Ok(latest_checkpoint_seq(&compacted) > before)
125    }

127    pub(super) async fn maybe_compact_context(
128        &self,
129        input: CompactionInput<'_>,
130    ) -> Result<Vec<StoredEvent>, EngineError> {
131        let Some(context_limit) = input.binding.descriptor.capabilities.limits.context else {
132            return Ok(input.events);
133        };
134        let config = &self.inner.config.runtime.context_compaction;
135        let trigger_tokens = resolve_compaction_trigger(context_limit, &config.trigger);
136        if !compaction_gate(input.force, config.auto_compaction, trigger_tokens) {
137            return Ok(input.events);
138        }
139        if !input.force {
140            let Some((usage_seq, observed_tokens)) = latest_real_usage(&input.events) else {
141                return Ok(input.events);
142            };
143            let last_checkpoint_seq = latest_checkpoint_seq(&input.events);
144            if usage_seq < last_checkpoint_seq {
145                return Ok(input.events);
146            }
147            if !usage_reaches_compaction_trigger(observed_tokens, trigger_tokens) {
148                return Ok(input.events);
149            }
150        }
151
152        let requested_input_through_seq = input.events.last().map_or(0, |event| event.seq);
153        let current_events = self.inner.store.get(input.session)?.log.events();
154        if checkpoint_covers_input(&current_events, requested_input_through_seq) {
155            return Ok(current_events);
156        }
157
158        let composed_prompt = self.run_agent_prompt(input.session, input.run)?;
159        let mut events = input.events.clone();
160        let mut context = assemble_model_context(
161            &events,
162            &self.inner.artifacts,
163            input.binding,
164            &composed_prompt,
165        )?;
166        let raw_fits = if let Some(raw_fits) = raw_fit_from_real_usage(
167            input.overflow_recovery,
168            latest_real_usage(&events).map(|(_, observed_tokens)| observed_tokens),
169            |tokens| compaction_input_fits(input.binding, input.internal_policy, tokens),
170        ) {
171            raw_fits
172        } else {
173            let raw_fit_tokens = if input.binding.descriptor.capabilities.compaction
174                == CompactionCapability::Native
175            {
176                self.estimated_request_tokens(input.session, &context.history, input.tools)?
177            } else {
178                let (history, _) = compaction_history(context.history.clone(), input.focus);
179                self.estimated_request_tokens(input.session, &history, input.tools)?
180            };
181            compaction_input_fits(input.binding, input.internal_policy, raw_fit_tokens)
182        };
183        let context_tokens_before = if raw_fits {
184            self.estimated_request_tokens(input.session, &context.history, input.tools)?
185        } else {
186            events = self
187                .stage_tool_output_elision(input.session, events, input.actor_direct)
188                .await?;
189            context = assemble_model_context(
190                &events,
191                &self.inner.artifacts,
192                input.binding,
193                &composed_prompt,
194            )?;
195            self.estimated_request_tokens(input.session, &context.history, input.tools)?
196        };
197        if !input.force && context_tokens_before < trigger_tokens {
198            return Ok(events);
199        }
200
201        // Rehydration runs outside a normal turn, so it uses the owner and binding that triggered
202        // this checkpoint.
203        let rehydration_turn_context = Arc::new(TurnAgentContext {
204            agent: input.owner_policy.agent.agent.clone(),
205            capabilities: input
206                .owner_policy
207                .model_capabilities(input.binding)
208                .ok_or(EngineError::NoRunnableModel)?,
209        });
210
211        let input_through_seq = events.last().map_or(0, |event| event.seq);
212        let previous = latest_checkpoint_seq(&events);
213        let source_from_seq = if previous == 0 {
214            1
215        } else {
216            previous.saturating_add(1)
217        };
218        let boundaries = ContextCheckpointBoundaries {
219            source_from_seq,
220            source_through_seq: input_through_seq,
221            input_through_seq,
222            prior_checkpoint_seq: (previous > 0).then_some(previous),
223        };
224        let summary_limit = SummaryByteLimit::new(config.max_summary_bytes as u64)
225            .map_err(|error| EngineError::from(ModelError::invalid_request(error.to_string())))?;
226        let native_checkpoint = if input.binding.descriptor.capabilities.compaction
227            == CompactionCapability::Native
228        {
229            if let Ok(model) = policy::resolve_model(input.binding, &input.owner_policy.runtime) {
230                let mut request =
231                    ModelRequest::new(context.history.clone()).with_tools(input.tools.to_vec());
232                if let Some(native_context) = context.native_context.clone() {
233                    request = request.with_native_context(native_context);
234                }
235                let request = model.prepare_request(request);
236                let mut compact_request = CompactionRequest::new(request);
237                let instructions = input.focus.map(str::to_owned);
238                compact_request = match input.binding.protocol_recipe.as_str() {
239                    "oven.openai.responses" => compact_request
240                        .with_openai_responses_compaction_options(
241                            OpenAiResponsesCompactionOptions {
242                                instructions,
243                                ..OpenAiResponsesCompactionOptions::default()
244                            },
245                        ),
246                    "oven.azure.openai.responses" => compact_request
247                        .with_azure_openai_compaction_options(AzureOpenAiCompactionOptions {
248                            instructions,
249                            ..AzureOpenAiCompactionOptions::default()
250                        }),
251                    _ => compact_request,
252                };
253                if model.model().supports_compaction(&compact_request) {
254                    let abort = AbortBridge::new(input.cancellation.child_token());
255                    match model.model().compact(compact_request, abort.signal()).await {
256                        Ok(result) => model_history::persist_native_context(
257                            result.native_context,
258                            input.binding,
259                        )
260                        .ok(),
261                        Err(_) => None,
262                    }
263                } else {
264                    None
265                }
266            } else {
267                None
268            }
269        } else {
270            None
271        };
272        if let Some(window) = native_checkpoint {
273            let retained_history = context.history.get(..1).unwrap_or(&context.history);
274            let input_tokens_after = estimated_request_tokens(retained_history, input.tools)?;
275            let budgets = ContextCheckpointBudgets {
276                context_limit_tokens: context_limit,
277                trigger_tokens: trigger_tokens.max(1).min(context_limit),
278                input_tokens_before: context_tokens_before,
279                input_tokens_after,
280                max_summary_bytes: summary_limit,
281            };
282            let commit = ContextCheckpointCommit {
283                checkpoint: ContextCheckpoint::NativeWindow { window },
284                boundaries: boundaries.clone(),
285                budgets,
286            };
287            if commit.validate_for_binding(input.binding).is_ok() {
288                self.append_compaction_event(
289                    input.session,
290                    Some(input.run),
291                    Event::ContextCheckpointCommitted { commit },
292                    input.actor_direct,
293                )
294                .await?;
295                return self
296                    .finalize_context_checkpoint(
297                        input,
298                        events,
299                        input_tokens_after,
300                        rehydration_turn_context,
301                    )
302                    .await;
303            }
304        }
305
306        let (history, instruction) = compaction_history(context.history, input.focus);
307        let input_tokens_before =
308            self.estimated_request_tokens(input.session, &history, input.tools)?;
309        let summary = self
310            .run_internal_history_agent(
311                input.session,
312                Some(input.run),
313                InternalAgentKind::ContextCompaction,
314                input.internal_policy,
315                InternalAgentHistoryInput {
316                    history,
317                    summary_source: instruction,
318                    tools: input.tools.to_vec(),
319                    reject_non_text: true,
320                },
321                InternalAgentExecution {
322                    cancellation: input.cancellation,
323                    actor_direct: input.actor_direct,
324                },
325            )
326            .await;
327        let Ok(summary) = summary else {
328            // Deliberately leave history fully intact when the raw context fit. Failed
329            // compaction no longer performs consolation elision on that path.
330            return Ok(events);
331        };
332        if summary.text.trim().is_empty() {
333            return Ok(events);
334        }
335        let checkpoint = InternalSummaryCheckpoint::new(
336            summary.text,
337            summary.invocation_id,
338            summary.internal_run_id,
339            summary_limit,
340        )
341        .map_err(|error| EngineError::from(ModelError::invalid_response(error.to_string())))?;
342        let input_tokens_after = estimated_tokens_for_bytes(
343            model_history::framed_compaction_summary(checkpoint.summary()).len(),
344        );
345        let budgets = ContextCheckpointBudgets {
346            context_limit_tokens: context_limit,
347            trigger_tokens: trigger_tokens.max(1).min(context_limit),
348            input_tokens_before,
349            input_tokens_after,
350            max_summary_bytes: summary_limit,
351        };
352        let commit = ContextCheckpointCommit {
353            checkpoint: ContextCheckpoint::InternalSummary { checkpoint },
354            boundaries,
355            budgets,
356        };
357        if commit.validate().is_err() {
358            return Ok(events);
359        }
360        self.append_compaction_event(
361            input.session,
362            Some(input.run),
363            Event::ContextCheckpointCommitted { commit },
364            input.actor_direct,
365        )
366        .await?;
367        self.finalize_context_checkpoint(
368            input,
369            events,
370            input_tokens_after,
371            rehydration_turn_context,
372        )
373        .await
374    }

376    async fn finalize_context_checkpoint(
377        &self,
378        input: CompactionInput<'_>,
379        mut events: Vec<StoredEvent>,
380        input_tokens_after: u64,
381        turn_context: Arc<TurnAgentContext>,
382    ) -> Result<Vec<StoredEvent>, EngineError> {
383        self.inner
384            .context_token_estimators
385            .lock()
386            .unwrap_or_else(|poisoned| poisoned.into_inner())
387            .entry(input.session)
388            .or_default()
389            .record_compaction(input_tokens_after);
390        let files = self
391            .rehydrated_files(RehydrationInput {
392                session: input.session,
393                run: input.run,
394                owner_policy: input.owner_policy,
395                cancellation: input.cancellation,
396                events: &events,
397                turn_context,
398            })
399            .await;
400        if !files.is_empty() {
401            self.append_compaction_event(
402                input.session,
403                Some(input.run),
404                Event::ContextRehydrated { files },
405                input.actor_direct,
406            )
407            .await?;
408        }
409        events = self.inner.store.get(input.session)?.log.events();
410        Ok(events)
411    }

413    async fn stage_tool_output_elision(
414        &self,
415        session: SessionId,
416        events: Vec<StoredEvent>,
417        actor_direct: bool,
418    ) -> Result<Vec<StoredEvent>, EngineError> {
419        let protected_turns = events
420            .iter()
421            .rev()
422            .filter_map(|event| match &event.payload {
423                Event::ModelTurnCommitted { model_turn_seq, .. } => Some(*model_turn_seq),
424                _ => None,
425            })
426            .take(2)
427            .collect::<HashSet<_>>();
428        let starts = events
429            .iter()
430            .filter_map(|event| match &event.payload {
431                Event::ToolCallStarted { start } => {
432                    Some((start.tool_call_id, start.owner.model_turn_seq))
433                }
434                _ => None,
435            })
436            .collect::<HashMap<_, _>>();
437        let already_elided = events
438            .iter()
439            .filter_map(|event| match event.payload {
440                Event::ToolOutputElided { tool_call_id, .. } => Some(tool_call_id),
441                _ => None,
442            })
443            .collect::<HashSet<_>>();
444        for event in &events {
445            let Event::ToolCallTerminated { termination } = &event.payload else {
446                continue;
447            };
448            let Some(result) = &termination.result else {
449                continue;
450            };
451            let Some(model_turn_seq) = starts.get(&termination.tool_call_id) else {
452                continue;
453            };
454            if !should_elide_tool_output(
455                *model_turn_seq,
456                &protected_turns,
457                already_elided.contains(&termination.tool_call_id),
458                result.output.len(),
459            ) {
460                continue;
461            }
462            let (retained, _) = self.inner.artifacts.retain(result.output.as_bytes())?;
463            self.append_compaction_event(
464                session,
465                event.run_id,
466                Event::ToolOutputElided {
467                    tool_call_id: termination.tool_call_id,
468                    original_bytes: result.output.len() as u64,
469                    retained,
470                },
471                actor_direct,
472            )
473            .await?;
474        }
475        Ok(self.inner.store.get(session)?.log.events())
476    }

478    fn estimated_request_tokens(
479        &self,
480        session: SessionId,
481        history: &[oven_sdk::HistoryTurn],
482        tools: &[ToolDefinition],
483    ) -> Result<u64, EngineError> {
484        let bytes = serialized_request_bytes(history, tools)?;
485        let calibrated = self
486            .inner
487            .context_token_estimators
488            .lock()
489            .unwrap_or_else(|poisoned| poisoned.into_inner())
490            .get(&session)
491            .copied()
492            .and_then(|estimator| estimator.estimated_context_tokens(bytes));
493        Ok(calibrated.unwrap_or_else(|| estimated_tokens_for_bytes(bytes)))
494    }

496    async fn rehydrated_files(&self, input: RehydrationInput<'_>) -> Vec<ContextRehydratedFile> {
497        let mut files = Vec::new();
498        let mut total = 0_usize;
499        for display_path in recent_read_candidates(input.events) {
500            if total >= REHYDRATION_MAX_TOTAL_BYTES {
501                break;
502            }
503            let call_id = ToolCallId::new_v7();
504            let prepared = self
505                .prepare_tool_call(
506                    input.session,
507                    input.run,
508                    ToolCall {
509                        id: call_id,
510                        name: "read".into(),
511                        arguments: serde_json::json!({
512                            "filePath": display_path,
513                            "limit": null,
514                            "offset": null
515                        }),
516                    },
517                    input.owner_policy,
518                    Arc::clone(&input.turn_context),
519                )
520                .await;
521            let Ok(prepared) = prepared.prepared else {
522                continue;
523            };
524            let Ok(session) = self.inner.store.get(input.session) else {
525                continue;
526            };
527            let permission_overlay = session.permission_overlay;
528            let permission = self.inner.permissions.decide_operation_with_overlay(
529                &input.owner_policy.agent,
530                Some(&permission_overlay),
531                &prepared.operation,
532                &prepared.policy_labels,
533                self.inner.store.cwd(),
534            );
535            if permission.effect != cookie_agent_protocol::PermissionEffect::Allow {
536                continue;
537            }
538            let Some(executor) = prepared.executor.lock().await.take() else {
539                continue;
540            };
541            let (progress_tx, _progress_rx) = mpsc::channel(1);
542            let result = executor
543                .execute(ToolExecutionContext {
544                    session: input.session,
545                    run: input.run,
546                    progress: ProgressSink::new(progress_tx, OutputHub::new(call_id, 1024)),
547                    cancellation: input.cancellation.child_token(),
548                    stdin: None,
549                    turn_context: Arc::clone(&input.turn_context),
550                    artifacts: self.inner.artifacts.clone(),
551                })
552                .await;
553            let Ok(result) = result else {
554                continue;
555            };
556            let remaining = REHYDRATION_MAX_TOTAL_BYTES.saturating_sub(total);
557            let content = truncate_utf8(&result.output, REHYDRATION_MAX_FILE_BYTES.min(remaining));
558            if content.is_empty() {
559                continue;
560            }
561            total = total.saturating_add(content.len());
562            files.push(ContextRehydratedFile {
563                path: safe_display(&display_path),
564                byte_length: content.len() as u64,
565                sha256: Sha256Digest::of_bytes(content.as_bytes()),
566                content,
567            });
568        }
569        files
570    }

626pub(super) fn resolve_compaction_trigger(
627    context_limit: u64,
628    trigger: &ContextCompactionTrigger,
629) -> u64 {
630    match trigger {
631        ContextCompactionTrigger::Percent { percent } => context_limit
632            .saturating_mul(u64::from(*percent))
633            .saturating_div(100),
634        ContextCompactionTrigger::BufferTokens { buffer_tokens } => {
635            context_limit.saturating_sub(*buffer_tokens)
636        }
637    }
638}
639
640fn compaction_gate(force: bool, auto: bool, trigger_tokens: u64) -> bool {
641    force || (auto && trigger_tokens > 0)
642}
643
644fn usage_reaches_compaction_trigger(observed_tokens: u64, trigger_tokens: u64) -> bool {
645    observed_tokens >= trigger_tokens
646}

648fn latest_real_usage(events: &[StoredEvent]) -> Option<(u64, u64)> {
649    let recorded = events.iter().rev().find_map(|event| match &event.payload {
650        Event::ModelUsageRecorded { usage, .. } => usage_total(event.seq, usage),
651        _ => None,
652    });
653    recorded.or_else(|| {
654        events.iter().rev().find_map(|event| match &event.payload {
655            Event::ModelTurnCommitted { turn, .. } => usage_total(event.seq, &turn.usage),
656            _ => None,
657        })
658    })
659}

661fn usage_total(seq: u64, usage: &cookie_agent_protocol::Usage) -> Option<(u64, u64)> {
662    let input = usage.input_tokens;
663    let output = usage.output_tokens;
664    (input.is_some() || output.is_some()).then(|| {
665        (
666            seq,
667            input
668                .unwrap_or_default()
669                .saturating_add(output.unwrap_or_default()),
670        )
671    })
672}

674pub(super) fn latest_checkpoint_seq(events: &[StoredEvent]) -> u64 {
675    events
676        .iter()
677        .rev()
678        .find_map(|event| {
679            matches!(event.payload, Event::ContextCheckpointCommitted { .. }).then_some(event.seq)
680        })
681        .unwrap_or(0)
682}

684fn checkpoint_covers_input(events: &[StoredEvent], input_through_seq: u64) -> bool {
685    events.iter().rev().any(|event| {
686        matches!(
687            &event.payload,
688            Event::ContextCheckpointCommitted { commit }
689                if commit.boundaries.input_through_seq >= input_through_seq
690        )
691    })
692}

716fn raw_fit_from_real_usage(
717    overflow_recovery: bool,
718    observed_tokens: Option<u64>,
719    fits: impl FnOnce(u64) -> bool,
720) -> Option<bool> {
721    if overflow_recovery {
722        Some(false)
723    } else {
724        observed_tokens.filter(|tokens| fits(*tokens)).map(|_| true)
725    }

728fn compaction_input_fits(
729    binding: &cookie_agent_protocol::FrozenModelBinding,
730    internal_policy: &FrozenInternalAgentPolicy,
731    input_tokens: u64,
732) -> bool {
733    if binding.descriptor.capabilities.compaction != CompactionCapability::Native {
734        return internal_policy
735            .models
736            .iter()
737            .any(|candidate| internal_agent_input_fits(input_tokens, candidate, internal_policy));
738    }
739    input_tokens <= native_compaction_input_budget(binding, internal_policy)
740}
741
742fn native_compaction_input_budget(
743    binding: &cookie_agent_protocol::FrozenModelBinding,
744    internal_policy: &FrozenInternalAgentPolicy,
745) -> u64 {
746    let output_reserve = internal_agent_output_limit(binding, internal_policy)
747        .unwrap_or(DEFAULT_COMPACTION_OUTPUT_RESERVE_TOKENS);
748    binding
749        .descriptor
750        .capabilities
751        .limits
752        .context
753        .unwrap_or(0)
754        .saturating_sub(output_reserve)
755        .max(1)
756}

789fn recent_read_candidates(events: &[StoredEvent]) -> Vec<String> {
790    let turns = events
791        .iter()
792        .filter_map(|event| match &event.payload {
793            Event::ModelTurnCommitted {
794                model_turn_seq,
795                turn,
796                ..
797            } => Some((*model_turn_seq, (event.run_id, turn))),
798            _ => None,
799        })
800        .collect::<HashMap<_, _>>();
801    let starts = events
802        .iter()
803        .filter_map(|event| match &event.payload {
804            Event::ToolCallStarted { start } => Some((
805                start.tool_call_id,
806                (
807                    event.run_id,
808                    start.owner.model_turn_seq,
809                    start.owner.content_index,
810                ),
811            )),
812            _ => None,
813        })
814        .collect::<HashMap<_, _>>();
815    let mut paths = Vec::new();
816    let mut seen = HashSet::new();
817    for event in events.iter().rev() {
818        let Event::ToolCallTerminated { termination } = &event.payload else {
819            continue;
820        };
821        if termination.result.is_none() {
822            continue;
823        }
824        let Some((start_run, model_turn_seq, content_index)) =
825            starts.get(&termination.tool_call_id)
826        else {
827            continue;
828        };
829        if event.run_id != *start_run {
830            continue;
831        }
832        let Some((turn_run, turn)) = turns.get(model_turn_seq) else {
833            continue;
834        };
835        if turn_run != start_run {
836            continue;
837        }
838        let Some(PersistedAssistantPart::ToolCall { name, input, .. }) =
839            turn.content.get(*content_index as usize)
840        else {
841            continue;
842        };
843        if name.as_str() != "read" {
844            continue;
845        }
846        let Some(path) = input.get("filePath").and_then(Value::as_str) else {
847            continue;
848        };
849        if seen.insert(path.to_owned()) {
850            paths.push(path.to_owned());
851        }
852        if paths.len() == REHYDRATION_MAX_FILES {
853            break;
854        }
855    }
856    paths.reverse();
857    paths
858}
