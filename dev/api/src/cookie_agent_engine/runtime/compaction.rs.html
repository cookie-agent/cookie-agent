93    pub(super) async fn compact_session_direct(
94        &self,
95        session: SessionId,
96        focus: Option<&str>,
97    ) -> Result<SessionCompactResult, EngineError> {
98        let projection = self.inner.store.get(session)?;
99        if projection.status == SessionStatus::Running {
100            return Err(EngineError::SessionRunning(session));
101        }
102        let events = projection.log.event_snapshot();
103        let run = projection
104            .log
105            .last_run_started()
106            .map(|(_, run, _)| run)
107            .ok_or(EngineError::NoRunnableModel)?;
108        let policy = self.historical_title_policy(&events, run)?;
109        let binding = active_compaction_binding(&policy, &events, run)?;
110        let internal_policy = self.internal_agent_policy(
111            InternalAgentKind::ContextCompaction,
112            &policy,
113            Some(binding),
114        )?;
115        let tools = self.tool_definitions(session, &policy)?;
116        let before = projection.log.latest_checkpoint_seq();
117        match self
118            .maybe_compact_context(CompactionInput {
119                session,
120                run,
121                cancellation: &CancellationToken::new(),
122                binding,
123                owner_policy: &policy,
124                internal_policy: &internal_policy,
125                tools: &tools,
126                events,
127                force: true,
128                overflow_recovery: false,
129                focus,
130                actor_direct: false,
131            })
132            .await
133        {
134            Ok(_) => {}
135            Err(EngineError::CompactionCancelled(reason)) => {
136                return Ok(SessionCompactResult {
137                    compacted: false,
138                    cancellation_reason: Some(reason),
139                });
140            }
141            Err(error) => return Err(error),
142        }
143        Ok(SessionCompactResult {
144            compacted: projection.log.latest_checkpoint_seq() > before,
145            cancellation_reason: None,
146        })
147    }

149    pub(super) async fn maybe_compact_context(
150        &self,
151        input: CompactionInput<'_>,
152    ) -> Result<Arc<[StoredEvent]>, EngineError> {
153        let Some(context_limit) = input.binding.descriptor.capabilities.limits.context else {
154            return Ok(input.events);
155        };
156        let config = &self.inner.config.runtime.context_compaction;
157        let trigger_tokens = resolve_compaction_trigger(context_limit, &config.trigger);
158        if !compaction_gate(input.force, config.auto_compaction, trigger_tokens) {
159            return Ok(input.events);
160        }
161        let projection = self.inner.store.get(input.session)?;
162        if !input.force {
163            let log = &projection.log;
164            let Some((usage_seq, observed_tokens)) = log.latest_real_usage() else {
165                return Ok(input.events);
166            };
167            let last_checkpoint_seq = log.latest_checkpoint_seq();
168            if usage_seq < last_checkpoint_seq {
169                return Ok(input.events);
170            }
171            if !usage_reaches_compaction_trigger(observed_tokens, trigger_tokens) {
172                return Ok(input.events);
173            }
174        }
175
176        let requested_input_through_seq = input.events.last().map_or(0, |event| event.seq);
177        let current_events = projection.log.event_snapshot();
178        if projection
179            .log
180            .checkpoint_covers_input(requested_input_through_seq)
181        {
182            return Ok(current_events);
183        }
184
185        let mut compaction_focus = input.focus.map(str::to_owned);
186        let mut additions = Vec::new();
187        let context_id = crate::plugin::plugin_context_id();
188        for plugin in self.inner.plugins.interception_plugins(
189            cookie_agent_protocol::ExtensionInterceptionHook::SessionBeforeCompact,
190        ) {
191            let result = self
192                .inner
193                .plugins
194                .intercept_named::<_, cookie_agent_protocol::ExtensionSessionBeforeCompactResult>(
195                    &plugin,
196                    cookie_agent_protocol::PLUGIN_INTERCEPT_SESSION_BEFORE_COMPACT_METHOD,
197                    &ExtensionSessionBeforeCompactParams {
198                        session_id: input.session,
199                        context_id: context_id.clone(),
200                        checkpoint_id: format!("{}:{requested_input_through_seq}", input.session),
201                        additions: additions.clone(),
202                        instructions: compaction_focus.clone(),
203                    },
204                    Some(input.session),
205                    Some(&context_id),
206                )
207                .await;
208            match result {
209                Ok(result) => {
210                    if result.cancel {
211                        let reason = result
212                            .reason
213                            .unwrap_or_else(|| "compaction cancelled by plugin".into());
214                        self.record_plugin_diagnostic(
215                            input.session,
216                            plugin,
217                            PluginDiagnosticKind::HookBlocked,
218                            reason.clone(),
219                        );
220                        return Err(EngineError::CompactionCancelled(reason));
221                    }
222                    if let Some(instructions) = result.instructions_override {
223                        if instructions.len() > 64 * 1024 {
224                            self.record_plugin_diagnostic(
225                                input.session,
226                                plugin.clone(),
227                                PluginDiagnosticKind::InvalidModification,
228                                "plugin compaction instruction override exceeds the 64 KiB limit"
229                                    .into(),
230                            );
231                        } else {
232                            compaction_focus = Some(instructions);
233                        }
234                    }
235                    if let Some(addendum) = result.addendum.filter(|value| !value.is_empty()) {
236                        let addition_bytes = additions.iter().map(String::len).sum::<usize>();
237                        if addition_bytes.saturating_add(addendum.len()) > 64 * 1024 {
238                            self.record_plugin_diagnostic(
239                                input.session,
240                                plugin,
241                                PluginDiagnosticKind::InvalidModification,
242                                "plugin compaction additions exceed the 64 KiB limit".into(),
243                            );
244                            continue;
245                        }
246                        let focus = compaction_focus.get_or_insert_with(String::new);
247                        if !focus.is_empty() {
248                            focus.push('\n');
249                        }
250                        focus.push_str(&addendum);
251                        additions.push(addendum);
252                    }
253                }
254                Err(error) => {
255                    let kind = if error.contains("crashed") || error.contains("not connected") {
256                        PluginDiagnosticKind::InterceptionCrash
257                    } else {
258                        PluginDiagnosticKind::InterceptionTimeout
259                    };
260                    self.record_plugin_diagnostic(input.session, plugin, kind, error);
261                }
262            }
263        }
264
265        let composed_prompt = self.run_agent_prompt(input.session, input.run)?;
266        let mut events = input.events.to_vec();
267        let mut context = assemble_model_context(
268            &events,
269            &self.inner.artifacts,
270            input.binding,
271            &composed_prompt,
272        )?;
273        let raw_fits = if let Some(raw_fits) = raw_fit_from_real_usage(
274            input.overflow_recovery,
275            projection
276                .log
277                .latest_real_usage()
278                .map(|(_, observed_tokens)| observed_tokens),
279            |tokens| compaction_input_fits(input.binding, input.internal_policy, tokens),
280        ) {
281            raw_fits
282        } else {
283            let raw_fit_tokens = if input.binding.descriptor.capabilities.compaction
284                == CompactionCapability::Native
285            {
286                self.estimated_request_tokens(input.session, &context.history, input.tools)?
287            } else {
288                let (history, _) = compaction_history(
289                    context.history.clone(),
290                    compaction_focus.as_deref(),
291                    &input.internal_policy.agent.composed_prompt,
292                );
293                self.estimated_request_tokens(input.session, &history, input.tools)?
294            };
295            compaction_input_fits(input.binding, input.internal_policy, raw_fit_tokens)
296        };
297        let context_tokens_before = if raw_fits {
298            self.estimated_request_tokens(input.session, &context.history, input.tools)?
299        } else {
300            events = self
301                .stage_tool_output_elision(input.session, events, input.actor_direct)
302                .await?;
303            context = assemble_model_context(
304                &events,
305                &self.inner.artifacts,
306                input.binding,
307                &composed_prompt,
308            )?;
309            self.estimated_request_tokens(input.session, &context.history, input.tools)?
310        };
311        if !input.force && context_tokens_before < trigger_tokens {
312            return Ok(Arc::from(events));
313        }
314
315        // Rehydration runs outside a normal turn, so it uses the owner and binding that triggered
316        // this checkpoint.
317        let rehydration_turn_context = Arc::new(TurnAgentContext {
318            agent: input.owner_policy.agent.agent.clone(),
319            capabilities: input
320                .owner_policy
321                .model_capabilities(input.binding)
322                .ok_or(EngineError::NoRunnableModel)?,
323        });
324
325        let input_through_seq = events.last().map_or(0, |event| event.seq);
326        let previous = projection.log.latest_checkpoint_seq();
327        let source_from_seq = if previous == 0 {
328            1
329        } else {
330            previous.saturating_add(1)
331        };
332        let boundaries = ContextCheckpointBoundaries {
333            source_from_seq,
334            source_through_seq: input_through_seq,
335            input_through_seq,
336            prior_checkpoint_seq: (previous > 0).then_some(previous),
337        };
338        let summary_limit = SummaryByteLimit::new(config.max_summary_bytes as u64)
339            .map_err(|error| EngineError::from(ModelError::invalid_request(error.to_string())))?;
340        let native_checkpoint = if input.binding.descriptor.capabilities.compaction
341            == CompactionCapability::Native
342        {
343            if let Ok(model) = policy::resolve_model(input.binding, &input.owner_policy.runtime) {
344                let mut request =
345                    ModelRequest::new(context.history.clone()).with_tools(input.tools.to_vec());
346                if let Some(native_context) = context.native_context.clone() {
347                    request = request.with_native_context(native_context);
348                }
349                let request = model.prepare_request_with_cache_strategy(
350                    request,
351                    input.owner_policy.prompt_cache_strategy.as_ref(),
352                );
353                let mut compact_request = CompactionRequest::new(request);
354                let instructions = compaction_focus.clone();
355                compact_request = match input.binding.protocol_recipe.as_str() {
356                    "oven.openai.responses" => compact_request
357                        .with_openai_responses_compaction_options(
358                            OpenAiResponsesCompactionOptions {
359                                instructions,
360                                ..OpenAiResponsesCompactionOptions::default()
361                            },
362                        ),
363                    "oven.azure.openai.responses" => compact_request
364                        .with_azure_openai_compaction_options(AzureOpenAiCompactionOptions {
365                            instructions,
366                            ..AzureOpenAiCompactionOptions::default()
367                        }),
368                    _ => compact_request,
369                };
370                if model.model().supports_compaction(&compact_request) {
371                    let abort = AbortBridge::new(input.cancellation.child_token());
372                    match model.model().compact(compact_request, abort.signal()).await {
373                        Ok(result) => model_history::persist_native_context(
374                            result.native_context,
375                            input.binding,
376                        )
377                        .ok(),
378                        Err(_) => None,
379                    }
380                } else {
381                    None
382                }
383            } else {
384                None
385            }
386        } else {
387            None
388        };
389        let checkpoint_prefix =
390            model_history::checkpoint_retained_history(&context.history, &events, None);
391        if let Some(window) = native_checkpoint {
392            let input_tokens_after = estimated_request_tokens(&checkpoint_prefix, input.tools)?;
393            let budgets = ContextCheckpointBudgets {
394                context_limit_tokens: context_limit,
395                trigger_tokens: trigger_tokens.max(1).min(context_limit),
396                input_tokens_before: context_tokens_before,
397                input_tokens_after,
398                max_summary_bytes: summary_limit,
399            };
400            let commit = ContextCheckpointCommit {
401                checkpoint: ContextCheckpoint::NativeWindow { window },
402                boundaries: boundaries.clone(),
403                budgets,
404            };
405            if commit.validate_for_binding(input.binding).is_ok() {
406                self.append_compaction_event(
407                    input.session,
408                    Some(input.run),
409                    Event::ContextCheckpointCommitted { commit },
410                    input.actor_direct,
411                )
412                .await?;
413                return self
414                    .finalize_context_checkpoint(
415                        input,
416                        events,
417                        input_tokens_after,
418                        rehydration_turn_context,
419                    )
420                    .await;
421            }
422        }
423
424        let (history, instruction) = compaction_history(
425            context.history,
426            compaction_focus.as_deref(),
427            &input.internal_policy.agent.composed_prompt,
428        );
429        let input_tokens_before =
430            self.estimated_request_tokens(input.session, &history, input.tools)?;
431        let summary = self
432            .run_internal_history_agent(
433                input.session,
434                Some(input.run),
435                InternalAgentKind::ContextCompaction,
436                input.internal_policy,
437                InternalAgentHistoryInput {
438                    history,
439                    summary_source: instruction,
440                    tools: input.tools.to_vec(),
441                    reject_non_text: true,
442                },
443                InternalAgentExecution {
444                    cancellation: input.cancellation,
445                    actor_direct: input.actor_direct,
446                },
447            )
448            .await;
449        let Ok(summary) = summary else {
450            // Deliberately leave history fully intact when the raw context fit. Failed
451            // compaction no longer performs consolation elision on that path.
452            return Ok(Arc::from(events));
453        };
454        if summary.text.trim().is_empty() {
455            return Ok(Arc::from(events));
456        }
457        let checkpoint = InternalSummaryCheckpoint::new(
458            summary.text,
459            summary.invocation_id,
460            summary.internal_run_id,
461            summary_limit,
462        )
463        .map_err(|error| EngineError::from(ModelError::invalid_response(error.to_string())))?;
464        let retained_history = model_history::checkpoint_retained_history(
465            &checkpoint_prefix,
466            &events,
467            Some(checkpoint.summary()),
468        );
469        let input_tokens_after = estimated_request_tokens(&retained_history, input.tools)?;
470        let budgets = ContextCheckpointBudgets {
471            context_limit_tokens: context_limit,
472            trigger_tokens: trigger_tokens.max(1).min(context_limit),
473            input_tokens_before,
474            input_tokens_after,
475            max_summary_bytes: summary_limit,
476        };
477        let commit = ContextCheckpointCommit {
478            checkpoint: ContextCheckpoint::InternalSummary { checkpoint },
479            boundaries,
480            budgets,
481        };
482        if commit.validate().is_err() {
483            return Ok(Arc::from(events));
484        }
485        self.append_compaction_event(
486            input.session,
487            Some(input.run),
488            Event::ContextCheckpointCommitted { commit },
489            input.actor_direct,
490        )
491        .await?;
492        self.finalize_context_checkpoint(
493            input,
494            events,
495            input_tokens_after,
496            rehydration_turn_context,
497        )
498        .await
499    }

501    async fn finalize_context_checkpoint(
502        &self,
503        input: CompactionInput<'_>,
504        events: Vec<StoredEvent>,
505        input_tokens_after: u64,
506        turn_context: Arc<TurnAgentContext>,
507    ) -> Result<Arc<[StoredEvent]>, EngineError> {
508        self.inner
509            .context_token_estimators
510            .lock()
511            .unwrap_or_else(|poisoned| poisoned.into_inner())
512            .entry(input.session)
513            .or_default()
514            .record_compaction(input_tokens_after);
515        let files = self
516            .rehydrated_files(RehydrationInput {
517                session: input.session,
518                run: input.run,
519                owner_policy: input.owner_policy,
520                cancellation: input.cancellation,
521                events: &events,
522                turn_context,
523            })
524            .await;
525        if !files.is_empty() {
526            self.append_compaction_event(
527                input.session,
528                Some(input.run),
529                Event::ContextRehydrated { files },
530                input.actor_direct,
531            )
532            .await?;
533        }
534        Ok(self.inner.store.get(input.session)?.log.event_snapshot())
535    }

537    async fn stage_tool_output_elision(
538        &self,
539        session: SessionId,
540        events: Vec<StoredEvent>,
541        actor_direct: bool,
542    ) -> Result<Vec<StoredEvent>, EngineError> {
543        let protected_turns = events
544            .iter()
545            .rev()
546            .filter_map(|event| match &event.payload {
547                Event::ModelTurnCommitted { model_turn_seq, .. } => Some(*model_turn_seq),
548                _ => None,
549            })
550            .take(2)
551            .collect::<HashSet<_>>();
552        let starts = events
553            .iter()
554            .filter_map(|event| match &event.payload {
555                Event::ToolCallStarted { start } => {
556                    Some((start.tool_call_id, start.owner.model_turn_seq))
557                }
558                _ => None,
559            })
560            .collect::<HashMap<_, _>>();
561        let already_elided = events
562            .iter()
563            .filter_map(|event| match event.payload {
564                Event::ToolOutputElided { tool_call_id, .. } => Some(tool_call_id),
565                _ => None,
566            })
567            .collect::<HashSet<_>>();
568        for event in &events {
569            let Event::ToolCallTerminated { termination } = &event.payload else {
570                continue;
571            };
572            let Some(result) = &termination.result else {
573                continue;
574            };
575            let Some(model_turn_seq) = starts.get(&termination.tool_call_id) else {
576                continue;
577            };
578            if !should_elide_tool_output(
579                *model_turn_seq,
580                &protected_turns,
581                already_elided.contains(&termination.tool_call_id),
582                result.output.len(),
583            ) {
584                continue;
585            }
586            let (retained, _) = self.inner.artifacts.retain(result.output.as_bytes())?;
587            self.append_compaction_event(
588                session,
589                event.run_id,
590                Event::ToolOutputElided {
591                    tool_call_id: termination.tool_call_id,
592                    original_bytes: result.output.len() as u64,
593                    retained,
594                },
595                actor_direct,
596            )
597            .await?;
598        }
599        Ok(self.inner.store.get(session)?.log.events())
600    }

602    fn estimated_request_tokens(
603        &self,
604        session: SessionId,
605        history: &[oven_sdk::HistoryTurn],
606        tools: &[ToolDefinition],
607    ) -> Result<u64, EngineError> {
608        let bytes = serialized_request_bytes(history, tools)?;
609        let calibrated = self
610            .inner
611            .context_token_estimators
612            .lock()
613            .unwrap_or_else(|poisoned| poisoned.into_inner())
614            .get(&session)
615            .copied()
616            .and_then(|estimator| estimator.estimated_context_tokens(bytes));
617        Ok(calibrated.unwrap_or_else(|| estimated_tokens_for_bytes(bytes)))
618    }

620    async fn rehydrated_files(&self, input: RehydrationInput<'_>) -> Vec<ContextRehydratedFile> {
621        let mut files = Vec::new();
622        let mut total = 0_usize;
623        for display_path in recent_read_candidates(input.events) {
624            if total >= REHYDRATION_MAX_TOTAL_BYTES {
625                break;
626            }
627            let call_id = ToolCallId::new_v7();
628            let prepared = self
629                .prepare_tool_call(
630                    input.session,
631                    input.run,
632                    ToolCall {
633                        id: call_id,
634                        name: "read".into(),
635                        arguments: serde_json::json!({
636                            "filePath": display_path,
637                            "limit": null,
638                            "offset": null
639                        }),
640                    },
641                    input.owner_policy,
642                    Arc::clone(&input.turn_context),
643                )
644                .await;
645            let Ok(prepared) = prepared.prepared else {
646                continue;
647            };
648            let Ok(session) = self.inner.store.get(input.session) else {
649                continue;
650            };
651            let permission_overlay = session.permission_overlay;
652            let grants = self.skill_grants_for_session(input.session);
653            let permission = self.inner.permissions.decide_operation_with_grants(
654                &input.owner_policy.agent,
655                Some(&permission_overlay),
656                grants.as_ref(),
657                &prepared.operation,
658                &prepared.policy_labels,
659                self.inner.store.cwd(),
660            );
661            if permission.effect != cookie_agent_protocol::PermissionEffect::Allow {
662                continue;
663            }
664            let Some(executor) = prepared.executor.lock().await.take() else {
665                continue;
666            };
667            let (progress_tx, _progress_rx) = mpsc::channel(1);
668            let result = executor
669                .execute(ToolExecutionContext {
670                    session: input.session,
671                    run: input.run,
672                    progress: ProgressSink::new(progress_tx, OutputHub::new(call_id, 1024)),
673                    cancellation: input.cancellation.child_token(),
674                    stdin: None,
675                    turn_context: Arc::clone(&input.turn_context),
676                    artifacts: self.inner.artifacts.clone(),
677                })
678                .await;
679            let Ok(result) = result else {
680                continue;
681            };
682            let remaining = REHYDRATION_MAX_TOTAL_BYTES.saturating_sub(total);
683            let content = truncate_utf8(&result.output, REHYDRATION_MAX_FILE_BYTES.min(remaining));
684            if content.is_empty() {
685                continue;
686            }
687            total = total.saturating_add(content.len());
688            files.push(ContextRehydratedFile {
689                path: safe_display(&display_path),
690                byte_length: content.len() as u64,
691                sha256: Sha256Digest::of_bytes(content.as_bytes()),
692                content,
693            });
694        }
695        files
696    }

752pub(super) fn resolve_compaction_trigger(
753    context_limit: u64,
754    trigger: &ContextCompactionTrigger,
755) -> u64 {
756    match trigger {
757        ContextCompactionTrigger::Percent { percent } => context_limit
758            .saturating_mul(u64::from(*percent))
759            .saturating_div(100),
760        ContextCompactionTrigger::BufferTokens { buffer_tokens } => {
761            context_limit.saturating_sub(*buffer_tokens)
762        }
763    }
764}

785fn serialized_request_bytes(
786    history: &[oven_sdk::HistoryTurn],
787    tools: &[ToolDefinition],
788) -> Result<usize, EngineError> {
789    let mut writer = CountingWriter::default();
790    serde_json::to_writer(&mut writer, &(history, tools))
791        .map_err(|error| EngineError::from(ModelError::invalid_request(error.to_string())))?;
792    Ok(writer.bytes)
793}
794
795#[derive(Default)]
796struct CountingWriter {
797    bytes: usize,
798}
799
800impl io::Write for CountingWriter {
801    fn write(&mut self, buffer: &[u8]) -> io::Result<usize> {
802        self.bytes = self.bytes.saturating_add(buffer.len());
803        Ok(buffer.len())
804    }
805
806    fn flush(&mut self) -> io::Result<()> {
807        Ok(())
808    }
809}

836fn compaction_input_fits(
837    binding: &cookie_agent_protocol::FrozenModelBinding,
838    internal_policy: &FrozenInternalAgentPolicy,
839    input_tokens: u64,
840) -> bool {
841    if binding.descriptor.capabilities.compaction != CompactionCapability::Native {
842        return internal_policy
843            .models
844            .iter()
845            .any(|candidate| internal_agent_input_fits(input_tokens, candidate, internal_policy));
846    }
847    input_tokens <= native_compaction_input_budget(binding, internal_policy)
848}

850fn native_compaction_input_budget(
851    binding: &cookie_agent_protocol::FrozenModelBinding,
852    internal_policy: &FrozenInternalAgentPolicy,
853) -> u64 {
854    let output_reserve = internal_agent_output_limit(binding, internal_policy)
855        .unwrap_or(DEFAULT_COMPACTION_OUTPUT_RESERVE_TOKENS);
856    binding
857        .descriptor
858        .capabilities
859        .limits
860        .context
861        .unwrap_or(0)
862        .saturating_sub(output_reserve)
863        .max(1)
864}

873fn compaction_history(
874    mut history: Vec<oven_sdk::HistoryTurn>,
875    focus: Option<&str>,
876    system_prompt: &str,
877) -> (Vec<oven_sdk::HistoryTurn>, String) {
878    let system = oven_sdk::HistoryTurn::system(oven_sdk::SystemMessage::new(vec![
879        oven_sdk::SystemPart::Text(oven_sdk::TextPart::new(system_prompt)),
880    ]));
881    if matches!(history.first(), Some(oven_sdk::HistoryTurn::System(_))) {
882        history[0] = system;
883    } else {
884        history.insert(0, system);
885    }
886    let instruction = compaction_instruction(focus);
887    history.push(oven_sdk::HistoryTurn::user(oven_sdk::UserMessage::new(
888        vec![oven_sdk::InputPart::Text(oven_sdk::TextPart::new(
889            instruction.clone(),
890        ))],
891    )));
892    (history, instruction)
893}

906fn recent_read_candidates(events: &[StoredEvent]) -> Vec<String> {
907    let turns = events
908        .iter()
909        .filter_map(|event| match &event.payload {
910            Event::ModelTurnCommitted {
911                model_turn_seq,
912                turn,
913                ..
914            } => Some((*model_turn_seq, (event.run_id, turn))),
915            _ => None,
916        })
917        .collect::<HashMap<_, _>>();
918    let starts = events
919        .iter()
920        .filter_map(|event| match &event.payload {
921            Event::ToolCallStarted { start } => Some((
922                start.tool_call_id,
923                (
924                    event.run_id,
925                    start.owner.model_turn_seq,
926                    start.owner.content_index,
927                ),
928            )),
929            _ => None,
930        })
931        .collect::<HashMap<_, _>>();
932    let mut paths = Vec::new();
933    let mut seen = HashSet::new();
934    for event in events.iter().rev() {
935        let Event::ToolCallTerminated { termination } = &event.payload else {
936            continue;
937        };
938        if termination.result.is_none() {
939            continue;
940        }
941        let Some((start_run, model_turn_seq, content_index)) =
942            starts.get(&termination.tool_call_id)
943        else {
944            continue;
945        };
946        if event.run_id != *start_run {
947            continue;
948        }
949        let Some((turn_run, turn)) = turns.get(model_turn_seq) else {
950            continue;
951        };
952        if turn_run != start_run {
953            continue;
954        }
955        let Some(PersistedAssistantPart::ToolCall { name, input, .. }) =
956            turn.content.get(*content_index as usize)
957        else {
958            continue;
959        };
960        if name.as_str() != "read" {
961            continue;
962        }
963        let Some(path) = input.get("filePath").and_then(Value::as_str) else {
964            continue;
965        };
966        if seen.insert(path.to_owned()) {
967            paths.push(path.to_owned());
968        }
969        if paths.len() == REHYDRATION_MAX_FILES {
970            break;
971        }
972    }
973    paths.reverse();
974    paths
975}
