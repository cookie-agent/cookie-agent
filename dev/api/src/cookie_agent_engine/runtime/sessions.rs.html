19    pub(super) fn rebuild_visible_tree_grants(&self) {
20        let invalidated = self.inner.grant_journal.invalidated_ids();
21        let grants = self
22            .inner
23            .store
24            .all()
25            .into_iter()
26            .flat_map(|session| session.log.events())
27            .filter_map(|event| match event.payload {
28                Event::TreeApprovalGrantCommitted { grant }
29                    if !invalidated.contains(&grant.grant_id) =>
30                {
31                    Some(grant)
32                }
33                _ => None,
34            });
35        self.inner.approvals.replace(grants);
36    }

38    pub fn create_session(&self, selection: RunSelection) -> Result<SessionMeta, EngineError> {
39        self.reconcile_provider_store()?;
40        let runtime = self.current_runtime();
41        if runtime.result.snapshot.models.is_empty()
42            || !runtime
43                .result
44                .snapshot
45                .agents
46                .iter()
47                .any(|agent| agent.runnable_as_root)
48        {
49            return Err(EngineError::NoRunnableModel);
50        }
51        let agents = Arc::clone(&runtime.agents);
52        let agent = resolve_agent(&agents, &selection.agent)?.clone();
53        if !agent.runnable_as_root {
54            return Err(EngineError::NoRunnableModel);
55        }
56        let policy = freeze_root_agent_policy(
57            &agent,
58            agents,
59            Arc::clone(&runtime),
60            &selection.model,
61            self.inner.config.runtime.delegation.max_depth,
62            policy::ResultLimits {
63                tool_output_max_lines: self.inner.config.runtime.tool_output.max_lines,
64                tool_output_max_bytes: self.inner.config.runtime.tool_output.max_bytes,
65            },
66            self.inner.config.runtime.prompt_caching.strategy(),
67        )?;
68        let id = SessionId::new_v7();
69        let cwd_identity = cwd_identity(self.inner.store.cwd())?;
70        let creation = Event::SessionCreated {
71            origin: SessionOrigin::Root,
72            cwd_identity: cwd_identity.clone(),
73            creation_selection: selection.clone(),
74            creation_agent: Box::new(policy.agent.clone()),
75            runtime_revision: runtime.result.snapshot.runtime_revision.clone(),
76            catalog_revision: runtime.result.snapshot.catalog_revision.clone(),
77            provider_state_revision: runtime.result.snapshot.provider_state_revision.clone(),
78            model_revision: runtime.result.snapshot.model_revision.clone(),
79            agent_revision: runtime.result.snapshot.agent_revision.clone(),
80            recipe_registry_revision: runtime.result.snapshot.recipe_registry_revision.clone(),
81            manifest_revision: runtime.current_manifest.revision.clone(),
82        };
83        self.inner.store.create(id, creation)?;
84        self.spawn_actor(id);
85        Ok(self.inner.store.get(id)?.metadata())
86    }

97    pub fn get_session(&self, id: SessionId) -> Result<SessionMeta, EngineError> {
98        let session = self.inner.store.get(id)?;
99        #[cfg(test)]
100        let hook = self
101            .inner
102            .read_only_reopen_hook
103            .lock()
104            .expect("read-only reopen hook lock poisoned")
105            .take();
106        #[cfg(test)]
107        if let Some(hook) = hook {
108            if let Some(reached) = hook
109                .reached
110                .lock()
111                .expect("read-only reopen reached lock poisoned")
112                .take()
113            {
114                let _ = reached.send(());
115            }
116            let _ = hook
117                .release
118                .lock()
119                .expect("read-only reopen release lock poisoned")
120                .recv();
121        }
122        Ok(session.metadata())
123    }

152    pub fn agent_usage(&self, agent_id: AgentId) -> AgentUsageResult {
153        let mut usage = UsageRollup::default();
154        for session in self.inner.store.all_summaries() {
155            if let Some(agent) = session.agent_usage.get(&agent_id) {
156                crate::usage::merge(&mut usage, agent);
157            }
158        }
159        AgentUsageResult {
160            agent_id,
161            usage: crate::usage::with_pricing(
162                usage,
163                &self.inner.config.runtime.pricing,
164                &self.catalog_pricing(),
165            ),

199    pub fn delegate_targets(&self, id: SessionId) -> Result<Vec<AgentId>, EngineError> {
200        let session = self.inner.store.get(id)?;
201        let active = self
202            .inner
203            .active
204            .lock()
205            .unwrap_or_else(|poisoned| poisoned.into_inner())
206            .values()
207            .find(|active| active.session == id)
208            .cloned();
209        let Some(active) = active else {
210            return Ok(Vec::new());
211        };
212        let depth = session_depth(&session.meta.origin);
213        let Some(delegation) = &active.policy.agent.delegation else {
214            return Ok(Vec::new());
215        };
216        if depth >= delegation.effective_depth_ceiling {
217            return Ok(Vec::new());
218        }
219        Ok(delegation
220            .targets
221            .iter()
222            .filter(|target| {
223                active.policy.registry.get(target).is_some_and(|agent| {
224                    agent.document.frontmatter.enabled
225                        && matches!(
226                            agent.document.frontmatter.mode,
227                            cookie_agent_config::AgentMode::Subagent
228                                | cookie_agent_config::AgentMode::All
229                        )
230                })
231            })
232            .cloned()
233            .collect())
234    }
235    #[must_use]
236    pub fn children(&self, id: SessionId) -> Vec<cookie_agent_protocol::ChildSummary> {
237        let known: HashSet<_> = self
238            .inner
239            .delegation_events
240            .entries()
241            .into_iter()
242            .map(|entry| {
243                (
244                    entry.reservation.invocation_id,
245                    entry.reservation.parent_session_id,
246                    entry.reservation.child_session_id,
247                )
248            })
249            .collect();
250        self.inner
251            .store
252            .all_summaries()
253            .into_iter()
254            .filter_map(|child| match child.meta.origin {
255                SessionOrigin::Delegated {
256                    parent_session_id,
257                    invocation_id,
258                    ..
259                } if parent_session_id == id
260                    && known.contains(&(
261                        invocation_id,
262                        parent_session_id,
263                        child.meta.session_id,
264                    )) =>
265                {
266                    Some(ChildSummary {
267                        session_id: child.meta.session_id,
268                        agent: child.meta.creation_selection.agent.clone(),
269                        title: child.meta.title.clone(),
270                        title_updated_seq: child.meta.title_updated_seq,
271                        status: child.meta.status,
272                        usage: child.usage,
273                    })
274                }
275                _ => None,
276            })
277            .collect()
278    }

336    pub async fn rename_session(
337        &self,
338        params: SessionRenameParams,
339    ) -> Result<SessionRenameResult, EngineError> {
340        let session_id = params.session_id;
341        let reset = matches!(params.change, SessionRenameChange::Reset);
342        let mut result = self
343            .request(session_id, |reply| SessionCommand::Rename { params, reply })
344            .await?;
345        if reset {
346            self.generate_title_after_reset(session_id).await?;
347            result.session = self.inner.store.get(session_id)?.metadata();
348        }
349        Ok(result)
350    }
