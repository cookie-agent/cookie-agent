14#[derive(Deserialize)]
15#[serde(deny_unknown_fields)]
16struct BashManifest {
17    title: serde_json::Value,
18    streams: BashStreams,
19}

21#[derive(Deserialize)]
22#[serde(deny_unknown_fields)]
23struct BashStreams {
24    stdout: CapturedArtifact,
25    stderr: CapturedArtifact,

28#[derive(Deserialize)]
29struct CapturedArtifact {

34impl Engine {
35    pub fn read_tool_result(
36        &self,
37        session: SessionId,
38        tool_call_id: ToolCallId,
39        stream: Option<&str>,
40        offset_lines: u64,
41        limit_lines: u64,
42    ) -> Result<ToolResultReadPage, ToolError> {
43        let projection = self
44            .inner
45            .store
46            .get(session)
47            .map_err(|error| ToolError::execution(error.to_string()))?;
48        let events = projection.log.events();
49        let result = events
50            .iter()
51            .rev()
52            .find_map(|event| match &event.payload {
53                EventPayload::ToolCallTerminated { termination }
54                    if termination.tool_call_id == tool_call_id =>
55                {
56                    termination.result.as_ref()
57                }
58                _ => None,
59            })
60            .ok_or_else(|| {
61                ToolError::execution(format!(
62                    "tool result for call {tool_call_id} is not visible in this session"
63                ))
64            })?;
65
66        if let Some(truncation) = &result.truncation {
67            let digest = artifact_digest(&truncation.retained)?;
68            if let Some(stream) = stream {
69                let manifest = self
70                    .inner
71                    .artifacts
72                    .read_paged(digest, 0, 1)
73                    .map_err(|error| ToolError::execution(error.to_string()))?;
74                let manifest: BashManifest = serde_json::from_str(&manifest.content)
75                    .map_err(|_| ToolError::execution("retained result has no bash streams"))?;
76                let _ = manifest.title;
77                let artifact = match stream {
78                    "stdout" => manifest.streams.stdout,
79                    "stderr" => manifest.streams.stderr,
80                    _ => return Err(ToolError::execution("stream must be stdout or stderr")),
81                };
82                if artifact.reference.uri != format!("artifact://sha256/{}", artifact.sha256) {
83                    return Err(ToolError::execution("bash stream artifact is invalid"));
84                }
85                let page = self
86                    .inner
87                    .artifacts
88                    .read_paged(&artifact.sha256, offset_lines, limit_lines)
89                    .map_err(|error| ToolError::execution(error.to_string()))?;
90                return Ok(ToolResultReadPage {
91                    content: page.content,
92                    next_offset_lines: page.next_offset_lines,
93                    source: format!("truncation.{stream}"),
94                });
95            }
96            let page = self
97                .inner
98                .artifacts
99                .read_paged(digest, offset_lines, limit_lines)
100                .map_err(|error| ToolError::execution(error.to_string()))?;
101            return Ok(ToolResultReadPage {
102                content: page.content,
103                next_offset_lines: page.next_offset_lines,
104                source: "truncation".into(),
105            });
106        }
107
108        if stream.is_some() {
109            return Err(ToolError::execution(
110                "stream is only available for retained bash results",
111            ));
112        }
113        if let Some(retained) = events.iter().rev().find_map(|event| match &event.payload {
114            EventPayload::ToolOutputElided {
115                tool_call_id: candidate,
116                retained,
117                ..
118            } if *candidate == tool_call_id => Some(retained),
119            _ => None,
120        }) {
121            let page = self
122                .inner
123                .artifacts
124                .read_paged(artifact_digest(retained)?, offset_lines, limit_lines)
125                .map_err(|error| ToolError::execution(error.to_string()))?;
126            return Ok(ToolResultReadPage {
127                content: page.content,
128                next_offset_lines: page.next_offset_lines,
129                source: "elision".into(),
130            });
131        }
132
133        let page = page_inline(&result.output, offset_lines, limit_lines)?;
134        Ok(ToolResultReadPage {
135            content: page.content,
136            next_offset_lines: page.next_offset_lines,
137            source: "inline".into(),
138        })
139    }
140}
141
142fn artifact_digest(reference: &ArtifactReference) -> Result<&str, ToolError> {
143    reference
144        .uri
145        .strip_prefix("artifact://sha256/")
146        .filter(|digest| {
147            digest.len() == 64
148                && digest
149                    .bytes()
150                    .all(|byte| byte.is_ascii_digit() || (b'a'..=b'f').contains(&byte))
151        })
152        .ok_or_else(|| ToolError::execution("tool result artifact reference is invalid"))
153}
154
155fn page_inline(
156    content: &str,
157    offset_lines: u64,
158    limit_lines: u64,
159) -> Result<super::artifacts::ArtifactPage, ToolError> {
160    if limit_lines == 0 {
161        return Err(ToolError::execution(
162            "tool result page limit must be positive",
163        ));
164    }
165    let offset = usize::try_from(offset_lines).unwrap_or(usize::MAX);
166    let limit = usize::try_from(limit_lines).unwrap_or(usize::MAX);
167    let mut lines = content.split_inclusive('\n').skip(offset);
168    let content = lines.by_ref().take(limit).collect::<String>();
169    let has_more = lines.next().is_some();
170    Ok(super::artifacts::ArtifactPage {
171        content,
172        next_offset_lines: has_more.then_some(offset_lines.saturating_add(limit_lines)),
173    })
174}
