13pub struct InProcessStream {
14    sender: mpsc::Sender<MessageFrame>,
15    receiver: mpsc::Receiver<MessageFrame>,
16}

18#[must_use]
19pub fn in_process_pair(capacity: usize) -> (InProcessStream, InProcessStream) {
20    let (client_to_server_tx, client_to_server_rx) = mpsc::channel(capacity);
21    let (server_to_client_tx, server_to_client_rx) = mpsc::channel(capacity);
22    (
23        InProcessStream {
24            sender: client_to_server_tx,
25            receiver: server_to_client_rx,
26        },
27        InProcessStream {
28            sender: server_to_client_tx,
29            receiver: client_to_server_rx,
30        },
31    )
32}

34#[async_trait]
35impl Transport for InProcessStream {
36    async fn send(&mut self, frame: MessageFrame) -> Result<(), TransportError> {
37        self.sender
38            .send(frame)
39            .await
40            .map_err(|_| TransportError::Closed)
41    }
42
43    async fn recv(&mut self) -> Result<Option<MessageFrame>, TransportError> {
44        Ok(self.receiver.recv().await)
45    }
46}

48/// Tokio WebSocket transport for protocol clients.
49pub struct WebSocketTransport {
50    socket: WebSocketStream<MaybeTlsStream<tokio::net::TcpStream>>,
51}
52
53impl WebSocketTransport {
54    /// Connect to a validated daemon endpoint using the standard bearer token.
55    pub async fn connect(url: &str) -> Result<Self, TransportError> {
56        let token = Zeroizing::new(
57            load_auth_token().map_err(|error| TransportError::Other(error.to_string()))?,
58        );
59        Self::connect_with_token(url, &token).await
60    }
61
62    /// Connect with an explicit bearer token, primarily for isolated clients and tests.
63    pub async fn connect_with_token(url: &str, token: &str) -> Result<Self, TransportError> {
64        validate_websocket_url(url).map_err(|error| TransportError::Other(error.to_string()))?;
65        let request = authenticated_request(url, token)?;
66        let (socket, _) = tokio_tungstenite::connect_async(request)
67            .await
68            .map_err(|error| TransportError::Other(error.to_string()))?;
69        Ok(Self { socket })
70    }
71}

73#[async_trait]
74impl Transport for WebSocketTransport {
75    async fn send(&mut self, frame: MessageFrame) -> Result<(), TransportError> {
76        let text = match frame {
77            MessageFrame::Text(text) => text,
78            MessageFrame::Value(value) => serde_json::to_string(&value)?,
79        };
80        self.socket
81            .send(Message::Text(text.into()))
82            .await
83            .map_err(|error| TransportError::Other(error.to_string()))
84    }
85
86    async fn recv(&mut self) -> Result<Option<MessageFrame>, TransportError> {
87        loop {
88            match self.socket.next().await {
89                Some(Ok(Message::Text(text))) => {
90                    return Ok(Some(MessageFrame::Text(text.to_string())));
91                }
92                Some(Ok(Message::Close(_))) | None => return Ok(None),
93                Some(Ok(_)) => {}
94                Some(Err(error)) => {
95                    return Err(TransportError::Other(error.to_string()));
96                }
97            }
98        }
99    }
100}

102pub(crate) fn authenticated_request(
103    url: &str,
104    token: &str,
105) -> Result<tokio_tungstenite::tungstenite::http::Request<()>, TransportError> {
106    if token.len() != 43
107        || !token
108            .bytes()
109            .all(|byte| byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'_'))
110    {
111        return Err(TransportError::Other(
112            "invalid daemon authentication token".into(),
113        ));
114    }
115    let mut request = url
116        .into_client_request()
117        .map_err(|error| TransportError::Other(error.to_string()))?;
118    let authorization = Zeroizing::new(format!("Bearer {token}"));
119    let value = authorization
120        .parse()
121        .map_err(|_| TransportError::Other("invalid authorization header".into()))?;
122    request.headers_mut().insert("authorization", value);
123    Ok(request)
124}

126#[cfg(test)]
127mod tests {
128    use super::authenticated_request;

130    #[test]
131    fn websocket_auth_uses_a_bearer_header_without_url_credentials() {
132        let token = "A".repeat(43);
133        let request =
134            authenticated_request("ws://127.0.0.1:7419/ws", &token).expect("authenticated request");
135        assert_eq!(request.uri().to_string(), "ws://127.0.0.1:7419/ws");
136        assert_eq!(
137            request
138                .headers()
139                .get("authorization")
140                .and_then(|value| value.to_str().ok()),
141            Some(format!("Bearer {token}").as_str())
142        );
143        assert!(!request.uri().to_string().contains(&token));
144        assert!(authenticated_request("ws://127.0.0.1:7419/ws", "sentinel-secret").is_err());
145    }
